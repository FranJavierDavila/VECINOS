     1→import type { Mesh, Scene, TransformNode } from '@babylonjs/core';
     2→import type { BlockInstance, Content, PlayerPosition, World } from '@/game/core/types';
     3→import { B } from './babylon';
     4→import { instantiateActorAsset, type ActorRig } from './assets';
     5→import { hash01, interiorRoute, interiorTarget, layoutGeometryOf, type Pt } from '@/game/interior';
     6→import { stairGeomOf, stairPathPoints } from '@/game/stairs';
     7→import type { ElevatorsDirector } from './elevator';
     8→
     9→/** Plano frontal de las viviendas: z local 0 de los interiores (la cara de la puerta). */
    10→const frontZOf = (roomZ: number, commonDepth: number): number => roomZ - commonDepth / 2;
    11→
    12→interface Actor {
    13→  /** Residente dueño del nodo (para las rutas de ascensor y depuración). */
    14→  id: string;
    15→  /** Nodo posicionable: cápsula greybox o raíz del modelo glTF (pivote en pies). */
    16→  node: Mesh | TransformNode;
    17→  height: number;
    18→  /** Offset Y del pivote: cápsula centrada (+h/2) o modelo con pies en el origen (+margen). */
    19→  pivotY: number;
    20→  /** Cola de tramos en curso: el actor la agota ANTES de capturar el destino del mundo de nuevo
    21→   *  (ir al pozo → subir → salir: un salto del sim se ve como un viaje, nunca un teletransporte). */
    22→  path: { x: number; y: number; z: number }[];
    23→  /** Vida interior (Fase C): cuenta atrás para el próximo punto de deambulación y nº de vuelta
    24→   *  (el punto es determinista por residente+ vuelta: sin RNG global ni capturas irreproducibles). */
    25→  wanderT: number;
    26→  wanderLap: number;
    27→  /** Objetivo interior VIGENTE (coords locales de la vivienda): se refresca con la cadencia del
    28→   *  catálogo (npcWanderSeconds), no cada tick — amortigua el parpadeo de actividades con chance. */
    29→  targetLocal: Pt | null;
    30→  lastLoc: string | null;
    31→  /** Dirección de cutscenes (Fase D): mientras está activa, la IA normal no captura destinos;
    32→   *  el actor solo consume la cola de tramos que le marca el director (nunca teletransporta). */
    33→  directed: boolean;
    34→  /** Rumbo actual (rad): el actor MIRA hacia donde camina (v1.2, credibilidad 2.5D).
    35→   *  0 = el modelo mira +X (perfil); ±π/2 = de espaldas/frente al entrar o salir de casa. */
    36→  yaw: number;
    37→  /** Rumbo pendiente de terminar: el actor remata el giro aunque se pare (nunca queda torcido). */
    38→  targetYaw: number | null;
    39→  /** Gesto social en curso (stateAnimations: talk/laugh/argue/phone/sit…) o null. */
    40→  gesture: string | null;
    41→  /** Cuenta atrás del gesto temporal (0 = persistente hasta que alguien lo quite). */
    42→  gestureT: number;
    43→  /** El objetivo interior vigente es un ancla de asiento (rules.view.seatAnchors): se sienta al llegar. */
    44→  seated: boolean;
    45→  /** Viaje en ascensor registrado (v1.5): { x: puerta; from/to: plantas }. null = sin viaje. */
    46→  elevatorReq: { x: number; from: number; to: number } | null;
    47→  /** Fase del viaje que se estaba atendiendo (para detectar transiciones y encolar tramos). */
    48→  ridePhase: string | null;
    49→  /** Estaba habilitado el frame anterior (v1.10): la REAPARICIÓN coloca al actor donde dice el
    50→   *  sim (vuelve de la calle), nunca caminando desde su última posición visible por la pared. */
    51→  enabledPrev: boolean;
    52→  /** Rig del arte final (null mientras carga o si hay fallback greybox). */
    53→  rig: ActorRig | null;
    54→}
    55→
    56→/**
    57→ * API de dirección de cutscenes (Fase D): toma actores, colócalos en escena y márcales rutas.
    58→ * Mientras un actor está "taken", el bucle normal de IA no le captura destinos: solo consume
    59→ * la cola de tramos (la misma locomoción interpolada de siempre, cero teletransportes).
    60→ */
    61→export interface ActorsDirector {
    62→  /** Posición del nodo (null si el actor no existe o está oculto). */
    63→  nodeOf(id: string): { x: number; y: number; z: number } | null;
    64→  /** Crea el actor si falta y lo marca como dirigido. false si el residente no está en escena. */
    65→  take(id: string): boolean;
    66→  /** Suelta al actor: la IA vuelve a decidir destino (cola y objetivo interior reseteados). */
    67→  release(id: string): void;
    68→  /** Coloca al actor en un punto de escena (staging teatral, decisión del contenido). */
    69→  stageAt(id: string, x: number, floor: number, z: number): void;
    70→  /** Encola la ruta del nodo actual al destino (pozo incluido si cambia la planta). */
    71→  walkTo(id: string, dest: { x: number; z: number; floor: number; inside?: { block: string; local: Pt } }): boolean;
    72→  /** true cuando el actor no tiene tramos pendientes (llegó — o no existe). */
    73→  idle(id: string): boolean;
    74→  /** Punto sobre la cabeza del actor (bocadillos v1.10); null si no está en escena. */
    75→  headOf(id: string): { x: number; y: number; z: number } | null;
    76→  /** Gesto social en curso (stateAnimations: talk/laugh/argue/phone…) o null. */
    77→  gestureOf(id: string): string | null;
    78→  /** true si el actor está tomado por el director de cutscenes (el staging manda). */
    79→  isDirected(id: string): boolean;
    80→  /**
    81→   * Estado social explícito (v1.2): talk/laugh/argue/phone/sit… según stateAnimations del
    82→   * manifiesto. Con `seconds` > 0 el gesto expira solo (líneas de diálogo); null lo suelta
    83→   * ya (al colgar el teléfono, al cerrar el modal). El clip que falte cae al default del rig.
    84→   */
    85→  setGesture(id: string, state: string | null, seconds?: number): void;
    86→}
    87→
    88→/**
    89→ * El jugador y los vecinos: cápsulas greybox o, si el manifiesto de assets declara un
    90→ * modelo glTF para el arquetipo (Content.assets.actors), el modelo con sus animaciones
    91→ * (idle/walk por desplazamiento y clip por actividad vía activityAnimations).
    92→ * Color y altura salen de la vista del arquetipo; radios y fallbacks, de content.rules.view.
    93→ *
    94→ * Dinámico (multijugador): la plantilla se sincroniza con world.residents en cada
    95→ * update — residentes nuevos reciben nodo y residentes que ya no existen lo pierden.
    96→ */
    97→export function createActors(scene: Scene, world: World, content: Content, elevators?: ElevatorsDirector | null) {
    98→  const { Color3, MeshBuilder, Quaternion, StandardMaterial, Vector3 } = B();
    99→  const v = content.rules.view;
   100→  const actors = new Map<string, Actor>();
   101→
   102→  /** Crea la cápsula de un residente (greybox/fallback) y lanza la carga del modelo si el manifiesto lo declara. */
   103→  const ensureActor = (id: string, archetype: string): Actor => {
   104→    const view = content.archetypes[archetype]?.view;
   105→    const height = view?.height ?? v.actorFallbackHeight;
   106→    const mesh = MeshBuilder.CreateCapsule(id, { height, radius: v.actorRadius }, scene);
   107→    const m = new StandardMaterial(`${id}_mat`, scene);
   108→    m.diffuseColor = Color3.FromHexString(view?.color ?? v.actorFallbackColor);
   109→    m.emissiveColor = m.diffuseColor.scale(0.25);
   110→    mesh.material = m;
   111→    mesh.isPickable = false;
   112→    const a: Actor = { id, node: mesh, height, pivotY: height / 2, path: [], wanderT: 0, wanderLap: 0, targetLocal: null, lastLoc: null, directed: false, yaw: 0, targetYaw: null, gesture: null, gestureT: 0, seated: false, elevatorReq: null, ridePhase: null, enabledPrev: false, rig: null };
   113→    actors.set(id, a);
   114→
   115→    // Arte final (opcional): modelo glTF del manifiesto; cualquier fallo → cápsula.
   116→    const def = content.assets?.actors?.[archetype];
   117→    if (def && content.assets) {
   118→      void instantiateActorAsset(scene, def, content.assets.base, `${id}:${archetype}`).then((rig) => {
   119→        // La escena pudo morir mientras cargaba (restart/Cargar partida): no adjuntar nada viejo.
   120→        if (rig && scene.isDisposed) {
   121→          rig.dispose();
   122→          return;
   123→        }
   124→        if (!rig) return; // fallback greybox permanente
   125→        const cur = actors.get(id);
   126→        if (!cur || cur.rig) {
   127→          rig.dispose(); // el residente se fue (o ya llegó otro rig): no dejar basura
   128→          return;
   129→        }
   130→        // PRESERVAR POSE (v1.10): el modelo puede terminar de cargar EN PLENO paseo (los assets
   131→        // llegan async). Sin esto, el swap cápsula→rig recolocaba el nodo a origen y el actor
   132→        // "saltaba" (el guard antiguo lo enmascaraba con OTRO teletransporte al objetivo: la
   133→        // vecina con la maleta cambiando de planta de golpe). Pies y rumbo se conservan 1:1.
   134→        const feetY = cur.node.position.y - cur.pivotY;
   135→        rig.root.position.set(cur.node.position.x, feetY + 0.02, cur.node.position.z);
   136→        if (rig.root.rotationQuaternion) Quaternion.RotationYawPitchRollToRef(cur.yaw, 0, 0, rig.root.rotationQuaternion);
   137→        else rig.root.rotation.y = cur.yaw;
   138→        // El modelo se authora a nominalHeight: escala uniforme a la altura del arquetipo.
   139→        rig.root.scaling.setAll(height / def.nominalHeight);
   140→        cur.node.dispose(false, true); // cápsula fuera (libera también su material propio)
   141→        cur.node = rig.root;
   142→        cur.pivotY = 0.02; // pies en el suelo
   143→        cur.rig = rig;
   144→      });
   145→    }
   146→    return a;
   147→  };
   148→
   149→  const goal = new Vector3();
   150→  /** true si la fase del viaje cambió desde el último frame (hay que recolocar la cola). */
   151→  const stalePhase = (a: Actor, phase: string): boolean => a.ridePhase !== phase;
   152→  /** Pozos verticales del mundo (features stairs/elevator), para el RITMO de los tramos verticales. */
   153→  const shafts = Object.values(world.blocks).filter((b) => {
   154→    const f = content.blocks[b.defId]!.features;
   155→    return !!f && (f.stairs === true || f.elevator === true);
   156→  });
   157→  /** ¿Pozo UTILIZABLE para rutear EN ESTE MOMENTO? (v1.10) La escalera siempre; el ascensor
   158→   *  SOLO si funciona (elevator_status 'ok'). Se comprueba EN CADA RUTA porque el estado del
   159→   *  sim cambia en vivo (avería/reparación): un filtro estático enlazaría llamadas al pozo
   160→   *  averiado que el director cancela al frame siguiente (bucle llamada/cancelación). */
   161→  const shaftUsable = (b: BlockInstance): boolean => {
   162→    const f = content.blocks[b.defId]!.features!;
   163→    if (f.stairs === true) return true;
   164→    return f.elevator === true && b.params.elevator_status === 'ok';
   165→  };
   166→  const shaftXOf = (b: BlockInstance): number => (b.cell.col + b.size.w / 2) * v.cellW;
   167→  /** Calado transitablo del bloque (vivienda 2× o bloque común). */
   168→  const depthOfBlock = (b: BlockInstance): number => (b.kind === 'dwelling' ? v.depth : v.commonDepth);
   169→  /** Plano frontal de las viviendas: z local 0 de los interiores y origen del pasillo. */
   170→  const frontZ = frontZOf(v.roomZ, v.commonDepth);
   171→
   172→  /**
   173→   * Ruta (cola de tramos) del nodo actual al destino: cambio de planta → pozo más cercano
   174→   * (andar hasta él, subir/bajar, salir), y dentro de una vivienda con geometría 2D la ruta
   175→   * pasa por los huecos del catálogo (nunca a través de tabiques). La comparten la IA normal
   176→   * y el director de cutscenes: una sola verdad de locomoción.
   177→   * Devuelve false SOLO cuando el destino exige cambiar de planta y no hay pozo utilizable
   178→   * (estado imposible: el llamador decide si recoloca sin teatro).
   179→   */
   180→  const buildRoute = (a: Actor, dest: { x: number; y: number; z: number; inside?: { block: BlockInstance; local: Pt } | null }): boolean => {
   181→    const node = a.node.position;
   182→    let startX = node.x;
   183→    let startZ = node.z;
   184→    let verticalOk = true;
   185→    if (Math.abs(dest.y - node.y) > 0.05) {
   186→      const floorNow = Math.round(node.y / v.cellH);
   187→      // Como el sim: ASCENSOR primero si funciona (coste 1 < 3 de la escalera); la escalera
   188→      // es la vía de emergencia cuando el núcleo está averiado (v1.10: estado comprobado EN VIVO).
   189→      const elevHere = shafts
   190→        .filter((s) => s.cell.floor === floorNow && content.blocks[s.defId]!.features!.elevator === true && s.params.elevator_status === 'ok')
   191→        .sort((s1, s2) => Math.abs(shaftXOf(s1) - startX) - Math.abs(shaftXOf(s2) - startX))[0];
   192→      const here = elevHere ?? shafts
   193→        .filter((s) => s.cell.floor === floorNow && shaftUsable(s))
   194→        .sort((s1, s2) => Math.abs(shaftXOf(s1) - startX) - Math.abs(shaftXOf(s2) - startX))[0];
   195→      if (here) {
   196→        const sx = shaftXOf(here);
   197→        if (Math.abs(sx - startX) > 0.3) {
   198→          a.path.push({ x: sx, y: node.y, z: v.corridorZ });
   199→        }
   200→        // ¿Escalera o ascensor? La escalera REAL (v1.4) va por los peldaños; el ascensor
   201→        // OPERATIVO (v1.5) registra una LLAMADA: el actor camina a la puerta y espera a la
   202→        // cabina (capacidad, puertas y semáforo los lleva el director del ascensor).
   203→        const feat = content.blocks[here.defId]!.features!;
   204→        const stairPts = feat.stairs === true ? stairPathPoints(v, sx, node.y, dest.y) : [];
   205→        if (stairPts.length > 1) {
   206→          for (const pt of stairPts) a.path.push({ x: pt.x, y: pt.y, z: pt.z });
   207→        } else if (elevators && feat.elevator === true) {
   208→          const from = Math.round(node.y / v.cellH);
   209→          const to = Math.round(dest.y / v.cellH);
   210→          if (to !== from) {
   211→            elevators.requestRide(a.id, here.cell.col, from, to);
   212→            a.elevatorReq = { x: sx, from, to };
   213→            a.ridePhase = null;
   214→          }
   215→        } else {
   216→          a.path.push({ x: sx, y: dest.y, z: v.corridorZ });
   217→        }
   218→        startX = sx;
   219→        startZ = v.corridorZ;
   220→      } else {
   221→        verticalOk = false;
   222→      }
   223→    }
   224→    const legs: Pt[] = [];
   225→    if (dest.inside) {
   226→      const b = dest.inside.block;
   227→      const geo = layoutGeometryOf(content, b);
   228→      if (geo) {
   229→        const x0 = b.cell.col * v.cellW;
   230→        const outside = startZ < frontZ - 0.01;
   231→        const fromLocal: Pt = outside
   232→          ? { x: Math.min(b.size.w * v.cellW - v.margin, Math.max(v.margin, startX - x0)), z: 0.45 }
   233→          : { x: startX - x0, z: startZ - frontZ };
   234→        const route = interiorRoute(geo, fromLocal, dest.inside.local);
   235→        if (route) {
   236→          if (outside) legs.push({ x: x0 + fromLocal.x, z: frontZ + 0.45 });
   237→          for (const p of route) legs.push({ x: x0 + p.x, z: frontZ + p.z });
   238→        } else {
   239→          legs.push({ x: (x0 + b.size.w * v.cellW) / 2, z: frontZ + 0.6 }); // sin ruta legal: al frente
   240→        }
   241→      } else {
   242→        legs.push({ x: dest.x, z: dest.z });
   243→      }
   244→    }
   245→    if (legs.length === 0) legs.push({ x: dest.x, z: dest.z });
   246→    for (const p of legs) a.path.push({ x: p.x, y: dest.y, z: p.z });
   247→    return verticalOk;
   248→  };
   249→
   250→  return {
   251→    /** playerResidentId identifica al residente del jugador local (multijugador: puede ser un invitado). */
   252→    update(dt: number, player: PlayerPosition, playerResidentId: string) {
   253→      // Altas y bajas: residentes sin nodo → crear; nodos huérfanos → dispose + remove.
   254→      for (const id in world.residents) {
   255→        if (!actors.has(id)) ensureActor(id, world.residents[id]!.archetype);
   256→      }
   257→      const gone: string[] = [];
   258→      for (const [id, a] of actors) {
   259→        if (!world.residents[id]) gone.push(id);
   260→      }
   261→      for (const id of gone) {
   262→        const a = actors.get(id)!;
   263→        actors.delete(id);
   264→        if (a.rig) a.rig.dispose();
   265→        else if (a.node instanceof B().Mesh) a.node.dispose(false, true);
   266→        else a.node.dispose(false, false);
   267→      }
   268→
   269→      const k = 1 - Math.exp(-dt * 10);
   270→      // Ritmos del catálogo (rules.view): crucero de vecinos y subida por pozo.
   271→      const npcSpeed = v.npcWalkSpeed;
   272→      for (const [id, a] of actors) {
   273→        const r = world.residents[id]!;
   274→        const isPlayer = id === playerResidentId;
   275→        const nowEnabled = isPlayer || (r.present && r.location !== null);
   276→        a.node.setEnabled(nowEnabled);
   277→        if (!nowEnabled) {
   278→          a.enabledPrev = false;
   279→          continue;
   280→        }
   281→        // En escena lo coloca el director (v1.10): marcar habilitado SIN snap — si no, al soltar
   282→        // al actor tras una cutscene, el primer frame normal lo trataría como "reaparición"
   283→        // y lo teletransportaría al objetivo en vez de rutear desde donde está.
   284→        if (a.directed) a.enabledPrev = true;
   285→        const node = a.node.position;
   286→        const prevX = node.x;
   287→        const prevZ = node.z;
   288→        if (isPlayer) {
   289→          const inside = player.mode === 'inside';
   290→          // Dentro de la vivienda el actor sigue la posición 2D (x + profundidad z del layout);
   291→          // en el cubo de escalera (v1.4) manda la Z absoluta del recorrido (pathZ).
   292→          const z = inside ? frontZ + Math.min(Math.max(player.z, 0), v.depth) : (player.pathZ ?? v.corridorZ);
   293→          goal.set(player.x, player.floor * v.cellH + (inside ? 0.12 : 0.14) + a.pivotY, z);
   294→          if (node.lengthSquared() === 0) node.copyFrom(goal);
   295→          node.addInPlace(goal.subtract(node).scale(Math.min(1, k * 2)));
   296→        } else {
   297→          const b = world.blocks[r.location!]!;
   298→          const home = world.blocks[r.home]!;
   299→          const inRoom = b.kind === 'dwelling';
   300→          const x0 = b.cell.col * v.cellW;
   301→          // ── Viaje en ascensor (v1.5): mientras hay viaje registrado, el director manda —
   302→          // esperar ante la puerta, entrar al hueco, ser llevado por la cabina y salir en la
   303→          // planta de destino. Si el viaje se cancela (avería), status() null → el flujo
   304→          // normal recaptura destino y la ruta pasará por la escalera.
   305→          const elev = elevators ?? undefined;
   306→          const ride = a.elevatorReq && elev ? elev.status(id) : null;
   307→          if (a.elevatorReq && !ride) {
   308→            a.elevatorReq = null;
   309→            a.ridePhase = null;
   310→            a.path = [];
   311→          }
   312→          if (a.elevatorReq && ride) {
   313→            if (stalePhase(a, ride.phase)) {
   314→              // Transición de fase: recolocar la cola según a dónde hay que caminar ahora.
   315→              if (ride.phase === 'boarding') {
   316→                a.path = [{ x: ride.slotX, y: Math.round(ride.floor) * v.cellH + 0.14 + a.pivotY, z: ride.slotZ }];
   317→              } else if (ride.phase === 'exiting') {
   318→                a.path = [{ x: a.elevatorReq.x, y: a.elevatorReq.to * v.cellH + 0.14 + a.pivotY, z: v.corridorZ }];
   319→              } else {
   320→                a.path = [];
   321→              }
   322→              a.ridePhase = ride.phase;
   323→            }
   324→            if (ride.phase === 'waiting') {
   325→              goal.set(a.elevatorReq.x, a.elevatorReq.from * v.cellH + 0.14 + a.pivotY, v.corridorZ);
   326→              if (node.lengthSquared() === 0) node.copyFrom(goal);
   327→              node.addInPlace(goal.subtract(node).scale(Math.min(1, k * 2)));
   328→            } else if (ride.phase === 'riding') {
   329→              // La cabina manda: posición directa en el hueco (sin teatro ni interpolación propia).
   330→              a.path = [];
   331→              node.set(ride.slotX, ride.floor * v.cellH + 0.14 + a.pivotY, ride.slotZ);
   332→            } else if (ride.phase === 'boarding' && a.path.length === 0) {
   333→              elev!.notifyBoarded(id); // ya está en su hueco: la puerta puede cerrar
   334→            } else if (ride.phase === 'exiting' && a.path.length === 0) {
   335→              a.elevatorReq = null;
   336→              a.ridePhase = null;
   337→            }
   338→          }
   339→          // Con dirección de cutscene (Fase D) el actor solo consume su cola de tramos:
   340→          // ni objetivo de IA ni captura de ruta (el director ya se la ha dado).
   341→          if (!a.directed && !a.elevatorReq) {
   342→            let gx = b.layer === 'corridor' ? (home.cell.col + 0.5) * v.cellW : (x0 + b.size.w * v.cellW) / 2;
   343→            let gz = inRoom ? frontZ + 0.6 : v.corridorZ;
   344→            // ── Vida interior (Fase C): en vivienda, el punto lo decide el contenido. La actividad
   345→            // con `anchor` (cama, tv, stove…) coloca al vecino en SU estancia por el catálogo de
   346→            // anclas (rooms.json); sin ancla, deambula por las estancias abiertas. El objetivo se
   347→            // REFRESCA con la cadencia npcWanderSeconds (no cada tick): una actividad con chance
   348→            // que parpadea en el sim no arrastra al vecino por la casa. Cerradas jamás son destino.
   349→            const geo = inRoom ? layoutGeometryOf(content, b) : null;
   350→            let localTarget: Pt | null = null;
   351→            if (a.lastLoc !== r.location) {
   352→              a.lastLoc = r.location;
   353→              a.targetLocal = null;
   354→              a.wanderT = 0;
   355→              a.seated = false;
   356→            }
   357→            if (geo) {
   358→              a.wanderT -= dt;
   359→              if (a.path.length === 0 && a.wanderT <= 0) {
   360→                a.wanderLap += 1;
   361→                a.wanderT = v.npcWanderSeconds * (0.7 + 0.6 * hash01(id, a.wanderLap));
   362→                const act = r.activity ? content.archetypes[r.archetype]?.activities[r.activity] : undefined;
   363→                const anchorKind = act?.anchor;
   364→                const anchored = anchorKind ? interiorTarget(content, geo, anchorKind, id, 0) : null;
   365→                a.targetLocal = anchored ?? interiorTarget(content, geo, undefined, id, a.wanderLap);
   366→                // Ancla de asiento (catálogo: rules.view.seatAnchors): al llegar se sienta (clip "sit").
   367→                a.seated = anchored !== null && !!anchorKind && v.seatAnchors.includes(anchorKind);
   368→              }
   369→              localTarget = a.targetLocal;
   370→            }
   371→            if (localTarget) {
   372→              gx = x0 + localTarget.x;
   373→              gz = frontZ + localTarget.z;
   374→            }
   375→            goal.set(gx, b.cell.floor * v.cellH + 0.14 + a.pivotY, gz);
   376→            // Reaparición (v1.10): el vecino vuelve de la calle o entra en escena — aparece
   377→            // DONDE DICE EL SIM, no caminando desde su última posición visible (por la pared).
   378→            if (!a.enabledPrev) {
   379→              a.enabledPrev = true;
   380→              node.copyFrom(goal);
   381→              a.path = [];
   382→            }
   383→            // ── Locomoción interpolada (nada de teletransportes): el destino del sim se recorre a
   384→            // velocidad de crucero del catálogo, tramo a tramo. El sim adelanta a los vecinos a
   385→            // saltos del grafo (varias plantas de golpe): la ruta se RECALCULA DESDE EL PUNTO
   386→            // ACTUAL (andar al pozo → subir/bajar → salir) y el viaje se ve entero, pieza a pieza.
   387→            // Solo un estado realmente imposible (cambio de planta sin pozo utilizable, v1.10)
   388→            // recoloca sin teatro. Con ruta en curso se consume SIEMPRE hasta el final.
   389→            if (a.path.length === 0) {
   390→              const routed = buildRoute(a, { x: goal.x, y: goal.y, z: goal.z, inside: geo && localTarget ? { block: b, local: localTarget } : null });
   391→              if (!routed) {
   392→                node.copyFrom(goal);
   393→                a.path = [];
   394→              }
   395→            }
   396→          }
   397→          if (a.path.length > 0) {
   398→            const w = a.path[0]!;
   399→            const wdh = Math.hypot(w.x - node.x, w.z - node.z);
   400→            const wdy = w.y - node.y;
   401→            if (Math.abs(wdy) > 0.02 && wdh < 0.3) {
   402→              // Tramo VERTICAL en el pozo: ritmo del catálogo (ascensor operativo → cabina).
   403→              const shaftHere = shafts
   404→                .filter((s) => s.cell.floor === b.cell.floor && Math.abs(shaftXOf(s) - node.x) < 0.3)
   405→                .sort((s1, s2) => Math.abs(shaftXOf(s1) - node.x) - Math.abs(shaftXOf(s2) - node.x))[0];
   406→              const feat = shaftHere ? content.blocks[shaftHere.defId]!.features : undefined;
   407→              const rate = feat?.elevator && shaftHere!.params.elevator_status === 'ok' ? v.cellH / v.elevatorSeconds : v.cellH / v.stairsSeconds;
   408→              node.y += Math.sign(wdy) * Math.min(Math.abs(wdy), rate * dt);
   409→            } else if (wdh > 0.001) {
   410→              // Tramo con componente horizontal: crucero con llegada suave (frena al acercarse,
   411→              // nunca sobrepasa). Con dy el tramo es 3D (pendiente de escalera, Task 29): el paso
   412→              // reparte X/Z/Y con la MISMA velocidad de crucero, así la subida es física.
   413→              const step = Math.min(npcSpeed * dt, (0.35 + wdh * 2.5) * dt, wdh);
   414→              const t = step / wdh;
   415→              node.x += (w.x - node.x) * t;
   416→              node.z += (w.z - node.z) * t;
   417→              node.y += wdy * t;
   418→            }
   419→            // Tramo terminado → siguiente tramo de la ruta.
   420→            if (Math.hypot(w.x - node.x, w.z - node.z) < 0.06 && Math.abs(w.y - node.y) < 0.03) a.path.shift();
   421→          }
   422→        }
   423→
   424→        // ── Rotación 2.5D (v1.2): el actor MIRA hacia donde camina. El pasillo es lateral (eje
   425→        // X, perfil); entrar/salir de una vivienda es profundidad (eje Z: de espaldas al alejarse,
   426→        // de frente al acercarse). Giro por el arco corto con el ritmo del catálogo
   427→        // (rules.view.actorTurnSpeed). El rumbo queda PENDIENTE hasta rematarse: aunque el actor
   428→        // se pare a mitad de giro, termina de encarar la dirección en la que iba (nadie se queda
   429→        // torcido en una puerta).
   430→        // Aplicación ROBUSTA (v1.3): si el nodo lleva rotationQuaternion (el loader glTF puede
   431→        // ponérselo), Babylon IGNORA rotation.y — hay que escribir el cuaternión; si no,
   432→        // rotation.y de toda la vida. En ambos casos SOLO yaw: nadie se ladea ni se tumba.
   433→        const mdx = node.x - prevX;
   434→        const mdz = node.z - prevZ;
   435→        if (Math.hypot(mdx, mdz) / Math.max(1e-4, dt) > 0.25) a.targetYaw = Math.atan2(-mdz, mdx);
   436→        if (a.targetYaw !== null) {
   437→          let dYaw = a.targetYaw - a.yaw;
   438→          dYaw = ((dYaw + Math.PI) % (2 * Math.PI) + 2 * Math.PI) % (2 * Math.PI) - Math.PI;
   439→          if (Math.abs(dYaw) < 0.02) {
   440→            a.yaw = a.targetYaw; // rematado
   441→            a.targetYaw = null;
   442→          } else {
   443→            a.yaw += dYaw * Math.min(1, dt * v.actorTurnSpeed);
   444→          }
   445→        }
   446→        const q = a.node.rotationQuaternion;
   447→        if (q) Quaternion.RotationYawPitchRollToRef(a.yaw, 0, 0, q);
   448→        else a.node.rotation.y = a.yaw;
   449→
   450→        // ── Animación (solo rigs) por prioridad de estados (v1.2) + SINCRONÍA DE ZANCADA (v1.3):
   451→        //   gesto explícito → run (speed > runSpeed) → walk (mientras se avanza) → clip de
   452→        //   actividad parado → sentado en ancla → idle. El ratio de playback sale de la
   453→        //   velocidad PLANA real (m/s en el suelo; la Y no mueve piernas) contra la velocidad
   454→        //   natural del clip (rules.view.animSync): las piernas acompañan al cuerpo, cero patinaje.
   455→        if (a.gestureT > 0) {
   456→          a.gestureT -= dt;
   457→          if (a.gestureT <= 0) a.gesture = null;
   458→        }
   459→        if (a.rig) {
   460→          const speedH = Math.hypot(node.x - prevX, node.z - prevZ) / Math.max(1e-4, dt);
   461→          const def = content.assets?.actors?.[r.archetype];
   462→          const activityClip = def?.activityAnimations?.[r.activity ?? ''] ?? null;
   463→          const walkClip = def?.walkAnimation ?? 'walk';
   464→          let clip: string | null;
   465→          if (a.gesture) clip = a.gesture;
   466→          else if (speedH > v.runSpeed) clip = 'run';
   467→          else if (speedH > 0.25) clip = walkClip;
   468→          else if (activityClip) clip = activityClip;
   469→          else if (a.seated && a.path.length === 0) clip = 'sit';
   470→          else clip = null;
   471→          const sync = v.animSync;
   472→          let rate = 1;
   473→          if (!a.gesture && clip === 'run') rate = speedH / sync.runRefSpeed;
   474→          else if (!a.gesture && clip === walkClip) rate = speedH / sync.walkRefSpeed;
   475→          rate = Math.min(sync.maxRate, Math.max(sync.minRate, rate));
   476→          a.rig.play(clip, rate);
   477→        }
   478→      }
   479→    },
   480→    /** Dirección de cutscenes (Fase D): toma/coloca/ruta/suelta actores como teatro. */
   481→    director: {
   482→      nodeOf(id: string): { x: number; y: number; z: number } | null {
   483→        const a = actors.get(id);
   484→        return a && a.node.isEnabled() ? { x: a.node.position.x, y: a.node.position.y, z: a.node.position.z } : null;
   485→      },
   486→      take(id: string): boolean {
   487→        const r = world.residents[id];
   488→        if (!r || !r.present || r.location === null) return false;
   489→        if (!actors.has(id)) ensureActor(id, r.archetype);
   490→        const a = actors.get(id)!;
   491→        a.directed = true;
   492→        a.seated = false;
   493→        a.node.setEnabled(true);
   494→        return true;
   495→      },
   496→      release(id: string): void {
   497→        const a = actors.get(id);
   498→        if (!a) return;
   499→        a.directed = false;
   500→        a.path = [];
   501→        a.targetLocal = null;
   502→        a.wanderT = 0;
   503→        a.seated = false;
   504→        a.gesture = null;
   505→        a.gestureT = 0;
   506→        a.targetYaw = null;
   507→        // Si estaba esperando el ascensor, se cancela la llamada (el director libera el hueco).
   508→        if (a.elevatorReq && elevators) {
   509→          elevators.cancel(a.id);
   510→          a.elevatorReq = null;
   511→          a.ridePhase = null;
   512→        }
   513→      },
   514→      stageAt(id: string, x: number, floor: number, z: number): void {
   515→        const a = actors.get(id);
   516→        if (!a) return;
   517→        a.path = [];
   518→        // El teatro manda: cualquier viaje en curso se cancela antes de colocar al actor.
   519→        if (a.elevatorReq && elevators) {
   520→          elevators.cancel(a.id);
   521→          a.elevatorReq = null;
   522→          a.ridePhase = null;
   523→        }
   524→        a.node.position.set(x, floor * v.cellH + 0.14 + a.pivotY, z);
   525→      },
   526→      walkTo(id: string, dest: { x: number; z: number; floor: number; inside?: { block: string; local: Pt } }): boolean {
   527→        const a = actors.get(id);
   528→        if (!a) return false;
   529→        const b = dest.inside ? world.blocks[dest.inside.block] : undefined;
   530→        const inside = b && dest.inside
   531→          ? {
   532→              block: b,
   533→              local: {
   534→                x: Math.min(b.size.w * v.cellW - 0.2, Math.max(0.2, dest.inside.local.x)),
   535→                z: Math.min(depthOfBlock(b) - 0.2, Math.max(0.2, dest.inside.local.z)),
   536→              },
   537→            }
   538→          : null;
   539→        buildRoute(a, { x: dest.x, y: dest.floor * v.cellH + 0.14 + a.pivotY, z: dest.z, inside });
   540→        return true;
   541→      },
   542→      idle(id: string): boolean {
   543→        const a = actors.get(id);
   544→        return !a || (a.path.length === 0 && !a.elevatorReq);
   545→      },
   546→      headOf(id: string): { x: number; y: number; z: number } | null {
   547→        const a = actors.get(id);
   548→        if (!a || !a.node.isEnabled()) return null;
   549→        const p = a.node.position;
   550→        return { x: p.x, y: p.y + a.height + 0.18, z: p.z };
   551→      },
   552→      gestureOf(id: string): string | null {
   553→        return actors.get(id)?.gesture ?? null;
   554→      },
   555→      isDirected(id: string): boolean {
   556→        return actors.get(id)?.directed ?? false;
   557→      },
   558→      setGesture(id: string, state: string | null, seconds = 0): void {
   559→        const a = actors.get(id);
   560→        if (!a) return;
   561→        a.gesture = state;
   562→        a.gestureT = state && seconds > 0 ? seconds : 0;
   563→      },
   564→    },
   565→  };
   566→}
