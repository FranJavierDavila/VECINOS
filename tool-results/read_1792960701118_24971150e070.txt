    98→  async init(canvas: HTMLCanvasElement): Promise<void> {
    99→    if (this.engine || this.initializing) return;
   100→    this.initializing = true;
   101→    this.canvas = canvas;
   102→
   103→    try {
   104→      // 1) Contenido externo + validación (los errores de referencias se avisan, no abortan).
   105→      const { content, issues } = await loadContent();
   106→      this._content = content;
   107→      this.cutscenePlayer = new CutscenePlayer(content);
   108→      preloadSounds();
   109→      useGameStore.setState({ touchHint: content.rules.touch.hint });
   110→      if (issues.length > 0) {
   111→        useGameStore.getState().pushToast(`Contenido inválido: ${issues.length} errores de validación`, 'bad');
   112→      }
   113→
   114→      // 2) Motor de render (WebGPU con fallback WebGL2). La calidad la elige el jugador en Opciones.
   115→      const { engine, backend } = await createEngine(canvas);
   116→      this.engine = engine;
   117→      engine.setHardwareScalingLevel(QUALITY_SCALE[loadOptions().quality]);
   118→      window.addEventListener('resize', this.onResize);
   119→      await this.restart();
   120→      useGameStore.getState().markReady(backend);
   121→      startGamepadPolling();
   122→      // Logros (v1.17): si el jugador cambia el idioma de la interfaz, los textos ya
   123→      // resueltos de los logros se re-pintan con el catálogo (los desbloqueos se quedan).
   124→      useGameStore.subscribe((s, prev) => {
   125→        if (s.lang !== prev.lang) this.rebuildAchievementTexts();
   126→      });
   127→      engine.runRenderLoop(this.frame);
   128→      // Gancho de depuración/pruebas E2E (no lo usa la UI).
   129→      (window as unknown as { __vecinos?: unknown }).__vecinos = { store: useGameStore, controller: this };
   130→    } catch (err) {
   131→      console.error('No se pudo arrancar el juego (contenido o motor de render)', err);
   132→      useGameStore.getState().setEngineError(true);
   133→    }
   134→  }

   159→  onKeyDown(code: string): void {
   160→    // Cutscene en curso: E/Espacio/Enter/Escape saltan el espectáculo; nada más pasa.
   161→    if (this.cutscenePlayer?.active) {
   162→      if (code === 'KeyE' || code === 'Space' || code === 'Enter' || code === 'Escape') this.cutsceneSkip();
   163→      return;
   164→    }
   165→    if (this.keys.has(code)) return;
   166→    this.keys.add(code);
   167→    const store = useGameStore.getState();
   168→    if (selectModalOpen(store)) return; // los modales gestionan sus propias teclas
   169→    if (code === 'KeyE') {
   170→      this.interact();
   171→      return;
   172→    }
   173→    if (code === 'KeyJ') {
   174→      // Presidencia (v1.18): J junto al tablón convoca una junta con la agenda del catálogo.
   175→      if (this.logic?.interactTarget()?.kind === 'noticeboard') this.convocaJunta();
   176→      return;
   177→    }
   178→    const cam = CAM_BY_KEY[code];
   179→    if (cam) this.setCamMode(cam);
   180→  }
   181→

   183→    this.keys.delete(code);

   196→      down: held('KeyS', 'ArrowDown') || this.touch.down,
   197→      run: held('ShiftLeft', 'ShiftRight') || this.touch.run,
   198→    };
   199→  }

   258→  /** Interacción (tecla E o botón virtual ✋): usar el objeto cercano, entrar o salir. */
   259→  interact(): void {
   260→    // El ✋ táctil también salta la cutscene (es la única acción habilitada mientras dura).
   261→    if (this.cutscenePlayer?.active) {
   262→      this.cutsceneSkip();
   263→      return;
   264→    }
   265→    if (!this.logic) return;
   266→    // Ascensor (v1.9): si el prompt anuncia ascensor (sin objetivo de E delante), la E abre el
   267→    // panel de destino. Misma condición que promptText → lo que ves es lo que la E ejecuta.
   268→    if (!this.logic.interactTarget()) {
   269→      const panel = this.logic.elevatorPanelTarget();
   270→      if (panel) {
   271→        useGameStore.getState().setElevatorPanel(panel);
   272→        return;
   273→      }
   274→    }
   275→    const r = this.logic.interact();
   276→    if (r.type === 'rearEnter') {
   277→      // Apilamiento 3D (v1.19): el swap lo ejecuta building.update por fases; aquí solo
   278→      // encuadra la cámara dentro del bloque trasero.
   279→      this.setCamMode('estancia');
   280→      this.pump();
   281→      return;
   282→    }
   283→    if (r.type === 'rearExit') {
   284→      this.setCamMode('planta');
   285→      this.pump();
   286→      return;
   287→    }
   288→    if (r.type === 'entered') this.setCamMode('estancia');
   289→    else if (r.type === 'exited') this.setCamMode('planta');
   290→    else if (r.type === 'used') this.presentInteraction(r.kind);
   291→    this.pump();
   292→  }

   316→  /**
   317→   * Lecturas sin estado (buzón/tablón): texto del catálogo aquí; work/sleep/knock llegan como
   318→   * eventos del sim (ui_toast / knock_result) y no se tratan en esta vía.
   319→   */
   320→  private presentInteraction(kind: 'laptop' | 'bed' | 'mailbox' | 'noticeboard' | 'resident'): void {
   321→    if (!this.world) return;
   322→    const ic = this.content.rules.interact;
   323→    if (kind === 'mailbox') {
   324→      const lines = ic.mailbox.lines;
   325→      useGameStore.getState().pushToast(lines[this.world.tick % lines.length]!, 'info');
   326→    } else if (kind === 'noticeboard') {
   327→      const elevatorBlock = Object.values(this.world.blocks).find((b) => b.params.elevator_status !== undefined);
   328→      const status = String(elevatorBlock?.params.elevator_status ?? 'ok');
   329→      let text = ic.noticeboard.template
   330→        .replace('{day}', String(this.world.clock.day))
   331→        .replace('{funds}', String(this.world.community.funds))
   332→        .replace('{elevator}', this.content.rules.labels.elevator[status] ?? status)
   333→        .replace('{weather}', this.content.rules.labels.weather[this.world.weather.state] ?? this.world.weather.state);
   334→      // Presidencia (v1.18): con el mandato activo, el tablón también convoca.
   335→      const cfg = this.content.rules.sim.campaign;
   336→      if (cfg && useGameStore.getState().campaign?.active) text += `\n🗳️ [J] Convocar junta de vecinos (${cfg.presidentLabel.toLowerCase()})`;
   337→      useGameStore.getState().pushToast(text, 'info');
   338→    }
   339→  }

   341→  /**
   342→   * Junta extraordinaria convocada por la presidencia desde el tablón (v1.18): la agenda
   343→   * sale del catálogo (rules.sim.campaign.presetAgenda) y la votación la resuelve el sim
   344→   * (mismo flujo que las juntas de misión). Cooldown del catálogo; solo con mandato activo.
   345→   */
   346→  private convocaJunta(): void {
   347→    const world = this.world;
   348→    const cfg = this.content.rules.sim.campaign;
   349→    if (!world || !cfg || world.junta) return;
   350→    if (!useGameStore.getState().campaign?.active) {
   351→      useGameStore.getState().pushToast('Solo la presidencia convoca juntas. De momento, eres un vecino más (con suerte).', 'info');
   352→      return;
   353→    }
   354→    if (world.tick - this.lastConvocaTick < cfg.convocationCooldownMinutes) {
   355→      const left = Math.ceil(cfg.convocationCooldownMinutes - (world.tick - this.lastConvocaTick));
   356→      useGameStore.getState().pushToast(`Juana dice que la última junta fue hace poco. Próxima disponible en ${left} min. La democracia necesita descansos.`, 'info');
   357→      return;
   358→    }
   359→    this.lastConvocaTick = world.tick;
   360→    world.junta = { agenda: cfg.presetAgenda.map((i) => ({ ...i })), votes: {}, playerVoted: false };
   361→    useGameStore.getState().setJunta(cfg.presetAgenda.map((i) => ({ id: i.id, title: i.title, description: i.description, cost: i.cost })));
   362→    useGameStore.getState().pushLog('🗳️ Junta extraordinaria convocada desde el tablón. Los vecinos, en camino.', 'info');
   363→  }

   436→  private connectNet(): void {
   437→    this.net = new NetSession({
   438→      onState: (s) => this.applyRemoteState(s),
   439→      onParticipants: (p) => useGameStore.getState().setMpParticipants(p),
   440→      onJoined: (j) => {
   441→        this.myResidentId = j.yourResidentId;
   442→        this.isHost = j.isHost;
   443→        useGameStore.getState().setMpSession({ code: j.code, isHost: j.isHost, myResidentId: j.yourResidentId });
   444→        useGameStore.getState().setMpParticipants(j.participants);
   445→      },
   446→      onError: (m) => {
   447→        useGameStore.getState().pushToast(m, 'bad');
   448→      },
   449→      onStatus: (st) => {
   450→        if (!this.net) return;
   451→        if (st === 'connected' && useGameStore.getState().mpStatus === 'connecting') return; // el ack decide
   452→        if (st === 'disconnected' && useGameStore.getState().mpStatus === 'connected') {
   453→          useGameStore.getState().setMpStatus('error');
   454→          useGameStore.getState().setMpError('Conexión perdida con el servidor de la comunidad.');
   455→        }
   456→      },
   457→    });
   458→  }

   471→  /** Adopta el estado del servidor (autoridad). Primera vez: reinicia la escena con ese mundo. */
   472→  private applyRemoteState(s: SessionStatePayload): void {
   473→    if (s.seq <= this.appliedSeq) return;
   474→    this.appliedSeq = s.seq;
   475→    if (!this.world || this.world.scenarioId !== s.world.scenarioId) {
   476→      void this.restart(s.world).then(() => {

   486→    Object.assign(this.world, s.world);
   487→    this.world.events = s.events.slice();

   489→    this.paintBuilding();
   490→    this.reconcile();
   491→    this.syncHud();

   495→  private reconcile(): void {
   496→    if (!this.logic || !this.world || !this.net) return;
   497→    const me = this.world.residents[this.actorId()];
   498→    if (!me) return;
   499→    const pos = this.logic.position();
   500→    const locBlock = me.location ? this.world.blocks[me.location] : null;
   501→    if (locBlock) {
   502→      const insideOk = pos.mode === 'inside' && pos.insideBlock === locBlock.id;
   503→      const corridorOk = pos.mode === 'corridor' && !pos.insideBlock;
   504→      if (!(locBlock.kind === 'dwelling' ? insideOk : corridorOk) || Math.abs(pos.floor - locBlock.cell.floor) > 0.01) {
   505→        this.logic.forceLocation(locBlock.id);
   506→      }
   507→    } else if (pos.mode === 'inside') {
   508→      this.logic.forceLocation(null);
   509→    }
   510→  }

   597→  async save(): Promise<void> {

   616→      store.setSaveState('error');

   656→  /** (Re)crea el mundo y la escena. Con `state` restaura una partida guardada. */
   657→  private async restart(state?: World): Promise<void> {
   658→    if (!this.engine || !this.canvas || !this._content) return;
   659→    // Ninguna cutscene sobrevive a un cambio de mundo: cola fuera y teatro desmontado.
   660→    this.cutsceneQueue = [];
   661→    this.cutscenePlayer?.abort();
   662→    const { Color4, DirectionalLight, HemisphericLight, Scene, UniversalCamera, Vector3 } = B();
   663→    const view = this.content.rules.view;
   664→    useGameStore.getState().resetUi();
   665→
   666→    this.world = state ?? createWorld(this.content, this.activeScenarioId());
   667→    this.scenarioOverrides = this.content.scenarios[this.world.scenarioId ?? '']?.minigameOverrides ?? {};
   668→    // Guardados antiguos sin `entry`: la entrada es la zona común de planta baja.
   669→    if (!this.world.entry) {
   670→      this.world.entry = Object.values(this.world.blocks).find((b) => b.cell.floor === 0 && b.kind !== 'dwelling')?.id ?? Object.keys(this.world.blocks)[0]!;
   671→    }
   672→    // Migración de estancias de guardados v0.1: params y ranuras de muebles según catálogo.
   673→    for (const b of Object.values(this.world.blocks)) migrateRooms(this.content, b.rooms);
   674→    this.logic = new PlayerLogic(this.world, this.content, {
   675→      actorId: this.myResidentId ?? undefined,
   676→      send: (cmd) => this.sendCommand(cmd),
   677→    });
   678→
   679→    this.scene?.dispose();
   680→    this.scene = new Scene(this.engine);
   681→    this.scene.clearColor = new Color4(view.sky.r, view.sky.g, view.sky.b, 1);
   682→    const hemi = new HemisphericLight('ambient', new Vector3(...view.ambient.dir), this.scene);
   683→    hemi.intensity = view.ambient.intensity;
   684→    const dirLight = new DirectionalLight('sun', new Vector3(...view.sun.dir), this.scene);
   685→    dirLight.intensity = view.sun.intensity;
   686→
   687→    this.building = createBuilding(this.scene, this.world, this.content);
   688→    this.elevators = createElevators(this.scene, this.world, this.content);
   689→    this.actors = createActors(this.scene, this.world, this.content, this.elevators);
   690→    // Fondo urbano (v1.20): cielo dinámico por hora + fachadas parallax; la luz del catálogo
   691→    // evoluciona con el sol (el edificio vive la misma hora que el cielo).
   692→    this.backdrop?.dispose();
   693→    this.backdrop = createBackdrop(this.scene, this.world, this.content, { hemi, dir: dirLight });
   694→    // Calle (v1.20): aceras, carretera y jardines con sus props; las farolas leen la luz
   695→    // de día del fondo para encenderse al anochecer.
   696→    this.exterior?.dispose();
   697→    this.exterior = createExterior(this.scene, this.world, this.content, this.backdrop);
   698→    // Ecosistema del edificio (v1.14): cerebro data-driven + teatro de extras. Se recrea con
   699→    // cada mundo (referencia al World vivo) y sigue la posición del jugador para las luces.
   700→    this.ecosystem?.dispose();
   701→    this.ecosystem = createEcosystem(this.scene, this.world, this.content, () => this.logic!.position());
   702→    // Sonda de QA (SOLO dev): escena y controlador accesibles para el E2E automatizado.
   703→    // En producción no se expone nada (la partida queda encapsulada igual que siempre).
   704→    if (process.env.NODE_ENV !== 'production') {
   705→      (window as unknown as { __vecinosQA?: { scene: unknown; controller: unknown; elevators: unknown; ecosystem: unknown } }).__vecinosQA = {
   706→        scene: this.scene,
   707→        controller: this,
   708→        elevators: this.elevators,
   709→        ecosystem: this.ecosystem,
   710→      };
   711→    }
   712→    // Sonda E2E de la calle y las charlas (v1.20, SOLO dev): acceso indirecto a los directores.
   713→    (window as unknown as { __vecinos20?: unknown }).__vecinos20 = {
   714→      chats: () => this.chats,
   715→      exterior: () => this.exterior,
   716→      backdrop: () => this.backdrop,
   717→    };
   718→    // Bocadillos de expresión (v1.10): capa DOM sobre el canvas, recreada con el mundo
   719→    // (referencia al World vivo). Sin emotes.json en el catálogo, director nulo sin capa.
   720→    this.bubbles?.dispose();
   721→    const stage = this.canvas?.parentElement;
   722→    this.bubbles = stage && this.actors ? createBubbles(stage, this.scene, this.engine, this.world, this.content, this.actors.director) : null;
   723→    // Charlas de cruce (v1.20): después de actores y bocadillos — son sus dos clientes.
   724→    this.chats?.dispose();
   725→    this.chats = createChats(this.world, this.content, this.actors!.director, this.bubbles);
   726→    const blocks = Object.values(this.world.blocks);
   727→    // El cubo de escalera (v1.4) sobresale spanW−1 celdas por la derecha del edificio:
   728→    // el encuadre lo incluye para que la torre no se corte en cámara.
   729→    const hasStairs = blocks.some((b) => this.content.blocks[b.defId]?.features?.stairs === true);
   730→    const cols = Math.max(...blocks.map((b) => b.cell.col + b.size.w)) + (hasStairs ? this.content.rules.view.stairs.spanW - 1 : 0);
   731→    this.rig = new CameraRig(
   732→      new UniversalCamera('cam', new Vector3(0, 0, -30), this.scene),
   733→      this.engine,
   734→      cols,
   735→      Math.max(...blocks.map((b) => b.cell.floor)) + 1,
   736→      view.cellW,
   737→      view.cellH,
   738→    );
   739→    // v1.20: plano frontal y calado del bloque principal — la cámara encuadra los bloques
   740→    // traseros (apilamiento 3D) a la misma distancia que una estancia exterior.
   741→    this.rig.metric = { frontZ: view.roomZ - view.commonDepth / 2, depth: view.depth };
   742→    // Re-sincronización de vista (v0.6-b): la cámara sigue la posición del mundo (restaurado o
   743→    // nuevo), no la preferencia anterior. Dentro de una vivienda → estancia; en pasillo, la
   744→    // preferencia salvo que fuera 'estancia' (ahí estás en el pasillo, no en una estancia).
   745→    const pos = this.logic.position();
   746→    const store = useGameStore.getState();
   747→    const wanted: CamMode = pos.mode === 'inside' ? 'estancia' : store.camMode === 'estancia' ? 'planta' : store.camMode;
   748→    this.rig.mode = wanted;
   749→    store.setCamMode(wanted);
   750→
   751→    this.clockAcc = 0;
   752→    this.keys.clear();
   753→    this.phoneLines = [];
   754→    this.phoneMission = '';
   755→    this.phoneQueue = [];
   756→    this.lastHudSync = 0;
   757→
   758→    // Re-llamada (Fase F): si la partida montada tenía pasos de teléfono pendientes, vuelven a
   759→    // sonar. La cola de openPhone ordena el resto sin que ninguna llamada se pierda.
   760→    if (this.world) {
   761→      for (const m of this.world.missions) {
   762→        if (m.state !== 'active') continue;
   763→        const step = this.content.missions[m.id]?.steps[m.step];
   764→        if (step?.type === 'phone_call') this.openPhone(m.id, step.dialog);
   765→      }
   766→    }
   767→
   768→    this.pump();
   769→    // Logros (v1.17): primera foto del catálogo para el panel (y desbloqueos de la partida restaurada).
   770→    this.syncAchievements();
   771→    // Campaña (v1.18): primer estado del mandato (sobrevive a Cargar vía localStorage).
   772→    syncCampaign(this.world!, this.content);
   773→  }

   779→  // ───────────────────────── Bucle de render ─────────────────────────
   780→
   781→  private frame = (): void => {
   782→    if (!this.engine || !this.scene || !this.world || !this.logic || !this.building || !this.actors || !this.rig || !this.elevators) return;
   783→    const dt = Math.min(0.1, this.engine.getDeltaTime() / 1000);
   784→    const store = useGameStore.getState();
   785→    const frozen = selectModalOpen(store) || store.paused;
   786→
   787→    let ticked = false;
   788→    if (!frozen) {
   789→      this.logic.update(dt, this.readInput());
   790→      // Ascensor (v1.5): el director mueve cabinas y puertas; el jugador llama con W/S en el
   791→      // núcleo y su escalada EXTERNA la gobierna la cabina (espera → embarco → viaje → salida).
   792→      this.elevators.update(dt);
   793→      const climb = this.logic.climbInfo();
   794→      if (climb) {
   795→        if (!this.logic.externalRequested) {
   796→          const zone = this.logic.zone();
   797→          this.elevators.requestRide(this.actorId(), zone.cell.col, climb.from, climb.to);
   798→          this.logic.externalRequested = true;
   799→        }
   800→        const st = this.elevators.status(this.actorId());
   801→        if (!st) {
   802→          this.logic.finishClimbAtNearest(); // fuera de servicio: aterriza en la planta más cercana (sin teletransporte)
   803→          this.logic.externalRequested = false;
   804→          this.playerBoardT = 0;
   805→        } else {
   806→          this.logic.setClimbProgress(st.phase === 'riding' ? st.progress : 0);
   807→          const inside = st.phase === 'boarding' || st.phase === 'riding';
   808→          this.logic.setClimbZ(inside ? this.content.rules.view.roomZ - this.content.rules.view.commonDepth / 2 + this.content.rules.view.elevator.rideZ : null);
   809→          // El jugador "da un paso" dentro de la cabina un instante tras abrirse la puerta
   810→          // (el zOverride ya lo acerca); avisar al director para que cierre y viaje.
   811→          if (st.phase === 'boarding') {
   812→            this.playerBoardT += dt;
   813→            if (this.playerBoardT > 1.0) this.elevators.notifyBoarded(this.actorId());
   814→          } else {
   815→            this.playerBoardT = 0;
   816→          }
   817→          if (st.phase === 'exiting') {
   818→            this.logic.finishClimb();
   819→            this.logic.externalRequested = false;
   820→          }
   821→        }
   822→      } else if (this.logic.externalRequested) {
   823→        this.logic.externalRequested = false;
   824→      }
   825→      // En remoto el servidor es la autoridad del reloj: nada de ticks locales.
   826→      if (!this.net) {
   827→        this.clockAcc += dt * store.speed;
   828→        let guard = 0;
   829→        while (this.clockAcc >= 1 && guard < 60) {
   830→          tick(this.world, this.content);
   831→          this.clockAcc -= 1;
   832→          ticked = true;
   833→          guard += 1;
   834→        }
   835→      }
   836→    }
   837→
   838→    if (this.world.events.length > 0) this.handleEvents();
   839→    if (ticked) this.paintBuilding();
   840→
   841→    const now = performance.now();
   842→    if (now - this.lastHudSync > 200) {
   843→      this.lastHudSync = now;
   844→      useGameStore.getState().setFps(Math.round(this.engine.getFps()));
   845→      this.syncHud();
   846→    }
   847→
   848→    // Logros (v1.17): evaluación perezosa a cadencia (~cada 2 s de tiempo real).
   849→    // El evaluador es puro y solo lee el mundo, así que da igual si hay modal o pausa.
   850→    this.achAcc += dt;
   851→    if (this.achAcc >= 2) {
   852→      this.achAcc = 0;
   853→      this.syncAchievements();
   854→      // Campaña (v1.18): el día de mandato avanza con el reloj — misma cadencia.
   855→      syncCampaign(this.world!, this.content);
   856→    }
   857→
   858→    const pos = this.logic.position();
   859→    this.actors.update(dt, pos, this.actorId());
   860→    // Fondo urbano (v1.20): el cielo, el sol y las nubes siguen la hora del sim; las farolas
   861→    // de la calle leen la luz de día resultante. Los modales no paran el ambiente (solo el reloj).
   862→    this.backdrop?.update(dt);
   863→    this.exterior?.update(dt);
   864→    // Charlas de cruce (v1.20): en multijugador quedan en pausa (mismo criterio que el ecosistema).
   865→    if (!this.net) this.chats?.update(dt);
   866→    // Cutscene (Fase D): sus beats mueven cámara/actores DESPUÉS de los eventos y ANTES del render.
   867→    this.cutscenePlayer?.update(dt);
   868→    // Bocadillos (v1.10): después de actores y cámara — proyectan la posición FINAL del frame.
   869→    this.bubbles?.update(dt, this.actorId());
   870→    // Ecosistema (v1.14): cerebro a cadencia + extras + luces + deterioro. En multijugador
   871→    // queda en pausa (el servidor es la autoridad; mismo criterio que partyDemo).
   872→    if (!this.net) this.ecosystem?.update(dt);
   873→    // Si la cutscene acaba y queda cola, encadena la siguiente aquí: no dependemos de que el sim
   874→    // genere otro evento para descolgar la siguiente (durante el teatro el sim está congelado).
   875→    if (this.cutsceneQueue.length > 0 && !this.cutscenePlayer?.active) this.pumpCutscene();
   876→    // LOD por distancia: solo con cámaras cercanas (planta/estancia); el plano general (edificio) muestra todo.
   877→    // Privacidad (v1.17): el tercer argumento es el bloque donde está el jugador — las fachadas
   878→    // de las viviendas reaccionan al instante cuando entra o sale.
   879→    // v1.19: el cuarto argumento alimenta carteles guía de trasteros y el apilamiento 3D.
   880→    this.syncRear(pos);
   881→    this.building.update(
   882→      dt,
   883→      useGameStore.getState().camMode === 'edificio' ? undefined : { x: pos.x, floor: pos.floor },
   884→      pos.insideBlock,
   885→      { x: pos.x, z: pos.z, insideBlock: pos.insideBlock },
   886→    );
   887→    this.rig.update(dt, pos);

   891→  /**
   892→   * Apilamiento 3D (v1.19): cada frame decide la fase del bloque trasero activo (y apaga el
   893→   * resto). Trastero propio: E manda ('inside' mientras rearCell). Cuarto trasero de la
   894→   * vivienda: asoma al acercarte a la puerta del fondo ('peek') y se activa al cruzar ('inside').
   895→   */
   896→  private syncRear(pos: ReturnType<PlayerLogic['position']>): void {
   897→    if (!this.building || !this.world || !this.logic) return;
   898→    let activeId: string | null = null;
   899→    let phase: 'idle' | 'peek' | 'inside' = 'idle';
   900→    if (this.logic.rearCell) {
   901→      activeId = this.logic.rearCell.block;
   902→      phase = 'inside';
   903→    } else if (pos.mode === 'inside' && pos.insideBlock) {
   904→      const b = this.world.blocks[pos.insideBlock];
   905→      if (b && b.kind === 'dwelling') {
   906→        const geo = layoutGeometryOf(this.content, b);
   907→        const depth = this.content.rules.view.depth;
   908→        const rear = geo?.rooms.find((r) => r.rect.z + r.rect.d > depth + 0.01);
   909→        if (geo && rear) {
   910→          const localX = pos.x - b.cell.col * this.content.rules.view.cellW;
   911→          const room = roomAt(geo, localX, pos.z);
   912→          const rearDoor = geo.doorways.find((d) => d.axis === 'x' && d.b === rear.key);
   913→          const near =
   914→            rearDoor &&
   915→            pos.z > depth - this.content.rules.view.rearBlock.peekDistance &&
   916→            Math.abs(localX - rearDoor.at) < rearDoor.width / 2 + 0.3;
   917→          if (room.key === rear.key) {
   918→            activeId = b.id;
   919→            phase = 'inside';
   920→          } else if (near) {
   921→            activeId = b.id;
   922→            phase = 'peek';
   923→          }
   924→        }
   925→      }
   926→    }
   927→    for (const b of Object.values(this.world.blocks)) {
   928→      if (!this.building.hasRear(b.id)) continue;
   929→      this.building.setRearPhase(b.id, b.id === activeId ? phase : 'idle');
   930→    }
   931→  }

  1225→        case 'mailbox':
  1226→          return ic.mailbox.prompt;
  1227→        case 'noticeboard':
  1228→          return ic.noticeboard.prompt;

  1230→          return ic.laptop;
  1231→        case 'bed':
  1232→          return ic.bed;
  1233→        case 'storageCell':
  1234→          return ic.storageEnter;
  1235→        case 'rearExit':
  1236→          return ic.rearExit;
  1237→      }
  1238→    }
  1239→    const zone = logic.zone();
  1240→    const feat = this.content.blocks[zone.defId]!.features;
  1241→    if (feat?.stairs || feat?.elevator) {
  1242→      const ic = this.content.rules.interact;
  1243→      // Lo que dice el prompt es lo que W/S hace aquí: ascensor operativo manda; si no hay
  1244→      // ascensor utilizable pero sí escalera, escalera; núcleo averiado sin escalera = aviso.
  1245→      if (feat.elevator && world.blocks[zone.id]!.params.elevator_status === 'ok') return ic.elevator;
  1246→      if (feat.stairs) return ic.stairs;

  1298→    });
  1299→    const travel = travelMinutes(buildCirculation(world, this.content), world.entry, home);
  1300→    const elevatorBlock = Object.values(world.blocks).find((b) => b.params.elevator_status !== undefined);
  1301→    useGameStore.getState().syncHud({
  1302→      clock: { ...world.clock },
  1303→      weather: world.weather.state,
  1304→      funds: world.community.funds,
  1305→      wallet: player.wallet,
  1306→      reputation: player.reputation,
  1307→      elevator: String(elevatorBlock?.params.elevator_status ?? 'ok'),
  1308→      homeLabel: blockLabel(home),
  1309→      travelText: Number.isFinite(travel) ? `del portal a tu puerta: ${travel} min` : null,
  1310→      objective: this.objectiveUi(),
  1311→      prompt: this.promptText(),
  1312→      params,
  1313→      missions,
  1314→      homes: this.homesUi(),
  1315→      furnitureConfig: { refundRatio: this.content.rules.furniture.refundRatio, roomUnlockPrice: this.content.rules.furniture.roomUnlock.price },
  1316→    });
  1317→  }

  1321→  /** Logro del catálogo → forma de UI con textos resueltos al idioma actual del store. */
  1322→  private achievementUi(def: AchievementDef, unlocked: boolean, newly = false): AchievementUi {
  1323→    const lang = useGameStore.getState().lang;
  1324→    return {
  1325→      id: def.id,
  1326→      icon: def.icon,
  1327→      name: def.name[lang] ?? def.name.es,
  1328→      desc: def.desc[lang] ?? def.desc.es,
  1329→      hidden: def.hidden === true,
  1330→      unlocked,
  1331→      newly,
  1332→    };
  1333→  }
  1334→
  1335→  /** Re-pinta la lista de logros (mismos desbloqueos) con los textos del idioma actual. */
  1336→  private rebuildAchievementTexts(): void {
  1337→    const catalog = this._content?.achievements;
  1338→    if (!catalog) return;
  1339→    const prev = useGameStore.getState().achievements;
  1340→    useGameStore.getState().setAchievements(
  1341→      catalog.achievements.map((def) => {
  1342→        const old = prev.find((a) => a.id === def.id);
  1343→        return this.achievementUi(def, this.unlockedAchievements.has(def.id), old?.newly === true);
  1344→      }),
  1345→    );
  1346→  }
  1347→
  1348→  /**
  1349→   * Evalúa el catálogo de logros contra el estado actual del mundo. Solo LEE el sim: el
  1350→   * evaluador es puro (src/game/support/achievements.ts) y aquí se persisten los
  1351→   * desbloqueos en el perfil (localStorage), se anuncian en el registro y se encola el
  1352→   * popup estilo Steam (uno en pantalla; el resto queda `newly` para AchievementPopup).
  1353→   * Sin catálogo (achievements.json ausente) no hay logros y no se toca nada.
  1354→   */
  1355→  private syncAchievements(): void {
  1356→    const catalog = this._content?.achievements;
  1357→    const world = this.world;
  1358→    if (!catalog || !world) return;
  1359→    const player = world.residents[this.actorId()] ?? world.residents[world.playerId];
  1360→    if (!player) return;
  1361→    const store = useGameStore.getState();
  1362→
  1363→    // Snapshot con las mismas fuentes que el HUD (día del reloj, funds, ascensor del núcleo…).
  1364→    const elevatorBlock = Object.values(world.blocks).find((b) => b.params.elevator_status !== undefined);
  1365→    let bestAffinity = 0;
  1366→    for (const rel of player.relations) bestAffinity = Math.max(bestAffinity, rel.affinity);
  1367→    const snap: AchievementSnapshot = {
  1368→      day: world.clock.day,
  1369→      tick: world.tick,
  1370→      wallet: player.wallet,
  1371→      funds: world.community.funds,
  1372→      reputation: player.reputation,
  1373→      elevatorStatus: String(elevatorBlock?.params.elevator_status ?? 'ok'),
  1374→      missionsDone: world.missions.filter((m) => m.state === 'done').map((m) => m.id),
  1375→      bestAffinity,
  1376→    };
  1377→
  1378→    const fresh = evaluateAchievements(catalog, snap, this.unlockedAchievements);
  1379→    if (fresh.length === 0 && store.achievements.length > 0) return; // nada nuevo y lista ya pintada
  1380→
  1381→    if (fresh.length > 0) {
  1382→      for (const id of fresh) this.unlockedAchievements.add(id);
  1383→      saveUnlocked([...this.unlockedAchievements]);
  1384→    }
  1385→    store.setAchievements(
  1386→      catalog.achievements.map((def) => this.achievementUi(def, this.unlockedAchievements.has(def.id), fresh.includes(def.id))),
  1387→    );
  1388→
  1389→    const lang = useGameStore.getState().lang;
  1390→    for (const id of fresh) {
  1391→      const def = catalog.achievements.find((a) => a.id === id);
  1392→      if (def) store.pushLog(`🏆 Logro desbloqueado: ${def.name[lang] ?? def.name.es}`, 'good');
  1393→    }
  1394→    if (fresh.length > 0 && useGameStore.getState().achievementPopup === null) {
  1395→      const first = catalog.achievements.find((a) => a.id === fresh[0]);
  1396→      if (first) useGameStore.getState().setAchievementPopup(this.achievementUi(first, true));
  1397→    }
  1398→  }
  1399→}

  1405→/** Crea (o reutiliza) el controlador y lo arranca sobre el canvas dado. */
  1406→export function createGameController(canvas: HTMLCanvasElement): GameController {
  1407→  if (!instance) {
  1408→    instance = new GameController();
  1409→    setController(instance); // la UI lo usa desde ya; sus acciones toleran mundo nulo
  1410→    void instance.init(canvas);
  1411→    // Gancho de depuración/E2E (solo lectura + teleport QoL de pruebas).
  1412→    const dbg = instance as unknown as { logic?: PlayerLogic; world?: World } | null;
  1413→    (globalThis as Record<string, unknown>).__vecinosDebug = () => ({
  1414→      player: dbg?.logic
  1415→        ? {
  1416→            x: +dbg.logic.x.toFixed(2),
  1417→            z: +dbg.logic.z.toFixed(2),
  1418→            floor: dbg.logic.floor,
  1419→            mode: dbg.logic.mode,
  1420→            insideBlock: dbg.logic.insideBlock,
  1421→            climbing: !!dbg.logic.position().climbing,
  1422→          }
  1423→        : null,
  1424→      clock: dbg?.world ? { ...dbg.world.clock } : null,
  1425→      camMode: useGameStore.getState().camMode,
  1426→      paused: useGameStore.getState().paused,
  1427→      simLocation: dbg?.world?.residents[dbg.world.playerId]?.location ?? null,
  1428→      missions: dbg?.world?.missions.map((m) => `${m.id}:${m.state}${m.state === 'active' ? `#${m.step}` : ''}`) ?? [],
  1429→      /** QoL de pruebas E2E: colocar al jugador ante la puerta de un bloque (recorrido legal del grafo). */
  1430→      tp: (blockId: string) => !!dbg?.logic?.teleportTo(blockId),
  1431→      /** QoL de pruebas E2E: comando del sim directo (diagnóstico de pasos). */
  1432→      cmd: (c: unknown) => {
  1433→        const w = dbg?.world;
  1434→        const ct = (dbg as unknown as { _content?: Content } | null)?._content;
  1435→        return w && ct ? JSON.stringify(dispatch(w, ct, c as Command)) : 'no-world';
  1436→      },
  1437→      /** QoL de pruebas E2E (pestaña en segundo plano: rAF suspendido): avanza n ticks del sim. */
  1438→      tickN: (n: number) => {
  1439→        const w = dbg?.world;
  1440→        const ct = (dbg as unknown as { _content?: Content } | null)?._content;
  1441→        if (!w || !ct) return 'no-world';
  1442→        for (let i = 0; i < n; i++) tick(w, ct);
  1443→        (dbg as unknown as { handleEvents?: () => void }).handleEvents?.();
  1444→        return JSON.stringify({ ...w.clock });
  1445→      },
  1446→      /** QoL de pruebas E2E: coloca la X del jugador (presentación) para apuntar interacciones fijas. */
  1447→      setX: (x: number) => {
  1448→        const lg = dbg?.logic;
  1449→        if (!lg) return 'no-logic';
  1450→        lg.x = x;
  1451→        return JSON.stringify({ x: lg.x });
  1452→      },
  1453→    });
  1454→  }
  1455→  return instance;
  1456→}
