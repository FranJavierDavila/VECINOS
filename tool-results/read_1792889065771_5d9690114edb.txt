    81→  async init(canvas: HTMLCanvasElement): Promise<void> {
    82→    if (this.engine || this.initializing) return;
    83→    this.initializing = true;
    84→    this.canvas = canvas;
    85→
    86→    try {
    87→      // 1) Contenido externo + validación (los errores de referencias se avisan, no abortan).
    88→      const { content, issues } = await loadContent();
    89→      this._content = content;
    90→      this.cutscenePlayer = new CutscenePlayer(content);
    91→      preloadSounds();
    92→      useGameStore.setState({ touchHint: content.rules.touch.hint });
    93→      if (issues.length > 0) {
    94→        useGameStore.getState().pushToast(`Contenido inválido: ${issues.length} errores de validación`, 'bad');
    95→      }
    96→
    97→      // 2) Motor de render (WebGPU con fallback WebGL2). La calidad la elige el jugador en Opciones.
    98→      const { engine, backend } = await createEngine(canvas);
    99→      this.engine = engine;
   100→      engine.setHardwareScalingLevel(QUALITY_SCALE[loadOptions().quality]);
   101→      window.addEventListener('resize', this.onResize);
   102→      await this.restart();
   103→      useGameStore.getState().markReady(backend);
   104→      startGamepadPolling();
   105→      engine.runRenderLoop(this.frame);
   106→      // Gancho de depuración/pruebas E2E (no lo usa la UI).
   107→      (window as unknown as { __vecinos?: unknown }).__vecinos = { store: useGameStore, controller: this };
   108→    } catch (err) {
   109→      console.error('No se pudo arrancar el juego (contenido o motor de render)', err);
   110→      useGameStore.getState().setEngineError(true);
   111→    }
   112→  }

   131→  onKeyDown(code: string): void {
   132→    // Cutscene en curso: E/Espacio/Enter/Escape saltan el espectáculo; nada más pasa.
   133→    if (this.cutscenePlayer?.active) {
   134→      if (code === 'KeyE' || code === 'Space' || code === 'Enter' || code === 'Escape') this.cutsceneSkip();
   135→      return;
   136→    }
   137→    if (this.keys.has(code)) return;
   138→    this.keys.add(code);
   139→    const store = useGameStore.getState();
   140→    if (selectModalOpen(store)) return; // los modales gestionan sus propias teclas
   141→    if (code === 'KeyE') {
   142→      this.interact();
   143→      return;
   144→    }
   145→    const cam = CAM_BY_KEY[code];
   146→    if (cam) this.setCamMode(cam);
   147→  }

   225→  /** Interacción (tecla E o botón virtual ✋): usar el objeto cercano, entrar o salir. */
   226→  interact(): void {
   227→    // El ✋ táctil también salta la cutscene (es la única acción habilitada mientras dura).
   228→    if (this.cutscenePlayer?.active) {
   229→      this.cutsceneSkip();
   230→      return;
   231→    }
   232→    if (!this.logic) return;
   233→    // Ascensor (v1.9): si el prompt anuncia ascensor (sin objetivo de E delante), la E abre el
   234→    // panel de destino. Misma condición que promptText → lo que ves es lo que la E ejecuta.
   235→    if (!this.logic.interactTarget()) {
   236→      const panel = this.logic.elevatorPanelTarget();
   237→      if (panel) {
   238→        useGameStore.getState().setElevatorPanel(panel);
   239→        return;
   240→      }
   241→    }
   242→    const r = this.logic.interact();
   243→    if (r.type === 'entered') this.setCamMode('estancia');
   244→    else if (r.type === 'exited') this.setCamMode('planta');
   245→    else if (r.type === 'used') this.presentInteraction(r.kind);
   246→    this.pump();
   247→  }

   364→  private connectNet(): void {
   365→    this.net = new NetSession({
   366→      onState: (s) => this.applyRemoteState(s),
   367→      onParticipants: (p) => useGameStore.getState().setMpParticipants(p),
   368→      onJoined: (j) => {
   369→        this.myResidentId = j.yourResidentId;
   370→        this.isHost = j.isHost;
   371→        useGameStore.getState().setMpSession({ code: j.code, isHost: j.isHost, myResidentId: j.yourResidentId });
   372→        useGameStore.getState().setMpParticipants(j.participants);
   373→      },
   374→      onError: (m) => {
   375→        useGameStore.getState().pushToast(m, 'bad');
   376→      },
   377→      onStatus: (st) => {
   378→        if (!this.net) return;
   379→        if (st === 'connected' && useGameStore.getState().mpStatus === 'connecting') return; // el ack decide
   380→        if (st === 'disconnected' && useGameStore.getState().mpStatus === 'connected') {
   381→          useGameStore.getState().setMpStatus('error');
   382→          useGameStore.getState().setMpError('Conexión perdida con el servidor de la comunidad.');
   383→        }
   384→      },
   385→    });
   386→  }

   399→  /** Adopta el estado del servidor (autoridad). Primera vez: reinicia la escena con ese mundo. */
   400→  private applyRemoteState(s: SessionStatePayload): void {
   401→    if (s.seq <= this.appliedSeq) return;
   402→    this.appliedSeq = s.seq;
   403→    if (!this.world || this.world.scenarioId !== s.world.scenarioId) {
   404→      void this.restart(s.world).then(() => {
   405→        if (!this.world) return;
   406→        this.world.events = s.events.slice();
   407→        this.handleEvents();
   408→        this.reconcile();
   409→        this.syncHud();
   410→      });
   411→      return;
   412→    }
   413→    // Aplica en sitio conservando la identidad del objeto (PlayerLogic/building/actors referencian el mismo World).
   414→    Object.assign(this.world, s.world);
   415→    this.world.events = s.events.slice();
   416→    this.handleEvents();
   417→    this.paintBuilding();
   418→    this.reconcile();
   419→    this.syncHud();
   420→  }

   422→  /** Corrige desvíos de posición del avatar local respecto al mundo del servidor (autoridad). */
   423→  private reconcile(): void {
   424→    if (!this.logic || !this.world || !this.net) return;
   425→    const me = this.world.residents[this.actorId()];
   426→    if (!me) return;
   427→    const pos = this.logic.position();
   428→    const locBlock = me.location ? this.world.blocks[me.location] : null;
   429→    if (locBlock) {
   430→      const insideOk = pos.mode === 'inside' && pos.insideBlock === locBlock.id;
   431→      const corridorOk = pos.mode === 'corridor' && !pos.insideBlock;
   432→      if (!(locBlock.kind === 'dwelling' ? insideOk : corridorOk) || Math.abs(pos.floor - locBlock.cell.floor) > 0.01) {
   433→        this.logic.forceLocation(locBlock.id);
   434→      }
   435→    } else if (pos.mode === 'inside') {
   436→      this.logic.forceLocation(null);
   437→    }
   438→  }

   440→  /** Siguiente frase del teléfono; al colgar responde la llamada en el sim y saca la siguiente de la cola. */
   441→  phoneNext(): void {
   442→    if (this.phoneLine < this.phoneLines.length - 1) {
   443→      this.phoneLine += 1;
   444→      this.showPhoneLine();
   445→      return;
   446→    }
   447→    const mission = this.phoneMission;
   448→    this.phoneLines = [];
   449→    this.phoneMission = '';
   450→    useGameStore.getState().setPhone(null);
   451→    this.actors?.director.setGesture(this.actorId(), null); // cuelgas: se suelta el teléfono
   452→    if (this.world && mission) {
   453→      this.sendCommand({ type: 'answer_call', mission });
   454→    }
   455→    const next = this.phoneQueue.shift();
   456→    if (next) this.openPhone(next.mission, next.dialogId);
   457→  }

   481→  placeFurniture(block: string, room: string, slot: string, item: string): void {

   584→  /** (Re)crea el mundo y la escena. Con `state` restaura una partida guardada. */
   585→  private async restart(state?: World): Promise<void> {
   586→    if (!this.engine || !this.canvas || !this._content) return;
   587→    // Ninguna cutscene sobrevive a un cambio de mundo: cola fuera y teatro desmontado.
   588→    this.cutsceneQueue = [];
   589→    this.cutscenePlayer?.abort();
   590→    const { Color4, DirectionalLight, HemisphericLight, Scene, UniversalCamera, Vector3 } = B();
   591→    const view = this.content.rules.view;
   592→    useGameStore.getState().resetUi();
   593→
   594→    this.world = state ?? createWorld(this.content, this.activeScenarioId());
   595→    this.scenarioOverrides = this.content.scenarios[this.world.scenarioId ?? '']?.minigameOverrides ?? {};
   596→    // Guardados antiguos sin `entry`: la entrada es la zona común de planta baja.
   597→    if (!this.world.entry) {
   598→      this.world.entry = Object.values(this.world.blocks).find((b) => b.cell.floor === 0 && b.kind !== 'dwelling')?.id ?? Object.keys(this.world.blocks)[0]!;
   599→    }
   600→    // Migración de estancias de guardados v0.1: params y ranuras de muebles según catálogo.
   601→    for (const b of Object.values(this.world.blocks)) migrateRooms(this.content, b.rooms);
   602→    this.logic = new PlayerLogic(this.world, this.content, {
   603→      actorId: this.myResidentId ?? undefined,
   604→      send: (cmd) => this.sendCommand(cmd),
   605→    });
   606→
   607→    this.scene?.dispose();
   608→    this.scene = new Scene(this.engine);
   609→    this.scene.clearColor = new Color4(view.sky.r, view.sky.g, view.sky.b, 1);
   610→    new HemisphericLight('ambient', new Vector3(...view.ambient.dir), this.scene).intensity = view.ambient.intensity;
   611→    new DirectionalLight('sun', new Vector3(...view.sun.dir), this.scene).intensity = view.sun.intensity;
   612→
   613→    this.building = createBuilding(this.scene, this.world, this.content);
   614→    this.elevators = createElevators(this.scene, this.world, this.content);
   615→    this.actors = createActors(this.scene, this.world, this.content, this.elevators);
   616→    // Ecosistema del edificio (v1.14): cerebro data-driven + teatro de extras. Se recrea con
   617→    // cada mundo (referencia al World vivo) y sigue la posición del jugador para las luces.
   618→    this.ecosystem?.dispose();
   619→    this.ecosystem = createEcosystem(this.scene, this.world, this.content, () => this.logic!.position());
   620→    // Sonda de QA (SOLO dev): escena y controlador accesibles para el E2E automatizado.
   621→    // En producción no se expone nada (la partida queda encapsulada igual que siempre).
   622→    if (process.env.NODE_ENV !== 'production') {
   623→      (window as unknown as { __vecinosQA?: { scene: unknown; controller: unknown; elevators: unknown; ecosystem: unknown } }).__vecinosQA = {
   624→        scene: this.scene,
   625→        controller: this,
   626→        elevators: this.elevators,
   627→        ecosystem: this.ecosystem,
   628→      };
   629→    }
   630→    // Bocadillos de expresión (v1.10): capa DOM sobre el canvas, recreada con el mundo
   631→    // (referencia al World vivo). Sin emotes.json en el catálogo, director nulo sin capa.
   632→    this.bubbles?.dispose();
   633→    const stage = this.canvas?.parentElement;
   634→    this.bubbles = stage && this.actors ? createBubbles(stage, this.scene, this.engine, this.world, this.content, this.actors.director) : null;
   635→    const blocks = Object.values(this.world.blocks);
   636→    // El cubo de escalera (v1.4) sobresale spanW−1 celdas por la derecha del edificio:
   637→    // el encuadre lo incluye para que la torre no se corte en cámara.
   638→    const hasStairs = blocks.some((b) => this.content.blocks[b.defId]?.features?.stairs === true);
   639→    const cols = Math.max(...blocks.map((b) => b.cell.col + b.size.w)) + (hasStairs ? this.content.rules.view.stairs.spanW - 1 : 0);
   640→    this.rig = new CameraRig(
   641→      new UniversalCamera('cam', new Vector3(0, 0, -30), this.scene),
   642→      this.engine,
   643→      cols,
   644→      Math.max(...blocks.map((b) => b.cell.floor)) + 1,
   645→      view.cellW,
   646→      view.cellH,
   647→    );
   648→    // Re-sincronización de vista (v0.6-b): la cámara sigue la posición del mundo (restaurado o
   649→    // nuevo), no la preferencia anterior. Dentro de una vivienda → estancia; en pasillo, la
   650→    // preferencia salvo que fuera 'estancia' (ahí estás en el pasillo, no en una estancia).
   651→    const pos = this.logic.position();
   652→    const store = useGameStore.getState();
   653→    const wanted: CamMode = pos.mode === 'inside' ? 'estancia' : store.camMode === 'estancia' ? 'planta' : store.camMode;
   654→    this.rig.mode = wanted;
   655→    store.setCamMode(wanted);
   656→
   657→    this.clockAcc = 0;
   658→    this.keys.clear();
   659→    this.phoneLines = [];
   660→    this.phoneMission = '';
   661→    this.phoneQueue = [];
   662→    this.lastHudSync = 0;
   663→
   664→    // Re-llamada (Fase F): si la partida montada tenía pasos de teléfono pendientes, vuelven a
   665→    // sonar. La cola de openPhone ordena el resto sin que ninguna llamada se pierda.
   666→    if (this.world) {
   667→      for (const m of this.world.missions) {
   668→        if (m.state !== 'active') continue;
   669→        const step = this.content.missions[m.id]?.steps[m.step];
   670→        if (step?.type === 'phone_call') this.openPhone(m.id, step.dialog);
   671→      }
   672→    }
   673→
   674→    this.pump();
   675→  }

   683→  private frame = (): void => {
   684→    if (!this.engine || !this.scene || !this.world || !this.logic || !this.building || !this.actors || !this.rig || !this.elevators) return;
   685→    const dt = Math.min(0.1, this.engine.getDeltaTime() / 1000);
   686→    const store = useGameStore.getState();
   687→    const frozen = selectModalOpen(store) || store.paused;
   688→
   689→    let ticked = false;
   690→    if (!frozen) {
   691→      this.logic.update(dt, this.readInput());
   692→      // Ascensor (v1.5): el director mueve cabinas y puertas; el jugador llama con W/S en el
   693→      // núcleo y su escalada EXTERNA la gobierna la cabina (espera → embarco → viaje → salida).
   694→      this.elevators.update(dt);
   695→      const climb = this.logic.climbInfo();
   696→      if (climb) {
   697→        if (!this.logic.externalRequested) {
   698→          const zone = this.logic.zone();
   699→          this.elevators.requestRide(this.actorId(), zone.cell.col, climb.from, climb.to);
   700→          this.logic.externalRequested = true;
   701→        }
   702→        const st = this.elevators.status(this.actorId());
   703→        if (!st) {
   704→          this.logic.finishClimbAtNearest(); // fuera de servicio: aterriza en la planta más cercana (sin teletransporte)
   705→          this.logic.externalRequested = false;
   706→          this.playerBoardT = 0;
   707→        } else {
   708→          this.logic.setClimbProgress(st.phase === 'riding' ? st.progress : 0);
   709→          const inside = st.phase === 'boarding' || st.phase === 'riding';
   710→          this.logic.setClimbZ(inside ? this.content.rules.view.roomZ - this.content.rules.view.commonDepth / 2 + this.content.rules.view.elevator.rideZ : null);
   711→          // El jugador "da un paso" dentro de la cabina un instante tras abrirse la puerta
   712→          // (el zOverride ya lo acerca); avisar al director para que cierre y viaje.
   713→          if (st.phase === 'boarding') {
   714→            this.playerBoardT += dt;
   715→            if (this.playerBoardT > 1.0) this.elevators.notifyBoarded(this.actorId());
   716→          } else {
   717→            this.playerBoardT = 0;
   718→          }
   719→          if (st.phase === 'exiting') {
   720→            this.logic.finishClimb();
   721→            this.logic.externalRequested = false;
   722→          }
   723→        }
   724→      } else if (this.logic.externalRequested) {
   725→        this.logic.externalRequested = false;
   726→      }
   727→      // En remoto el servidor es la autoridad del reloj: nada de ticks locales.
   728→      if (!this.net) {
   729→        this.clockAcc += dt * store.speed;
   730→        let guard = 0;
   731→        while (this.clockAcc >= 1 && guard < 60) {
   732→          tick(this.world, this.content);
   733→          this.clockAcc -= 1;
   734→          ticked = true;
   735→          guard += 1;
   736→        }
   737→      }
   738→    }
   739→
   740→    if (this.world.events.length > 0) this.handleEvents();
   741→    if (ticked) this.paintBuilding();
   742→
   743→    const now = performance.now();
   744→    if (now - this.lastHudSync > 200) {
   745→      this.lastHudSync = now;
   746→      useGameStore.getState().setFps(Math.round(this.engine.getFps()));
   747→      this.syncHud();
   748→    }
   749→
   750→    const pos = this.logic.position();
   751→    this.actors.update(dt, pos, this.actorId());
   752→    // Cutscene (Fase D): sus beats mueven cámara/actores DESPUÉS de los eventos y ANTES del render.
   753→    this.cutscenePlayer?.update(dt);
   754→    // Bocadillos (v1.10): después de actores y cámara — proyectan la posición FINAL del frame.
   755→    this.bubbles?.update(dt, this.actorId());
   756→    // Ecosistema (v1.14): cerebro a cadencia + extras + luces + deterioro. En multijugador
   757→    // queda en pausa (el servidor es la autoridad; mismo criterio que partyDemo).
   758→    if (!this.net) this.ecosystem?.update(dt);
   759→    // Si la cutscene acaba y queda cola, encadena la siguiente aquí: no dependemos de que el sim
   760→    // genere otro evento para descolgar la siguiente (durante el teatro el sim está congelado).
   761→    if (this.cutsceneQueue.length > 0 && !this.cutscenePlayer?.active) this.pumpCutscene();
   762→    // LOD por distancia: solo con cámaras cercanas (planta/estancia); el plano general (edificio) muestra todo.
   763→    // Privacidad (v1.17): el tercer argumento es el bloque donde está el jugador — las fachadas
   764→    // de las viviendas reaccionan al instante cuando entra o sale.
   765→    this.building.update(dt, useGameStore.getState().camMode === 'edificio' ? undefined : { x: pos.x, floor: pos.floor }, pos.insideBlock);
   766→    this.rig.update(dt, pos);
   767→    this.scene.render();
   768→  };

   894→          break;
   895→        case 'junta_closed':
   896→          store.setJunta(null);
   897→          break;
   898→        case 'stairs_grumble':
   899→          // Malestar vecinal (v1.5): texto del catálogo con el nombre del vecino que sube a pie.
   900→          store.pushLog(String(this.content.rules.sim.elevator.grumbleText).replace('{name}', this.nameOf(d.resident)), 'info');
   901→          break;
   902→        case 'minigame_requested': {
   903→          const def = this.content.minigames[String(d.minigame)];
   904→          if (!def) break;
   905→          // Overrides numéricos del escenario activo (QA/E2E): solo config, nunca efectos ni textos.
   906→          const ov = this.scenarioOverrides[def.id];
   907→          const effective = ov ? ({ ...def, config: { ...def.config, ...ov } } as MinigameDef) : def;
   908→          store.setMinigame({ ...effective, emoji: effective.emoji ?? '🎮', instruction: effective.instruction ?? null });
   909→          break;
   910→        }
   911→        case 'minigame_finished': {
   912→          const name = this.content.minigames[String(d.minigame)]?.name ?? String(d.minigame);
   913→          store.pushLog(d.success ? `🎮 Minijuego superado: ${name}` : `🎮 Minijuego fallido: ${name}`, d.success ? 'good' : 'bad');
   914→          break;
   915→        }
   916→        case 'ui_toast':
   917→          store.pushToast(String(d.text), (d.tone as Tone | undefined) ?? 'info');
   918→          break;
   919→        case 'finale':
   920→          store.setFinale(String(d.text));
   921→          break;
   922→        case 'funds_changed': {
   923→          const amount = typeof d.amount === 'number' ? d.amount : 0;
   924→          if ((d.to === 'player' || d.to === 'community') && amount !== 0) {
   925→            store.pushLog(amount > 0 ? `+${amount} € (fondos)` : `${amount} € (fondos)`, amount > 0 ? 'good' : 'bad');
   926→          }
   927→          break;
   928→        }
   929→        default: {
   930→          // Umbral de canal con texto del catálogo (odors, wifi y futuros canales): totalmente data-driven.
   931→          if (d.uiLog === true && typeof d.message === 'string') {
   932→            store.pushLog(String(d.message).replace('{block}', blockLabel(String(d.block))), (d.tone as Tone | undefined) ?? 'info');
   933→          }
   934→          break; // resident_moved, param_changed, activity_ended... demasiado ruido para el registro
   935→        }
   936→      }
   937→      // Cutscenes (Fase D): después del reparto, ¿alguna quiere este evento como disparador?
   938→      this.maybeCutscene(e);
   939→      // Bocadillos (v1.10): reacciones expresivas al evento según emotes.json del catálogo.
   940→      this.bubbles?.onEvent(e);
   941→    }
   942→    this.pumpCutscene();
   943→  }

   947→  /** Si el evento encaja con el trigger de alguna cutscene del catálogo, se encola (máx. 3). */
   948→  private maybeCutscene(e: GameEvent): void {
   949→    const defs = Object.values(this.content.cutscenes);
   950→    if (defs.length === 0) return;
   951→    const d = e.data as Record<string, unknown>;
   952→    const hit = defs.find((c) => {
   953→      const t = c.trigger;
   954→      if (t.on !== e.type) return false;
   955→      if (t.mission !== undefined && t.mission !== String(d.mission ?? '')) return false;
   956→      if (t.step !== undefined && t.step !== String(d.step ?? '')) return false;
   957→      return true;
   958→    });
   959→    if (!hit) return;
   960→    if (this.cutsceneQueue.length >= 3) return; // nunca más de 3 esperando: el teatro no atravaca la partida
   961→    this.cutsceneQueue.push({
   962→      def: hit,
   963→      ctx: {
   964→        mission: typeof d.mission === 'string' ? d.mission : null,
   965→        resident: typeof d.resident === 'string' ? d.resident : null,
   966→        block: typeof d.block === 'string' ? d.block : null,
   967→      },
   968→    });
   969→  }
   970→
   971→  /** Lanza la siguiente cutscene en cola si no hay nada en pantalla (también un modal la pospone). */
   972→  private pumpCutscene(): void {
   973→    if (!this.cutscenePlayer || !this.world || !this.rig || !this.actors || !this.engine || !this.logic) return;
   974→    if (this.cutscenePlayer.active || this.cutsceneQueue.length === 0) return;
   975→    if (selectModalOpen(useGameStore.getState())) return; // no se roba la pantalla a un modal
   976→    const next = this.cutsceneQueue.shift()!;
   977→    this.cutscenePlayer.start(next.def, next.ctx, {
   978→      engine: this.engine,
   979→      rig: this.rig,
   980→      actors: this.actors.director,
   981→      world: this.world,
   982→      playerPos: () => this.logic!.position(),
   983→    });
   984→  }

   988→  private openPhone(mission: string, dialogId: string): void {
   989→    const dlg = this.content.dialogs[dialogId];
   990→    if (!dlg || dlg.lines.length === 0) return;
   991→    // Si ya hay una llamada en pantalla, la nueva espera en cola (dedupe por misión).
   992→    if (this.phoneMission) {
   993→      if (this.phoneMission !== mission && !this.phoneQueue.some((q) => q.mission === mission)) {
   994→        this.phoneQueue.push({ mission, dialogId });
   995→      }
   996→      return;
   997→    }
   998→    this.phoneMission = mission;
   999→    this.phoneLines = dlg.lines;
  1000→    this.phoneLine = 0;
  1001→    this.actors?.director.setGesture(this.actorId(), 'phone'); // el jugador atiende con el móvil a la oreja
  1002→    this.showPhoneLine();
  1003→  }

  1009→  }

  1020→  private objectiveUi(): { title: string; text: string } | null {
  1021→    if (!this.world) return null;
  1022→    const o = activeObjective(this.world, this.content);
  1023→    if (!o) return null;
  1024→    if (o.state === 'awaiting_choice') return { title: o.title, text: 'El edificio contiene la respiración. Decide qué hacer.' };
  1025→    const s = o.step!;
  1026→    const left = o.untilTick !== null ? ` · quedan ${Math.max(0, o.untilTick - this.world.tick)} min` : '';
  1027→    switch (s.type) {
  1028→      case 'phone_call':
  1029→        return { title: o.title, text: 'Suena el teléfono. Atiéndelo.' };
  1030→      case 'reach':
  1031→        return { title: o.title, text: `Llega a la puerta (${this.blockName(s.block ?? '')}).` };
  1032→      case 'enter':
  1033→        return { title: o.title, text: 'Entra en tu piso con E.' };
  1034→      case 'catch_in_act':
  1035→        return { title: o.title, text: `Pilla al responsable in fraganti${left}.` };
  1036→      default:
  1037→        return { title: o.title, text: o.title };
  1038→    }
  1039→  }

  1041→  private promptText(): string | null {
  1042→    const logic = this.logic;
  1043→    const world = this.world;
  1044→    if (!logic || !world) return null;
  1045→    if (logic.position().climbing) return null;
  1046→    // El prompt es SIEMPRE el objetivo real de la tecla E (misma fuente, cero divergencias).
  1047→    const t = logic.interactTarget();
  1048→    if (t) {
  1049→      const ic = this.content.rules.interact;
  1050→      switch (t.kind) {
  1051→        case 'door': {
  1052→          const door = world.blocks[t.blockId]!;
  1053→          if (logic.position().mode === 'inside') return ic.exit;
  1054→          return door.access === 'locked'
  1055→            ? ic.doorLocked.replace('{block}', this.blockName(door.id))
  1056→            : ic.doorEnter.replace('{block}', this.blockName(door.id));
  1057→        }
  1058→        case 'resident':
  1059→          return ic.greet.replace('{name}', this.nameOf(t.residentId));
  1060→        case 'mailbox':
  1061→          return ic.mailbox.prompt;
  1062→        case 'noticeboard':
  1063→          return ic.noticeboard.prompt;
  1064→        case 'laptop':
  1065→          return ic.laptop;
  1066→        case 'bed':
  1067→          return ic.bed;
  1068→      }
  1069→    }
  1070→    const zone = logic.zone();
  1071→    const feat = this.content.blocks[zone.defId]!.features;
  1072→    if (feat?.stairs || feat?.elevator) {
  1073→      const ic = this.content.rules.interact;
  1074→      // Lo que dice el prompt es lo que W/S hace aquí: ascensor operativo manda; si no hay
  1075→      // ascensor utilizable pero sí escalera, escalera; núcleo averiado sin escalera = aviso.
  1076→      if (feat.elevator && world.blocks[zone.id]!.params.elevator_status === 'ok') return ic.elevator;
  1077→      if (feat.stairs) return ic.stairs;
  1078→      return ic.elevatorBroken;
  1079→    }
  1080→    return null;
  1081→  }

  1083→  /** Viviendas del jugador para el editor de interiores (foto serializable, posiciones ya en %). */
  1084→  private homesUi() {
  1085→    const world = this.world!;
  1086→    const cellH = this.content.rules.view.cellH;
  1087→    const mine = this.actorId();
  1088→    return Object.values(world.blocks)
  1089→      .filter((b) => b.kind === 'dwelling' && b.ownerId === mine)
  1090→      .sort((a, b) => a.id.localeCompare(b.id))
  1091→      .map((b) => ({
  1092→        id: b.id,
  1093→        label: blockLabel(b.id),
  1094→        rooms: b.rooms.map((room) => {
  1095→          const rdef = this.content.rooms[room.defId];
  1096→          const slots = (rdef?.furnitureSlots ?? []).map((s) => {
  1097→            const itemId = room.furniture[s.id] ?? '';
  1098→            const fdef = itemId ? this.content.furniture[itemId] : undefined;
  1099→            return {
  1100→              id: s.id,
  1101→              kind: s.kind,
  1102→              left: Math.min(92, Math.max(2, (s.pos[0] / Math.max(0.001, rdef?.size.w ?? 1)) * 100)),
  1103→              bottom: Math.min(76, Math.max(0, (s.pos[1] / Math.max(0.001, cellH)) * 100)),
  1104→              item: fdef ? { id: fdef.id, name: fdef.name, emoji: fdef.emoji ?? '📦', color: fdef.view.color } : null,
  1105→            };
  1106→          });
  1107→          return { key: room.key, label: rdef?.label ?? room.key, access: room.access, slots };
  1108→        }),
  1109→      }));
  1110→  }

  1112→  private syncHud(): void {
  1113→    const world = this.world;
  1114→    if (!world) return;
  1115→    const player = world.residents[this.actorId()] ?? world.residents[world.playerId];
  1116→    if (!player) return;
  1117→    const home = player.home;
  1118→    const params = visibleParams(world, this.content, home).map((p) => ({ name: p.name, label: p.def.label ?? p.name, value: String(p.value) }));
  1119→    const missions = world.missions.map((m) => {
  1120→      const def = this.content.missions[m.id]!;
  1121→      const step = m.state === 'active' ? def.steps[m.step] : undefined;
  1122→      return {
  1123→        id: m.id,
  1124→        title: def.title,
  1125→        state: m.state,
  1126→        hint: step?.hint ?? null,
  1127→        remaining: m.until !== null ? `quedan ${Math.max(0, m.until - world.tick)} min` : null,
  1128→      };
  1129→    });
  1130→    const travel = travelMinutes(buildCirculation(world, this.content), world.entry, home);
  1131→    const elevatorBlock = Object.values(world.blocks).find((b) => b.params.elevator_status !== undefined);
  1132→    useGameStore.getState().syncHud({
  1133→      clock: { ...world.clock },
  1134→      weather: world.weather.state,
  1135→      funds: world.community.funds,
  1136→      wallet: player.wallet,
  1137→      reputation: player.reputation,
  1138→      elevator: String(elevatorBlock?.params.elevator_status ?? 'ok'),
  1139→      homeLabel: blockLabel(home),
  1140→      travelText: Number.isFinite(travel) ? `del portal a tu puerta: ${travel} min` : null,
  1141→      objective: this.objectiveUi(),
  1142→      prompt: this.promptText(),
  1143→      params,
  1144→      missions,
  1145→      homes: this.homesUi(),
  1146→      furnitureConfig: { refundRatio: this.content.rules.furniture.refundRatio, roomUnlockPrice: this.content.rules.furniture.roomUnlock.price },
  1147→    });
  1148→  }
