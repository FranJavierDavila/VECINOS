    55→  async init(canvas: HTMLCanvasElement): Promise<void> {
    56→    if (this.engine || this.initializing) return;
    57→    this.initializing = true;
    58→    this.canvas = canvas;
    59→
    60→    try {
    61→      // 1) Contenido externo + validación (los errores de referencias se avisan, no abortan).
    62→      const { content, issues } = await loadContent();
    63→      this._content = content;
    64→      preloadSounds();
    65→      useGameStore.setState({ touchHint: content.rules.touch.hint });
    66→      if (issues.length > 0) {
    67→        useGameStore.getState().pushToast(`Contenido inválido: ${issues.length} errores de validación`, 'bad');
    68→      }
    69→
    70→      // 2) Motor de render (WebGPU con fallback WebGL2).
    71→      const { engine, backend } = await createEngine(canvas);
    72→      this.engine = engine;
    73→      window.addEventListener('resize', this.onResize);
    74→      await this.restart();
    75→      useGameStore.getState().markReady(backend);
    76→      engine.runRenderLoop(this.frame);
    77→      // Gancho de depuración/pruebas E2E (no lo usa la UI).
    78→      (window as unknown as { __vecinos?: unknown }).__vecinos = { store: useGameStore, controller: this };
    79→    } catch (err) {
    80→      console.error('No se pudo arrancar el juego (contenido o motor de render)', err);
    81→      useGameStore.getState().setEngineError(true);
    82→    }
    83→  }

    96→  onKeyDown(code: string): void {
    97→    if (this.keys.has(code)) return;
    98→    this.keys.add(code);
    99→    const store = useGameStore.getState();
   100→    if (selectModalOpen(store)) return; // los modales gestionan sus propias teclas
   101→    if (code === 'KeyE') {
   102→      this.interact();
   103→      return;
   104→    }
   105→    const cam = CAM_BY_KEY[code];
   106→    if (cam) this.setCamMode(cam);
   107→  }

   117→  private readInput(): Input {
   118→    const held = (...codes: string[]) => codes.some((c) => this.keys.has(c));
   119→    return {
   120→      left: held('KeyA', 'ArrowLeft') || this.touch.left,
   121→      right: held('KeyD', 'ArrowRight') || this.touch.right,
   122→      up: held('KeyW', 'ArrowUp') || this.touch.up,
   123→      down: held('KeyS', 'ArrowDown') || this.touch.down,
   124→    };
   125→  }

   248→  async save(): Promise<void> {
   249→    if (!this.world) return;
   250→    const store = useGameStore.getState();
   251→    store.setSaveState('saving');
   252→    drainEvents(this.world); // la cola de eventos no entra en el guardado
   253→    try {
   254→      const res = await fetch('/api/save', {
   255→        method: 'POST',
   256→        headers: { 'Content-Type': 'application/json' },
   257→        body: JSON.stringify({ slot: 'auto', state: this.world }),
   258→      });
   259→      if (!res.ok) throw new Error(String(res.status));
   260→      store.setSaveState('saved', new Date().toLocaleTimeString('es-ES', { hour: '2-digit', minute: '2-digit' }));
   261→      store.pushToast('Partida guardada. El acta queda en el tablón (y en SQLite).', 'good');
   262→    } catch {
   263→      store.setSaveState('error');
   264→      store.pushToast('No se pudo guardar la partida. ¿El administrador se habrá ido a comer?', 'bad');
   265→    }
   266→  }

   268→  async load(): Promise<void> {
   269→    const store = useGameStore.getState();
   270→    try {
   271→      const res = await fetch('/api/save?slot=auto');
   272→      const json = (await res.json()) as { state?: unknown };
   273→      if (!json.state || typeof json.state !== 'object') {
   274→        store.pushToast('Todavía no hay ninguna partida guardada.', 'info');
   275→        return;
   276→      }
   277→      if (!this._content) {
   278→        store.pushToast('El contenido todavía se está cargando. Espera un momento.', 'info');
   279→        return;
   280→      }
   281→      await this.restart(json.state as World);
   282→      store.pushToast('Partida cargada. El vecindario, exactamente como lo dejaste.', 'good');
   283→    } catch {
   284→      store.pushToast('No se pudo cargar la partida.', 'bad');
   285→    }
   286→  }

   299→  /** (Re)crea el mundo y la escena. Con `state` restaura una partida guardada. */
   300→  private async restart(state?: World): Promise<void> {
   301→    if (!this.engine || !this.canvas || !this._content) return;
   302→    const { Color4, DirectionalLight, HemisphericLight, Scene, UniversalCamera, Vector3 } = B();
   303→    const view = this.content.rules.view;
   304→    useGameStore.getState().resetUi();
   305→
   306→    this.world = state ?? createWorld(this.content, this.activeScenarioId());
   307→    this.scenarioOverrides = this.content.scenarios[this.world.scenarioId ?? '']?.minigameOverrides ?? {};
   308→    // Guardados antiguos sin `entry`: la entrada es la zona común de planta baja.
   309→    if (!this.world.entry) {
   310→      this.world.entry = Object.values(this.world.blocks).find((b) => b.cell.floor === 0 && b.kind !== 'dwelling')?.id ?? Object.keys(this.world.blocks)[0]!;
   311→    }
   312→    // Migración de estancias de guardados v0.1: params y ranuras de muebles según catálogo.
   313→    for (const b of Object.values(this.world.blocks)) migrateRooms(this.content, b.rooms);
   314→    this.logic = new PlayerLogic(this.world, this.content);
   315→
   316→    this.scene?.dispose();
   317→    this.scene = new Scene(this.engine);
   318→    this.scene.clearColor = new Color4(view.sky.r, view.sky.g, view.sky.b, 1);
   319→    new HemisphericLight('ambient', new Vector3(...view.ambient.dir), this.scene).intensity = view.ambient.intensity;
   320→    new DirectionalLight('sun', new Vector3(...view.sun.dir), this.scene).intensity = view.sun.intensity;
   321→
   322→    this.building = createBuilding(this.scene, this.world, this.content);
   323→    this.actors = createActors(this.scene, this.world, this.content);
   324→    const blocks = Object.values(this.world.blocks);
   325→    this.rig = new CameraRig(
   326→      new UniversalCamera('cam', new Vector3(0, 0, -30), this.scene),
   327→      this.engine,
   328→      Math.max(...blocks.map((b) => b.cell.col + b.size.w)),
   329→      Math.max(...blocks.map((b) => b.cell.floor)) + 1,
   330→      view.cellW,
   331→      view.cellH,
   332→    );
   333→    this.rig.mode = useGameStore.getState().camMode;
   334→
   335→    this.clockAcc = 0;
   336→    this.keys.clear();
   337→    this.phoneLines = [];
   338→    this.phoneMission = '';
   339→    this.lastHudSync = 0;
   340→    this.pump();
   341→  }

   347→  // ───────────────────────── Bucle de render ─────────────────────────

   349→  private frame = (): void => {
   350→    if (!this.engine || !this.scene || !this.world || !this.logic || !this.building || !this.actors || !this.rig) return;
   351→    const dt = Math.min(0.1, this.engine.getDeltaTime() / 1000);
   352→    const store = useGameStore.getState();
   353→    const frozen = selectModalOpen(store) || store.paused;
   354→
   355→    let ticked = false;
   356→    if (!frozen) {
   357→      this.logic.update(dt, this.readInput());
   358→      this.clockAcc += dt * store.speed;
   359→      let guard = 0;
   360→      while (this.clockAcc >= 1 && guard < 60) {
   361→        tick(this.world, this.content);
   362→        this.clockAcc -= 1;
   363→        ticked = true;
   364→        guard += 1;
   365→      }
   366→    }
   367→
   368→    if (this.world.events.length > 0) this.handleEvents();
   369→    if (ticked) this.paintBuilding();
   370→
   371→    const now = performance.now();
   372→    if (now - this.lastHudSync > 200) {
   373→      this.lastHudSync = now;
   374→      this.syncHud();
   375→    }
   376→
   377→    const pos = this.logic.position();
   378→    this.actors.update(dt, pos);
   379→    // LOD por distancia: solo con cámaras cercanas (planta/estancia); el plano general (edificio) muestra todo.
   380→    this.building.update(dt, useGameStore.getState().camMode === 'edificio' ? undefined : { x: pos.x, floor: pos.floor });
   381→    this.rig.update(dt, pos);
   382→    this.scene.render();
   383→  };

   390→    this.syncHud();
   391→  }

   414→  private handleEvents(): void {
   415→    if (!this.world) return;
   416→    const store = useGameStore.getState();
   417→    for (const e of drainEvents(this.world)) {
   418→      const d = e.data as Record<string, unknown>;
   419→      eventSound(e.type, typeof d.type === 'string' ? d.type : undefined);
   420→      switch (e.type) {
   421→        case 'step_started':
   422→          if (d.type === 'phone_call') this.openPhone(String(d.mission), String(d.dialog));
   423→          break;
   424→        case 'choice_required': {
   425→          const choices = (d.choices as { id: string; label: string; hint?: string | null }[] | undefined) ?? [];
   426→          store.setChoices({ mission: String(d.mission), title: String(d.title), choices: choices.map((c) => ({ id: c.id, label: c.label, hint: c.hint ?? null })) });
   427→          store.pushLog(`⚖️ Decisión pendiente: ${String(d.title)}`, 'info');
   428→          break;
   429→        }
   430→        case 'door_locked': {
   431→          this.building?.shake(String(d.block));
   432→          const price = typeof d.price === 'number' ? d.price : null;
   433→          store.pushToast(`${blockLabel(String(d.block))}: puerta con candado${d.reason === 'for_sale' && price !== null ? ` (a la venta por ${price} €)` : ''}`, 'info');
   434→          break;
   435→        }
   436→        case 'neighbor_arrived':
   437→          store.pushToast(`Nuevo vecino en ${blockLabel(String(d.block))}: ${this.nameOf(d.resident)}`, 'good');
   438→          store.pushLog(`${this.nameOf(d.resident)} se muda a ${blockLabel(String(d.block))}. El edificio ya no es lo que era. Era mejor.`, 'info');
   439→          break;
   440→        case 'mission_activated':
   441→          store.pushLog(`📋 Nueva misión: ${String(d.title)}`, 'info');
   442→          break;
   443→        case 'mission_completed':
   444→          store.pushToast(`Misión completada: ${this.titleOf(d.mission)}`, 'good');
   445→          store.pushLog(`✔ Misión completada: ${this.titleOf(d.mission)}`, 'good');
   446→          break;
   447→        case 'mission_failed':
   448→          store.pushToast(`Misión fallida: ${this.titleOf(d.mission)}`, 'bad');
   449→          store.pushLog(`✘ Misión fallida: ${this.titleOf(d.mission)}`, 'bad');
   450→          break;
   451→        case 'block_unlocked':
   452→          store.pushToast(`${blockLabel(String(d.block))} desbloqueado`, 'good');
   453→          break;
   454→        case 'block_bought':
   455→          store.pushLog(`🔑 ${blockLabel(String(d.block))} comprado por ${String(d.price)} €`, 'good');
   456→          break;
   457→        case 'block_for_sale':
   458→          store.pushLog(`🪧 Cartel de "SE VENDE" en ${blockLabel(String(d.block))} (${String(d.price)} €)`, 'info');
   459→          break;
   460→        case 'noise_complaint': // texto data-driven: lo pinta el uiLog del catálogo (case mantenido por compatibilidad de eventos antiguos)
   461→          if (d.uiLog === true) break;
   462→          store.pushLog(`🔊 Ruido por encima del límite en ${blockLabel(String(d.block))}. Alguien aprieta los dientes.`, 'bad');
   463→          break;
   464→        case 'leak_detected':
   465→          if (d.uiLog === true) break;
   466→          store.pushLog(`💧 Goteo detectado en ${blockLabel(String(d.block))}. El cubo del salón hace ojitos.`, 'bad');
   467→          break;
   468→        case 'furniture_placed':
   469→          store.pushLog(`${String(d.emoji ?? '📦')} ${String(d.name)} colocado en ${blockLabel(String(d.block))} (${String(d.room)}).`, 'info');
   470→          break;
   471→        case 'furniture_removed':
   472→          store.pushLog(`${String(d.emoji ?? '📦')} ${String(d.name)} retirado de ${blockLabel(String(d.block))} (+${String(d.refund)} €).`, 'info');
   473→          break;
   474→        case 'room_renovated':
   475→          store.pushLog(`🧱 Reforma terminada en ${blockLabel(String(d.block))}. Hay estancia nueva que amueblar.`, 'good');
   476→          break;
   477→        case 'weather_changed':
   478→          store.pushLog(`El cielo cambia: ${this.content.rules.labels.weather[String(d.state)] ?? String(d.state)}`, 'info');
   479→          break;
   480→        case 'rent_collected':
   481→          store.pushToast(`+${String(d.amount)} € de alquiler (${blockLabel(String(d.block))})`, 'good');
   482→          break;
   483→        case 'work_done':
   484→          store.pushLog(`💻 Cerrada una "reunión que podía ser un correo": +${String(d.amount)} €`, 'info');
   485→          break;
   486→        case 'activity_started':
   487→          if (d.activity !== 'idle') store.pushLog(`${this.nameOf(d.resident)} ${this.activityLabel(d.resident, String(d.activity))}`, 'info');
   488→          break;
   489→        case 'junta_requested': {
   490→          const agenda = (d.agenda as { id: string; title: string; description: string; cost: number }[] | undefined) ?? [];
   491→          store.setJunta(agenda.map((i) => ({ id: i.id, title: i.title, description: i.description, cost: i.cost })));
   492→          break;
   493→        }
   494→        case 'junta_result':
   495→          store.pushToast(
   496→            d.approved ? `Aprobado: ${String(d.title)} (${String(d.yes)} sí / ${String(d.no)} no)` : `Rechazado: ${String(d.title)} (${String(d.yes)} sí / ${String(d.no)} no)`,
   497→            d.approved ? ('good' as Tone) : ('bad' as Tone),
   498→          );
   499→          break;
   500→        case 'minigame_requested': {
   501→          const def = this.content.minigames[String(d.minigame)];
   502→          if (!def) break;
   503→          // Overrides numéricos del escenario activo (QA/E2E): solo config, nunca efectos ni textos.
   504→          const ov = this.scenarioOverrides[def.id];
   505→          const effective = ov ? ({ ...def, config: { ...def.config, ...ov } } as MinigameDef) : def;
   506→          store.setMinigame({ ...effective, emoji: effective.emoji ?? '🎮', instruction: effective.instruction ?? null });
   507→          break;
   508→        }
   509→        case 'minigame_finished': {
   510→          const name = this.content.minigames[String(d.minigame)]?.name ?? String(d.minigame);
   511→          store.pushLog(d.success ? `🎮 Minijuego superado: ${name}` : `🎮 Minijuego fallido: ${name}`, d.success ? 'good' : 'bad');
   512→          break;
   513→        }
   514→        case 'ui_toast':
   515→          store.pushToast(String(d.text), (d.tone as Tone | undefined) ?? 'info');
   516→          break;
   517→        case 'finale':
   518→          store.setFinale(String(d.text));
   519→          break;
   520→        case 'funds_changed': {
   521→          const amount = typeof d.amount === 'number' ? d.amount : 0;
   522→          if ((d.to === 'player' || d.to === 'community') && amount !== 0) {
   523→            store.pushLog(amount > 0 ? `+${amount} € (fondos)` : `${amount} € (fondos)`, amount > 0 ? 'good' : 'bad');
   524→          }
   525→          break;
   526→        }
   527→        default: {
   528→          // Umbral de canal con texto del catálogo (odors, wifi y futuros canales): totalmente data-driven.
   529→          if (d.uiLog === true && typeof d.message === 'string') {
   530→            store.pushLog(String(d.message).replace('{block}', blockLabel(String(d.block))), (d.tone as Tone | undefined) ?? 'info');
   531→          }
   532→          break; // resident_moved, param_changed, activity_ended... demasiado ruido para el registro
   533→        }
   534→      }
   535→    }
   536→  }

   557→  private objectiveUi(): { title: string; text: string } | null {
   558→    if (!this.world) return null;
   559→    const o = activeObjective(this.world, this.content);
   560→    if (!o) return null;
   561→    if (o.state === 'awaiting_choice') return { title: o.title, text: 'El edificio contiene la respiración. Decide qué hacer.' };
   562→    const s = o.step!;
   563→    const left = o.untilTick !== null ? ` · quedan ${Math.max(0, o.untilTick - this.world.tick)} min` : '';
   564→    switch (s.type) {
   565→      case 'phone_call':
   566→        return { title: o.title, text: 'Suena el teléfono. Atiéndelo.' };
   567→      case 'reach':
   568→        return { title: o.title, text: `Llega a la puerta (${blockLabel(s.block ?? '')}).` };
   569→      case 'enter':
   570→        return { title: o.title, text: 'Entra en tu piso con E.' };
   571→      case 'catch_in_act':
   572→        return { title: o.title, text: `Pilla al responsable in fraganti${left}.` };
   573→      default:
   574→        return { title: o.title, text: o.title };
   575→    }
   576→  }
   577→
   578→  private promptText(): string | null {
   579→    const logic = this.logic;
   580→    const world = this.world;
   581→    if (!logic || !world) return null;
   582→    const p = logic.position();
   583→    if (p.mode === 'inside') return 'E · salir a la escalera';
   584→    if (p.climbing) return null;
   585→    const door = logic.doorInFront();
   586→    if (door) return door.access === 'locked' ? `E · ${blockLabel(door.id)}: candado` : `E · entrar en ${blockLabel(door.id)}`;
   587→    const zone = logic.zone();
   588→    if (this.content.blocks[zone.defId]!.features?.stairs) {
   589→      return world.blocks[zone.id]!.params.elevator_status === 'ok' ? 'W/S · ascensor' : 'W/S · escalera (el ascensor no funciona)';
   590→    }
   591→    return null;
   592→  }
   593→
   594→  /** Viviendas del jugador para el editor de interiores (foto serializable, posiciones ya en %). */
   595→  private homesUi() {
   596→    const world = this.world!;
   597→    const cellH = this.content.rules.view.cellH;
   598→    return Object.values(world.blocks)
   599→      .filter((b) => b.kind === 'dwelling' && b.ownerId === world.playerId)
   600→      .sort((a, b) => a.id.localeCompare(b.id))
   601→      .map((b) => ({
   602→        id: b.id,
   603→        label: blockLabel(b.id),
   604→        rooms: b.rooms.map((room) => {
   605→          const rdef = this.content.rooms[room.defId];
   606→          const slots = (rdef?.furnitureSlots ?? []).map((s) => {
   607→            const itemId = room.furniture[s.id] ?? '';
   608→            const fdef = itemId ? this.content.furniture[itemId] : undefined;
   609→            return {
   610→              id: s.id,
   611→              kind: s.kind,
   612→              left: Math.min(92, Math.max(2, (s.pos[0] / Math.max(0.001, rdef?.size.w ?? 1)) * 100)),
   613→              bottom: Math.min(76, Math.max(0, (s.pos[1] / Math.max(0.001, cellH)) * 100)),
   614→              item: fdef ? { id: fdef.id, name: fdef.name, emoji: fdef.emoji ?? '📦', color: fdef.view.color } : null,
   615→            };
   616→          });
   617→          return { key: room.key, label: rdef?.label ?? room.key, access: room.access, slots };
   618→        }),
   619→      }));
   620→  }
   621→
   622→  private syncHud(): void {
   623→    const world = this.world;
   624→    if (!world) return;
   625→    const player = world.residents[world.playerId]!;
   626→    const home = player.home;
   627→    const params = visibleParams(world, this.content, home).map((p) => ({ name: p.name, label: p.def.label ?? p.name, value: String(p.value) }));
   628→    const missions = world.missions.map((m) => {
   629→      const def = this.content.missions[m.id]!;
   630→      const step = m.state === 'active' ? def.steps[m.step] : undefined;
   631→      return {
   632→        id: m.id,
   633→        title: def.title,
   634→        state: m.state,
   635→        hint: step?.hint ?? null,
   636→        remaining: m.until !== null ? `quedan ${Math.max(0, m.until - world.tick)} min` : null,
   637→      };
   638→    });
   639→    const travel = travelMinutes(buildCirculation(world, this.content), world.entry, home);
   640→    const elevatorBlock = Object.values(world.blocks).find((b) => b.params.elevator_status !== undefined);
   641→    useGameStore.getState().syncHud({
   642→      clock: { ...world.clock },
   643→      weather: world.weather.state,
   644→      funds: world.community.funds,
   645→      wallet: player.wallet,
   646→      reputation: player.reputation,
   647→      elevator: String(elevatorBlock?.params.elevator_status ?? 'ok'),
   648→      homeLabel: blockLabel(home),
   649→      travelText: Number.isFinite(travel) ? `del portal a tu puerta: ${travel} min` : null,
   650→      objective: this.objectiveUi(),
   651→      prompt: this.promptText(),
   652→      params,
   653→      missions,
   654→      homes: this.homesUi(),
   655→      furnitureConfig: { refundRatio: this.content.rules.furniture.refundRatio, roomUnlockPrice: this.content.rules.furniture.roomUnlock.price },
   656→    });
   657→  }
