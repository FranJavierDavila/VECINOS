   921→  function color(b: BlockInstance): Color3 {
   922→    const base = Color3.FromHexString(content.blocks[b.defId]!.view?.color ?? '#cccccc');
   923→    if (b.kind === 'dwelling' && b.access === 'locked') return Color3.FromHexString(v.lockedColor);
   924→    if (b.kind === 'dwelling' && b.occupancy !== 'occupied') return Color3.Lerp(base, Color3.White(), v.vacantTint);
   925→    if (b.params.elevator_status !== undefined && b.params.elevator_status !== 'ok') return Color3.FromHexString(v.brokenColor);
   926→    return base;
   927→  }

   930→  function stateTint(b: BlockInstance): Color3 | null {
   931→    if (b.kind === 'dwelling' && b.access === 'locked') return Color3.FromHexString(v.lockedColor);
   932→    if (b.kind === 'dwelling' && b.occupancy !== 'occupied') return Color3.Lerp(Color3.FromHexString(content.blocks[b.defId]!.view?.color ?? '#cccccc'), Color3.White(), v.vacantTint);

   937→  /** Emissive del bloque: canales de influencia → emissive según color del canal en el catálogo. Se acumulan. */
   938→  function applyEmissive(view: BlockView, b: BlockInstance) {
   939→    let e = new Color3(0, 0, 0);
   940→    for (const [channel, value] of Object.entries(world.channels)) {
   941→      const level = value[b.id];
   942→      if (!level) continue;
   943→      const g = glow[channel];
   944→      if (!g) continue;
   945→      e = e.add(g.scale(Math.min(1, level)));
   946→    }
   947→    for (const m of view.mats) m.emissiveColor = e;
   948→  }

   950→  /** Color difuso/albedo por material: greybox plano (tintStrength 1) o mezcla sobre el arte. */
   951→  function applyTint(view: BlockView, b: BlockInstance) {
   952→    const state = color(b);
   953→    const override = stateTint(b);
   954→    for (let i = 0; i < view.mats.length; i++) {
   955→      const m = view.mats[i]!;
   956→      const base = view.baseColor[i]!;
   957→      const c = view.tintStrength >= 1 ? state : override ? Color3.Lerp(base, override, view.tintStrength) : base;
   958→      if (m.diffuseColor) m.diffuseColor = c;
   959→      else if (m.albedoColor) m.albedoColor = c;
   960→    }
   961→  }
   962→
   963→  // ── Privacidad de viviendas (v1.17): conjunto de bloques con alguien dentro (el jugador
   964→  //  vía playerInside + cualquier residente presente cuyo location sea el bloque). Se
   965→  //  recalcula por frame en update(): la fachada reacciona al instante cuando un vecino
   966→  //  llega a casa o el jugador entra, sin esperar al siguiente tick.
   967→  let playerInside: string | null = null;
   968→  const privacyOn = v.privacy.enabled;
   969→  const computeOccupancy = (): Set<string> => {
   970→    const occ = new Set<string>();
   971→    if (playerInside) occ.add(playerInside);
   972→    for (const r of Object.values(world.residents)) {
   973→      if (r.present && r.location) occ.add(r.location);
   974→    }
   975→    return occ;
   976→  };
   977→
   978→  /** Visibilidad efectiva de fachada/puerta/candado: candado solo si access='locked';
   979→   *  fachada cerrada si está bloqueada O si la privacidad la cierra (nadie dentro).
   980→   *  paint() la fija por el acceso y el LOD puede ocultarlos encima. */
   981→  function applyPanels(view: BlockView, b: BlockInstance, occ?: Set<string>) {
   982→    const locked = b.kind === 'dwelling' && b.access === 'locked';
   983→    let privacyClosed = false;
   984→    if (privacyOn && b.kind === 'dwelling') {
   985→      const set = occ ?? computeOccupancy();
   986→      privacyClosed = !set.has(b.id);
   987→    }
   988→    const wall = (locked || privacyClosed) && !view.lodHiddenPanels;
   989→    view.front?.setEnabled(wall);
   990→    view.door?.setEnabled(wall);
   991→    view.handle?.setEnabled(wall);
   992→    view.lock?.setEnabled(locked && !view.lodHiddenPanels);
   993→  }
   994→
   995→  /**
   996→   * LOD por distancia (rules.view.lod). Se ejecuta SIEMPRE al final de update(): con foco degrada
   997→   * cada bloque según su distancia, y sin foco (o con lod.enabled=false) restaura el estado completo.
   998→   * Idempotente y sin allocations por frame: solo aritmética, booleanos y asignaciones.
   999→   *
  1000→   * Métrica (Vista · LOD por distancia): dx = |centroX del bloque − focus.x| en metros y
  1001→   * dy = |planta del bloque − focus.floor| * view.cellH en metros; dist = Math.hypot(dx, dy).
  1002→   * Nivel efectivo = todos los niveles de lod.levels cuyo `dist` ≤ distBloque (el `hide` es
  1003→   * acumulado: un flag se oculta si cualquier nivel alcanzado lo lista; sin nivel → todo visible).
  1004→   */
  1005→  function applyLod(focus: { x: number; floor: number } | null) {
  1006→    const lod = v.lod;
  1007→    for (const id in world.blocks) {
  1008→      const b = world.blocks[id]!;
  1009→      const view = views.get(id)!;
  1010→      let hideEdges = false;
  1011→      let hideGlow = false;
  1012→      let hidePanels = false;
  1013→      if (lod.enabled && focus) {
  1014→        const dx = Math.abs(view.centerX - focus.x);
  1015→        const dy = Math.abs(view.floor - focus.floor) * v.cellH;
  1016→        const dist = Math.hypot(dx, dy);
  1017→        for (let i = 0; i < lod.levels.length; i++) {
  1018→          const level = lod.levels[i];
  1019→          if (level.dist > dist) continue;
  1020→          for (let j = 0; j < level.hide.length; j++) {
  1021→            const flag = level.hide[j];
  1022→            if (flag === 'edges') hideEdges = true;
  1023→            else if (flag === 'glow') hideGlow = true;
  1024→            else if (flag === 'panels') hidePanels = true;
  1025→          }
  1026→        }
  1027→      }
  1028→      // 'edges': interruptor nativo barato — el renderingGroup consulta edgesRenderer.isEnabled cada
  1029→      // frame, así que nunca se destruye ni recrea el EdgesRenderer (disable/enableEdgesRendering sí
  1030→      // lo harían: el primero lo dispone y el segundo recomputaría las aristas). El grosor/color
  1031→      // viven en la malla y sobreviven al apagado. Los modelos glTF no tienen edges (edgesMesh null).
  1032→      if (view.edgesMesh?.edgesRenderer) view.edgesMesh.edgesRenderer.isEnabled = !hideEdges;
  1033→      if (view.edgesMesh2?.edgesRenderer) view.edgesMesh2.edgesRenderer.isEnabled = !hideEdges;
  1034→      // 'glow': pisa el emissive CADA frame (paint() pudo repintar los canales este tick); al soltarlo,
  1035→      // una única llamada repinta el cálculo de paint para no esperar al próximo tick.
  1036→      if (hideGlow) {
  1037→        view.glowForced = true;
  1038→        for (let i = 0; i < view.mats.length; i++) view.mats[i]!.emissiveColor = view.baseEmissive[i]!;
  1039→      } else if (view.glowForced) {
  1040→        view.glowForced = false;
  1041→        applyEmissive(view, b);
  1042→      }
  1043→      // 'panels': solo gira la bandera (la visibilidad la calcula applyPanels); en la transición se
  1044→      // aplica ya para no esperar al próximo paint(), que reproducirá el mismo resultado.
  1045→      if (view.lodHiddenPanels !== hidePanels) {
  1046→        view.lodHiddenPanels = hidePanels;
  1047→        applyPanels(view, b);
  1048→      }
  1049→    }
  1050→  }
  1051→
  1052→  return {
  1053→    paint() {
  1054→      for (const b of Object.values(world.blocks)) {
  1055→        const view = views.get(b.id)!;
  1056→        applyTint(view, b);
  1057→        applyEmissive(view, b);
  1058→        applyPanels(view, b);
  1059→      }
  1060→      // Interiores 2D (Fase C): suelo de cada estancia según su acceso + tapas de puertas cerradas.
  1061→      for (const [blockId, interior] of interiors) {
  1062→        const block = world.blocks[blockId];
  1063→        if (!block) continue;
  1064→        for (const f of interior.floors) {
  1065→          const room = block.rooms.find((r) => r.key === f.roomKey);
  1066→          const locked = room ? room.access !== 'unlocked' : true;
  1067→          const hex = locked ? v.lockedColor : content.rooms[f.defId]?.view?.color ?? '#cccccc';
  1068→          f.mat.diffuseColor = Color3.FromHexString(hex);
  1069→        }
  1070→        for (const c of interior.covers) {
  1071→          const anyLocked = c.rooms.some((key) => block.rooms.find((r) => r.key === key)?.access !== 'unlocked');
  1072→          c.mesh.setEnabled(anyLocked);
  1073→        }
  1074→      }
  1075→      // Muebles del editor de interiores: visibles/escala/color según furniture.view (catálogo).
  1076→      for (const [key, fv] of furnViews) {
  1077→        const [blockId, roomKey, slotId] = key.split(':');
  1078→        const room = world.blocks[blockId ?? '']?.rooms.find((r) => r.key === roomKey);
  1079→        const itemId = room?.furniture[slotId ?? ''] ?? '';
  1080→        const fdef = itemId ? content.furniture[itemId] : undefined;
  1081→        if (!fdef) {
  1082→          fv.mesh.setEnabled(false);
  1083→          continue;
  1084→        }
  1085→        const [iw, ih, idz] = fdef.view.size;
  1086→        fv.mesh.setEnabled(true);
  1087→        fv.mesh.scaling.set(iw, ih, idz);
  1088→        // Con y de anclaje (muebles de pared) cuelga a esa altura; si no, apoyado en el suelo.
  1089→        const yBase = fv.y !== null ? fv.bottom + v.slab + v.wall + fv.y : fv.bottom + v.slab + v.wall;
  1090→        fv.mesh.position.set(fv.x, yBase + ih / 2, fv.z);
  1091→        fv.mat.diffuseColor = Color3.FromHexString(fdef.view.color);
  1092→      }
  1093→    },
  1094→    /** Sacude el candado de una puerta cerrada. */
  1095→    shake(blockId: string) {
  1096→      shakes.set(blockId, 0.4);
  1097→    },
  1098→    /**
  1099→     * Apilamiento 3D (v1.19): fase del bloque trasero de `blockId`.
  1100→     *  · 'idle'   — bloque inactivo (todo como siempre).
  1101→     *  · 'peek'   — el jugador se acerca a la puerta del fondo: el bloque trasero asoma por
  1102→     *               el hueco (transparencia → visible) y la hoja se abre.
  1103→     *  · 'inside' — cruzaste: el bloque principal SE DESVANECE y el trasero queda en primer
  1104→     *               plano con su contorno rojo.
  1105→     */
  1106→    setRearPhase(blockId: string, phase: 'idle' | 'peek' | 'inside'): void {
  1107→      const r = rears.get(blockId);
  1108→      if (r) r.phase = phase;
  1109→    },
  1110→    hasRear(blockId: string): boolean {
  1111→      return rears.has(blockId);
  1112→    },
  1113→    /**
  1114→     * dt en segundos; `focus` es el punto de interés (posición del jugador) para el
  1115→     * LOD por distancia (rules.view.lod): applyLod() lo aplica SIEMPRE al final del frame
  1116→     * (con focus undefined, p. ej. cámara 'edificio', restaura el estado completo).
  1117→     * `inside` es el bloque donde está el jugador ahora (pos.insideBlock): alimenta la
  1118→     * privacidad y re-evalúa las fachadas CADA frame — un vecino que llega a casa abre
  1119→     * su interior al instante, sin esperar al siguiente tick.
  1120→     * `pos` (v1.19) alimenta los carteles guía de trasteros y el apilamiento 3D.
  1121→     */
  1122→    update(dt: number, focus?: { x: number; floor: number }, inside?: string | null, pos?: { x: number; z: number; insideBlock: string | null }) {
  1123→      if (focus) lastFocus = focus;
  1124→      if (inside !== undefined) playerInside = inside;
  1125→      // PRIVACIDAD (v1.17) cada frame: la fachada de una vivienda sin nadie dentro se cierra
  1126→      // con su puerta al pasillo y se abre en cuanto alguien entra (propietario, visita…
  1127→      // o un ladrón). Vive en update() y no en paint() para reaccionar incluso con el
  1128→      // mundo congelado (modal abierto) y sin esperar al siguiente tick.
  1129→      if (privacyOn) {
  1130→        const occ = computeOccupancy();
  1131→        for (const b of Object.values(world.blocks)) {
  1132→          if (b.kind !== 'dwelling') continue;
  1133→          applyPanels(views.get(b.id)!, b, occ);
  1134→        }
  1135→      }
  1136→      // CARTELES GUÍA (v1.19): visibles solo con el jugador dentro del bloque de trasteros;
  1137→      // texto = los trasteros que tienes a cada lado según la profundidad del pasillo.
  1138→      const sgG = colBlocks.length > 0 ? storageGeomOf(v) : null;
  1139→      for (const [blkId, sign] of guideSigns) {
  1140→        const active = !!pos && pos.insideBlock === blkId && !!sgG;
  1141→        const plate = scene.getMeshByName(`${blkId}_guia`);
  1142→        const frame = scene.getMeshByName(`${blkId}_guia_marco`);
  1143→        plate?.setEnabled(active);
  1144→        frame?.setEnabled(active);
  1145→        if (!active || !sgG) continue;
  1146→        const localZ = pos!.z; // en modo 'inside' de trasteros, z es profundidad local
  1147→        const usable = sgG.depth - 2 * sgG.wallT;
  1148→        const perSide = new Map<string, number>();
  1149→        for (const u of sign.plan) perSide.set(u.side, (perSide.get(u.side) ?? 0) + 1);
  1150→        const labelOf = (side: 'left' | 'right'): string => {
  1151→          const count = perSide.get(side) ?? 0;
  1152→          if (count === 0) return tr.emptyLabel;
  1153→          const idx = Math.max(0, Math.min(count - 1, Math.floor((localZ - sgG.wallT) / (usable / count))));
  1154→          const unit = sign.plan.find((u) => u.side === side && u.index === idx);
  1155→          return unit ? blockLabel(`blk_p${unit.floor}_${sign.ownTag}`) : tr.emptyLabel;
  1156→        };
  1157→        const left = labelOf('left');
  1158→        const right = labelOf('right');
  1159→        const key = `${left}|${right}`;
  1160→        if (sign.last !== key) {
  1161→          sign.last = key;
  1162→          const ctx = sign.tex.getContext() as unknown as CanvasRenderingContext2D;
  1163→          ctx.fillStyle = tr.guide.bg;
  1164→          ctx.fillRect(0, 0, 256, 60);
  1165→          ctx.font = 'bold 30px Arial, sans-serif';
  1166→          ctx.textBaseline = 'middle';
  1167→          ctx.fillStyle = tr.guide.text;
  1168→          ctx.textAlign = 'left';
  1169→          ctx.fillText('◀', 10, 32);
  1170→          ctx.textAlign = 'center';
  1171→          const ownLeft = left.includes(sign.ownTag.toUpperCase());
  1172→          const ownRight = right.includes(sign.ownTag.toUpperCase());
  1173→          ctx.fillStyle = ownLeft ? tr.ownBg : tr.guide.text;
  1174→          ctx.fillText((ownLeft ? '★ ' : '') + left, 88, 32);
  1175→          ctx.fillStyle = ownRight ? tr.ownBg : tr.guide.text;
  1176→          ctx.fillText((ownRight ? '★ ' : '') + right, 176, 32);
  1177→          ctx.fillStyle = tr.guide.text;
  1178→          ctx.textAlign = 'right';
  1179→          ctx.fillText('▶', 248, 32);
  1180→          sign.tex.update(true);
  1181→        }
  1182→      }
  1183→      // APILAMIENTO 3D (v1.19): avanzar los fundidos de los bloques traseros.
  1184→      for (const r of rears.values()) {
  1185→        const mainT = r.phase === 'inside' ? 0 : 1;
  1186→        const rearT = r.phase === 'idle' ? 0 : 1;
  1187→        // v1.20: la PARED DIVISORIA del bloque trasero desaparece al entrar (fase 'inside').
  1188→        // Se evalúa SIEMPRE (incluso con los fundidos ya asentados): es un cambio de fase,
  1189→        // no de alpha — el early-continue de abajo no la vería.
  1190→        const frontOpen = r.phase === 'inside';
  1191→        for (const f of r.frontParts) f.setEnabled(!frontOpen);
  1192→        if (r.settled && r.mainAlpha === mainT && r.rearAlpha === rearT) continue;
  1193→        const mRate = dt / Math.max(0.05, rb.fadeSeconds);
  1194→        const rRate = dt / Math.max(0.05, rb.peekSeconds);
  1195→        r.mainAlpha += Math.sign(mainT - r.mainAlpha) * Math.min(Math.abs(mainT - r.mainAlpha), mRate);
  1196→        r.rearAlpha += Math.sign(rearT - r.rearAlpha) * Math.min(Math.abs(rearT - r.rearAlpha), rRate);
  1197→        r.settled = r.mainAlpha === mainT && r.rearAlpha === rearT;
  1198→        const mainKids = r.mainRoot.getChildMeshes();
  1199→        for (const k of mainKids) k.visibility = r.mainAlpha;
  1200→        r.mainRoot.setEnabled(r.mainAlpha > 0.002);
  1201→        const rearKids = r.node.getChildMeshes();
  1202→        for (const k of rearKids) k.visibility = r.rearAlpha;
  1203→        r.node.setEnabled(r.rearAlpha > 0.002);
  1204→        for (const o of r.outline) o.setEnabled(r.mainAlpha < 0.55 && r.rearAlpha > 0.4);
  1205→        if (r.doorLeaf) r.doorLeaf.setEnabled(r.rearAlpha < 0.2);
  1206→      }
  1207→      for (const [id, left] of shakes) {
  1208→        const view = views.get(id)!;
  1209→        const t = left - dt;
  1210→        if (t <= 0) {
  1211→          shakes.delete(id);
  1212→          view.lock?.position.set(view.lockBase.x, view.lockBase.y, view.lock.position.z);
  1213→        } else {
  1214→          shakes.set(id, t);
  1215→          view.lock?.position.set(view.lockBase.x + Math.sin(t * 60) * 0.06, view.lockBase.y, view.lock.position.z);
  1216→        }
  1217→      }
  1218→      // LOD SIEMPRE al final del frame: con foco aplica el nivel efectivo por bloque,
  1219→      // sin foco restaura edges/emissive/paneles al estado completo.
  1220→      applyLod(focus ?? null);
  1221→    },
  1222→  };
  1223→}
