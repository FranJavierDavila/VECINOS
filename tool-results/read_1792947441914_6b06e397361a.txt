    12→const frontZOf = (roomZ: number, commonDepth: number): number => roomZ - commonDepth / 2;

    14→interface Actor {
    15→  /** Residente dueño del nodo (para las rutas de ascensor y depuración). */
    16→  id: string;
    17→  /** Nodo posicionable: cápsula greybox o raíz del modelo glTF (pivote en pies). */
    18→  node: Mesh | TransformNode;
    19→  height: number;
    20→  /** Offset Y del pivote: cápsula centrada (+h/2) o modelo con pies en el origen (+margen). */
    21→  pivotY: number;
    22→  /** Cola de tramos en curso: el actor la agota ANTES de capturar el destino del mundo de nuevo
    23→   *  (ir al pozo → subir → salir: un salto del sim se ve como un viaje, nunca un teletransporte). */
    24→  path: { x: number; y: number; z: number }[];
    25→  /** Vida interior (Fase C): cuenta atrás para el próximo punto de deambulación y nº de vuelta
    26→   *  (el punto es determinista por residente+ vuelta: sin RNG global ni capturas irreproducibles). */
    27→  wanderT: number;
    28→  wanderLap: number;
    29→  /** Objetivo interior VIGENTE (coords locales de la vivienda): se refresca con la cadencia del
    30→   *  catálogo (npcWanderSeconds), no cada tick — amortigua el parpadeo de actividades con chance. */
    31→  targetLocal: Pt | null;
    32→  lastLoc: string | null;
    33→  /** Dirección de cutscenes (Fase D): mientras está activa, la IA normal no captura destinos;
    34→   *  el actor solo consume la cola de tramos que le marca el director (nunca teletransporta). */
    35→  directed: boolean;
    36→  /** Rumbo actual (rad): el actor MIRA hacia donde camina (v1.2, credibilidad 2.5D).
    37→   *  0 = el modelo mira +X (perfil); ±π/2 = de espaldas/frente al entrar o salir de casa. */
    38→  yaw: number;
    39→  /** Rumbo pendiente de terminar: el actor remata el giro aunque se pare (nunca queda torcido). */
    40→  targetYaw: number | null;
    41→  /** Gesto social en curso (stateAnimations: talk/laugh/argue/phone/sit…) o null. */
    42→  gesture: string | null;
    43→  /** Cuenta atrás del gesto temporal (0 = persistente hasta que alguien lo quite). */
    44→  gestureT: number;
    45→  /** El objetivo interior vigente es un ancla de asiento (rules.view.seatAnchors): se sienta al llegar. */
    46→  seated: boolean;
    47→  /** Viaje en ascensor registrado (v1.5): { x: puerta; from/to: plantas }. null = sin viaje. */
    48→  elevatorReq: { x: number; from: number; to: number } | null;
    49→  /** Fase del viaje que se estaba atendiendo (para detectar transiciones y encolar tramos). */
    50→  ridePhase: string | null;
    51→  /** Estaba habilitado el frame anterior (v1.10): la REAPARICIÓN coloca al actor donde dice el
    52→   *  sim (vuelve de la calle), nunca caminando desde su última posición visible por la pared. */
    53→  enabledPrev: boolean;
    54→  /** Trastero ocupado (v1.18): id del bloque por columna cuyo compartimento habitó por última
    55→   *  vez. Mientras su ubicación siga siendo ese bloque, el nodo permanece oculto (está DENTRO
    56→   *  de su celda cerrada); al cambiar, reaparece por la puerta con la lógica de siempre. */
    57→  stowedBlock: string | null;
    58→  /** Rig del arte final (null mientras carga o si hay fallback greybox). */
    59→  rig: ActorRig | null;
    60→}

    62→/**
    63→ * API de dirección de cutscenes (Fase D): toma actores, colócalos en escena y márcales rutas.
    64→ * Mientras un actor está "taken", el bucle normal de IA no le captura destinos: solo consume
    65→ * la cola de tramos (la misma locomoción interpolada de siempre, cero teletransportes).
    66→ */
    67→export interface ActorsDirector {
    68→  /** Posición del nodo (null si el actor no existe o está oculto). */
    69→  nodeOf(id: string): { x: number; y: number; z: number } | null;
    70→  /** Crea el actor si falta y lo marca como dirigido. false si el residente no está en escena. */
    71→  take(id: string): boolean;
    72→  /** Suelta al actor: la IA vuelve a decidir destino (cola y objetivo interior reseteados). */
    73→  release(id: string): void;
    74→  /** Coloca al actor en un punto de escena (staging teatral, decisión del contenido). */
    75→  stageAt(id: string, x: number, floor: number, z: number): void;
    76→  /** Encola la ruta del nodo actual al destino (pozo incluido si cambia la planta). */
    77→  walkTo(id: string, dest: { x: number; z: number; floor: number; inside?: { block: string; local: Pt } }): boolean;
    78→  /** true cuando el actor no tiene tramos pendientes (llegó — o no existe). */
    79→  idle(id: string): boolean;
    80→  /** Punto sobre la cabeza del actor (bocadillos v1.10); null si no está en escena. */
    81→  headOf(id: string): { x: number; y: number; z: number } | null;
    82→  /** Gesto social en curso (stateAnimations: talk/laugh/argue/phone…) o null. */
    83→  gestureOf(id: string): string | null;
    84→  /** true si el actor está tomado por el director de cutscenes (el staging manda). */
    85→  isDirected(id: string): boolean;
    86→  /**
    87→   * Estado social explícito (v1.2): talk/laugh/argue/phone/sit… según stateAnimations del
    88→   * manifiesto. Con `seconds` > 0 el gesto expira solo (líneas de diálogo); null lo suelta
    89→   * ya (al colgar el teléfono, al cerrar el modal). El clip que falte cae al default del rig.
    90→   */
    91→  setGesture(id: string, state: string | null, seconds?: number): void;
    92→}

    94→/**
    95→ * El jugador y los vecinos: cápsulas greybox o, si el manifiesto de assets declara un
    96→ * modelo glTF para el arquetipo (Content.assets.actors), el modelo con sus animaciones
    97→ * (idle/walk por desplazamiento y clip por actividad vía activityAnimations).
    98→ * Color y altura salen de la vista del arquetipo; radios y fallbacks, de content.rules.view.
    99→ *
   100→ * Dinámico (multijugador): la plantilla se sincroniza con world.residents en cada
   101→ * update — residentes nuevos reciben nodo y residentes que ya no existen lo pierden.
   102→ */
   103→export function createActors(scene: Scene, world: World, content: Content, elevators?: ElevatorsDirector | null) {
   104→  const { Color3, MeshBuilder, Quaternion, StandardMaterial, Vector3 } = B();
   105→  const v = content.rules.view;
   106→  const actors = new Map<string, Actor>();
   107→
   108→  /** Crea la cápsula de un residente (greybox/fallback) y lanza la carga del modelo si el manifiesto lo declara. */
   109→  const ensureActor = (id: string, archetype: string): Actor => {
   110→    const view = content.archetypes[archetype]?.view;
   111→    const height = view?.height ?? v.actorFallbackHeight;
   112→    const mesh = MeshBuilder.CreateCapsule(id, { height, radius: v.actorRadius }, scene);
   113→    const m = new StandardMaterial(`${id}_mat`, scene);
   114→    m.diffuseColor = Color3.FromHexString(view?.color ?? v.actorFallbackColor);
   115→    m.emissiveColor = m.diffuseColor.scale(0.25);
   116→    mesh.material = m;
   117→    mesh.isPickable = false;
   118→    const a: Actor = { id, node: mesh, height, pivotY: height / 2, path: [], wanderT: 0, wanderLap: 0, targetLocal: null, lastLoc: null, directed: false, yaw: 0, targetYaw: null, gesture: null, gestureT: 0, seated: false, elevatorReq: null, ridePhase: null, enabledPrev: false, stowedBlock: null, rig: null };
   119→    actors.set(id, a);
   120→
   121→    // Arte final (opcional): modelo glTF del manifiesto; cualquier fallo → cápsula.
   122→    const def = content.assets?.actors?.[archetype];
   123→    if (def && content.assets) {
   124→      void instantiateActorAsset(scene, def, content.assets.base, `${id}:${archetype}`).then((rig) => {
   125→        // La escena pudo morir mientras cargaba (restart/Cargar partida): no adjuntar nada viejo.
   126→        if (rig && scene.isDisposed) {
   127→          rig.dispose();
   128→          return;
   129→        }
   130→        if (!rig) return; // fallback greybox permanente
   131→        const cur = actors.get(id);
   132→        if (!cur || cur.rig) {
   133→          rig.dispose(); // el residente se fue (o ya llegó otro rig): no dejar basura
   134→          return;
   135→        }
   136→        // PRESERVAR POSE (v1.10): el modelo puede terminar de cargar EN PLENO paseo (los assets
   137→        // llegan async). Sin esto, el swap cápsula→rig recolocaba el nodo a origen y el actor
   138→        // "saltaba" (el guard antiguo lo enmascaraba con OTRO teletransporte al objetivo: la
   139→        // vecina con la maleta cambiando de planta de golpe). Pies y rumbo se conservan 1:1.
   140→        const feetY = cur.node.position.y - cur.pivotY;
   141→        rig.root.position.set(cur.node.position.x, feetY + 0.02, cur.node.position.z);
   142→        if (rig.root.rotationQuaternion) Quaternion.RotationYawPitchRollToRef(cur.yaw, 0, 0, rig.root.rotationQuaternion);
   143→        else rig.root.rotation.y = cur.yaw;
   144→        // El modelo se authora a nominalHeight: escala uniforme a la altura del arquetipo.
   145→        rig.root.scaling.setAll(height / def.nominalHeight);
   146→        cur.node.dispose(false, true); // cápsula fuera (libera también su material propio)
   147→        cur.node = rig.root;
   148→        cur.pivotY = 0.02; // pies en el suelo
   149→        cur.rig = rig;
   150→      });
   151→    }
   152→    return a;
   153→  };
   154→
   155→  const goal = new Vector3();
   156→  /** true si la fase del viaje cambió desde el último frame (hay que recolocar la cola). */
   157→  const stalePhase = (a: Actor, phase: string): boolean => a.ridePhase !== phase;
   158→  /** Pozos verticales del mundo (features stairs/elevator), para el RITMO de los tramos verticales. */
   159→  const shafts = Object.values(world.blocks).filter((b) => {
   160→    const f = content.blocks[b.defId]!.features;
   161→    return !!f && (f.stairs === true || f.elevator === true);
   162→  });
   163→  /** ¿Pozo UTILIZABLE para rutear EN ESTE MOMENTO? (v1.10) La escalera siempre; el ascensor
   164→   *  SOLO si funciona (elevator_status 'ok'). Se comprueba EN CADA RUTA porque el estado del
   165→   *  sim cambia en vivo (avería/reparación): un filtro estático enlazaría llamadas al pozo
   166→   *  averiado que el director cancela al frame siguiente (bucle llamada/cancelación). */
   167→  const shaftUsable = (b: BlockInstance): boolean => {
   168→    const f = content.blocks[b.defId]!.features!;
   169→    if (f.stairs === true) return true;
   170→    return f.elevator === true && b.params.elevator_status === 'ok';
   171→  };
   172→  const shaftXOf = (b: BlockInstance): number => (b.cell.col + b.size.w / 2) * v.cellW;
   173→  /** Calado transitablo del bloque (vivienda 2×; naves del cierre v1.16 a calado de vivienda;
   174→   *  trasteros por columna v1.18: depth × blockDepthFactor).
   175→   *  v1.19: en vivienda manda el LAYOUT — las salas traseras viven más allá de view.depth. */
   176→  const depthOfBlock = (b: BlockInstance): number => {
   177→    if (b.kind === 'dwelling') {
   178→      const layoutId = content.blocks[b.defId]?.container?.layout;
   179→      const layout = layoutId ? content.layouts[layoutId] : undefined;
   180→      const rects = (layout?.rooms ?? []).map((r) => r.rect).filter((r): r is NonNullable<typeof r> => !!r);
   181→      if (rects.length > 0) return Math.max(...rects.map((r) => r.z + r.d));
   182→      return v.depth;
   183→    }
   184→    if (b.defId === 'trasteros_planta' || b.defId === 'bajocubierta') return v.depth;
   185→    if (b.defId === v.trasteros.blockDefId) return v.depth * v.trasteros.blockDepthFactor;
   186→    if (b.defId === 'tejado') return v.depth + v.roof.overhang * 2;
   187→    return v.commonDepth;
   188→  };
   189→  /** Plano frontal de los bloques (compartido con storage.ts/building.ts). */
   190→  /**
   191→   * Punto de puerta del trastero de un residente en el bloque por columna de su letra
   192→   * (v1.18): pasillo central hasta su celda. null si no aplica (bloque equivocado, planta
   193→   * sin vivienda en el plan…).
   194→   */
   195→  const storageDoorOf = (resident: ResidentInstance): { x: number; z: number } | null => {
   196→    const home = world.blocks[resident.home];
   197→    if (!home) return null;
   198→    const tag = home.id.slice(home.id.lastIndexOf('_') + 1);
   199→    const store = Object.values(world.blocks).find(
   200→      (b) => isStorageColumnBlock(b, v) && b.id.endsWith(`_${tag}`),
   201→    );
   202→    if (!store) return null;
   203→    const colTag = store.id.slice(store.id.lastIndexOf('_') + 1);
   204→    const floors = Object.values(world.blocks)
   205→      .filter((d) => d.kind === 'dwelling' && d.id.endsWith(`_${colTag}`))
   206→      .map((d) => d.cell.floor)
   207→      .sort((a, z) => a - z);
   208→    return storageDoorPoint(v, store, floors, home.cell.floor);
   209→  };
   210→  /** Plano frontal de las viviendas: z local 0 de los interiores y origen del pasillo. */
   211→  const frontZ = frontZOf(v.roomZ, v.commonDepth);
   212→  /** Bloque main bajo los pies (v1.14): para salir de la banda de viviendas por SU puerta. */
   213→  const hostSpanAt = (x: number, floor: number): { x0: number; w: number } | null => {
   214→    for (const b of Object.values(world.blocks)) {
   215→      if (b.cell.floor !== floor || b.layer !== 'main') continue;
   216→      const x0 = b.cell.col * v.cellW;
   217→      const w = b.size.w * v.cellW;
   218→      if (x >= x0 && x <= x0 + w) return { x0, w };
   219→    }
   220→    return null;
   221→  };
   222→
   223→  /**
   224→   * Ruta (cola de tramos) del nodo actual al destino: cambio de planta → pozo más cercano
   225→   * (andar hasta él por el pasillo, subir/bajar, salir), y dentro de una vivienda con
   226→   * geometría 2D la ruta pasa por los huecos del catálogo (nunca a través de tabiques).
   227→   * La comparten la IA normal y el director de cutscenes: una sola verdad de locomoción.
   228→   *
   229→   * v1.14 — LEY DEL PASILLO (rutas blindadas): todo el tramo horizontal lo calcula
   230→   * horizontalLegs() (@/game/routing): el avance en X solo ocurre sobre la línea del
   231→   * pasillo y la banda de bloques solo se entra/sale en vertical en la x del bloque
   232→   * propio — jamás una diagonal atravesando el 2C (o el núcleo del ascensor) para
   233→   * llegar al 2B. Devuelve false SOLO cuando el destino exige cambiar de planta y no
   234→   * hay pozo utilizable (estado imposible: el llamador decide si recoloca sin teatro).
   235→   */
   236→  const buildRoute = (a: Actor, dest: { x: number; y: number; z: number; inside?: { block: BlockInstance; local: Pt } | null }): boolean => {
   237→    const node = a.node.position;
   238→    let startX = node.x;
   239→    let startZ = node.z;
   240→    let verticalOk = true;
   241→    if (Math.abs(dest.y - node.y) > 0.05) {
   242→      const floorNow = Math.round(node.y / v.cellH);
   243→      // Como el sim: ASCENSOR primero si funciona (coste 1 < 3 de la escalera); la escalera
   244→      // es la vía de emergencia cuando el núcleo está averiado (v1.10: estado comprobado EN VIVO).
   245→      const elevHere = shafts
   246→        .filter((s) => s.cell.floor === floorNow && content.blocks[s.defId]!.features!.elevator === true && s.params.elevator_status === 'ok')
   247→        .sort((s1, s2) => Math.abs(shaftXOf(s1) - startX) - Math.abs(shaftXOf(s2) - startX))[0];
   248→      const here = elevHere ?? shafts
   249→        .filter((s) => s.cell.floor === floorNow && shaftUsable(s))
   250→        .sort((s1, s2) => Math.abs(shaftXOf(s1) - startX) - Math.abs(shaftXOf(s2) - startX))[0];
   251→      if (here) {
   252→        const sx = shaftXOf(here);
   253→        // Salir del interior por la puerta propia (si procede) y andar hasta el pozo SOLO
   254→        // por la línea del pasillo: la diagonal directa cruzaba viviendas vecinas.
   255→        for (const l of horizontalLegs({ view: v, from: { x: startX, z: startZ }, dest: { x: sx, z: v.corridorZ }, hostSpan: hostSpanAt(startX, floorNow) })) {
   256→          a.path.push({ x: l.x, y: node.y, z: l.z });
   257→        }
   258→        // ¿Escalera o ascensor? La escalera REAL (v1.4) va por los peldaños; el ascensor
   259→        // OPERATIVO (v1.5) registra una LLAMADA: el actor camina a la puerta y espera a la
   260→        // cabina (capacidad, puertas y semáforo los lleva el director del ascensor).
   261→        const feat = content.blocks[here.defId]!.features!;
   262→        const stairPts = feat.stairs === true ? stairPathPoints(v, sx, node.y, dest.y) : [];
   263→        if (stairPts.length > 1) {
   264→          for (const pt of stairPts) a.path.push({ x: pt.x, y: pt.y, z: pt.z });
   265→        } else if (elevators && feat.elevator === true) {
   266→          const from = Math.round(node.y / v.cellH);
   267→          const to = Math.round(dest.y / v.cellH);
   268→          if (to !== from) {
   269→            elevators.requestRide(a.id, here.cell.col, from, to);
   270→            a.elevatorReq = { x: sx, from, to };
   271→            a.ridePhase = null;
   272→          }
   273→        } else {
   274→          a.path.push({ x: sx, y: dest.y, z: v.corridorZ });
   275→        }
   276→        startX = sx;
   277→        startZ = v.corridorZ;
   278→      } else {
   279→        verticalOk = false;
   280→      }
   281→    }
   282→    // Tramos horizontales al destino bajo la ley del pasillo (misma función que el enfoque
   283→    // al pozo: entrada al bloque solo en vertical, en la x del destino, o ruta interior).
   284→    const destGeo = dest.inside ? layoutGeometryOf(content, dest.inside.block) : null;
   285→    const legs = horizontalLegs({
   286→      view: v,
   287→      from: { x: startX, z: startZ },
   288→      dest: { x: dest.x, z: dest.z },
   289→      destInside:
   290→        dest.inside && destGeo
   291→          ? { x0: dest.inside.block.cell.col * v.cellW, w: dest.inside.block.size.w * v.cellW, geo: destGeo, target: dest.inside.local }
   292→          : null,
   293→      hostSpan: hostSpanAt(startX, Math.round(dest.y / v.cellH)),
   294→    });
   295→    for (const p of legs) a.path.push({ x: p.x, y: dest.y, z: p.z });
   296→    return verticalOk;
   297→  };
   298→
   299→  return {
   300→    /** playerResidentId identifica al residente del jugador local (multijugador: puede ser un invitado). */
   301→    update(dt: number, player: PlayerPosition, playerResidentId: string) {
   302→      // Altas y bajas: residentes sin nodo → crear; nodos huérfanos → dispose + remove.
   303→      for (const id in world.residents) {
   304→        if (!actors.has(id)) ensureActor(id, world.residents[id]!.archetype);
   305→      }
   306→      const gone: string[] = [];
   307→      for (const [id, a] of actors) {
   308→        if (!world.residents[id]) gone.push(id);
   309→      }
   310→      for (const id of gone) {
   311→        const a = actors.get(id)!;
   312→        actors.delete(id);
   313→        if (a.rig) a.rig.dispose();
   314→        else if (a.node instanceof B().Mesh) a.node.dispose(false, true);
   315→        else a.node.dispose(false, false);
   316→      }
   317→
   318→      const k = 1 - Math.exp(-dt * 10);
   319→      // Ritmos del catálogo (rules.view): crucero de vecinos y subida por pozo.
   320→      const npcSpeed = v.npcWalkSpeed;
   321→      for (const [id, a] of actors) {
   322→        const r = world.residents[id]!;
   323→        const isPlayer = id === playerResidentId;
   324→        const nowEnabled = isPlayer || (r.present && r.location !== null);
   325→        // ── TRASTEROS POR COLUMNA (v1.18): mientras el vecino sigue "en su bloque de
   326→        //  trasteros", el nodo permanece oculto — está DENTRO de su celda cerrada. En cuanto
   327→        //  el sim lo muda (otra ubicación o se va), sale del stow y reaparece por su puerta.
   328→        const stowedHere = a.stowedBlock !== null && r.present && r.location === a.stowedBlock;
   329→        if (a.stowedBlock !== null && !stowedHere) a.stowedBlock = null;
   330→        a.node.setEnabled(nowEnabled && !stowedHere);
   331→        if (!nowEnabled || stowedHere) {
   332→          a.enabledPrev = false;
   333→          continue;
   334→        }
   335→        // En escena lo coloca el director (v1.10): marcar habilitado SIN snap — si no, al soltar
   336→        // al actor tras una cutscene, el primer frame normal lo trataría como "reaparición"
   337→        // y lo teletransportaría al objetivo en vez de rutear desde donde está.
   338→        if (a.directed) a.enabledPrev = true;
   339→        const node = a.node.position;
   340→        const prevX = node.x;
   341→        const prevZ = node.z;
   342→        if (isPlayer) {
   343→          const inside = player.mode === 'inside';
   344→          // Apilamiento 3D (v1.19): dentro del trastero propio el cuerpo vive en el bloque
   345→          // trasero (posición que dicta PlayerLogic), no en el pasillo del sim.
   346→          if (player.rear) {
   347→            goal.set(player.rear.x, player.floor * v.cellH + 0.14 + a.pivotY, player.rear.z);
   348→            if (node.lengthSquared() === 0) node.copyFrom(goal);
   349→            node.addInPlace(goal.subtract(node).scale(Math.min(1, k * 2)));
   350→            // Skipear el resto del bloque isPlayer cayendo al flujo común de posición.
   351→          } else {
   352→          // Dentro de la vivienda el actor sigue la posición 2D (x + profundidad z del layout);
   353→          // en el cubo de escalera (v1.4) manda la Z absoluta del recorrido (pathZ).
   354→          // En pasillo (v1.19) la Z del jugador es ABSOLUTA (dos carriles: interior/exterior).
   355→          const insideBlock = player.insideBlock ? world.blocks[player.insideBlock] : null;
   356→          const insideDepth = insideBlock ? depthOfBlock(insideBlock) : v.depth;
   357→          const z = inside ? frontZ + Math.min(Math.max(player.z, 0), insideDepth) : (player.pathZ ?? player.z);
   358→          goal.set(player.x, player.floor * v.cellH + (inside ? 0.12 : 0.14) + a.pivotY, z);
   359→          if (node.lengthSquared() === 0) node.copyFrom(goal);
   360→          node.addInPlace(goal.subtract(node).scale(Math.min(1, k * 2)));
   361→          }
   362→        } else {
   363→          const b = world.blocks[r.location!]!;
   364→          const home = world.blocks[r.home]!;
   365→          const inRoom = b.kind === 'dwelling';
   366→          const x0 = b.cell.col * v.cellW;
   367→          // ── Trastero propio (v1.18): si el vecino está en el bloque por columna de su
   368→          //  letra, su objetivo es LA PUERTA DE SU CELDA (pasillo central): llega y entra
   369→          //  (stow: nodo oculto dentro del compartimento cerrado).
   370→          const storeDoor = isStorageColumnBlock(b, v) ? storageDoorOf(r) : null;
   371→          // ── Viaje en ascensor (v1.5): mientras hay viaje registrado, el director manda —
   372→          // esperar ante la puerta, entrar al hueco, ser llevado por la cabina y salir en la
   373→          // planta de destino. Si el viaje se cancela (avería), status() null → el flujo
   374→          // normal recaptura destino y la ruta pasará por la escalera.
   375→          const elev = elevators ?? undefined;
   376→          const ride = a.elevatorReq && elev ? elev.status(id) : null;
   377→          if (a.elevatorReq && !ride) {
   378→            a.elevatorReq = null;
   379→            a.ridePhase = null;
   380→            a.path = [];
   381→          }
   382→          if (a.elevatorReq && ride) {
   383→            if (stalePhase(a, ride.phase)) {
   384→              // Transición de fase: recolocar la cola según a dónde hay que caminar ahora.
   385→              if (ride.phase === 'boarding') {
   386→                a.path = [{ x: ride.slotX, y: Math.round(ride.floor) * v.cellH + 0.14 + a.pivotY, z: ride.slotZ }];
   387→              } else if (ride.phase === 'exiting') {
   388→                a.path = [{ x: a.elevatorReq.x, y: a.elevatorReq.to * v.cellH + 0.14 + a.pivotY, z: v.corridorZ }];
   389→              } else {
   390→                a.path = [];
   391→              }
   392→              a.ridePhase = ride.phase;
   393→            }
   394→            if (ride.phase === 'waiting') {
   395→              goal.set(a.elevatorReq.x, a.elevatorReq.from * v.cellH + 0.14 + a.pivotY, v.corridorZ);
   396→              if (node.lengthSquared() === 0) node.copyFrom(goal);
   397→              node.addInPlace(goal.subtract(node).scale(Math.min(1, k * 2)));
   398→            } else if (ride.phase === 'riding') {
   399→              // La cabina manda: posición directa en el hueco (sin teatro ni interpolación propia).
   400→              a.path = [];
   401→              node.set(ride.slotX, ride.floor * v.cellH + 0.14 + a.pivotY, ride.slotZ);
   402→            } else if (ride.phase === 'boarding' && a.path.length === 0) {
   403→              elev!.notifyBoarded(id); // ya está en su hueco: la puerta puede cerrar
   404→            } else if (ride.phase === 'exiting' && a.path.length === 0) {
   405→              a.elevatorReq = null;
   406→              a.ridePhase = null;
   407→            }
   408→          }
   409→          // Con dirección de cutscene (Fase D) el actor solo consume su cola de tramos:
   410→          // ni objetivo de IA ni captura de ruta (el director ya se la ha dado).
   411→          if (!a.directed && !a.elevatorReq) {
   412→            let gx = b.layer === 'corridor' ? (home.cell.col + 0.5) * v.cellW : (x0 + b.size.w * v.cellW) / 2;
   413→            let gz = inRoom ? frontZ + 0.6 : v.corridorZ;
   414→            if (storeDoor) {
   415→              // Trastero por columna (v1.18): se camina POR EL PASILLO CENTRAL hasta SU puerta.
   416→              gx = storeDoor.x;
   417→              gz = storeDoor.z;
   418→            }
   419→            // ── Vida interior (Fase C): en vivienda, el punto lo decide el contenido. La actividad
   420→            // con `anchor` (cama, tv, stove…) coloca al vecino en SU estancia por el catálogo de
   421→            // anclas (rooms.json); sin ancla, deambula por las estancias abiertas. El objetivo se
   422→            // REFRESCA con la cadencia npcWanderSeconds (no cada tick): una actividad con chance
   423→            // que parpadea en el sim no arrastra al vecino por la casa. Cerradas jamás son destino.
   424→            const geo = inRoom ? layoutGeometryOf(content, b) : null;
   425→            let localTarget: Pt | null = null;
   426→            if (a.lastLoc !== r.location) {
   427→              a.lastLoc = r.location;
   428→              a.targetLocal = null;
   429→              a.wanderT = 0;
   430→              a.seated = false;
   431→            }
   432→            if (geo) {
   433→              a.wanderT -= dt;
   434→              if (a.path.length === 0 && a.wanderT <= 0) {
   435→                a.wanderLap += 1;
   436→                a.wanderT = v.npcWanderSeconds * (0.7 + 0.6 * hash01(id, a.wanderLap));
   437→                const act = r.activity ? content.archetypes[r.archetype]?.activities[r.activity] : undefined;
   438→                const anchorKind = act?.anchor;
   439→                const anchored = anchorKind ? interiorTarget(content, geo, anchorKind, id, 0) : null;
   440→                a.targetLocal = anchored ?? interiorTarget(content, geo, undefined, id, a.wanderLap);
   441→                // Ancla de asiento (catálogo: rules.view.seatAnchors): al llegar se sienta (clip "sit").
   442→                a.seated = anchored !== null && !!anchorKind && v.seatAnchors.includes(anchorKind);
   443→              }
   444→              localTarget = a.targetLocal;
   445→            }
   446→            if (localTarget) {
   447→              gx = x0 + localTarget.x;
   448→              gz = frontZ + localTarget.z;
   449→            }
   450→            goal.set(gx, b.cell.floor * v.cellH + 0.14 + a.pivotY, gz);
   451→            // Reaparición (v1.10): el vecino vuelve de la calle o entra en escena — aparece
   452→            // DONDE DICE EL SIM, no caminando desde su última posición visible (por la pared).
   453→            // CON TRASTERO (v1.18): sale en la línea del pasillo ANTE SU BLOQUE y se lo ve
   454→            // entrar y recorrer el pasillo central hasta su celda (nunca pop dentro del stow).
   455→            if (!a.enabledPrev) {
   456→              a.enabledPrev = true;
   457→              if (storeDoor) {
   458→                node.set((x0 + b.size.w * v.cellW) / 2, goal.y, v.corridorZ);
   459→                a.path = [];
   460→              } else {
   461→                node.copyFrom(goal);
   462→                a.path = [];
   463→              }
   464→            }
   465→            // ── Locomoción interpolada (nada de teletransportes): el destino del sim se recorre a
   466→            // velocidad de crucero del catálogo, tramo a tramo. El sim adelanta a los vecinos a
   467→            // saltos del grafo (varias plantas de golpe): la ruta se RECALCULA DESDE EL PUNTO
   468→            // ACTUAL (andar al pozo → subir/bajar → salir) y el viaje se ve entero, pieza a pieza.
   469→            // Solo un estado realmente imposible (cambio de planta sin pozo utilizable, v1.10)
   470→            // recoloca sin teatro. Con ruta en curso se consume SIEMPRE hasta el final.
   471→            if (a.path.length === 0) {
   472→              // STOW (v1.18): llegó a la puerta de su trastero — entra y se oculta (la celda
   473→              // es cerrada: el interior jamás se ve). enabledPrev=false para que, cuando el
   474→              // sim lo mude, la reaparición lo coloque donde toque.
   475→              if (storeDoor && a.stowedBlock === null && Math.hypot(node.x - storeDoor.x, node.z - storeDoor.z) < 0.35) {
   476→                a.stowedBlock = b.id;
   477→                a.node.setEnabled(false);
   478→                a.enabledPrev = false;
   479→                a.path = [];
   480→                continue;
   481→              }
   482→              const routed = buildRoute(a, { x: goal.x, y: goal.y, z: goal.z, inside: geo && localTarget ? { block: b, local: localTarget } : null });
   483→              if (!routed) {
   484→                node.copyFrom(goal);
   485→                a.path = [];
   486→              }
   487→            }
   488→          }
   489→          if (a.path.length > 0) {
   490→            const w = a.path[0]!;
   491→            const wdh = Math.hypot(w.x - node.x, w.z - node.z);
   492→            const wdy = w.y - node.y;
   493→            if (Math.abs(wdy) > 0.02 && wdh < 0.3) {
   494→              // Tramo VERTICAL en el pozo: ritmo del catálogo (ascensor operativo → cabina).
   495→              const shaftHere = shafts
   496→                .filter((s) => s.cell.floor === b.cell.floor && Math.abs(shaftXOf(s) - node.x) < 0.3)
   497→                .sort((s1, s2) => Math.abs(shaftXOf(s1) - node.x) - Math.abs(shaftXOf(s2) - node.x))[0];
   498→              const feat = shaftHere ? content.blocks[shaftHere.defId]!.features : undefined;
   499→              const rate = feat?.elevator && shaftHere!.params.elevator_status === 'ok' ? v.cellH / v.elevatorSeconds : v.cellH / v.stairsSeconds;
   500→              node.y += Math.sign(wdy) * Math.min(Math.abs(wdy), rate * dt);
   501→            } else if (wdh > 0.001) {
   502→              // Tramo con componente horizontal: crucero con llegada suave (frena al acercarse,
   503→              // nunca sobrepasa). Con dy el tramo es 3D (pendiente de escalera, Task 29): el paso
   504→              // reparte X/Z/Y con la MISMA velocidad de crucero, así la subida es física.
   505→              const step = Math.min(npcSpeed * dt, (0.35 + wdh * 2.5) * dt, wdh);
   506→              const t = step / wdh;
   507→              node.x += (w.x - node.x) * t;
   508→              node.z += (w.z - node.z) * t;
   509→              node.y += wdy * t;
   510→            }
   511→            // Tramo terminado → siguiente tramo de la ruta.
   512→            if (Math.hypot(w.x - node.x, w.z - node.z) < 0.06 && Math.abs(w.y - node.y) < 0.03) a.path.shift();
   513→          }
   514→        }
   515→
   516→        // ── Rotación 2.5D (v1.2): el actor MIRA hacia donde camina. El pasillo es lateral (eje
   517→        // X, perfil); entrar/salir de una vivienda es profundidad (eje Z: de espaldas al alejarse,
   518→        // de frente al acercarse). Giro por el arco corto con el ritmo del catálogo
   519→        // (rules.view.actorTurnSpeed). El rumbo queda PENDIENTE hasta rematarse: aunque el actor
   520→        // se pare a mitad de giro, termina de encarar la dirección en la que iba (nadie se queda
   521→        // torcido en una puerta).
   522→        // Aplicación ROBUSTA (v1.3): si el nodo lleva rotationQuaternion (el loader glTF puede
   523→        // ponérselo), Babylon IGNORA rotation.y — hay que escribir el cuaternión; si no,
   524→        // rotation.y de toda la vida. En ambos casos SOLO yaw: nadie se ladea ni se tumba.
   525→        const mdx = node.x - prevX;
   526→        const mdz = node.z - prevZ;
   527→        if (Math.hypot(mdx, mdz) / Math.max(1e-4, dt) > 0.25) a.targetYaw = Math.atan2(-mdz, mdx);
   528→        if (a.targetYaw !== null) {
   529→          let dYaw = a.targetYaw - a.yaw;
   530→          dYaw = ((dYaw + Math.PI) % (2 * Math.PI) + 2 * Math.PI) % (2 * Math.PI) - Math.PI;
   531→          if (Math.abs(dYaw) < 0.02) {
   532→            a.yaw = a.targetYaw; // rematado
   533→            a.targetYaw = null;
   534→          } else {
   535→            a.yaw += dYaw * Math.min(1, dt * v.actorTurnSpeed);
   536→          }
   537→        }
   538→        const q = a.node.rotationQuaternion;
   539→        if (q) Quaternion.RotationYawPitchRollToRef(a.yaw, 0, 0, q);
   540→        else a.node.rotation.y = a.yaw;
   541→
   542→        // ── Animación (solo rigs) por prioridad de estados (v1.2) + SINCRONÍA DE ZANCADA (v1.3):
   543→        //   gesto explícito → run (speed > runSpeed) → walk (mientras se avanza) → clip de
   544→        //   actividad parado → sentado en ancla → idle. El ratio de playback sale de la
   545→        //   velocidad PLANA real (m/s en el suelo; la Y no mueve piernas) contra la velocidad
   546→        //   natural del clip (rules.view.animSync): las piernas acompañan al cuerpo, cero patinaje.
   547→        if (a.gestureT > 0) {
   548→          a.gestureT -= dt;
   549→          if (a.gestureT <= 0) a.gesture = null;
   550→        }
   551→        if (a.rig) {
   552→          const speedH = Math.hypot(node.x - prevX, node.z - prevZ) / Math.max(1e-4, dt);
   553→          const def = content.assets?.actors?.[r.archetype];
   554→          const activityClip = def?.activityAnimations?.[r.activity ?? ''] ?? null;
   555→          const walkClip = def?.walkAnimation ?? 'walk';
   556→          let clip: string | null;
   557→          if (a.gesture) clip = a.gesture;
   558→          else if (speedH > v.runSpeed) clip = 'run';
   559→          else if (speedH > 0.25) clip = walkClip;
   560→          else if (activityClip) clip = activityClip;
   561→          else if (a.seated && a.path.length === 0) clip = 'sit';
   562→          else clip = null;
   563→          const sync = v.animSync;
   564→          let rate = 1;
   565→          if (!a.gesture && clip === 'run') rate = speedH / sync.runRefSpeed;
   566→          else if (!a.gesture && clip === walkClip) rate = speedH / sync.walkRefSpeed;
   567→          rate = Math.min(sync.maxRate, Math.max(sync.minRate, rate));
   568→          a.rig.play(clip, rate);
   569→        }
   570→      }
   571→    },
   572→    /** Dirección de cutscenes (Fase D): toma/coloca/ruta/suelta actores como teatro. */
   573→    director: {
   574→      nodeOf(id: string): { x: number; y: number; z: number } | null {
   575→        const a = actors.get(id);
   576→        return a && a.node.isEnabled() ? { x: a.node.position.x, y: a.node.position.y, z: a.node.position.z } : null;
   577→      },
   578→      take(id: string): boolean {
   579→        const r = world.residents[id];
   580→        if (!r || !r.present || r.location === null) return false;
   581→        if (!actors.has(id)) ensureActor(id, r.archetype);
   582→        const a = actors.get(id)!;
   583→        a.directed = true;
   584→        a.seated = false;
   585→        a.node.setEnabled(true);
   586→        return true;
   587→      },
   588→      release(id: string): void {
   589→        const a = actors.get(id);
   590→        if (!a) return;
   591→        a.directed = false;
   592→        a.path = [];
   593→        a.targetLocal = null;
   594→        a.wanderT = 0;
   595→        a.seated = false;
   596→        a.gesture = null;
   597→        a.gestureT = 0;
   598→        a.targetYaw = null;
   599→        // Si estaba esperando el ascensor, se cancela la llamada (el director libera el hueco).
   600→        if (a.elevatorReq && elevators) {
   601→          elevators.cancel(a.id);
   602→          a.elevatorReq = null;
   603→          a.ridePhase = null;
   604→        }
   605→      },
   606→      stageAt(id: string, x: number, floor: number, z: number): void {
   607→        const a = actors.get(id);
   608→        if (!a) return;
   609→        a.path = [];
   610→        // El teatro manda: cualquier viaje en curso se cancela antes de colocar al actor.
   611→        if (a.elevatorReq && elevators) {
   612→          elevators.cancel(a.id);
   613→          a.elevatorReq = null;
   614→          a.ridePhase = null;
   615→        }
   616→        a.node.position.set(x, floor * v.cellH + 0.14 + a.pivotY, z);
   617→      },
   618→      walkTo(id: string, dest: { x: number; z: number; floor: number; inside?: { block: string; local: Pt } }): boolean {
   619→        const a = actors.get(id);
   620→        if (!a) return false;
   621→        const b = dest.inside ? world.blocks[dest.inside.block] : undefined;
   622→        const inside = b && dest.inside
   623→          ? {
   624→              block: b,
   625→              local: {
   626→                x: Math.min(b.size.w * v.cellW - 0.2, Math.max(0.2, dest.inside.local.x)),
   627→                z: Math.min(depthOfBlock(b) - 0.2, Math.max(0.2, dest.inside.local.z)),
   628→              },
   629→            }
   630→          : null;
   631→        buildRoute(a, { x: dest.x, y: dest.floor * v.cellH + 0.14 + a.pivotY, z: dest.z, inside });
   632→        return true;
   633→      },
   634→      idle(id: string): boolean {
   635→        const a = actors.get(id);
   636→        return !a || (a.path.length === 0 && !a.elevatorReq);
   637→      },
   638→      headOf(id: string): { x: number; y: number; z: number } | null {
   639→        const a = actors.get(id);
   640→        if (!a || !a.node.isEnabled()) return null;
   641→        const p = a.node.position;
   642→        return { x: p.x, y: p.y + a.height + 0.18, z: p.z };
   643→      },
   644→      gestureOf(id: string): string | null {
   645→        return actors.get(id)?.gesture ?? null;
   646→      },
   647→      isDirected(id: string): boolean {
   648→        return actors.get(id)?.directed ?? false;
   649→      },
   650→      setGesture(id: string, state: string | null, seconds = 0): void {
   651→        const a = actors.get(id);
   652→        if (!a) return;
   653→        a.gesture = state;
   654→        a.gestureT = state && seconds > 0 ? seconds : 0;
   655→      },
   656→    },
   657→  };
   658→}
