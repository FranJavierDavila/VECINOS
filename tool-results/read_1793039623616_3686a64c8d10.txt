    83→    83→export function createBuilding(scene: Scene, world: World, content: Content) {
    84→    84→  const { Color3, Color4, MeshBuilder, StandardMaterial, TransformNode } = B();
    85→    85→  const v = content.rules.view;
    86→    86→  const glow = Object.fromEntries(Object.entries(v.channelGlow).map(([k, hex]) => [k, Color3.FromHexString(hex)])) as Record<string, Color3>;
    87→    87→  const views = new Map<string, BlockView>();
    88→    88→  const shakes = new Map<string, number>();
    89→    89→  /** Raíz de escena por bloque (v1.19): colgar aquí TODA malla del bloque permite el
    90→    90→   *  desvanecimiento completo del bloque principal en el apilamiento 3D (swap de bloques). */
    91→    91→  const roots = new Map<string, TransformNode>();
    92→    92→  const attach = (blockId: string, mesh: Mesh): Mesh => {
    93→    93→    const r = roots.get(blockId);
    94→    94→    if (r) mesh.parent = r;
    95→    95→    return mesh;
    96→    96→  };
    97→    97→  /** Interiores 2D (Fase C) por bloque: suelos de estancia y tapas de puertas cerradas. */
    98→    98→  const interiors = new Map<string, InteriorView>();
    99→    99→  /** Foco del LOD (posición del jugador): lo alimenta update() y lo consume paint(). */
   100→   100→  let lastFocus: { x: number; floor: number } | null = null;
   101→   101→  // Plano frontal de las estancias: la cara de la puerta. Con viviendas del doble de calado
   102→   102→  // (v0.6-b), el extra crece hacia el FONDO (+Z) para no envolver el pasillo ni al núcleo.
   103→   103→  // v1.19: los calados del catálogo YA incluyen el +1 módulo (rules.view.depthModule) —
   104→   104→  // depth 4.2 en viviendas y commonDepth 2.8 en comunes son los valores finales.
   105→   105→  const frontZ = v.roomZ - v.commonDepth / 2;
   106→   106→  /** Calado por bloque: viviendas 2× (mobiliario interior); naves del cierre (v1.16) y el
   107→   107→   *  tejado (con aleros) a calado de vivienda; bloques comunes, el de siempre. */
   108→   108→  const depthOf = (b: BlockInstance): number => {
   109→   109→    if (b.kind === 'dwelling') return v.depth;
   110→   110→    if (b.defId === 'trasteros_planta' || b.defId === 'bajocubierta') return v.depth;
   111→   111→    if (b.defId === v.trasteros.blockDefId) return v.depth * v.trasteros.blockDepthFactor;
   112→   112→    if (b.defId === 'tejado') return v.depth + v.roof.overhang * 2;
   113→   113→    return v.commonDepth;
   114→   114→  };
   115→   115→  const mat = (name: string) => new StandardMaterial(name, scene);
   116→   116→  /** Texturas de puerta cacheadas por fichero (v1.24→v1.25): FULL-BLEED de verdad. El arte
   117→   117→   *  llega con pared/suelo cocinados alrededor de la hoja: flood-fill contra los colores de
   118→   118→   *  las esquinas (stripCorners) + recorte a la caja del contenido (alphaBoundsOf) → la
   119→   119→   *  textura ES la hoja y desaparecen los elementos antiguos que se veían alrededor. */
   120→   120→  const doorArtCache = new Map<string, DynamicTexture>();
   121→   121→  const doorArtTex = (sc: Scene, file: string): DynamicTexture => {
   122→   122→    const hit = doorArtCache.get(file);
   123→   123→    if (hit) return hit;
   124→   124→    const dt = new (B().DynamicTexture)(`puerta_tex_${doorArtCache.size}`, { width: 384, height: 768 }, sc, true);
   125→   125→    doorArtCache.set(file, dt);
   126→   126→    const img = new Image();
   127→   127→    img.onload = () => {
   128→   128→      const big = document.createElement('canvas');
   129→   129→      big.width = img.naturalWidth;
   130→   130→      big.height = img.naturalHeight;
   131→   131→      const bx = big.getContext('2d', { willReadFrequently: true });
   132→   132→      if (!bx) return;
   133→   133→      bx.drawImage(img, 0, 0);
   134→   134→      stripCorners(big, 0.14);
   135→   135→      const b = alphaBoundsOf(big);
   136→   136→      const dctx = dt.getContext() as unknown as CanvasRenderingContext2D;
   137→   137→      dctx.clearRect(0, 0, 384, 768);
   138→   138→      if (b) dctx.drawImage(big, b.x0, b.y0, b.x1 - b.x0, b.y1 - b.y0, 0, 0, 384, 768);
   139→   139→      else dctx.drawImage(img, 0, 0, img.naturalWidth, img.naturalHeight, 0, 0, 384, 768);
   140→   140→      dt.update(true);
   141→   141→    };
   142→   142→    img.src = file;
   143→   143→    return dt;
   144→   144→  };
   145→   145→  /** Candado DIBUJADO (v1.25): silueta real de candado en un DynamicTexture — nada de cajas
   146→   146→   *  3D. Colores del catálogo (lockColor); grillete de acero fijo. MISMO PATRÓN que los
   147→   147→   *  sprites del diorama: se recorta el quad a la caja del contenido (alphaBounds) — el
   148→   148→   *  canvas subido no puede confiar en el alpha fuera del contenido (fondo a negro). */
   149→   149→  let lockTex: DynamicTexture | null = null;
   150→   150→  let lockTexAr = 0.8;
   151→   151→  const padlockTex = (): DynamicTexture => {
   152→   152→    if (lockTex) return lockTex;
   153→   153→    const dt = new (B().DynamicTexture)('candado_tex', { width: 128, height: 128 }, scene, true);
   154→   154→    dt.hasAlpha = true;
   155→   155→    const c = document.createElement('canvas');
   156→   156→    c.width = 128;
   157→   157→    c.height = 128;
   158→   158→    const ctx = c.getContext('2d', { willReadFrequently: true });
   159→   159→    if (!ctx) return dt;
   160→   160→    ctx.clearRect(0, 0, 128, 128);
   161→   161→    const body = Color3.FromHexString(v.lockColor).toHexString();
   162→   162→    // Grillete: arco de acero.
   163→   163→    ctx.strokeStyle = '#8d939c';
   164→   164→    ctx.lineWidth = 11;
   165→   165→    ctx.beginPath();
   166→   166→    ctx.arc(64, 52, 26, Math.PI, 0);
   167→   167→    ctx.stroke();
   168→   168→    // Cuerpo de latón con borde.
   169→   169→    ctx.fillStyle = body;
   170→   170→    ctx.strokeStyle = 'rgba(60,40,10,0.9)';
   171→   171→    ctx.lineWidth = 4;
   172→   172→    const r = 12;
   173→   173→    ctx.beginPath();
   174→   174→    ctx.moveTo(24, 124);
   175→   175→    ctx.lineTo(24, 66 + r);
   176→   176→    ctx.arcTo(24, 66, 24 + r, 66, r);
   177→   177→    ctx.lineTo(104 - r, 66);
   178→   178→    ctx.arcTo(104, 66, 104, 66 + r, r);
   179→   179→    ctx.lineTo(104, 124);
   180→   180→    ctx.closePath();
   181→   181→    ctx.fill();
   182→   182→    ctx.stroke();
   183→   183→    // Ojo de la cerradura.
   184→   184→    ctx.fillStyle = 'rgba(50,32,8,0.95)';
   185→   185→    ctx.beginPath();
   186→   186→    ctx.arc(64, 90, 8, 0, Math.PI * 2);
   187→   187→    ctx.fill();
   188→   188→    ctx.fillRect(60, 90, 8, 20);
   189→   189→    // Recorte a la caja del contenido (patrón diorama) + volcado a la textura cuadrada.
   190→   190→    const b = alphaBoundsOf(c);
   191→   191→    const dctx = dt.getContext() as unknown as CanvasRenderingContext2D;
   192→   192→    dctx.clearRect(0, 0, 128, 128);
   193→   193→    if (b) {
   194→   194→      const bw = Math.max(1, b.x1 - b.x0);
   195→   195→      const bh = Math.max(1, b.y1 - b.y0);
   196→   196→      lockTexAr = bw / bh;
   197→   197→      dctx.drawImage(c, b.x0, b.y0, bw, bh, 0, 0, 128, 128);
   198→   198→    } else {
   199→   199→      dctx.drawImage(c, 0, 0);
   200→   200→    }
   201→   201→    dt.update(true);
   202→   202→    lockTex = dt;
   203→   203→    return dt;
   204→   204→  };
   205→   205→  /** Apliques de las puertas de vivienda: el foco cálido amaanece con la noche (update). */
   206→   206→  const doorLampMats: { mat: StandardMaterial; hex: string }[] = [];
   207→   207→  /** Dioramas interiores (v1.23): null sin catálogo interiors.json — greybox como siempre. */
   208→   208→  const diro: DioramaDirector | null = createDioramaDirector(scene, content);
   209→   209→  const box = (name: string, w: number, h: number, d: number, x: number, y: number, z: number, m: StandardMaterial) => {
   210→   210→    const mesh = MeshBuilder.CreateBox(name, { width: w, height: h, depth: d }, scene);
   211→   211→    mesh.position.set(x, y, z);
   212→   212→    mesh.material = m;
   213→   213→    return mesh;
   214→   214→  };
   215→   215→
   216→   216→  for (const b of Object.values(world.blocks)) {
   217→   217→    const m = mat(`${b.id}_mat`);
   218→   218→    const cx = (b.cell.col + b.size.w / 2) * v.cellW;
   219→   219→    const bottom = b.cell.floor * v.cellH;
   220→   220→    const w = b.size.w * v.cellW - v.gapW;
   221→   221→    const h = v.cellH - v.gapH;
   222→   222→    const d = depthOf(b);
   223→   223→    const cz = frontZ + d / 2;
   224→   224→    const view: BlockView = {
   225→   225→      greybox: [],
   226→   226→      mats: [m],
   227→   227→      baseEmissive: [m.emissiveColor.clone()],
   228→   228→      baseColor: [Color3.FromHexString(content.blocks[b.defId]!.view?.color ?? '#cccccc')],
   229→   229→      tintStrength: 1,
   230→   230→      model: null,
   231→   231→      front: null,
   232→   232→      door: null,
   233→   233→      handle: null,
   234→   234→      lock: null,
   235→   235→      lockBase: { x: cx, y: bottom + v.cellH / 2 - 0.2 },
   236→   236→      centerX: cx,
   237→   237→      floor: b.cell.floor,
   238→   238→      edgesMesh: null,
   239→   239→      edgesMesh2: null,
   240→   240→      glowForced: false,
   241→   241→      lodHiddenPanels: false,
   242→   242→      doorDecor: [],
   243→   243→      fixedDiffuse: null,
   244→   244→    };
   245→   245→    const root = new TransformNode(`${b.id}_root`, scene);
   246→   246→    roots.set(b.id, root);
   247→   247→    if (b.layer === 'corridor') {
   248→   248→      // Pasillo exterior (data-driven): tira transitable saliente hacia la calle, con el borde
   249→   249→      // trasero a ras de fachada (frontZ) y barandilla baja en el borde exterior — sitio de
   250→   250→      // sobra para que los personajes se crucen o charlen sin bailar el vacío.
   251→   251→      const d = v.corridorDepth;
   252→   252→      view.greybox.push(box(b.id, b.size.w * v.cellW, v.slab * 2, d, cx, bottom + v.slab, frontZ - d / 2, m));
   253→   253→      if (v.corridorRail > 0) {
   254→   254→        view.greybox.push(
   255→   255→          box(`${b.id}_barandilla`, b.size.w * v.cellW, v.corridorRail, v.wall, cx, bottom + v.slab * 2 + v.corridorRail / 2, frontZ - d, m),
   256→   256→        );
   257→   257→      }
   258→   258→    } else if (b.kind === 'dwelling') {
   259→   259→      const y = bottom + v.cellH / 2;
   260→   260→      const pv = v.privacy;
   261→   261→      // v1.25 — acabado de la vivienda: pared GRIS CLARITO fija (el tintado de estado no la
   262→   262→      // repinta; el glow de canales sigue) + ZÓCALO gris de 25 cm desde el suelo + marcos de
   263→   263→      // puerta DEL COLOR del zócalo. Todo del catálogo (rules.view.privacy).
   264→   264→      view.fixedDiffuse = Color3.FromHexString(pv.wallColor ?? '#dbd8d2');
   265→   265→      // ── Puerta a sala TRASERA (v1.19, apilamiento 3D): la pared del fondo se parte con el
   266→   266→      // hueco del layout (doorway hacia una estancia más profunda que el bloque principal).
   267→   267→      const layoutId0 = content.blocks[b.defId]?.container?.layout;
   268→   268→      const layout0 = layoutId0 ? content.layouts[layoutId0] : undefined;
   269→   269→      const rearKeys = new Set((layout0?.rooms ?? []).filter((r) => r.rect && r.rect.z + r.rect.d > v.depth + 0.01).map((r) => r.key));
   270→   270→      const rearDoor = (layout0?.doorways ?? []).find((dw) => dw.axis === 'x' && rearKeys.has(dw.b));
   271→   271→      const x0w = cx - w / 2;
   272→   272→      const x1w = cx + w / 2;
   273→   273→      if (rearDoor) {
   274→   274→        const gapA = Math.max(x0w, x0w + rearDoor.at - rearDoor.width / 2);
   275→   275→        const gapB = Math.min(x1w, x0w + rearDoor.at + rearDoor.width / 2);
   276→   276→        const doorH = Math.min(h - 0.15, 1.95);
   277→   277→        const lintelH = h - doorH;
   278→   278→        view.edgesMesh = box(`${b.id}_fondo_i`, gapA - x0w, h, v.wall, (x0w + gapA) / 2, y, frontZ + d, m);
   279→   279→        view.edgesMesh2 = box(`${b.id}_fondo_d`, x1w - gapB, h, v.wall, (gapB + x1w) / 2, y, frontZ + d, m);
   280→   280→        view.greybox.push(
   281→   281→          view.edgesMesh,
   282→   282→          view.edgesMesh2,
   283→   283→          box(`${b.id}_fondo_dintel`, gapB - gapA, lintelH, v.wall, (gapA + gapB) / 2, y + h / 2 - lintelH / 2, frontZ + d, m),
   284→   284→          box(`${b.id}_izq`, v.wall, h, d, cx - w / 2, y, cz, m),
   285→   285→          box(`${b.id}_der`, v.wall, h, d, cx + w / 2, y, cz, m),
   286→   286→          box(`${b.id}_suelo`, w, v.wall, d, cx, bottom + v.slab + v.wall / 2, cz, m),
   287→   287→          box(`${b.id}_techo`, w, v.wall, d, cx, bottom + v.cellH - v.slab - v.wall / 2, cz, m),
   288→   288→        );
   289→   289→      } else {
   290→   290→        view.edgesMesh = box(`${b.id}_fondo`, w, h, v.wall, cx, y, frontZ + d, m);
   291→   291→        view.greybox.push(
   292→   292→          view.edgesMesh,
   293→   293→          box(`${b.id}_izq`, v.wall, h, d, cx - w / 2, y, cz, m),
   294→   294→          box(`${b.id}_der`, v.wall, h, d, cx + w / 2, y, cz, m),
   295→   295→          box(`${b.id}_suelo`, w, v.wall, d, cx, bottom + v.slab + v.wall / 2, cz, m),
   296→   296→          box(`${b.id}_techo`, w, v.wall, d, cx, bottom + v.cellH - v.slab - v.wall / 2, cz, m),
   297→   297→        );
   298→   298→      }
   299→   299→      view.front = box(`${b.id}_puerta`, w, h, v.wall, cx, y, frontZ, m);
   300→   300→      // ── ZÓCALO (v1.25): banda gris de ~25 cm desde el suelo. Continúa en el MARCO de la
   301→   301→      //  puerta (mismo material): la pieza que perimetra la puerta es del color del zócalo.
   302→   302→      const skH = pv.skirtingH ?? 0.25;
   303→   303→      const skMat = mat(`${b.id}_zocalo_mat`);
   304→   304→      skMat.diffuseColor = Color3.FromHexString(pv.skirtingColor ?? '#a3a19b');
   305→   305→      skMat.specularColor = new Color3(0.02, 0.02, 0.02);
   306→   306→      const ySk = bottom + v.slab + v.wall + skH / 2;
   307→   307→      view.greybox.push(
   308→   308→        attach(b.id, box(`${b.id}_zocalo_frente`, w, skH, v.wall + 0.02, cx, ySk, frontZ, skMat)),
   309→   309→        attach(b.id, box(`${b.id}_zocalo_izq`, v.wall + 0.02, skH, d - 2 * v.wall, cx - w / 2 + v.wall / 2, ySk, cz, skMat)),
   310→   310→        attach(b.id, box(`${b.id}_zocalo_der`, v.wall + 0.02, skH, d - 2 * v.wall, cx + w / 2 - v.wall / 2, ySk, cz, skMat)),
   311→   311→      );
   312→   312→      // Candado DIBUJADO (v1.25): quad con silueta real de candado — los candados en forma
   313→   313→      // de caja desaparecen. Cuelga de la puerta de privacidad y sigue sacudiéndose.
   314→   314→      const lh = pv.lockH ?? 0.28;
   315→   315→      const lw = lh * Math.max(0.5, Math.min(1.2, lockTexAr));
