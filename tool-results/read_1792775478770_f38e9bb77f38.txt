    62→  async init(canvas: HTMLCanvasElement): Promise<void> {
    63→    if (this.engine || this.initializing) return;
    64→    this.initializing = true;
    65→    this.canvas = canvas;
    66→
    67→    try {
    68→      // 1) Contenido externo + validación (los errores de referencias se avisan, no abortan).
    69→      const { content, issues } = await loadContent();
    70→      this._content = content;
    71→      preloadSounds();
    72→      useGameStore.setState({ touchHint: content.rules.touch.hint });
    73→      if (issues.length > 0) {
    74→        useGameStore.getState().pushToast(`Contenido inválido: ${issues.length} errores de validación`, 'bad');
    75→      }
    76→
    77→      // 2) Motor de render (WebGPU con fallback WebGL2).
    78→      const { engine, backend } = await createEngine(canvas);
    79→      this.engine = engine;
    80→      window.addEventListener('resize', this.onResize);
    81→      await this.restart();
    82→      useGameStore.getState().markReady(backend);
    83→      engine.runRenderLoop(this.frame);
    84→      // Gancho de depuración/pruebas E2E (no lo usa la UI).
    85→      (window as unknown as { __vecinos?: unknown }).__vecinos = { store: useGameStore, controller: this };
    86→    } catch (err) {
    87→      console.error('No se pudo arrancar el juego (contenido o motor de render)', err);
    88→      useGameStore.getState().setEngineError(true);
    89→    }
    90→  }

   281→  private connectNet(): void {
   282→    this.net = new NetSession({
   283→      onState: (s) => this.applyRemoteState(s),
   284→      onParticipants: (p) => useGameStore.getState().setMpParticipants(p),
   285→      onJoined: (j) => {
   286→        this.myResidentId = j.yourResidentId;
   287→        this.isHost = j.isHost;
   288→        useGameStore.getState().setMpSession({ code: j.code, isHost: j.isHost, myResidentId: j.yourResidentId });
   289→        useGameStore.getState().setMpParticipants(j.participants);
   290→      },
   291→      onError: (m) => {
   292→        useGameStore.getState().pushToast(m, 'bad');
   293→      },
   294→      onStatus: (st) => {
   295→        if (!this.net) return;
   296→        if (st === 'connected' && useGameStore.getState().mpStatus === 'connecting') return; // el ack decide
   297→        if (st === 'disconnected' && useGameStore.getState().mpStatus === 'connected') {
   298→          useGameStore.getState().setMpStatus('error');
   299→          useGameStore.getState().setMpError('Conexión perdida con el servidor de la comunidad.');
   300→        }
   301→      },
   302→    });
   303→  }

   316→  /** Adopta el estado del servidor (autoridad). Primera vez: reinicia la escena con ese mundo. */
   317→  private applyRemoteState(s: SessionStatePayload): void {
   318→    if (s.seq <= this.appliedSeq) return;
   319→    this.appliedSeq = s.seq;
   320→    if (!this.world || this.world.scenarioId !== s.world.scenarioId) {
   321→      void this.restart(s.world).then(() => {
   322→        if (!this.world) return;
   323→        this.world.events = s.events.slice();
   324→        this.handleEvents();
   325→        this.reconcile();
   326→        this.syncHud();
   327→      });
   328→      return;
   329→    }
   330→    // Aplica en sitio conservando la identidad del objeto (PlayerLogic/building/actors referencian el mismo World).
   331→    Object.assign(this.world, s.world);
   332→    this.world.events = s.events.slice();
   333→    this.handleEvents();
   334→    this.paintBuilding();
   335→    this.reconcile();
   336→    this.syncHud();
   337→  }

   339→  /** Corrige desvíos de posición del avatar local respecto al mundo del servidor (autoridad). */
   340→  private reconcile(): void {
   341→    if (!this.logic || !this.world || !this.net) return;
   342→    const me = this.world.residents[this.actorId()];
   343→    if (!me) return;
   344→    const pos = this.logic.position();
   345→    const locBlock = me.location ? this.world.blocks[me.location] : null;
   346→    if (locBlock) {
   347→      const insideOk = pos.mode === 'inside' && pos.insideBlock === locBlock.id;
   348→      const corridorOk = pos.mode === 'corridor' && !pos.insideBlock;
   349→      if (!(locBlock.kind === 'dwelling' ? insideOk : corridorOk) || Math.abs(pos.floor - locBlock.cell.floor) > 0.01) {
   350→        this.logic.forceLocation(locBlock.id);
   351→      }
   352→    } else if (pos.mode === 'inside') {
   353→      this.logic.forceLocation(null);
   354→    }
   355→  }

   437→  async save(): Promise<void> {
   438→    if (!this.world) return;
   439→    const store = useGameStore.getState();
   440→    if (this.net) {
   441→      store.pushToast(this.content.rules.multiplayer.texts.remoteSaveDisabled, 'info');
   442→      return;
   443→    }
   444→    store.setSaveState('saving');
   445→    drainEvents(this.world); // la cola de eventos no entra en el guardado
   446→    try {
   447→      const res = await fetch('/api/save', {
   448→        method: 'POST',
   449→        headers: { 'Content-Type': 'application/json' },
   450→        body: JSON.stringify({ slot: 'auto', state: this.world }),
   451→      });
   452→      if (!res.ok) throw new Error(String(res.status));
   453→      store.setSaveState('saved', new Date().toLocaleTimeString('es-ES', { hour: '2-digit', minute: '2-digit' }));
   454→      store.pushToast('Partida guardada. El acta queda en el tablón (y en SQLite).', 'good');
   455→    } catch {
   456→      store.setSaveState('error');
   457→      store.pushToast('No se pudo guardar la partida. ¿El administrador se habrá ido a comer?', 'bad');
   458→    }
   459→  }

   461→  async load(): Promise<void> {
   462→    const store = useGameStore.getState();
   463→    if (this.net) {
   464→      store.pushToast(this.content.rules.multiplayer.texts.remoteSaveDisabled, 'info');
   465→      return;
   466→    }
   467→    try {
   468→      const res = await fetch('/api/save?slot=auto');
   469→      const json = (await res.json()) as { state?: unknown };
   470→      if (!json.state || typeof json.state !== 'object') {
   471→        store.pushToast('Todavía no hay ninguna partida guardada.', 'info');
   472→        return;
   473→      }
   474→      if (!this._content) {
   475→        store.pushToast('El contenido todavía se está cargando. Espera un momento.', 'info');
   476→        return;
   477→      }
   478→      await this.restart(json.state as World);
   479→      store.pushToast('Partida cargada. El vecindario, exactamente como lo dejaste.', 'good');
   480→    } catch {
   481→      store.pushToast('No se pudo cargar la partida.', 'bad');
   482→    }
   483→  }

   496→  /** (Re)crea el mundo y la escena. Con `state` restaura una partida guardada. */
   497→  private async restart(state?: World): Promise<void> {
   498→    if (!this.engine || !this.canvas || !this._content) return;
   499→    const { Color4, DirectionalLight, HemisphericLight, Scene, UniversalCamera, Vector3 } = B();
   500→    const view = this.content.rules.view;
   501→    useGameStore.getState().resetUi();
   502→
   503→    this.world = state ?? createWorld(this.content, this.activeScenarioId());
   504→    this.scenarioOverrides = this.content.scenarios[this.world.scenarioId ?? '']?.minigameOverrides ?? {};
   505→    // Guardados antiguos sin `entry`: la entrada es la zona común de planta baja.
   506→    if (!this.world.entry) {
   507→      this.world.entry = Object.values(this.world.blocks).find((b) => b.cell.floor === 0 && b.kind !== 'dwelling')?.id ?? Object.keys(this.world.blocks)[0]!;
   508→    }
   509→    // Migración de estancias de guardados v0.1: params y ranuras de muebles según catálogo.
   510→    for (const b of Object.values(this.world.blocks)) migrateRooms(this.content, b.rooms);
   511→    this.logic = new PlayerLogic(this.world, this.content, {
   512→      actorId: this.myResidentId ?? undefined,
   513→      send: (cmd) => this.sendCommand(cmd),
   514→    });
   515→
   516→    this.scene?.dispose();
   517→    this.scene = new Scene(this.engine);
   518→    this.scene.clearColor = new Color4(view.sky.r, view.sky.g, view.sky.b, 1);
   519→    new HemisphericLight('ambient', new Vector3(...view.ambient.dir), this.scene).intensity = view.ambient.intensity;
   520→    new DirectionalLight('sun', new Vector3(...view.sun.dir), this.scene).intensity = view.sun.intensity;
   521→
   522→    this.building = createBuilding(this.scene, this.world, this.content);
   523→    this.actors = createActors(this.scene, this.world, this.content);
   524→    const blocks = Object.values(this.world.blocks);
   525→    this.rig = new CameraRig(
   526→      new UniversalCamera('cam', new Vector3(0, 0, -30), this.scene),
   527→      this.engine,
   528→      Math.max(...blocks.map((b) => b.cell.col + b.size.w)),
   529→      Math.max(...blocks.map((b) => b.cell.floor)) + 1,
   530→      view.cellW,
   531→      view.cellH,
   532→    );
   533→    // Re-sincronización de vista (v0.6-b): la cámara sigue la posición del mundo (restaurado o
   534→    // nuevo), no la preferencia anterior. Dentro de una vivienda → estancia; en pasillo, la
   535→    // preferencia salvo que fuera 'estancia' (ahí estás en el pasillo, no en una estancia).
   536→    const pos = this.logic.position();
   537→    const store = useGameStore.getState();
   538→    const wanted: CamMode = pos.mode === 'inside' ? 'estancia' : store.camMode === 'estancia' ? 'planta' : store.camMode;
   539→    this.rig.mode = wanted;
   540→    store.setCamMode(wanted);
   541→
   542→    this.clockAcc = 0;
   543→    this.keys.clear();
   544→    this.phoneLines = [];
   545→    this.phoneMission = '';
   546→    this.lastHudSync = 0;
   547→    this.pump();
   548→  }

   554→  // ───────────────────────── Bucle de render ─────────────────────────
   555→
   556→  private frame = (): void => {
   557→    if (!this.engine || !this.scene || !this.world || !this.logic || !this.building || !this.actors || !this.rig) return;
   558→    const dt = Math.min(0.1, this.engine.getDeltaTime() / 1000);
   559→    const store = useGameStore.getState();
   560→    const frozen = selectModalOpen(store) || store.paused;
   561→
   562→    let ticked = false;
   563→    if (!frozen) {
   564→      this.logic.update(dt, this.readInput());
   565→      // En remoto el servidor es la autoridad del reloj: nada de ticks locales.
   566→      if (!this.net) {
   567→        this.clockAcc += dt * store.speed;
   568→        let guard = 0;
   569→        while (this.clockAcc >= 1 && guard < 60) {
   570→          tick(this.world, this.content);
   571→          this.clockAcc -= 1;
   572→          ticked = true;
   573→          guard += 1;
   574→        }
   575→      }
   576→    }
   577→
   578→    if (this.world.events.length > 0) this.handleEvents();
   579→    if (ticked) this.paintBuilding();
   580→
   581→    const now = performance.now();
   582→    if (now - this.lastHudSync > 200) {
   583→      this.lastHudSync = now;
   584→      this.syncHud();
   585→    }
   586→
   587→    const pos = this.logic.position();
   588→    this.actors.update(dt, pos, this.actorId());
   589→    // LOD por distancia: solo con cámaras cercanas (planta/estancia); el plano general (edificio) muestra todo.
   590→    this.building.update(dt, useGameStore.getState().camMode === 'edificio' ? undefined : { x: pos.x, floor: pos.floor });
   591→    this.rig.update(dt, pos);
   592→    this.scene.render();
   593→  };

   624→  private handleEvents(): void {
   625→    if (!this.world) return;
   626→    const store = useGameStore.getState();
   627→    for (const e of drainEvents(this.world)) {
   628→      const d = e.data as Record<string, unknown>;
   629→      eventSound(e.type, typeof d.type === 'string' ? d.type : undefined);
   630→      switch (e.type) {
   631→        case 'step_started':
   632→          if (d.type === 'phone_call') this.openPhone(String(d.mission), String(d.dialog));
   633→          break;
   634→        case 'choice_required': {
   635→          const choices = (d.choices as { id: string; label: string; hint?: string | null }[] | undefined) ?? [];
   636→          store.setChoices({ mission: String(d.mission), title: String(d.title), choices: choices.map((c) => ({ id: c.id, label: c.label, hint: c.hint ?? null })) });
   637→          store.pushLog(`⚖️ Decisión pendiente: ${String(d.title)}`, 'info');
   638→          break;
   639→        }
   640→        case 'door_locked': {
   641→          this.building?.shake(String(d.block));
   642→          const price = typeof d.price === 'number' ? d.price : null;
   643→          store.pushToast(`${blockLabel(String(d.block))}: puerta con candado${d.reason === 'for_sale' && price !== null ? ` (a la venta por ${price} €)` : ''}`, 'info');
   644→          break;
   645→        }
   646→        case 'neighbor_arrived':
   647→          store.pushToast(`Nuevo vecino en ${blockLabel(String(d.block))}: ${this.nameOf(d.resident)}`, 'good');
   648→          store.pushLog(`${this.nameOf(d.resident)} se muda a ${blockLabel(String(d.block))}. El edificio ya no es lo que era. Era mejor.`, 'info');
   649→          break;
   650→        case 'mission_activated':
   651→          store.pushLog(`📋 Nueva misión: ${String(d.title)}`, 'info');
   652→          break;
   653→        case 'mission_completed':
   654→          store.pushToast(`Misión completada: ${this.titleOf(d.mission)}`, 'good');
   655→          store.pushLog(`✔ Misión completada: ${this.titleOf(d.mission)}`, 'good');
   656→          break;
   657→        case 'mission_failed':
   658→          store.pushToast(`Misión fallida: ${this.titleOf(d.mission)}`, 'bad');
   659→          store.pushLog(`✘ Misión fallida: ${this.titleOf(d.mission)}`, 'bad');
   660→          break;
   661→        case 'block_unlocked':
   662→          store.pushToast(`${blockLabel(String(d.block))} desbloqueado`, 'good');
   663→          break;
   664→        case 'block_bought':
   665→          store.pushLog(`🔑 ${blockLabel(String(d.block))} comprado por ${String(d.price)} €`, 'good');
   666→          break;
   667→        case 'block_for_sale':
   668→          store.pushLog(`🪧 Cartel de "SE VENDE" en ${blockLabel(String(d.block))} (${String(d.price)} €)`, 'info');
   669→          break;
   670→        case 'knock_result':
   671→          store.pushToast(`🗣️ ${String(d.name)}: «${String(d.line)}»`, d.asleep ? 'info' : 'good');
   672→          break;
   673→        case 'noise_complaint': // texto data-driven: lo pinta el uiLog del catálogo (case mantenido por compatibilidad de eventos antiguos)
   674→          if (d.uiLog === true) break;
   675→          store.pushLog(`🔊 Ruido por encima del límite en ${blockLabel(String(d.block))}. Alguien aprieta los dientes.`, 'bad');
   676→          break;
   677→        case 'leak_detected':
   678→          if (d.uiLog === true) break;
   679→          store.pushLog(`💧 Goteo detectado en ${blockLabel(String(d.block))}. El cubo del salón hace ojitos.`, 'bad');
   680→          break;
   681→        case 'furniture_placed':
   682→          store.pushLog(`${String(d.emoji ?? '📦')} ${String(d.name)} colocado en ${blockLabel(String(d.block))} (${String(d.room)}).`, 'info');
   683→          break;
   684→        case 'furniture_removed':
   685→          store.pushLog(`${String(d.emoji ?? '📦')} ${String(d.name)} retirado de ${blockLabel(String(d.block))} (+${String(d.refund)} €).`, 'info');
   686→          break;
   687→        case 'room_renovated':
   688→          store.pushLog(`🧱 Reforma terminada en ${blockLabel(String(d.block))}. Hay estancia nueva que amueblar.`, 'good');
   689→          break;
   690→        case 'weather_changed':
   691→          store.pushLog(`El cielo cambia: ${this.content.rules.labels.weather[String(d.state)] ?? String(d.state)}`, 'info');
   692→          break;
   693→        case 'rent_collected':
   694→          store.pushToast(`+${String(d.amount)} € de alquiler (${blockLabel(String(d.block))})`, 'good');
   695→          break;
   696→        case 'work_done':
   697→          store.pushLog(`💻 Cerrada una "reunión que podía ser un correo": +${String(d.amount)} €`, 'info');
   698→          break;
   699→        case 'activity_started':
   700→          if (d.activity !== 'idle') store.pushLog(`${this.nameOf(d.resident)} ${this.activityLabel(d.resident, String(d.activity))}`, 'info');
   701→          break;
   702→        case 'junta_requested': {
   703→          const agenda = (d.agenda as { id: string; title: string; description: string; cost: number }[] | undefined) ?? [];
   704→          store.setJunta(agenda.map((i) => ({ id: i.id, title: i.title, description: i.description, cost: i.cost })));
   705→          break;
   706→        }
   707→        case 'junta_result':
   708→          store.pushToast(
   709→            d.approved ? `Aprobado: ${String(d.title)} (${String(d.yes)} sí / ${String(d.no)} no)` : `Rechazado: ${String(d.title)} (${String(d.yes)} sí / ${String(d.no)} no)`,
   710→            d.approved ? ('good' as Tone) : ('bad' as Tone),
   711→          );
   712→          break;
   713→        case 'junta_vote_registered':
   714→          store.pushLog(`🗳️ ${this.nameOf(d.resident)} ha depositado su voto. Pendientes: ${String(d.pendingHumans)}`, 'info');
   715→          break;
   716→        case 'junta_closed':
   717→          store.setJunta(null);
   718→          break;
   719→        case 'minigame_requested': {
   720→          const def = this.content.minigames[String(d.minigame)];
   721→          if (!def) break;
   722→          // Overrides numéricos del escenario activo (QA/E2E): solo config, nunca efectos ni textos.
   723→          const ov = this.scenarioOverrides[def.id];
   724→          const effective = ov ? ({ ...def, config: { ...def.config, ...ov } } as MinigameDef) : def;
   725→          store.setMinigame({ ...effective, emoji: effective.emoji ?? '🎮', instruction: effective.instruction ?? null });
   726→          break;
   727→        }
   728→        case 'minigame_finished': {
   729→          const name = this.content.minigames[String(d.minigame)]?.name ?? String(d.minigame);
   730→          store.pushLog(d.success ? `🎮 Minijuego superado: ${name}` : `🎮 Minijuego fallido: ${name}`, d.success ? 'good' : 'bad');
   731→          break;
   732→        }
   733→        case 'ui_toast':
   734→          store.pushToast(String(d.text), (d.tone as Tone | undefined) ?? 'info');
   735→          break;
   736→        case 'finale':
   737→          store.setFinale(String(d.text));
   738→          break;
   739→        case 'funds_changed': {
   740→          const amount = typeof d.amount === 'number' ? d.amount : 0;
   741→          if ((d.to === 'player' || d.to === 'community') && amount !== 0) {
   742→            store.pushLog(amount > 0 ? `+${amount} € (fondos)` : `${amount} € (fondos)`, amount > 0 ? 'good' : 'bad');
   743→          }
   744→          break;
   745→        }
   746→        default: {
   747→          // Umbral de canal con texto del catálogo (odors, wifi y futuros canales): totalmente data-driven.
   748→          if (d.uiLog === true && typeof d.message === 'string') {
   749→            store.pushLog(String(d.message).replace('{block}', blockLabel(String(d.block))), (d.tone as Tone | undefined) ?? 'info');
   750→          }
   751→          break; // resident_moved, param_changed, activity_ended... demasiado ruido para el registro
   752→        }
   753→      }
   754→    }
   755→  }

   776→  private objectiveUi(): { title: string; text: string } | null {
   777→    if (!this.world) return null;
   778→    const o = activeObjective(this.world, this.content);
   779→    if (!o) return null;
   780→    if (o.state === 'awaiting_choice') return { title: o.title, text: 'El edificio contiene la respiración. Decide qué hacer.' };
   781→    const s = o.step!;
   782→    const left = o.untilTick !== null ? ` · quedan ${Math.max(0, o.untilTick - this.world.tick)} min` : '';
   783→    switch (s.type) {
   784→      case 'phone_call':
   785→        return { title: o.title, text: 'Suena el teléfono. Atiéndelo.' };
   786→      case 'reach':
   787→        return { title: o.title, text: `Llega a la puerta (${blockLabel(s.block ?? '')}).` };
   788→      case 'enter':
   789→        return { title: o.title, text: 'Entra en tu piso con E.' };
   790→      case 'catch_in_act':
   791→        return { title: o.title, text: `Pilla al responsable in fraganti${left}.` };
   792→      default:
   793→        return { title: o.title, text: o.title };
   794→    }
   795→  }

   797→  private promptText(): string | null {
   798→    const logic = this.logic;
   799→    const world = this.world;
   800→    if (!logic || !world) return null;
   801→    if (logic.position().climbing) return null;
   802→    // El prompt es SIEMPRE el objetivo real de la tecla E (misma fuente, cero divergencias).
   803→    const t = logic.interactTarget();
   804→    if (t) {
   805→      const ic = this.content.rules.interact;
   806→      switch (t.kind) {
   807→        case 'door': {
   808→          const door = world.blocks[t.blockId]!;
   809→          if (logic.position().mode === 'inside') return ic.exit;
   810→          return door.access === 'locked'
   811→            ? ic.doorLocked.replace('{block}', blockLabel(door.id))
   812→            : ic.doorEnter.replace('{block}', blockLabel(door.id));
   813→        }
   814→        case 'resident':
   815→          return ic.greet.replace('{name}', this.nameOf(t.residentId));
   816→        case 'mailbox':
   817→          return ic.mailbox.prompt;
   818→        case 'noticeboard':
   819→          return ic.noticeboard.prompt;
   820→        case 'laptop':
   821→          return ic.laptop;
   822→        case 'bed':
   823→          return ic.bed;
   824→      }
   825→    }
   826→    const zone = logic.zone();
   827→    const feat = this.content.blocks[zone.defId]!.features;
   828→    if (feat?.stairs || feat?.elevator) {
   829→      const ic = this.content.rules.interact;
   830→      // Lo que dice el prompt es lo que W/S hace aquí: ascensor operativo manda; si no hay
   831→      // ascensor utilizable pero sí escalera, escalera; núcleo averiado sin escalera = aviso.
   832→      if (feat.elevator && world.blocks[zone.id]!.params.elevator_status === 'ok') return ic.elevator;
   833→      if (feat.stairs) return ic.stairs;
   834→      return ic.elevatorBroken;
   835→    }
   836→    return null;
   837→  }

   839→  /** Viviendas del jugador para el editor de interiores (foto serializable, posiciones ya en %). */
   840→  private homesUi() {
   841→    const world = this.world!;
   842→    const cellH = this.content.rules.view.cellH;
   843→    const mine = this.actorId();
   844→    return Object.values(world.blocks)
   845→      .filter((b) => b.kind === 'dwelling' && b.ownerId === mine)
   846→      .sort((a, b) => a.id.localeCompare(b.id))
   847→      .map((b) => ({
   848→        id: b.id,
   849→        label: blockLabel(b.id),
   850→        rooms: b.rooms.map((room) => {
   851→          const rdef = this.content.rooms[room.defId];
   852→          const slots = (rdef?.furnitureSlots ?? []).map((s) => {
   853→            const itemId = room.furniture[s.id] ?? '';
   854→            const fdef = itemId ? this.content.furniture[itemId] : undefined;
   855→            return {
   856→              id: s.id,
   857→              kind: s.kind,
   858→              left: Math.min(92, Math.max(2, (s.pos[0] / Math.max(0.001, rdef?.size.w ?? 1)) * 100)),
   859→              bottom: Math.min(76, Math.max(0, (s.pos[1] / Math.max(0.001, cellH)) * 100)),
   860→              item: fdef ? { id: fdef.id, name: fdef.name, emoji: fdef.emoji ?? '📦', color: fdef.view.color } : null,
   861→            };
   862→          });
   863→          return { key: room.key, label: rdef?.label ?? room.key, access: room.access, slots };
   864→        }),
   865→      }));
   866→  }

   868→  private syncHud(): void {
   869→    const world = this.world;
   870→    if (!world) return;
   871→    const player = world.residents[this.actorId()] ?? world.residents[world.playerId];
   872→    if (!player) return;
   873→    const home = player.home;
   874→    const params = visibleParams(world, this.content, home).map((p) => ({ name: p.name, label: p.def.label ?? p.name, value: String(p.value) }));
   875→    const missions = world.missions.map((m) => {
   876→      const def = this.content.missions[m.id]!;
   877→      const step = m.state === 'active' ? def.steps[m.step] : undefined;
   878→      return {
   879→        id: m.id,
   880→        title: def.title,
   881→        state: m.state,
   882→        hint: step?.hint ?? null,
   883→        remaining: m.until !== null ? `quedan ${Math.max(0, m.until - world.tick)} min` : null,
   884→      };
   885→    });
   886→    const travel = travelMinutes(buildCirculation(world, this.content), world.entry, home);
   887→    const elevatorBlock = Object.values(world.blocks).find((b) => b.params.elevator_status !== undefined);
   888→    useGameStore.getState().syncHud({
   889→      clock: { ...world.clock },
   890→      weather: world.weather.state,
   891→      funds: world.community.funds,
   892→      wallet: player.wallet,
   893→      reputation: player.reputation,
   894→      elevator: String(elevatorBlock?.params.elevator_status ?? 'ok'),
   895→      homeLabel: blockLabel(home),
   896→      travelText: Number.isFinite(travel) ? `del portal a tu puerta: ${travel} min` : null,
   897→      objective: this.objectiveUi(),
   898→      prompt: this.promptText(),
   899→      params,
   900→      missions,
   901→      homes: this.homesUi(),
   902→      furnitureConfig: { refundRatio: this.content.rules.furniture.refundRatio, roomUnlockPrice: this.content.rules.furniture.roomUnlock.price },
   903→    });
   904→  }
