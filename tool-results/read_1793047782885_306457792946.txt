    21→  /** Materiales que reciben glow de canales y tintado de estado (1 en greybox, N en modelos). */

    23→  /** Emissive base por material (lo que paint() deja sin canales); lo usa el LOD 'glow'. */

    25→  /** Color base por material (view.color en greybox; albedo capturado en modelos). */

    44→  edgesMesh: Mesh | null;

    74→/**
    75→ * Una caja por bloque (placeholder greybox del futuro prefab) y, si el manifiesto de assets
    76→ * (Content.assets.blocks) declara un modelo glTF para el tipo de bloque, el modelo en su lugar
    77→ * (con la caja oculta). TODA la geometría y los colores salen de content.rules.view +
    78→ * block.view + assets: sustituir el greybox por arte final no toca este código.
    79→ * Las viviendas se dibujan en corte (interior visible) SOLO mientras hay alguien dentro;
    80→ * vacías, la fachada se cierra con su puerta al pasillo (rules.view.privacy) — y las
    81→ * bloqueadas, con panel y candado como siempre.
    82→ */
    83→export function createBuilding(scene: Scene, world: World, content: Content) {
    84→  const { Color3, Color4, MeshBuilder, StandardMaterial, TransformNode } = B();
    85→  const v = content.rules.view;
    86→  const glow = Object.fromEntries(Object.entries(v.channelGlow).map(([k, hex]) => [k, Color3.FromHexString(hex)])) as Record<string, Color3>;
    87→  const views = new Map<string, BlockView>();
    88→  const shakes = new Map<string, number>();
    89→  /** Raíz de escena por bloque (v1.19): colgar aquí TODA malla del bloque permite el
    90→   *  desvanecimiento completo del bloque principal en el apilamiento 3D (swap de bloques). */
    91→  const roots = new Map<string, TransformNode>();
    92→  const attach = (blockId: string, mesh: Mesh): Mesh => {
    93→    const r = roots.get(blockId);
    94→    if (r) mesh.parent = r;
    95→    return mesh;
    96→  };
    97→  /** Interiores 2D (Fase C) por bloque: suelos de estancia y tapas de puertas cerradas. */
    98→  const interiors = new Map<string, InteriorView>();
    99→  /** Foco del LOD (posición del jugador): lo alimenta update() y lo consume paint(). */
   100→  let lastFocus: { x: number; floor: number } | null = null;
   101→  // Plano frontal de las estancias: la cara de la puerta. Con viviendas del doble de calado
   102→  // (v0.6-b), el extra crece hacia el FONDO (+Z) para no envolver el pasillo ni al núcleo.
   103→  // v1.19: los calados del catálogo YA incluyen el +1 módulo (rules.view.depthModule) —
   104→  // depth 4.2 en viviendas y commonDepth 2.8 en comunes son los valores finales.
   105→  const frontZ = v.roomZ - v.commonDepth / 2;
   106→  /** Calado por bloque: viviendas 2× (mobiliario interior); naves del cierre (v1.16) y el
   107→   *  tejado (con aleros) a calado de vivienda; bloques comunes, el de siempre. */
   108→  const depthOf = (b: BlockInstance): number => {
   109→    if (b.kind === 'dwelling') return v.depth;
   110→    if (b.defId === 'trasteros_planta' || b.defId === 'bajocubierta') return v.depth;
   111→    if (b.defId === v.trasteros.blockDefId) return v.depth * v.trasteros.blockDepthFactor;
   112→    if (b.defId === 'tejado') return v.depth + v.roof.overhang * 2;
   113→    return v.commonDepth;
   114→  };
   115→  const mat = (name: string) => new StandardMaterial(name, scene);
   116→  /**
   117→   * V1.26 — OPACIDAD TOTAL para hojas de puerta: los DynamicTexture con píxeles alpha=0 se
   118→   * renderizan a NEGRO en esta config (WebGPU/CDN — diagnóstico v1.25). Las artes de puerta
   119→   * traen sombras cocinadas que el flood-fill de esquinas deja a medias: las rendijas
   120→   * transparentes sobrevivientes dentro del recorte se rellenan con el color opaco más
   121→   * cercano (dilatación horizontal en dos pasadas) — la textura queda 100% opaca y el
   122→   * perímetro de la hoja deja de verse negro/rojizo.
   123→   */
   124→  const dilateOpaque = (c: HTMLCanvasElement): void => {
   125→    const ctx = c.getContext('2d', { willReadFrequently: true });
   126→    if (!ctx) return;
   127→    const W = c.width;
   128→    const H = c.height;
   129→    const img = ctx.getImageData(0, 0, W, H);
   130→    const d = img.data;
   131→    for (let y = 0; y < H; y++) {
   132→      const row = y * W;
   133→      let pr = -1;
   134→      let pg = -1;
   135→      let pb = -1;
   136→      for (let x = 0; x < W; x++) {
   137→        const p = (row + x) * 4;
   138→        if (d[p + 3]! > 250) {
   139→          pr = d[p]!;
   140→          pg = d[p + 1]!;
   141→          pb = d[p + 2]!;
   142→        } else if (pr >= 0) {
   143→          d[p] = pr;
   144→          d[p + 1] = pg!;
   145→          d[p + 2] = pb!;
   146→          d[p + 3] = 255;
   147→        }
   148→      }
   149→      pr = -1;
   150→      pg = -1;
   151→      pb = -1;
   152→      for (let x = W - 1; x >= 0; x--) {
   153→        const p = (row + x) * 4;
   154→        if (d[p + 3]! > 250) {
   155→          pr = d[p]!;
   156→          pg = d[p + 1]!;
   157→          pb = d[p + 2]!;
   158→        } else if (pr >= 0) {
   159→          d[p] = pr;
   160→          d[p + 1] = pg!;
   161→          d[p + 2] = pb!;
   162→          d[p + 3] = 255;
   163→        }
   164→      }
   165→    }
   166→    ctx.putImageData(img, 0, 0);
   167→  };
   168→  /** Texturas de puerta cacheadas por fichero (v1.24→v1.26): FULL-BLEED y OPACA. El arte
   169→   *  llega con pared/suelo cocinados alrededor de la hoja: flood-fill contra los colores de
   170→   *  las esquinas (stripCorners) + recorte a la caja del contenido (alphaBoundsOf) +
   171→   *  dilatación opaca (dilateOpaque) → la textura ES la hoja, sin rendijas negras. */
   172→  const doorArtCache = new Map<string, DynamicTexture>();
   173→  const doorArtTex = (sc: Scene, file: string): DynamicTexture => {
   174→    const hit = doorArtCache.get(file);
   175→    if (hit) return hit;
   176→    const dt = new (B().DynamicTexture)(`puerta_tex_${doorArtCache.size}`, { width: 384, height: 768 }, sc, true);
   177→    doorArtCache.set(file, dt);
   178→    const img = new Image();
   179→    img.onload = () => {
   180→      const big = document.createElement('canvas');
   181→      big.width = img.naturalWidth;
   182→      big.height = img.naturalHeight;
   183→      const bx = big.getContext('2d', { willReadFrequently: true });
   184→      if (!bx) return;
   185→      bx.drawImage(img, 0, 0);
   186→      stripCorners(big, 0.14);
   187→      const b = alphaBoundsOf(big);
   188→      if (b) bx.drawImage(big, b.x0, b.y0, b.x1 - b.x0, b.y1 - b.y0, 0, 0, 384, 768);
   189→      else bx.drawImage(img, 0, 0, img.naturalWidth, img.naturalHeight, 0, 0, 384, 768);
   190→      dilateOpaque(big);
   191→      const dctx = dt.getContext() as unknown as CanvasRenderingContext2D;
   192→      dctx.clearRect(0, 0, 384, 768);
   193→      dctx.drawImage(big, 0, 0, 384, 768);
   194→      dt.update(true);
   195→    };
   196→    img.src = file;
   197→    return dt;
   198→  };
   199→  /** Candado DIBUJADO (v1.25→v1.26): silueta real de candado — nada de cajas 3D. Colores
   200→   *  del catálogo (lockColor); grillete de acero fijo. v1.26: la subida sale por
   201→   *  TEXTURE(dataURL) — los DynamicTexture desde canvas pierden el canal alpha en esta
   202→   *  config (WebGPU/CDN: píxeles alpha=0 → NEGRO) y el candado se veía con halo negro. */
   203→  let lockTex: Texture | null = null;
   204→  let lockTexAr = 0.8;
   205→  const canvasTex = (name: string, c: HTMLCanvasElement): Texture => {
   206→    const t = new (B().Texture)(c.toDataURL('image/png'), scene, false, true);
   207→    t.name = name;
   208→    t.hasAlpha = true;
   209→    return t;
   210→  };
   211→  const padlockTex = (): Texture => {
   212→    if (lockTex) return lockTex;
   213→    const c = document.createElement('canvas');
   214→    c.width = 128;
   215→    c.height = 128;
   216→    const ctx = c.getContext('2d', { willReadFrequently: true });
   217→    if (!ctx) {
   218→      const fallback = canvasTex('candado_tex', c);
   219→      lockTex = fallback;
   220→      return fallback;
   221→    }
   222→    ctx.clearRect(0, 0, 128, 128);
   223→    const body = Color3.FromHexString(v.lockColor).toHexString();
   224→    // Grillete: arco de acero.
   225→    ctx.strokeStyle = '#8d939c';
   226→    ctx.lineWidth = 11;
   227→    ctx.beginPath();
   228→    ctx.arc(64, 52, 26, Math.PI, 0);
   229→    ctx.stroke();
   230→    // Cuerpo de latón con borde.
   231→    ctx.fillStyle = body;
   232→    ctx.strokeStyle = 'rgba(60,40,10,0.9)';
   233→    ctx.lineWidth = 4;
   234→    const r = 12;
   235→    ctx.beginPath();
   236→    ctx.moveTo(24, 124);
   237→    ctx.lineTo(24, 66 + r);
   238→    ctx.arcTo(24, 66, 24 + r, 66, r);
   239→    ctx.lineTo(104 - r, 66);
   240→    ctx.arcTo(104, 66, 104, 66 + r, r);
   241→    ctx.lineTo(104, 124);
   242→    ctx.closePath();
   243→    ctx.fill();
   244→    ctx.stroke();
   245→    // Ojo de la cerradura.
   246→    ctx.fillStyle = 'rgba(50,32,8,0.95)';
   247→    ctx.beginPath();
   248→    ctx.arc(64, 90, 8, 0, Math.PI * 2);
   249→    ctx.fill();
   250→    ctx.fillRect(60, 90, 8, 20);
   251→    // Recorte a la caja del contenido: el quad queda del tamaño exacto de la silueta.
   252→    const b = alphaBoundsOf(c);
   253→    const out = document.createElement('canvas');
   254→    out.width = 128;
   255→    out.height = 128;
   256→    const octx = out.getContext('2d');
   257→    if (b && octx) {
   258→      const bw = Math.max(1, b.x1 - b.x0);
   259→      const bh = Math.max(1, b.y1 - b.y0);
   260→      lockTexAr = bw / bh;
   261→      octx.drawImage(c, b.x0, b.y0, bw, bh, 0, 0, bw, bh);
   262→    } else {
   263→      octx?.drawImage(c, 0, 0);
   264→    }
   265→    lockTex = canvasTex('candado_tex', out);
   266→    return lockTex;
   267→  };
   268→  /** Apliques de las puertas de vivienda: el foco cálido amaanece con la noche (update). */
   269→  const doorLampMats: { mat: StandardMaterial; hex: string }[] = [];
   270→  /** Dioramas interiores (v1.23): null sin catálogo interiors.json — greybox como siempre. */
   271→  const diro: DioramaDirector | null = createDioramaDirector(scene, content);
   272→  const box = (name: string, w: number, h: number, d: number, x: number, y: number, z: number, m: StandardMaterial) => {
   273→    const mesh = MeshBuilder.CreateBox(name, { width: w, height: h, depth: d }, scene);
   274→    mesh.position.set(x, y, z);
   275→    mesh.material = m;
   276→    return mesh;
   277→  };
   278→
   279→  for (const b of Object.values(world.blocks)) {
   280→    const m = mat(`${b.id}_mat`);
   281→    const cx = (b.cell.col + b.size.w / 2) * v.cellW;
   282→    const bottom = b.cell.floor * v.cellH;
   283→    const w = b.size.w * v.cellW - v.gapW;
   284→    const h = v.cellH - v.gapH;
   285→    const d = depthOf(b);
   286→    const cz = frontZ + d / 2;
   287→    const view: BlockView = {
   288→      greybox: [],
   289→      mats: [m],
   290→      baseEmissive: [m.emissiveColor.clone()],
   291→      baseColor: [Color3.FromHexString(content.blocks[b.defId]!.view?.color ?? '#cccccc')],
   292→      tintStrength: 1,
   293→      model: null,
   294→      front: null,
   295→      door: null,
   296→      handle: null,
   297→      lock: null,
   298→      lockBase: { x: cx, y: bottom + v.cellH / 2 - 0.2 },
   299→      centerX: cx,
   300→      floor: b.cell.floor,
   301→      edgesMesh: null,
   302→      edgesMesh2: null,
   303→      glowForced: false,
   304→      lodHiddenPanels: false,
   305→      doorDecor: [],
   306→      fixedDiffuse: null,
   307→    };
   308→    const root = new TransformNode(`${b.id}_root`, scene);
   309→    roots.set(b.id, root);
   310→    if (b.layer === 'corridor') {
   311→      // Pasillo exterior (data-driven): tira transitable saliente hacia la calle, con el borde
   312→      // trasero a ras de fachada (frontZ) y barandilla baja en el borde exterior — sitio de
   313→      // sobra para que los personajes se crucen o charlen sin bailar el vacío.
   314→      const d = v.corridorDepth;
   315→      view.greybox.push(box(b.id, b.size.w * v.cellW, v.slab * 2, d, cx, bottom + v.slab, frontZ - d / 2, m));
   316→      if (v.corridorRail > 0) {
   317→        view.greybox.push(
   318→          box(`${b.id}_barandilla`, b.size.w * v.cellW, v.corridorRail, v.wall, cx, bottom + v.slab * 2 + v.corridorRail / 2, frontZ - d, m),
   319→        );
   320→      }
   321→    } else if (b.kind === 'dwelling') {
   322→      const y = bottom + v.cellH / 2;
   323→      const pv = v.privacy;
   324→      // v1.25 — acabado de la vivienda: pared GRIS CLARITO fija (el tintado de estado no la
   325→      // repinta; el glow de canales sigue) + ZÓCALO gris de 25 cm desde el suelo + marcos de
   326→      // puerta DEL COLOR del zócalo. Todo del catálogo (rules.view.privacy).
   327→      view.fixedDiffuse = Color3.FromHexString(pv.wallColor ?? '#dbd8d2');
   328→      // ── Puerta a sala TRASERA (v1.19, apilamiento 3D): la pared del fondo se parte con el
   329→      // hueco del layout (doorway hacia una estancia más profunda que el bloque principal).
   330→      const layoutId0 = content.blocks[b.defId]?.container?.layout;
   331→      const layout0 = layoutId0 ? content.layouts[layoutId0] : undefined;
   332→      const rearKeys = new Set((layout0?.rooms ?? []).filter((r) => r.rect && r.rect.z + r.rect.d > v.depth + 0.01).map((r) => r.key));
   333→      const rearDoor = (layout0?.doorways ?? []).find((dw) => dw.axis === 'x' && rearKeys.has(dw.b));
   334→      const x0w = cx - w / 2;
   335→      const x1w = cx + w / 2;
   336→      if (rearDoor) {
   337→        const gapA = Math.max(x0w, x0w + rearDoor.at - rearDoor.width / 2);
   338→        const gapB = Math.min(x1w, x0w + rearDoor.at + rearDoor.width / 2);
   339→        const doorH = Math.min(h - 0.15, 1.95);
   340→        const lintelH = h - doorH;
   341→        view.edgesMesh = box(`${b.id}_fondo_i`, gapA - x0w, h, v.wall, (x0w + gapA) / 2, y, frontZ + d, m);
   342→        view.edgesMesh2 = box(`${b.id}_fondo_d`, x1w - gapB, h, v.wall, (gapB + x1w) / 2, y, frontZ + d, m);
   343→        view.greybox.push(
   344→          view.edgesMesh,
   345→          view.edgesMesh2,
   346→          box(`${b.id}_fondo_dintel`, gapB - gapA, lintelH, v.wall, (gapA + gapB) / 2, y + h / 2 - lintelH / 2, frontZ + d, m),
   347→          box(`${b.id}_izq`, v.wall, h, d, cx - w / 2, y, cz, m),
   348→          box(`${b.id}_der`, v.wall, h, d, cx + w / 2, y, cz, m),
   349→          box(`${b.id}_suelo`, w, v.wall, d, cx, bottom + v.slab + v.wall / 2, cz, m),
   350→          box(`${b.id}_techo`, w, v.wall, d, cx, bottom + v.cellH - v.slab - v.wall / 2, cz, m),
   351→        );
   352→      } else {
   353→        view.edgesMesh = box(`${b.id}_fondo`, w, h, v.wall, cx, y, frontZ + d, m);
   354→        view.greybox.push(
   355→          view.edgesMesh,
   356→          box(`${b.id}_izq`, v.wall, h, d, cx - w / 2, y, cz, m),
   357→          box(`${b.id}_der`, v.wall, h, d, cx + w / 2, y, cz, m),
   358→          box(`${b.id}_suelo`, w, v.wall, d, cx, bottom + v.slab + v.wall / 2, cz, m),
   359→          box(`${b.id}_techo`, w, v.wall, d, cx, bottom + v.cellH - v.slab - v.wall / 2, cz, m),
   360→        );
   361→      }
   362→      view.front = box(`${b.id}_puerta`, w, h, v.wall, cx, y, frontZ, m);
   363→      // ── ZÓCALO (v1.25→v1.26): banda gris de ~25 cm desde el suelo, CORTADA en la puerta
   364→      //  (dos tramos, sin cruzar el acceso — petición expresa del vecino). Continúa en el
   365→      //  MARCO de la puerta (mismo material). Todo del catálogo (rules.view.privacy).
   366→      const skH = pv.skirtingH ?? 0.25;
   367→      const skMat = mat(`${b.id}_zocalo_mat`);
   368→      skMat.diffuseColor = Color3.FromHexString(pv.skirtingColor ?? '#a3a19b');
   369→      skMat.specularColor = new Color3(0.02, 0.02, 0.02);
   370→      const ySk = bottom + v.slab + v.wall + skH / 2;
   371→      const cutSkirting = pv.skirtingDoorGap ?? false;
   372→      const skL = cutSkirting ? Math.max(x0w, cx - pv.doorW / 2) : x0w;
   373→      const skR = cutSkirting ? Math.min(x1w, cx + pv.doorW / 2) : x1w;
   374→      if (skL - x0w > 0.02) {
   375→        view.greybox.push(attach(b.id, box(`${b.id}_zocalo_frente_i`, skL - x0w, skH, v.wall + 0.02, (x0w + skL) / 2, ySk, frontZ, skMat)));

  1620→  /** Emissive del bloque: canales de influencia → emissive según color del canal en el catálogo. Se acumulan. */
  1621→  function applyEmissive(view: BlockView, b: BlockInstance) {
  1622→    let e = new Color3(0, 0, 0);
  1623→    for (const [channel, value] of Object.entries(world.channels)) {
  1624→      const level = value[b.id];
  1625→      if (!level) continue;
  1626→      const g = glow[channel];
  1627→      if (!g) continue;
  1628→      e = e.add(g.scale(Math.min(1, level)));
  1629→    }
  1630→    for (const m of view.mats) m.emissiveColor = e;
  1631→  }

  1633→  /** Color difuso/albedo por material: greybox plano (tintStrength 1) o mezcla sobre el arte.
  1634→   *  v1.25: fixedDiffuse (pared gris de vivienda) se impone al tintado — el estado lo cuentan
  1635→   *  la puerta, el candado y la placa; el emissive de canales sigue funcionando igual. */
  1636→  function applyTint(view: BlockView, b: BlockInstance) {
  1637→    const state = color(b);
  1638→    const override = stateTint(b);
  1639→    for (let i = 0; i < view.mats.length; i++) {
  1640→      const m = view.mats[i]!;
  1641→      const base = view.baseColor[i]!;
  1642→      if (view.fixedDiffuse) {
  1643→        if (m.diffuseColor) m.diffuseColor = view.fixedDiffuse;
  1644→        else if (m.albedoColor) m.albedoColor = view.fixedDiffuse;
  1645→        continue;
  1646→      }
  1647→      const c = view.tintStrength >= 1 ? state : override ? Color3.Lerp(base, override, view.tintStrength) : base;
  1648→      if (m.diffuseColor) m.diffuseColor = c;
  1649→      else if (m.albedoColor) m.albedoColor = c;
  1650→    }
  1651→  }

  1653→  // ── Privacidad de viviendas (v1.17): conjunto de bloques con alguien dentro (el jugador

  1660→  // por frame y NIVELES del catálogo (privacy.night) — las viviendas despiertas se VEN

  1677→  /** Visibilidad efectiva de fachada/puerta/candado: candado solo si access='locked';
  1678→   *  fachada cerrada si está bloqueada O si la privacidad la cierra (nadie dentro).
  1679→   *  paint() la fija por el acceso y el LOD puede ocultarlos encima. */
  1680→  function applyPanels(view: BlockView, b: BlockInstance, occ?: Set<string>) {
  1681→    const locked = b.kind === 'dwelling' && b.access === 'locked';
  1682→    let privacyClosed = false;
  1683→    if (privacyOn && b.kind === 'dwelling') {

  1688→    view.front?.setEnabled(wall);
  1689→    view.door?.setEnabled(wall);
  1690→    view.handle?.setEnabled(wall);

  1692→    view.lock?.setEnabled(locked && !view.lodHiddenPanels);
  1693→    // Diorama (v1.23): visibilidad ESPEJO de la fachada — fachada cerrada = interior oculto,
  1694→    // fachada abierta = diorama a la vista (y cero draw calls de interiores tapados).

  1699→  /**
  1700→   * LOD por distancia (rules.view.lod). Se ejecuta SIEMPRE al final de update(): con foco degrada
  1701→   * cada bloque según su distancia, y sin foco (o con lod.enabled=false) restaura el estado completo.
  1702→   * Idempotente y sin allocations por frame: solo aritmética, booleanos y asignaciones.
  1703→   *
  1704→   * Métrica (Vista · LOD por distancia): dx = |centroX del bloque − focus.x| en metros y
  1705→   * dy = |planta del bloque − focus.floor| * view.cellH en metros; dist = Math.hypot(dx, dy).
  1706→   * Nivel efectivo = todos los niveles de lod.levels cuyo `dist` ≤ distBloque (el `hide` es
  1707→   * acumulado: un flag se oculta si cualquier nivel alcanzado lo lista; sin nivel → todo visible).
  1708→   */
  1709→  function applyLod(focus: { x: number; floor: number } | null) {
  1710→    const lod = v.lod;
  1711→    for (const id in world.blocks) {
  1712→      const b = world.blocks[id]!;
  1713→      const view = views.get(id)!;
  1714→      let hideEdges = false;
  1715→      let hideGlow = false;
  1716→      let hidePanels = false;
  1717→      if (lod.enabled && focus) {
  1718→        const dx = Math.abs(view.centerX - focus.x);
  1719→        const dy = Math.abs(view.floor - focus.floor) * v.cellH;
  1720→        const dist = Math.hypot(dx, dy);
  1721→        for (let i = 0; i < lod.levels.length; i++) {
  1722→          const level = lod.levels[i];
  1723→          if (level.dist > dist) continue;
  1724→          for (let j = 0; j < level.hide.length; j++) {
  1725→            const flag = level.hide[j];
  1726→            if (flag === 'edges') hideEdges = true;
  1727→            else if (flag === 'glow') hideGlow = true;
  1728→            else if (flag === 'panels') hidePanels = true;
  1729→          }
  1730→        }
  1731→      }
  1732→      // 'edges': interruptor nativo barato — el renderingGroup consulta edgesRenderer.isEnabled cada
  1733→      // frame, así que nunca se destruye ni recrea el EdgesRenderer (disable/enableEdgesRendering sí
  1734→      // lo harían: el primero lo dispone y el segundo recomputaría las aristas). El grosor/color
  1735→      // viven en la malla y sobreviven al apagado. Los modelos glTF no tienen edges (edgesMesh null).
  1736→      if (view.edgesMesh?.edgesRenderer) view.edgesMesh.edgesRenderer.isEnabled = !hideEdges;
  1737→      if (view.edgesMesh2?.edgesRenderer) view.edgesMesh2.edgesRenderer.isEnabled = !hideEdges;
  1738→      // 'glow': pisa el emissive CADA frame (paint() pudo repintar los canales este tick); al soltarlo,
  1739→      // una única llamada repinta el cálculo de paint para no esperar al próximo tick.
  1740→      if (hideGlow) {
  1741→        view.glowForced = true;
  1742→        for (let i = 0; i < view.mats.length; i++) view.mats[i]!.emissiveColor = view.baseEmissive[i]!;
  1743→      } else if (view.glowForced) {
  1744→        view.glowForced = false;
  1745→        applyEmissive(view, b);
  1746→      }
  1747→      // 'panels': solo gira la bandera (la visibilidad la calcula applyPanels); en la transición se
  1748→      // aplica ya para no esperar al próximo paint(), que reproducirá el mismo resultado.
  1749→      if (view.lodHiddenPanels !== hidePanels) {
  1750→        view.lodHiddenPanels = hidePanels;
  1751→        applyPanels(view, b);
  1752→      }
  1753→    }
  1754→  }

  1804→    /**
  1805→     * dt en segundos; `focus` es el punto de interés (posición del jugador) para el

  1808→     * `inside` es el bloque donde está el jugador ahora (pos.insideBlock): alimenta la

  1811→     * `pos` (v1.19) alimenta los carteles guía de trasteros y el apilamiento 3D.

  1813→    update(dt: number, focus?: { x: number; floor: number }, inside?: string | null, pos?: { x: number; z: number; insideBlock: string | null }, env?: { daylight: number; movers?: { x: number; y: number; z: number }[] }) {
  1814→      if (focus) lastFocus = focus;
  1815→      if (inside !== undefined) playerInside = inside;
  1816→      // PRIVACIDAD (v1.17) cada frame: la fachada de una vivienda sin nadie dentro se cierra
  1817→      // con su puerta al pasillo y se abre en cuanto alguien entra (propietario, visita…
  1818→      // o un ladrón). Vive en update() y no en paint() para reaccionar incluso con el
  1819→      // mundo congelado (modal abierto) y sin esperar al siguiente tick.
  1820→      const occSet = computeOccupancy();
  1821→      if (privacyOn) {
  1822→        for (const b of Object.values(world.blocks)) {
  1823→          if (b.kind !== 'dwelling') continue;
  1824→          applyPanels(views.get(b.id)!, b, occSet);
  1825→        }
  1826→      }
  1827→      // LUCES INTERIORES NOCTURNAS (v1.21): al caer el sol, las estancias que se ven abiertas
  1828→      // (alguien dentro) se ILUMINAN salvo que su inquilino duerma — entonces, a oscuras.
  1829→      // Colores cacheados: cero allocations por frame. De día, lámparas apagadas.
  1830→      {
  1831→        const day = env?.daylight ?? 1;
  1832→        const litWorld = day < nightCfg.dayCutoff;
  1833→        // Apliques de las puertas (v1.24): foco cálido al anochecer, apagados de día.
  1834→        const lampGlow = litWorld ? 1.0 : 0.06;
  1835→        for (const { mat: lm, hex } of doorLampMats) lm.emissiveColor = Color3.FromHexString(hex).scale(lampGlow);
  1836→        // PUERTAS DE ESCALERA (v1.24): se abren antes de que alguien entre o salga (cualquier
  1837→        // mover cerca del umbral) y la ILUMINACIÓN COMPARTIDA de TODOS los cubos se enciende
  1838→        // de noche si algún vecino se mueve por las escaleras (un solo nivel para todos).
  1839→        const movers = env?.movers ?? [];
  1840→        const al = v.stairs.activityLight;
  1841→        for (const d of stairDoors) {
  1842→          let near = false;
  1843→          for (const mv of movers) {
  1844→            if (
  1845→              Math.abs(mv.x - d.doorX) < 0.85 &&
  1846→              Math.abs(mv.z - d.doorZ) < stF.openAhead &&
  1847→              Math.abs(mv.y - d.floorY) < v.cellH * 0.6
  1848→            ) {
  1849→              near = true;
  1850→              break;
  1851→            }
  1852→          }
  1853→          const next = d.openT + (near ? dt : -dt * 0.8) * stF.openSpeed;
  1854→          d.openT = Math.min(1, Math.max(0, next));
  1855→          if (d.openT !== (near ? 1 : 0) || d.hinge.rotation.y !== 0) {
  1856→            // Bisagras a la izquierda (vista pasillo = x mayor): girar +Y mete la hoja
  1857→            // hacia el interior del cubo; espejo para hinge 'right'.
  1858→            const sign = stF.hinge === 'left' ? 1 : -1;
  1859→            d.hinge.rotation.y = sign * stF.openAngle * d.openT;
  1860→          }

  1863→          let busy = false;

  1866→              for (const fp of stairFootprints) {
  1867→                if (
  1868→                  mv.x >= fp.x0 &&
  1869→                  mv.x <= fp.x1 &&
  1870→                  mv.z > frontZ + 0.3 &&
  1871→                  mv.z < frontZ + v.commonDepth - 0.1 &&
  1872→                  Math.abs(mv.y - fp.floorY) < v.cellH * 0.6
  1873→                ) {
  1874→                  busy = true;
  1875→                  break;
  1876→                }
  1877→              }
  1878→              if (busy) break;
  1879→            }
  1880→          }
  1881→          const next = stairLightLevel + (busy ? dt : -dt) / Math.max(0.4, al.fade);
  1882→          stairLightLevel = Math.min(1, Math.max(0, next));
  1883→          const lc = Color3.FromHexString(al.color);

  1885→          // en las escaleras de noche, muros/forjados/tramos brillan un poco — el efecto
  1886→          // pedido: "zonas más claras con algo de autoiluminación, como los interiores".

  1888→          const wg = al.wallGlow ?? 0;

  1890→            stairWallGlow.copyFromFloats(lc.r * wg * stairLightLevel, lc.g * wg * stairLightLevel, lc.b * wg * stairLightLevel);
  1891→            for (const { mat: wm } of stairWallMats) wm.emissiveColor = stairWallGlow;
  1892→          }
  1893→        }
  1894→        for (const [blockId, interior] of interiors) {
  1895→          let lit = false;
  1896→          if (litWorld && occSet.has(blockId)) {
  1897→            lit = true;
  1898→            for (const r of Object.values(world.residents)) {
  1899→              if (r.present && r.location === blockId && r.activity === 'sleep') {
  1900→                lit = false;
  1901→                break;
  1902→              }
  1903→            }
  1904→          }
  1905→          for (const lamp of interior.lamps) {
  1906→            lamp.mesh.setEnabled(lit);
  1907→            if (lit) lamp.mat.emissiveColor = nightLampLit;
  1908→          }
  1909→          for (const f of interior.floors) f.mat.emissiveColor = lit ? nightFloorGlow : nightDark;
  1910→          // Diorama (v1.23): el arte ilustrado sigue la misma luz que suelos y lámparas.
  1911→          for (const m of interior.dioMats) m.emissiveColor = lit ? nightDioGlow : nightDark;
  1912→        }
  1913→      }
  1914→      // CARTELES GUÍA (v1.19): visibles solo con el jugador dentro del bloque de trasteros;
  1915→      // texto = los trasteros que tienes a cada lado según la profundidad del pasillo.
  1916→      const sgG = colBlocks.length > 0 ? storageGeomOf(v) : null;
  1917→      for (const [blkId, sign] of guideSigns) {
  1918→        const active = !!pos && pos.insideBlock === blkId && !!sgG;
  1919→        const plate = scene.getMeshByName(`${blkId}_guia`);
  1920→        const frame = scene.getMeshByName(`${blkId}_guia_marco`);
  1921→        plate?.setEnabled(active);
  1922→        frame?.setEnabled(active);
  1923→        if (!active || !sgG) continue;
  1924→        const localZ = pos!.z; // en modo 'inside' de trasteros, z es profundidad local
  1925→        const usable = sgG.depth - 2 * sgG.wallT;
  1926→        const perSide = new Map<string, number>();
  1927→        for (const u of sign.plan) perSide.set(u.side, (perSide.get(u.side) ?? 0) + 1);
  1928→        const labelOf = (side: 'left' | 'right'): string => {
  1929→          const count = perSide.get(side) ?? 0;
  1930→          if (count === 0) return tr.emptyLabel;
  1931→          const idx = Math.max(0, Math.min(count - 1, Math.floor((localZ - sgG.wallT) / (usable / count))));
  1932→          const unit = sign.plan.find((u) => u.side === side && u.index === idx);
  1933→          return unit ? blockLabel(`blk_p${unit.floor}_${sign.ownTag}`) : tr.emptyLabel;
