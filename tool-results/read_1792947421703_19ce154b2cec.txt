   361→/**

   370→/**
   371→ * Beat de cutscene: la unión mínima para teatralizar misiones. Todo lo temporal
   372→ * es opcional y cae a rules.cutscene (defaults del catálogo).
   373→ */
   374→export type CutsceneBeat =
   375→  /** Cámara: vuela al encuadre (edificio entero, jugador o bloque) y aguanta `hold` s. */
   376→  | { type: 'cam'; focus: 'wide' | 'player' | CutsceneSpot; viewH?: number; hold?: number }
   377→  /** Actor: se coloca en `from` (si hay) y anda hasta `to` por pozo/interior (nunca teletransporta).
   378→   *  Mientras camina, la cámara le SIGUE (travelling); viewH ajusta lo cerrado del plano. */
   379→  | { type: 'walk'; who: string; from?: CutsceneSpot; to: CutsceneSpot; max?: number; viewH?: number }
   380→  /** Subtítulo cinematográfico con marcadores {resident} y {block}. */
   381→  | { type: 'line'; speaker?: string; text: string; seconds?: number; who?: string; gesture?: string }
   382→  /** Pausa teatral. */
   383→  | { type: 'wait'; seconds: number }
   384→  /** Sonido del catálogo procedural (sounds.json). */
   385→  | { type: 'sound'; id: string };
   386→
   387→/**
   388→ * Cutscene data-driven (Fase D): teatro de presentación sobre eventos del sim.
   389→ * El trigger filtra por tipo de evento y (opcional) misión/paso; el resto es secuencia
   390→ * de beats. Cero lógica de juego: al acabar, el mundo sigue exactamente igual.
   391→ */
   392→export interface CutsceneDef {
   393→  id: string;
   394→  trigger: { on: 'neighbor_arrived' | 'mission_activated' | 'step_started' | 'step_completed' | 'mission_completed' | 'mission_failed'; mission?: string; step?: string };
   395→  beats: CutsceneBeat[];
   396→}

   400→/**
   401→ * Catálogo de bocadillos (public/content/emotes.json): qué expresión muestra cada
   402→ * personaje sobre su cabeza según su estado — ánimo, actividad, gesto social o eventos
   403→ * del sim. 100% data-driven: el código de presentación solo recita este catálogo.
   404→ */
   405→export interface EmotesCatalog {
   406→  /** Ajustes visuales del bocadillo (proyección, tamaño y ritmo de decisión). */
   407→  bubble: {
   408→    /** Altura del bocadillo sobre la cabeza (m). */
   409→    offsetY: number;
   410→    /** Tamaño del glifo en píxeles CSS. */
   411→    sizePx: number;
   412→    /** Distancia máxima a la cámara (m) para mostrar el bocadillo. */
   413→    maxDistance: number;
   414→    /** Duración de la animación de aparición (ms). */
   415→    popMs: number;
   416→    /** Cadencia de reevaluación del estado (ms): amortigua parpadeos del sim. */
   417→    decisionMs: number;
   418→    /** Vista EDIFICIO (v1.17, opcional): tope de glifos visibles en el plano general —
   419→     *  el edificio entero no se convierte en una sopa de emoticonos. El texto (`say`) se
   420→     *  calla siempre en esa vista. */
   421→    buildingMaxGlyphs?: number;
   422→  };
   423→  /**
   424→   * Bocadillos con TEXTO (v1.15, opcional): las actividades con líneas `say` del arquetipo
   425→   * se dicen en un bocadillo legible sobre la cabeza, ciclando por sus líneas
   426→   * (decir → silencio con glifo → línea siguiente → pausa larga). Sin esta sección,
   427→   * los bocadillos son solo glifos (v1.10).
   428→   */
   429→  speech?: {
   430→    /** Segundos que se muestra cada línea. */
   431→    lineSeconds: number;
   432→    /** Segundos de silencio entre líneas (luce el glifo). */
   433→    gapSeconds: number;
   434→    /** Pausa larga tras decir todas las líneas antes de repetir el ciclo. */
   435→    cycleGap: number;
   436→    /** Ancho máximo del bocadillo de texto (px CSS). */
   437→    maxPx: number;
   438→    /** Tamaño de letra del texto (px CSS). */
   439→    fontSizePx: number;
   440→  };
   441→  /** Qué necesidades componen el "ánimo" (media ponderada 0..100) y con qué peso. */
   442→  mood: { needs: Record<string, number> };
   443→  /** Emotes por banda de ánimo: gana el de mayor prioridad cuyo rango [min,max] contenga el ánimo (rangos abiertos si falta min/max). */
   444→  moods: { id: string; glyph: string; priority: number; label?: string; min?: number; max?: number }[];
   445→  /** Emote por actividad del arquetipo (id de actividad → glifo). Las no listadas no muestran bocadillo. */
   446→  activities: { id: string; glyph: string; priority: number; label?: string }[];
   447→  /** Emote por gesto social en curso (stateAnimations: talk/laugh/argue/phone…). */
   448→  gestures: { id: string; glyph: string; priority: number; label?: string }[];
   449→  /** Emotes temporales por evento del sim (bocadillo que aparece y expira solo). */
   450→  events: {
   451→    /** Tipo de evento del sim (p. ej. stairs_grumble, knock_result). */
   452→    on: string;
   453→    glyph: string;
   454→    priority: number;
   455→    /** A quién le aparece: el residente del evento, el jugador, o los vecinos del bloque. */
   456→    who: 'resident' | 'player' | 'blockResidents';
   457→    /** Segundos que dura el bocadillo. */
   458→    seconds: number;
   459→    /** Glifo alternativo si un campo del evento es truthy (p. ej. asleep en knock_result). */
   460→    altGlyph?: { when: string; glyph: string };
   461→    /** Enfriamiento por residente y emote (s): evita bocadillos en cadena del mismo evento. */
   462→    cooldown?: number;
   463→  }[];
   464→}
