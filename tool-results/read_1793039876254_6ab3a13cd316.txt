    14→interface Actor {
    15→  /** Residente dueño del nodo (para las rutas de ascensor y depuración). */
    16→  id: string;
    17→  /** Nodo posicionable: cápsula greybox o raíz del modelo glTF (pivote en pies). */
    18→  node: Mesh | TransformNode;
    19→  height: number;
    20→  /** Offset Y del pivote: cápsula centrada (+h/2) o modelo con pies en el origen (+margen). */
    21→  pivotY: number;
    22→  /** Cola de tramos en curso: el actor la agota ANTES de capturar el destino del mundo de nuevo
    23→   *  (ir al pozo → subir → salir: un salto del sim se ve como un viaje, nunca un teletransporte). */
    24→  path: { x: number; y: number; z: number }[];
    25→  /** Vida interior (Fase C): cuenta atrás para el próximo punto de deambulación y nº de vuelta
    26→   *  (el punto es determinista por residente+ vuelta: sin RNG global ni capturas irreproducibles). */
    27→  wanderT: number;
    28→  wanderLap: number;
    29→  /** Objetivo interior VIGENTE (coords locales de la vivienda): se refresca con la cadencia del
    30→   *  catálogo (npcWanderSeconds), no cada tick — amortigua el parpadeo de actividades con chance. */
    31→  targetLocal: Pt | null;
    32→  lastLoc: string | null;
    33→  /** Dirección de cutscenes (Fase D): mientras está activa, la IA normal no captura destinos;
    34→   *  el actor solo consume la cola de tramos que le marca el director (nunca teletransporta). */
    35→  directed: boolean;
    36→  /** Rumbo actual (rad): el actor MIRA hacia donde camina (v1.2, credibilidad 2.5D).
    37→   *  0 = el modelo mira +X (perfil); ±π/2 = de espaldas/frente al entrar o salir de casa. */
    38→  yaw: number;
    39→  /** Rumbo pendiente de terminar: el actor remata el giro aunque se pare (nunca queda torcido). */
    40→  targetYaw: number | null;
    41→  /** Gesto social en curso (stateAnimations: talk/laugh/argue/phone/sit…) o null. */
    42→  gesture: string | null;
    43→  /** Cuenta atrás del gesto temporal (0 = persistente hasta que alguien lo quite). */
    44→  gestureT: number;
    45→  /** El objetivo interior vigente es un ancla de asiento (rules.view.seatAnchors): se sienta al llegar. */
    46→  seated: boolean;
    47→  /** Viaje en ascensor registrado (v1.5): { x: puerta; from/to: plantas }. null = sin viaje. */
    48→  elevatorReq: { x: number; from: number; to: number } | null;
    49→  /** Fase del viaje que se estaba atendiendo (para detectar transiciones y encolar tramos). */
    50→  ridePhase: string | null;
    51→  /** Estaba habilitado el frame anterior (v1.10): la REAPARICIÓN coloca al actor donde dice el
    52→   *  sim (vuelve de la calle), nunca caminando desde su última posición visible por la pared. */
    53→  enabledPrev: boolean;
    54→  /** Trastero ocupado (v1.18): id del bloque por columna cuyo compartimento habitó por última
    55→   *  vez. Mientras su ubicación siga siendo ese bloque, el nodo permanece oculto (está DENTRO
    56→   *  de su celda cerrada); al cambiar, reaparece por la puerta con la lógica de siempre. */
    57→  stowedBlock: string | null;
    58→  /** Rig del arte final (null mientras carga o si hay fallback greybox). */
    59→  rig: ActorRig | null;
    60→  /** Pausa de charla (v1.20): mientras > 0, el actor NO avanza (ni ruta ni objetivo) —
    61→   *  está parado charlando con otro vecino; la presentación lo suelta al terminar. */
    62→  chatT: number;
    63→}

    65→/**
    66→ * API de dirección de cutscenes (Fase D): toma actores, colócalos en escena y márcales rutas.
    67→ * Mientras un actor está "taken", el bucle normal de IA no le captura destinos: solo consume
    68→ * la cola de tramos (la misma locomoción interpolada de siempre, cero teletransportes).
    69→ */
    70→export interface ActorsDirector {
    71→  /** Posición del nodo (null si el actor no existe o está oculto). */
    72→  nodeOf(id: string): { x: number; y: number; z: number } | null;
    73→  /** Crea el actor si falta y lo marca como dirigido. false si el residente no está en escena. */
    74→  take(id: string): boolean;
    75→  /** Suelta al actor: la IA vuelve a decidir destino (cola y objetivo interior reseteados). */
    76→  release(id: string): void;
    77→  /** Coloca al actor en un punto de escena (staging teatral, decisión del contenido). */
    78→  stageAt(id: string, x: number, floor: number, z: number): void;
    79→  /** Encola la ruta del nodo actual al destino (pozo incluido si cambia la planta). */
    80→  walkTo(id: string, dest: { x: number; z: number; floor: number; inside?: { block: string; local: Pt } }): boolean;
    81→  /** true cuando el actor no tiene tramos pendientes (llegó — o no existe). */
    82→  idle(id: string): boolean;
    83→  /** Punto sobre la cabeza del actor (bocadillos v1.10); null si no está en escena. */
    84→  headOf(id: string): { x: number; y: number; z: number } | null;
    85→  /** Gesto social en curso (stateAnimations: talk/laugh/argue/phone…) o null. */
    86→  gestureOf(id: string): string | null;
    87→  /** true si el actor está tomado por el director de cutscenes (el staging manda). */
    88→  isDirected(id: string): boolean;
    89→  /** true si el actor está en un viaje de ascensor (v1.20: las charlas no lo pausan). */
    90→  isRiding(id: string): boolean;
    91→  /**
    92→   * Estado social explícito (v1.2): talk/laugh/argue/phone/sit… según stateAnimations del
    93→   * manifiesto. Con `seconds` > 0 el gesto expira solo (líneas de diálogo); null lo suelta
    94→   * ya (al colgar el teléfono, al cerrar el modal). El clip que falte cae al default del rig.
    95→   */
    96→  setGesture(id: string, state: string | null, seconds?: number): void;
    97→  /** Pausa de charla (v1.20): congela la locomoción del actor `seconds` (0 lo suelta). */
    98→  setHold(id: string, seconds: number): void;
    99→  /** Gira el actor hacia un punto del mundo (charla de cruce: mirarse al hablar). */
   100→  faceToward(id: string, x: number, z: number): void;
   101→}

   103→/**
   104→ * El jugador y los vecinos: cápsulas greybox o, si el manifiesto de assets declara un
   105→ * modelo glTF para el arquetipo (Content.assets.actors), el modelo con sus animaciones
   106→ * (idle/walk por desplazamiento y clip por actividad vía activityAnimations).
   107→ * Color y altura salen de la vista del arquetipo; radios y fallbacks, de content.rules.view.
   108→ *
   109→ * Dinámico (multijugador): la plantilla se sincroniza con world.residents en cada
   110→ * update — residentes nuevos reciben nodo y residentes que ya no existen lo pierden.
   111→ */
   112→export function createActors(scene: Scene, world: World, content: Content, elevators?: ElevatorsDirector | null) {
   113→  const { Color3, MeshBuilder, Quaternion, StandardMaterial, Vector3 } = B();
   114→  const v = content.rules.view;
   115→  const actors = new Map<string, Actor>();
   116→
   117→  /** Crea la cápsula de un residente (greybox/fallback) y lanza la carga del modelo si el manifiesto lo declara. */
   118→  const ensureActor = (id: string, archetype: string): Actor => {
   119→    const view = content.archetypes[archetype]?.view;
   120→    const height = view?.height ?? v.actorFallbackHeight;
   121→    const mesh = MeshBuilder.CreateCapsule(id, { height, radius: v.actorRadius }, scene);
   122→    const m = new StandardMaterial(`${id}_mat`, scene);
   123→    m.diffuseColor = Color3.FromHexString(view?.color ?? v.actorFallbackColor);
   124→    m.emissiveColor = m.diffuseColor.scale(0.25);
   125→    mesh.material = m;
   126→    mesh.isPickable = false;
   127→    const a: Actor = { id, node: mesh, height, pivotY: height / 2, path: [], wanderT: 0, wanderLap: 0, targetLocal: null, lastLoc: null, directed: false, yaw: 0, targetYaw: null, gesture: null, gestureT: 0, seated: false, elevatorReq: null, ridePhase: null, enabledPrev: false, stowedBlock: null, rig: null, chatT: 0 };
   128→    actors.set(id, a);
   129→
   130→    // Arte final (opcional): modelo glTF del manifiesto; cualquier fallo → cápsula.
   131→    const def = content.assets?.actors?.[archetype];
   132→    if (def && content.assets) {
   133→      void instantiateActorAsset(scene, def, content.assets.base, `${id}:${archetype}`).then((rig) => {
   134→        // La escena pudo morir mientras cargaba (restart/Cargar partida): no adjuntar nada viejo.
   135→        if (rig && scene.isDisposed) {
   136→          rig.dispose();
   137→          return;
   138→        }
   139→        if (!rig) return; // fallback greybox permanente
   140→        const cur = actors.get(id);
   141→        if (!cur || cur.rig) {
   142→          rig.dispose(); // el residente se fue (o ya llegó otro rig): no dejar basura
   143→          return;
   144→        }
   145→        // PRESERVAR POSE (v1.10): el modelo puede terminar de cargar EN PLENO paseo (los assets
   146→        // llegan async). Sin esto, el swap cápsula→rig recolocaba el nodo a origen y el actor
   147→        // "saltaba" (el guard antiguo lo enmascaraba con OTRO teletransporte al objetivo: la
   148→        // vecina con la maleta cambiando de planta de golpe). Pies y rumbo se conservan 1:1.
   149→        const feetY = cur.node.position.y - cur.pivotY;
   150→        rig.root.position.set(cur.node.position.x, feetY + 0.02, cur.node.position.z);
   151→        if (rig.root.rotationQuaternion) Quaternion.RotationYawPitchRollToRef(cur.yaw, 0, 0, rig.root.rotationQuaternion);
   152→        else rig.root.rotation.y = cur.yaw;
   153→        // El modelo se authora a nominalHeight: escala uniforme a la altura del arquetipo.
   154→        rig.root.scaling.setAll(height / def.nominalHeight);
   155→        cur.node.dispose(false, true); // cápsula fuera (libera también su material propio)
   156→        cur.node = rig.root;
   157→        cur.pivotY = 0.02; // pies en el suelo
   158→        cur.rig = rig;
   159→      });
   160→    }
   161→    return a;
   162→  };
   163→
   164→  const goal = new Vector3();
   165→  /** true si la fase del viaje cambió desde el último frame (hay que recolocar la cola). */
   166→  const stalePhase = (a: Actor, phase: string): boolean => a.ridePhase !== phase;
   167→  /** Pozos verticales del mundo (features stairs/elevator), para el RITMO de los tramos verticales. */
   168→  const shafts = Object.values(world.blocks).filter((b) => {
   169→    const f = content.blocks[b.defId]!.features;
   170→    return !!f && (f.stairs === true || f.elevator === true);
   171→  });
   172→  /** ¿Pozo UTILIZABLE para rutear EN ESTE MOMENTO? (v1.10) La escalera siempre; el ascensor
   173→   *  SOLO si funciona (elevator_status 'ok'). Se comprueba EN CADA RUTA porque el estado del
   174→   *  sim cambia en vivo (avería/reparación): un filtro estático enlazaría llamadas al pozo
   175→   *  averiado que el director cancela al frame siguiente (bucle llamada/cancelación). */
   176→  const shaftUsable = (b: BlockInstance): boolean => {
   177→    const f = content.blocks[b.defId]!.features!;
   178→    if (f.stairs === true) return true;
   179→    return f.elevator === true && b.params.elevator_status === 'ok';
   180→  };
   181→  const shaftXOf = (b: BlockInstance): number => (b.cell.col + b.size.w / 2) * v.cellW;
   182→  /** Calado transitablo del bloque (vivienda 2×; naves del cierre v1.16 a calado de vivienda;
   183→   *  trasteros por columna v1.18: depth × blockDepthFactor).
   184→   *  v1.19: en vivienda manda el LAYOUT — las salas traseras viven más allá de view.depth. */
   185→  const depthOfBlock = (b: BlockInstance): number => {
   186→    if (b.kind === 'dwelling') {
   187→      const layoutId = content.blocks[b.defId]?.container?.layout;
   188→      const layout = layoutId ? content.layouts[layoutId] : undefined;
   189→      const rects = (layout?.rooms ?? []).map((r) => r.rect).filter((r): r is NonNullable<typeof r> => !!r);
   190→      if (rects.length > 0) return Math.max(...rects.map((r) => r.z + r.d));
   191→      return v.depth;
   192→    }
   193→    if (b.defId === 'trasteros_planta' || b.defId === 'bajocubierta') return v.depth;
   194→    if (b.defId === v.trasteros.blockDefId) return v.depth * v.trasteros.blockDepthFactor;
   195→    if (b.defId === 'tejado') return v.depth + v.roof.overhang * 2;
   196→    return v.commonDepth;
   197→  };
   198→  /** Plano frontal de los bloques (compartido con storage.ts/building.ts). */
   199→  /**
   200→   * Punto de puerta del trastero de un residente en el bloque por columna de su letra
   201→   * (v1.18): pasillo central hasta su celda. null si no aplica (bloque equivocado, planta
   202→   * sin vivienda en el plan…).
   203→   */
   204→  const storageDoorOf = (resident: ResidentInstance): { x: number; z: number } | null => {
   205→    const home = world.blocks[resident.home];
   206→    if (!home) return null;
   207→    const tag = home.id.slice(home.id.lastIndexOf('_') + 1);
   208→    const store = Object.values(world.blocks).find(
   209→      (b) => isStorageColumnBlock(b, v) && b.id.endsWith(`_${tag}`),
   210→    );
   211→    if (!store) return null;
   212→    const colTag = store.id.slice(store.id.lastIndexOf('_') + 1);
   213→    const floors = Object.values(world.blocks)
   214→      .filter((d) => d.kind === 'dwelling' && d.id.endsWith(`_${colTag}`))
   215→      .map((d) => d.cell.floor)
   216→      .sort((a, z) => a - z);
   217→    return storageDoorPoint(v, store, floors, home.cell.floor);
   218→  };
   219→  /** Plano frontal de las viviendas: z local 0 de los interiores y origen del pasillo. */
   220→  const frontZ = frontZOf(v.roomZ, v.commonDepth);
   221→  /** Bloque main bajo los pies (v1.14): para salir de la banda de viviendas por SU puerta. */
   222→  const hostSpanAt = (x: number, floor: number): { x0: number; w: number } | null => {
   223→    for (const b of Object.values(world.blocks)) {
   224→      if (b.cell.floor !== floor || b.layer !== 'main') continue;
   225→      const x0 = b.cell.col * v.cellW;
   226→      const w = b.size.w * v.cellW;
   227→      if (x >= x0 && x <= x0 + w) return { x0, w };
   228→    }
   229→    return null;
   230→  };
   231→
   232→  /**
   233→   * Ruta (cola de tramos) del nodo actual al destino: cambio de planta → pozo más cercano
   234→   * (andar hasta él por el pasillo, subir/bajar, salir), y dentro de una vivienda con
   235→   * geometría 2D la ruta pasa por los huecos del catálogo (nunca a través de tabiques).
   236→   * La comparten la IA normal y el director de cutscenes: una sola verdad de locomoción.
   237→   *
   238→   * v1.14 — LEY DEL PASILLO (rutas blindadas): todo el tramo horizontal lo calcula
   239→   * horizontalLegs() (@/game/routing): el avance en X solo ocurre sobre la línea del
   240→   * pasillo y la banda de bloques solo se entra/sale en vertical en la x del bloque
   241→   * propio — jamás una diagonal atravesando el 2C (o el núcleo del ascensor) para
   242→   * llegar al 2B. Devuelve false SOLO cuando el destino exige cambiar de planta y no
   243→   * hay pozo utilizable (estado imposible: el llamador decide si recoloca sin teatro).
   244→   */
   245→  const buildRoute = (a: Actor, dest: { x: number; y: number; z: number; inside?: { block: BlockInstance; local: Pt } | null }): boolean => {
   246→    const node = a.node.position;
   247→    let startX = node.x;
   248→    let startZ = node.z;
   249→    let verticalOk = true;
   250→    if (Math.abs(dest.y - node.y) > 0.05) {
   251→      const floorNow = Math.round(node.y / v.cellH);
   252→      // Como el sim: ASCENSOR primero si funciona (coste 1 < 3 de la escalera); la escalera
   253→      // es la vía de emergencia cuando el núcleo está averiado (v1.10: estado comprobado EN VIVO).
   254→      const elevHere = shafts
   255→        .filter((s) => s.cell.floor === floorNow && content.blocks[s.defId]!.features!.elevator === true && s.params.elevator_status === 'ok')
   256→        .sort((s1, s2) => Math.abs(shaftXOf(s1) - startX) - Math.abs(shaftXOf(s2) - startX))[0];
   257→      const here = elevHere ?? shafts
   258→        .filter((s) => s.cell.floor === floorNow && shaftUsable(s))
   259→        .sort((s1, s2) => Math.abs(shaftXOf(s1) - startX) - Math.abs(shaftXOf(s2) - startX))[0];
   260→      if (here) {
   261→        const sx = shaftXOf(here);
   262→        // Salir del interior por la puerta propia (si procede) y andar hasta el pozo SOLO
   263→        // por la línea del pasillo: la diagonal directa cruzaba viviendas vecinas.
   264→        for (const l of horizontalLegs({ view: v, from: { x: startX, z: startZ }, dest: { x: sx, z: v.corridorZ }, hostSpan: hostSpanAt(startX, floorNow) })) {
   265→          a.path.push({ x: l.x, y: node.y, z: l.z });
   266→        }
   267→        // ¿Escalera o ascensor? La escalera REAL (v1.4) va por los peldaños; el ascensor
   268→        // OPERATIVO (v1.5) registra una LLAMADA: el actor camina a la puerta y espera a la
   269→        // cabina (capacidad, puertas y semáforo los lleva el director del ascensor).
   270→        const feat = content.blocks[here.defId]!.features!;
   271→        if (feat.stairs === true) {
   272→          // v1.23 — ESCALERA REAL POR PLANTAS: la escalera física conecta plantas ADYACENTES.
   273→          // Llamar a stairPathPoints con una subida de varias plantas estiraba cada peldaño a
   274→          // media planta (el vecino flotaba y atravesaba forjados). Ahora se ENCADENA un vuelo
   275→          // por planta: sube a la planta vecina, desembarca (el path termina en la línea del
   276→          // pasillo, exactamente donde arranca el siguiente vuelo) y repite hasta el destino.
   277→          const frac = node.y - Math.round(node.y / v.cellH) * v.cellH; // offset intra-planta (forjado + pivote)
   278→          const startFloor = Math.round(node.y / v.cellH);
   279→          const targetFloor = Math.round(dest.y / v.cellH);
   280→          const dir = targetFloor >= startFloor ? 1 : -1;
   281→          let curY = node.y;
   282→          let curFloor = startFloor;
   283→          while (curFloor !== targetFloor) {
   284→            curFloor += dir;
   285→            const nextY = curFloor * v.cellH + frac;
   286→            const flight = stairPathPoints(v, sx, curY, nextY);
   287→            if (flight.length > 1) {
   288→              // Enganche sin duplicar el punto donde ya está el actor (el desembarco del
   289→              // vuelo anterior ES el arranque del siguiente).
   290→              if (a.path.length > 0 || curY !== node.y) flight.shift();
   291→              for (const pt of flight) a.path.push({ x: pt.x, y: pt.y, z: pt.z });
   292→              curY = nextY;
   293→            }
   294→          }
   295→          if (a.path.length === 0) a.path.push({ x: sx, y: dest.y, z: v.corridorZ });
   296→        } else if (elevators && feat.elevator === true) {
   297→          const from = Math.round(node.y / v.cellH);
   298→          const to = Math.round(dest.y / v.cellH);
   299→          if (to !== from) {
   300→            elevators.requestRide(a.id, here.cell.col, from, to);
   301→            a.elevatorReq = { x: sx, from, to };
   302→            a.ridePhase = null;
   303→          }
   304→        } else {
   305→          a.path.push({ x: sx, y: dest.y, z: v.corridorZ });
   306→        }
   307→        startX = sx;
   308→        startZ = v.corridorZ;
   309→      } else {
   310→        verticalOk = false;
   311→      }
   312→    }
   313→    // Tramos horizontales al destino bajo la ley del pasillo (misma función que el enfoque
   314→    // al pozo: entrada al bloque solo en vertical, en la x del destino, o ruta interior).
   315→    const destGeo = dest.inside ? layoutGeometryOf(content, dest.inside.block) : null;
   316→    const legs = horizontalLegs({
   317→      view: v,
   318→      from: { x: startX, z: startZ },
   319→      dest: { x: dest.x, z: dest.z },
   320→      destInside:
   321→        dest.inside && destGeo
   322→          ? { x0: dest.inside.block.cell.col * v.cellW, w: dest.inside.block.size.w * v.cellW, geo: destGeo, target: dest.inside.local }
   323→          : null,
   324→      hostSpan: hostSpanAt(startX, Math.round(dest.y / v.cellH)),
   325→    });
   326→    for (const p of legs) a.path.push({ x: p.x, y: dest.y, z: p.z });
   327→    return verticalOk;
   328→  };
   329→
   330→  return {
   331→    /** playerResidentId identifica al residente del jugador local (multijugador: puede ser un invitado). */
   332→    update(dt: number, player: PlayerPosition, playerResidentId: string) {
   333→      // Altas y bajas: residentes sin nodo → crear; nodos huérfanos → dispose + remove.
   334→      for (const id in world.residents) {
   335→        if (!actors.has(id)) ensureActor(id, world.residents[id]!.archetype);
   336→      }
   337→      const gone: string[] = [];
   338→      for (const [id, a] of actors) {
   339→        if (!world.residents[id]) gone.push(id);
   340→      }
   341→      for (const id of gone) {
   342→        const a = actors.get(id)!;
   343→        actors.delete(id);
   344→        if (a.rig) a.rig.dispose();
   345→        else if (a.node instanceof B().Mesh) a.node.dispose(false, true);
   346→        else a.node.dispose(false, false);
   347→      }
   348→
   349→      const k = 1 - Math.exp(-dt * 10);
   350→      // Ritmos del catálogo (rules.view): crucero de vecinos y subida por pozo.
   351→      const npcSpeed = v.npcWalkSpeed;
   352→      for (const [id, a] of actors) {
   353→        const r = world.residents[id]!;
   354→        const isPlayer = id === playerResidentId;
   355→        const nowEnabled = isPlayer || (r.present && r.location !== null);
   356→        // ── TRASTEROS POR COLUMNA (v1.18): mientras el vecino sigue "en su bloque de
   357→        //  trasteros", el nodo permanece oculto — está DENTRO de su celda cerrada. En cuanto
   358→        //  el sim lo muda (otra ubicación o se va), sale del stow y reaparece por su puerta.
   359→        const stowedHere = a.stowedBlock !== null && r.present && r.location === a.stowedBlock;
   360→        if (a.stowedBlock !== null && !stowedHere) a.stowedBlock = null;
   361→        a.node.setEnabled(nowEnabled && !stowedHere);
   362→        if (!nowEnabled || stowedHere) {
   363→          a.enabledPrev = false;
   364→          continue;
   365→        }
   366→        // En escena lo coloca el director (v1.10): marcar habilitado SIN snap — si no, al soltar
   367→        // al actor tras una cutscene, el primer frame normal lo trataría como "reaparición"
   368→        // y lo teletransportaría al objetivo en vez de rutear desde donde está.
   369→        if (a.directed) a.enabledPrev = true;
   370→        const node = a.node.position;
   371→        const prevX = node.x;
   372→        const prevZ = node.z;
   373→        // Charla de cruce (v1.20): mientras dura, el actor NO avanza — se queda donde está,
   374→        // mira a su vecino y conversa. Ni rutas ni objetivos: al soltar, la IA recaptura.
   375→        if (a.chatT > 0) a.chatT -= dt;
   376→        const chatting = a.chatT > 0;
   377→        if (isPlayer) {
   378→          const inside = player.mode === 'inside';
   379→          // Apilamiento 3D (v1.19): dentro del trastero propio el cuerpo vive en el bloque
   380→          // trasero (posición que dicta PlayerLogic), no en el pasillo del sim.
   381→          if (player.rear) {
   382→            goal.set(player.rear.x, player.floor * v.cellH + 0.14 + a.pivotY, player.rear.z);
   383→            if (node.lengthSquared() === 0) node.copyFrom(goal);
   384→            node.addInPlace(goal.subtract(node).scale(Math.min(1, k * 2)));
   385→            // Skipear el resto del bloque isPlayer cayendo al flujo común de posición.
   386→          } else {
   387→          // Dentro de la vivienda el actor sigue la posición 2D (x + profundidad z del layout);
   388→          // en el cubo de escalera (v1.4) manda la Z absoluta del recorrido (pathZ).
   389→          // En pasillo (v1.19) la Z del jugador es ABSOLUTA (dos carriles: interior/exterior).
   390→          const insideBlock = player.insideBlock ? world.blocks[player.insideBlock] : null;
   391→          const insideDepth = insideBlock ? depthOfBlock(insideBlock) : v.depth;
   392→          const z = inside ? frontZ + Math.min(Math.max(player.z, 0), insideDepth) : (player.pathZ ?? player.z);
   393→          goal.set(player.x, player.floor * v.cellH + (inside ? 0.12 : 0.14) + a.pivotY, z);
   394→          if (node.lengthSquared() === 0) node.copyFrom(goal);
   395→          node.addInPlace(goal.subtract(node).scale(Math.min(1, k * 2)));
   396→          }
   397→        } else if (!chatting) {
   398→          const b = world.blocks[r.location!]!;
   399→          const home = world.blocks[r.home]!;
   400→          const inRoom = b.kind === 'dwelling';
   401→          const x0 = b.cell.col * v.cellW;
   402→          // ── Trastero propio (v1.18): si el vecino está en el bloque por columna de su
   403→          //  letra, su objetivo es LA PUERTA DE SU CELDA (pasillo central): llega y entra
   404→          //  (stow: nodo oculto dentro del compartimento cerrado).
   405→          const storeDoor = isStorageColumnBlock(b, v) ? storageDoorOf(r) : null;
   406→          // ── Viaje en ascensor (v1.5): mientras hay viaje registrado, el director manda —
   407→          // esperar ante la puerta, entrar al hueco, ser llevado por la cabina y salir en la
   408→          // planta de destino. Si el viaje se cancela (avería), status() null → el flujo
   409→          // normal recaptura destino y la ruta pasará por la escalera.
   410→          const elev = elevators ?? undefined;
   411→          const ride = a.elevatorReq && elev ? elev.status(id) : null;
   412→          if (a.elevatorReq && !ride) {
   413→            a.elevatorReq = null;
   414→            a.ridePhase = null;
   415→            a.path = [];
   416→          }
   417→          if (a.elevatorReq && ride) {
   418→            if (stalePhase(a, ride.phase)) {
   419→              // Transición de fase: recolocar la cola según a dónde hay que caminar ahora.
   420→              if (ride.phase === 'boarding') {
   421→                a.path = [{ x: ride.slotX, y: Math.round(ride.floor) * v.cellH + 0.14 + a.pivotY, z: ride.slotZ }];
   422→              } else if (ride.phase === 'exiting') {
   423→                a.path = [{ x: a.elevatorReq.x, y: a.elevatorReq.to * v.cellH + 0.14 + a.pivotY, z: v.corridorZ }];
   424→              } else {
   425→                a.path = [];
   426→              }
   427→              a.ridePhase = ride.phase;
   428→            }
   429→            if (ride.phase === 'waiting') {
   430→              goal.set(a.elevatorReq.x, a.elevatorReq.from * v.cellH + 0.14 + a.pivotY, v.corridorZ);
   431→              if (node.lengthSquared() === 0) node.copyFrom(goal);
   432→              node.addInPlace(goal.subtract(node).scale(Math.min(1, k * 2)));
   433→            } else if (ride.phase === 'riding') {
   434→              // La cabina manda: posición directa en el hueco (sin teatro ni interpolación propia).
   435→              a.path = [];
   436→              node.set(ride.slotX, ride.floor * v.cellH + 0.14 + a.pivotY, ride.slotZ);
   437→            } else if (ride.phase === 'boarding' && a.path.length === 0) {
   438→              elev!.notifyBoarded(id); // ya está en su hueco: la puerta puede cerrar
   439→            } else if (ride.phase === 'exiting' && a.path.length === 0) {
   440→              a.elevatorReq = null;
   441→              a.ridePhase = null;
   442→            }
   443→          }
   444→          // Con dirección de cutscene (Fase D) el actor solo consume su cola de tramos:
   445→          // ni objetivo de IA ni captura de ruta (el director ya se la ha dado).
   446→          if (!a.directed && !a.elevatorReq) {
   447→            let gx = b.layer === 'corridor' ? (home.cell.col + 0.5) * v.cellW : (x0 + b.size.w * v.cellW) / 2;
   448→            let gz = inRoom ? frontZ + 0.6 : v.corridorZ;
   449→            if (storeDoor) {
   450→              // Trastero por columna (v1.18): se camina POR EL PASILLO CENTRAL hasta SU puerta.
   451→              gx = storeDoor.x;
   452→              gz = storeDoor.z;
   453→            }
   454→            // ── Vida interior (Fase C): en vivienda, el punto lo decide el contenido. La actividad
   455→            // con `anchor` (cama, tv, stove…) coloca al vecino en SU estancia por el catálogo de
   456→            // anclas (rooms.json); sin ancla, deambula por las estancias abiertas. El objetivo se
   457→            // REFRESCA con la cadencia npcWanderSeconds (no cada tick): una actividad con chance
   458→            // que parpadea en el sim no arrastra al vecino por la casa. Cerradas jamás son destino.
   459→            const geo = inRoom ? layoutGeometryOf(content, b) : null;
   460→            let localTarget: Pt | null = null;
   461→            if (a.lastLoc !== r.location) {
   462→              a.lastLoc = r.location;
   463→              a.targetLocal = null;
   464→              a.wanderT = 0;
   465→              a.seated = false;
   466→            }
   467→            if (geo) {
   468→              a.wanderT -= dt;
   469→              if (a.path.length === 0 && a.wanderT <= 0) {
   470→                a.wanderLap += 1;
   471→                a.wanderT = v.npcWanderSeconds * (0.7 + 0.6 * hash01(id, a.wanderLap));
   472→                const act = r.activity ? content.archetypes[r.archetype]?.activities[r.activity] : undefined;
   473→                const anchorKind = act?.anchor;
   474→                const anchored = anchorKind ? interiorTarget(content, geo, anchorKind, id, 0) : null;
   475→                a.targetLocal = anchored ?? interiorTarget(content, geo, undefined, id, a.wanderLap);
   476→                // Ancla de asiento (catálogo: rules.view.seatAnchors): al llegar se sienta (clip "sit").
   477→                a.seated = anchored !== null && !!anchorKind && v.seatAnchors.includes(anchorKind);
   478→              }
   479→              localTarget = a.targetLocal;
   480→            }
   481→            if (localTarget) {
   482→              gx = x0 + localTarget.x;
   483→              gz = frontZ + localTarget.z;
   484→            }
   485→            goal.set(gx, b.cell.floor * v.cellH + 0.14 + a.pivotY, gz);
   486→            // Reaparición (v1.10): el vecino vuelve de la calle o entra en escena — aparece
   487→            // DONDE DICE EL SIM, no caminando desde su última posición visible (por la pared).
   488→            // CON TRASTERO (v1.18): sale en la línea del pasillo ANTE SU BLOQUE y se lo ve
   489→            // entrar y recorrer el pasillo central hasta su celda (nunca pop dentro del stow).
   490→            if (!a.enabledPrev) {
   491→              a.enabledPrev = true;
   492→              if (storeDoor) {
   493→                node.set((x0 + b.size.w * v.cellW) / 2, goal.y, v.corridorZ);
   494→                a.path = [];
   495→              } else {
   496→                node.copyFrom(goal);
   497→                a.path = [];
   498→              }
   499→            }
   500→            // ── Locomoción interpolada (nada de teletransportes): el destino del sim se recorre a
   501→            // velocidad de crucero del catálogo, tramo a tramo. El sim adelanta a los vecinos a
   502→            // saltos del grafo (varias plantas de golpe): la ruta se RECALCULA DESDE EL PUNTO
   503→            // ACTUAL (andar al pozo → subir/bajar → salir) y el viaje se ve entero, pieza a pieza.
   504→            // Solo un estado realmente imposible (cambio de planta sin pozo utilizable, v1.10)
   505→            // recoloca sin teatro. Con ruta en curso se consume SIEMPRE hasta el final.
   506→            if (a.path.length === 0) {
   507→              // STOW (v1.18): llegó a la puerta de su trastero — entra y se oculta (la celda
   508→              // es cerrada: el interior jamás se ve). enabledPrev=false para que, cuando el
   509→              // sim lo mude, la reaparición lo coloque donde toque.
   510→              if (storeDoor && a.stowedBlock === null && Math.hypot(node.x - storeDoor.x, node.z - storeDoor.z) < 0.35) {
   511→                a.stowedBlock = b.id;
   512→                a.node.setEnabled(false);
   513→                a.enabledPrev = false;
   514→                a.path = [];
   515→                continue;
   516→              }
   517→              const routed = buildRoute(a, { x: goal.x, y: goal.y, z: goal.z, inside: geo && localTarget ? { block: b, local: localTarget } : null });
   518→              if (!routed) {
   519→                node.copyFrom(goal);
   520→                a.path = [];
   521→              }
   522→            }
   523→          }
   524→          if (a.path.length > 0) {
   525→            const w = a.path[0]!;
   526→            const wdh = Math.hypot(w.x - node.x, w.z - node.z);
   527→            const wdy = w.y - node.y;
   528→            if (Math.abs(wdy) > 0.02 && wdh < 0.05) {
   529→              // Tramo VERTICAL PURO (v1.23, umbral 0.05): solo el arranque/parada vertical del
   530→              // fallback sin escalera. El umbral anterior (0.3) atrapaba los PELDAÑOS de la
   531→              // escalera real (run ≈ 0.27 m): el vecino subía en vertical como un ascensor y
   532→              // LUEGO patinaba en horizontal hasta el siguiente peldaño — el "sube mal".
   533→              const shaftHere = shafts
   534→                .filter((s) => s.cell.floor === b.cell.floor && Math.abs(shaftXOf(s) - node.x) < 0.3)
   535→                .sort((s1, s2) => Math.abs(shaftXOf(s1) - node.x) - Math.abs(shaftXOf(s2) - node.x))[0];
   536→              const feat = shaftHere ? content.blocks[shaftHere.defId]!.features : undefined;
   537→              const rate = feat?.elevator && shaftHere!.params.elevator_status === 'ok' ? v.cellH / v.elevatorSeconds : v.cellH / v.stairsSeconds;
   538→              node.y += Math.sign(wdy) * Math.min(Math.abs(wdy), rate * dt);
   539→            } else if (wdh > 0.001) {
   540→              // Tramo con componente horizontal: crucero con llegada suave (frena al acercarse,
   541→              // nunca sobrepasa). Con dy el tramo es 3D (pendiente de escalera, Task 29): el paso
   542→              // reparte X/Z/Y con la MISMA velocidad de crucero, así la subida es física.
   543→              const step = Math.min(npcSpeed * dt, (0.35 + wdh * 2.5) * dt, wdh);
   544→              const t = step / wdh;
   545→              node.x += (w.x - node.x) * t;
   546→              node.z += (w.z - node.z) * t;
   547→              node.y += wdy * t;
   548→            }
   549→            // Tramo terminado → siguiente tramo de la ruta.
   550→            if (Math.hypot(w.x - node.x, w.z - node.z) < 0.06 && Math.abs(w.y - node.y) < 0.03) a.path.shift();
   551→          }
   552→        }
   553→
   554→        // ── Rotación 2.5D (v1.2): el actor MIRA hacia donde camina. El pasillo es lateral (eje
   555→        // X, perfil); entrar/salir de una vivienda es profundidad (eje Z: de espaldas al alejarse,
   556→        // de frente al acercarse). Giro por el arco corto con el ritmo del catálogo
   557→        // (rules.view.actorTurnSpeed). El rumbo queda PENDIENTE hasta rematarse: aunque el actor
   558→        // se pare a mitad de giro, termina de encarar la dirección en la que iba (nadie se queda
   559→        // torcido en una puerta).
   560→        // Aplicación ROBUSTA (v1.3): si el nodo lleva rotationQuaternion (el loader glTF puede
   561→        // ponérselo), Babylon IGNORA rotation.y — hay que escribir el cuaternión; si no,
   562→        // rotation.y de toda la vida. En ambos casos SOLO yaw: nadie se ladea ni se tumba.
   563→        const mdx = node.x - prevX;
   564→        const mdz = node.z - prevZ;
   565→        if (Math.hypot(mdx, mdz) / Math.max(1e-4, dt) > 0.25) a.targetYaw = Math.atan2(-mdz, mdx);
   566→        if (a.targetYaw !== null) {
   567→          let dYaw = a.targetYaw - a.yaw;
   568→          dYaw = ((dYaw + Math.PI) % (2 * Math.PI) + 2 * Math.PI) % (2 * Math.PI) - Math.PI;
   569→          if (Math.abs(dYaw) < 0.02) {
   570→            a.yaw = a.targetYaw; // rematado
   571→            a.targetYaw = null;
   572→          } else {
   573→            a.yaw += dYaw * Math.min(1, dt * v.actorTurnSpeed);
   574→          }
   575→        }
   576→        const q = a.node.rotationQuaternion;
   577→        if (q) Quaternion.RotationYawPitchRollToRef(a.yaw, 0, 0, q);
   578→        else a.node.rotation.y = a.yaw;
   579→
   580→        // ── Animación (solo rigs) por prioridad de estados (v1.2) + SINCRONÍA DE ZANCADA (v1.3):
   581→        //   gesto explícito → run (speed > runSpeed) → walk (mientras se avanza) → clip de
   582→        //   actividad parado → sentado en ancla → idle. El ratio de playback sale de la
   583→        //   velocidad PLANA real (m/s en el suelo; la Y no mueve piernas) contra la velocidad
   584→        //   natural del clip (rules.view.animSync): las piernas acompañan al cuerpo, cero patinaje.
   585→        if (a.gestureT > 0) {
   586→          a.gestureT -= dt;
   587→          if (a.gestureT <= 0) a.gesture = null;
   588→        }
   589→        if (a.rig) {
   590→          const speedH = Math.hypot(node.x - prevX, node.z - prevZ) / Math.max(1e-4, dt);
   591→          const def = content.assets?.actors?.[r.archetype];
   592→          const activityClip = def?.activityAnimations?.[r.activity ?? ''] ?? null;
   593→          const walkClip = def?.walkAnimation ?? 'walk';
   594→          let clip: string | null;
   595→          if (a.gesture) clip = a.gesture;
   596→          else if (speedH > v.runSpeed) clip = 'run';
   597→          else if (speedH > 0.25) clip = walkClip;
   598→          else if (activityClip) clip = activityClip;
   599→          else if (a.seated && a.path.length === 0) clip = 'sit';
   600→          else clip = null;
   601→          const sync = v.animSync;
   602→          let rate = 1;
   603→          if (!a.gesture && clip === 'run') rate = speedH / sync.runRefSpeed;
   604→          else if (!a.gesture && clip === walkClip) rate = speedH / sync.walkRefSpeed;
   605→          rate = Math.min(sync.maxRate, Math.max(sync.minRate, rate));
   606→          a.rig.play(clip, rate);
   607→        }
   608→      }
   609→    },
   610→    /** Dirección de cutscenes (Fase D): toma/coloca/ruta/suelta actores como teatro. */
   611→    director: {
   612→      nodeOf(id: string): { x: number; y: number; z: number } | null {
   613→        const a = actors.get(id);
   614→        return a && a.node.isEnabled() ? { x: a.node.position.x, y: a.node.position.y, z: a.node.position.z } : null;
   615→      },
   616→      take(id: string): boolean {
   617→        const r = world.residents[id];
   618→        if (!r || !r.present || r.location === null) return false;
   619→        if (!actors.has(id)) ensureActor(id, r.archetype);
   620→        const a = actors.get(id)!;
   621→        a.directed = true;
   622→        a.seated = false;
   623→        a.node.setEnabled(true);
   624→        return true;
   625→      },
   626→      release(id: string): void {
   627→        const a = actors.get(id);
   628→        if (!a) return;
   629→        a.directed = false;
   630→        a.path = [];
   631→        a.targetLocal = null;
   632→        a.wanderT = 0;
   633→        a.seated = false;
   634→        a.gesture = null;
   635→        a.gestureT = 0;
   636→        a.targetYaw = null;
   637→        // Si estaba esperando el ascensor, se cancela la llamada (el director libera el hueco).
   638→        if (a.elevatorReq && elevators) {
   639→          elevators.cancel(a.id);
   640→          a.elevatorReq = null;
   641→          a.ridePhase = null;
   642→        }
   643→      },
   644→      stageAt(id: string, x: number, floor: number, z: number): void {
   645→        const a = actors.get(id);
   646→        if (!a) return;
   647→        a.path = [];
   648→        // El teatro manda: cualquier viaje en curso se cancela antes de colocar al actor.
   649→        if (a.elevatorReq && elevators) {
   650→          elevators.cancel(a.id);
   651→          a.elevatorReq = null;
   652→          a.ridePhase = null;
   653→        }
   654→        a.node.position.set(x, floor * v.cellH + 0.14 + a.pivotY, z);
   655→      },
   656→      walkTo(id: string, dest: { x: number; z: number; floor: number; inside?: { block: string; local: Pt } }): boolean {
   657→        const a = actors.get(id);
   658→        if (!a) return false;
   659→        const b = dest.inside ? world.blocks[dest.inside.block] : undefined;
   660→        const inside = b && dest.inside
   661→          ? {
   662→              block: b,
   663→              local: {
   664→                x: Math.min(b.size.w * v.cellW - 0.2, Math.max(0.2, dest.inside.local.x)),
   665→                z: Math.min(depthOfBlock(b) - 0.2, Math.max(0.2, dest.inside.local.z)),
   666→              },
   667→            }
   668→          : null;
   669→        buildRoute(a, { x: dest.x, y: dest.floor * v.cellH + 0.14 + a.pivotY, z: dest.z, inside });
   670→        return true;
   671→      },
   672→      idle(id: string): boolean {
   673→        const a = actors.get(id);
   674→        return !a || (a.path.length === 0 && !a.elevatorReq);
   675→      },
   676→      headOf(id: string): { x: number; y: number; z: number } | null {
   677→        const a = actors.get(id);
   678→        if (!a || !a.node.isEnabled()) return null;
   679→        const p = a.node.position;
   680→        return { x: p.x, y: p.y + a.height + 0.18, z: p.z };
   681→      },
   682→      gestureOf(id: string): string | null {
   683→        return actors.get(id)?.gesture ?? null;
   684→      },
   685→      isDirected(id: string): boolean {
   686→        return actors.get(id)?.directed ?? false;
   687→      },
   688→      isRiding(id: string): boolean {
   689→        return actors.get(id)?.elevatorReq !== null && actors.get(id)?.elevatorReq !== undefined;
   690→      },
   691→      setGesture(id: string, state: string | null, seconds = 0): void {
   692→        const a = actors.get(id);
   693→        if (!a) return;
   694→        a.gesture = state;
   695→        a.gestureT = state && seconds > 0 ? seconds : 0;
   696→      },
   697→      /** Pausa de charla (v1.20): congela la locomoción; 0 suelta. */
   698→      setHold(id: string, seconds: number): void {
   699→        const a = actors.get(id);
   700→        if (!a) return;
   701→        a.chatT = Math.max(0, seconds);
   702→        if (a.chatT === 0) {
   703→          a.targetLocal = null;
   704→          a.wanderT = 0;
   705→        }
   706→      },
   707→      /** Gira hacia un punto del mundo (arco corto con el ritmo del catálogo). */
   708→      faceToward(id: string, x: number, z: number): void {
   709→        const a = actors.get(id);
   710→        if (!a) return;
   711→        const dx = x - a.node.position.x;
   712→        const dz = z - a.node.position.z;
   713→        if (Math.hypot(dx, dz) < 0.05) return;
   714→        a.targetYaw = Math.atan2(-dz, dx);
   715→      },
   716→    },
   717→  };
   718→}
