    72→    72→  async init(canvas: HTMLCanvasElement): Promise<void> {
    73→    73→    if (this.engine || this.initializing) return;
    74→    74→    this.initializing = true;
    75→    75→    this.canvas = canvas;

    80→    80→      this._content = content;
    81→    81→      this.cutscenePlayer = new CutscenePlayer(content);
    82→    82→      preloadSounds();

    91→    91→      engine.setHardwareScalingLevel(QUALITY_SCALE[loadOptions().quality]);
    92→    92→      window.addEventListener('resize', this.onResize);
    93→    93→      await this.restart();
    94→    94→      useGameStore.getState().markReady(backend);
    95→    95→      startGamepadPolling();
    96→    96→      engine.runRenderLoop(this.frame);
    97→    97→      // Gancho de depuración/pruebas E2E (no lo usa la UI).
    98→    98→      (window as unknown as { __vecinos?: unknown }).__vecinos = { store: useGameStore, controller: this };

   102→   102→    }

   120→   120→    if (this.cutscenePlayer?.active) {

   131→   131→    }
   132→   132→    const cam = CAM_BY_KEY[code];
   133→   133→    if (cam) this.setCamMode(cam);

   326→   326→  private connectNet(): void {
   327→   327→    this.net = new NetSession({
   328→   328→      onState: (s) => this.applyRemoteState(s),
   329→   329→      onParticipants: (p) => useGameStore.getState().setMpParticipants(p),
   330→   330→      onJoined: (j) => {
   331→   331→        this.myResidentId = j.yourResidentId;
   332→   332→        this.isHost = j.isHost;
   333→   333→        useGameStore.getState().setMpSession({ code: j.code, isHost: j.isHost, myResidentId: j.yourResidentId });
   334→   334→        useGameStore.getState().setMpParticipants(j.participants);
   335→   335→      },
   336→   336→      onError: (m) => {
   337→   337→        useGameStore.getState().pushToast(m, 'bad');
   338→   338→      },
   339→   339→      onStatus: (st) => {
   340→   340→        if (!this.net) return;
   341→   341→        if (st === 'connected' && useGameStore.getState().mpStatus === 'connecting') return; // el ack decide
   342→   342→        if (st === 'disconnected' && useGameStore.getState().mpStatus === 'connected') {
   343→   343→          useGameStore.getState().setMpStatus('error');
   344→   344→          useGameStore.getState().setMpError('Conexión perdida con el servidor de la comunidad.');
   345→   345→        }
   346→   346→      },
   347→   347→    });
   348→   348→  }

   355→   355→      return;

   357→   357→    store.setMpStatus('connected');

   359→   359→  }

   361→   361→  /** Adopta el estado del servidor (autoridad). Primera vez: reinicia la escena con ese mundo. */
   362→   362→  private applyRemoteState(s: SessionStatePayload): void {
   363→   363→    if (s.seq <= this.appliedSeq) return;
   364→   364→    this.appliedSeq = s.seq;
   365→   365→    if (!this.world || this.world.scenarioId !== s.world.scenarioId) {
   366→   366→      void this.restart(s.world).then(() => {
   367→   367→        if (!this.world) return;
   368→   368→        this.world.events = s.events.slice();
   369→   369→        this.handleEvents();
   370→   370→        this.reconcile();
   371→   371→        this.syncHud();
   372→   372→      });
   373→   373→      return;
   374→   374→    }
   375→   375→    // Aplica en sitio conservando la identidad del objeto (PlayerLogic/building/actors referencian el mismo World).
   376→   376→    Object.assign(this.world, s.world);
   377→   377→    this.world.events = s.events.slice();
   378→   378→    this.handleEvents();
   379→   379→    this.paintBuilding();
   380→   380→    this.reconcile();
   381→   381→    this.syncHud();
   382→   382→  }

   384→   384→  /** Corrige desvíos de posición del avatar local respecto al mundo del servidor (autoridad). */
   385→   385→  private reconcile(): void {
   386→   386→    if (!this.logic || !this.world || !this.net) return;
   387→   387→    const me = this.world.residents[this.actorId()];
   388→   388→    if (!me) return;
   389→   389→    const pos = this.logic.position();
   390→   390→    const locBlock = me.location ? this.world.blocks[me.location] : null;
   391→   391→    if (locBlock) {
   392→   392→      const insideOk = pos.mode === 'inside' && pos.insideBlock === locBlock.id;
   393→   393→      const corridorOk = pos.mode === 'corridor' && !pos.insideBlock;
   394→   394→      if (!(locBlock.kind === 'dwelling' ? insideOk : corridorOk) || Math.abs(pos.floor - locBlock.cell.floor) > 0.01) {
   395→   395→        this.logic.forceLocation(locBlock.id);
   396→   396→      }
   397→   397→    } else if (pos.mode === 'inside') {
   398→   398→      this.logic.forceLocation(null);
   399→   399→    }
   400→   400→  }

   402→   402→  /** Siguiente frase del teléfono; al colgar responde la llamada en el sim y saca la siguiente de la cola. */
   403→   403→  phoneNext(): void {
   404→   404→    if (this.phoneLine < this.phoneLines.length - 1) {
   405→   405→      this.phoneLine += 1;
   406→   406→      this.showPhoneLine();
   407→   407→      return;
   408→   408→    }

   546→   546→  /** (Re)crea el mundo y la escena. Con `state` restaura una partida guardada. */
   547→   547→  private async restart(state?: World): Promise<void> {
   548→   548→    if (!this.engine || !this.canvas || !this._content) return;
   549→   549→    // Ninguna cutscene sobrevive a un cambio de mundo: cola fuera y teatro desmontado.
   550→   550→    this.cutsceneQueue = [];
   551→   551→    this.cutscenePlayer?.abort();
   552→   552→    const { Color4, DirectionalLight, HemisphericLight, Scene, UniversalCamera, Vector3 } = B();
   553→   553→    const view = this.content.rules.view;
   554→   554→    useGameStore.getState().resetUi();
   555→   555→
   556→   556→    this.world = state ?? createWorld(this.content, this.activeScenarioId());
   557→   557→    this.scenarioOverrides = this.content.scenarios[this.world.scenarioId ?? '']?.minigameOverrides ?? {};
   558→   558→    // Guardados antiguos sin `entry`: la entrada es la zona común de planta baja.
   559→   559→    if (!this.world.entry) {
   560→   560→      this.world.entry = Object.values(this.world.blocks).find((b) => b.cell.floor === 0 && b.kind !== 'dwelling')?.id ?? Object.keys(this.world.blocks)[0]!;
   561→   561→    }
   562→   562→    // Migración de estancias de guardados v0.1: params y ranuras de muebles según catálogo.
   563→   563→    for (const b of Object.values(this.world.blocks)) migrateRooms(this.content, b.rooms);
   564→   564→    this.logic = new PlayerLogic(this.world, this.content, {
   565→   565→      actorId: this.myResidentId ?? undefined,
   566→   566→      send: (cmd) => this.sendCommand(cmd),
   567→   567→    });
   568→   568→
   569→   569→    this.scene?.dispose();
   570→   570→    this.scene = new Scene(this.engine);
   571→   571→    this.scene.clearColor = new Color4(view.sky.r, view.sky.g, view.sky.b, 1);
   572→   572→    new HemisphericLight('ambient', new Vector3(...view.ambient.dir), this.scene).intensity = view.ambient.intensity;
   573→   573→    new DirectionalLight('sun', new Vector3(...view.sun.dir), this.scene).intensity = view.sun.intensity;
   574→   574→
   575→   575→    this.building = createBuilding(this.scene, this.world, this.content);
   576→   576→    this.actors = createActors(this.scene, this.world, this.content);
   577→   577→    const blocks = Object.values(this.world.blocks);
   578→   578→    this.rig = new CameraRig(
   579→   579→      new UniversalCamera('cam', new Vector3(0, 0, -30), this.scene),
   580→   580→      this.engine,
   581→   581→      Math.max(...blocks.map((b) => b.cell.col + b.size.w)),
   582→   582→      Math.max(...blocks.map((b) => b.cell.floor)) + 1,
   583→   583→      view.cellW,
   584→   584→      view.cellH,
   585→   585→    );
   586→   586→    // Re-sincronización de vista (v0.6-b): la cámara sigue la posición del mundo (restaurado o
   587→   587→    // nuevo), no la preferencia anterior. Dentro de una vivienda → estancia; en pasillo, la
   588→   588→    // preferencia salvo que fuera 'estancia' (ahí estás en el pasillo, no en una estancia).
   589→   589→    const pos = this.logic.position();
   590→   590→    const store = useGameStore.getState();
   591→   591→    const wanted: CamMode = pos.mode === 'inside' ? 'estancia' : store.camMode === 'estancia' ? 'planta' : store.camMode;
   592→   592→    this.rig.mode = wanted;
   593→   593→    store.setCamMode(wanted);
   594→   594→
   595→   595→    this.clockAcc = 0;
   596→   596→    this.keys.clear();
   597→   597→    this.phoneLines = [];
   598→   598→    this.phoneMission = '';
   599→   599→    this.phoneQueue = [];
   600→   600→    this.lastHudSync = 0;
   601→   601→
   602→   602→    // Re-llamada (Fase F): si la partida montada tenía pasos de teléfono pendientes, vuelven a
   603→   603→    // sonar. La cola de openPhone ordena el resto sin que ninguna llamada se pierda.
   604→   604→    if (this.world) {
   605→   605→      for (const m of this.world.missions) {
   606→   606→        if (m.state !== 'active') continue;
   607→   607→        const step = this.content.missions[m.id]?.steps[m.step];
   608→   608→        if (step?.type === 'phone_call') this.openPhone(m.id, step.dialog);
   609→   609→      }
   610→   610→    }
   611→   611→
   612→   612→    this.pump();
   613→   613→  }

   619→   619→  // ───────────────────────── Bucle de render ─────────────────────────
   620→   620→
   621→   621→  private frame = (): void => {
   622→   622→    if (!this.engine || !this.scene || !this.world || !this.logic || !this.building || !this.actors || !this.rig) return;
   623→   623→    const dt = Math.min(0.1, this.engine.getDeltaTime() / 1000);
   624→   624→    const store = useGameStore.getState();
   625→   625→    const frozen = selectModalOpen(store) || store.paused;
   626→   626→
   627→   627→    let ticked = false;
   628→   628→    if (!frozen) {
   629→   629→      this.logic.update(dt, this.readInput());
   630→   630→      // En remoto el servidor es la autoridad del reloj: nada de ticks locales.
   631→   631→      if (!this.net) {
   632→   632→        this.clockAcc += dt * store.speed;
   633→   633→        let guard = 0;
   634→   634→        while (this.clockAcc >= 1 && guard < 60) {
   635→   635→          tick(this.world, this.content);
   636→   636→          this.clockAcc -= 1;
   637→   637→          ticked = true;
   638→   638→          guard += 1;
   639→   639→        }
   640→   640→      }
   641→   641→    }
   642→   642→
   643→   643→    if (this.world.events.length > 0) this.handleEvents();
   644→   644→    if (ticked) this.paintBuilding();
   645→   645→
   646→   646→    const now = performance.now();
   647→   647→    if (now - this.lastHudSync > 200) {
   648→   648→      this.lastHudSync = now;
   649→   649→      useGameStore.getState().setFps(Math.round(this.engine.getFps()));
   650→   650→      this.syncHud();
   651→   651→    }
   652→   652→
   653→   653→    const pos = this.logic.position();
   654→   654→    this.actors.update(dt, pos, this.actorId());
   655→   655→    // Cutscene (Fase D): sus beats mueven cámara/actores DESPUÉS de los eventos y ANTES del render.
   656→   656→    this.cutscenePlayer?.update(dt);
   657→   657→    // Si la cutscene acaba y queda cola, encadena la siguiente aquí: no dependemos de que el sim
   658→   658→    // genere otro evento para descolgar la siguiente (durante el teatro el sim está congelado).
   659→   659→    if (this.cutsceneQueue.length > 0 && !this.cutscenePlayer?.active) this.pumpCutscene();
   660→   660→    // LOD por distancia: solo con cámaras cercanas (planta/estancia); el plano general (edificio) muestra todo.
   661→   661→    this.building.update(dt, useGameStore.getState().camMode === 'edificio' ? undefined : { x: pos.x, floor: pos.floor });
   662→   662→    this.rig.update(dt, pos);
   663→   663→    this.scene.render();
   664→   664→  };

   782→   782→        case 'junta_result':
   783→   783→          store.pushToast(
   784→   784→            d.approved ? `Aprobado: ${String(d.title)} (${String(d.yes)} sí / ${String(d.no)} no)` : `Rechazado: ${String(d.title)} (${String(d.yes)} sí / ${String(d.no)} no)`,
   785→   785→            d.approved ? ('good' as Tone) : ('bad' as Tone),
   786→   786→          );
   787→   787→          break;
   788→   788→        case 'junta_vote_registered':
   789→   789→          store.pushLog(`🗳️ ${this.nameOf(d.resident)} ha depositado su voto. Pendientes: ${String(d.pendingHumans)}`, 'info');
   790→   790→          break;
   791→   791→        case 'junta_closed':
   792→   792→          store.setJunta(null);
   793→   793→          break;
   794→   794→        case 'minigame_requested': {
   795→   795→          const def = this.content.minigames[String(d.minigame)];
   796→   796→          if (!def) break;
   797→   797→          // Overrides numéricos del escenario activo (QA/E2E): solo config, nunca efectos ni textos.
   798→   798→          const ov = this.scenarioOverrides[def.id];
   799→   799→          const effective = ov ? ({ ...def, config: { ...def.config, ...ov } } as MinigameDef) : def;
   800→   800→          store.setMinigame({ ...effective, emoji: effective.emoji ?? '🎮', instruction: effective.instruction ?? null });
   801→   801→          break;
   802→   802→        }
   803→   803→        case 'minigame_finished': {
   804→   804→          const name = this.content.minigames[String(d.minigame)]?.name ?? String(d.minigame);
   805→   805→          store.pushLog(d.success ? `🎮 Minijuego superado: ${name}` : `🎮 Minijuego fallido: ${name}`, d.success ? 'good' : 'bad');
   806→   806→          break;
   807→   807→        }
   808→   808→        case 'ui_toast':
   809→   809→          store.pushToast(String(d.text), (d.tone as Tone | undefined) ?? 'info');
   810→   810→          break;
   811→   811→        case 'finale':
   812→   812→          store.setFinale(String(d.text));
   813→   813→          break;
   814→   814→        case 'funds_changed': {
   815→   815→          const amount = typeof d.amount === 'number' ? d.amount : 0;
   816→   816→          if ((d.to === 'player' || d.to === 'community') && amount !== 0) {
   817→   817→            store.pushLog(amount > 0 ? `+${amount} € (fondos)` : `${amount} € (fondos)`, amount > 0 ? 'good' : 'bad');
   818→   818→          }
   819→   819→          break;
   820→   820→        }
   821→   821→        default: {
   822→   822→          // Umbral de canal con texto del catálogo (odors, wifi y futuros canales): totalmente data-driven.
   823→   823→          if (d.uiLog === true && typeof d.message === 'string') {
   824→   824→            store.pushLog(String(d.message).replace('{block}', blockLabel(String(d.block))), (d.tone as Tone | undefined) ?? 'info');
   825→   825→          }
   826→   826→          break; // resident_moved, param_changed, activity_ended... demasiado ruido para el registro
   827→   827→        }
   828→   828→      }
   829→   829→      // Cutscenes (Fase D): después del reparto, ¿alguna quiere este evento como disparador?
   830→   830→      this.maybeCutscene(e);
   831→   831→    }
   832→   832→    this.pumpCutscene();
   833→   833→  }

   837→   837→  /** Si el evento encaja con el trigger de alguna cutscene del catálogo, se encola (máx. 3). */
   838→   838→  private maybeCutscene(e: GameEvent): void {
   839→   839→    const defs = Object.values(this.content.cutscenes);
   840→   840→    if (defs.length === 0) return;
   841→   841→    const d = e.data as Record<string, unknown>;
   842→   842→    const hit = defs.find((c) => {
   843→   843→      const t = c.trigger;
   844→   844→      if (t.on !== e.type) return false;
   845→   845→      if (t.mission !== undefined && t.mission !== String(d.mission ?? '')) return false;
   846→   846→      if (t.step !== undefined && t.step !== String(d.step ?? '')) return false;
   847→   847→      return true;
   848→   848→    });
   849→   849→    if (!hit) return;
   850→   850→    if (this.cutsceneQueue.length >= 3) return; // nunca más de 3 esperando: el teatro no atravaca la partida
   851→   851→    this.cutsceneQueue.push({
   852→   852→      def: hit,
   853→   853→      ctx: {
   854→   854→        mission: typeof d.mission === 'string' ? d.mission : null,
   855→   855→        resident: typeof d.resident === 'string' ? d.resident : null,
   856→   856→        block: typeof d.block === 'string' ? d.block : null,
   857→   857→      },
   858→   858→    });
   859→   859→  }

   861→   861→  /** Lanza la siguiente cutscene en cola si no hay nada en pantalla (también un modal la pospone). */
   862→   862→  private pumpCutscene(): void {
   863→   863→    if (!this.cutscenePlayer || !this.world || !this.rig || !this.actors || !this.engine || !this.logic) return;
   864→   864→    if (this.cutscenePlayer.active || this.cutsceneQueue.length === 0) return;
   865→   865→    if (selectModalOpen(useGameStore.getState())) return; // no se roba la pantalla a un modal
   866→   866→    const next = this.cutsceneQueue.shift()!;
   867→   867→    this.cutscenePlayer.start(next.def, next.ctx, {
   868→   868→      engine: this.engine,
   869→   869→      rig: this.rig,
   870→   870→      actors: this.actors.director,
   871→   871→      world: this.world,
   872→   872→      playerPos: () => this.logic!.position(),
   873→   873→    });
   874→   874→  }

   878→   878→  private openPhone(mission: string, dialogId: string): void {
   879→   879→    const dlg = this.content.dialogs[dialogId];
   880→   880→    if (!dlg || dlg.lines.length === 0) return;
   881→   881→    // Si ya hay una llamada en pantalla, la nueva espera en cola (dedupe por misión).
   882→   882→    if (this.phoneMission) {
   883→   883→      if (this.phoneMission !== mission && !this.phoneQueue.some((q) => q.mission === mission)) {
   884→   884→        this.phoneQueue.push({ mission, dialogId });
   885→   885→      }
   886→   886→      return;
   887→   887→    }
   888→   888→    this.phoneMission = mission;
   889→   889→    this.phoneLines = dlg.lines;
   890→   890→    this.phoneLine = 0;
   891→   891→    this.actors?.director.setGesture(this.actorId(), 'phone'); // el jugador atiende con el móvil a la oreja
   892→   892→    this.showPhoneLine();

   910→   910→  private objectiveUi(): { title: string; text: string } | null {
   911→   911→    if (!this.world) return null;
   912→   912→    const o = activeObjective(this.world, this.content);
   913→   913→    if (!o) return null;
   914→   914→    if (o.state === 'awaiting_choice') return { title: o.title, text: 'El edificio contiene la respiración. Decide qué hacer.' };
   915→   915→    const s = o.step!;
   916→   916→    const left = o.untilTick !== null ? ` · quedan ${Math.max(0, o.untilTick - this.world.tick)} min` : '';
   917→   917→    switch (s.type) {
   918→   918→      case 'phone_call':
   919→   919→        return { title: o.title, text: 'Suena el teléfono. Atiéndelo.' };
   920→   920→      case 'reach':
   921→   921→        return { title: o.title, text: `Llega a la puerta (${this.blockName(s.block ?? '')}).` };
   922→   922→      case 'enter':
   923→   923→        return { title: o.title, text: 'Entra en tu piso con E.' };
   924→   924→      case 'catch_in_act':
   925→   925→        return { title: o.title, text: `Pilla al responsable in fraganti${left}.` };
   926→   926→      default:
   927→   927→        return { title: o.title, text: o.title };
   928→   928→    }
   929→   929→  }

   931→   931→  private promptText(): string | null {
   932→   932→    const logic = this.logic;
   933→   933→    const world = this.world;
   934→   934→    if (!logic || !world) return null;
   935→   935→    if (logic.position().climbing) return null;
   936→   936→    // El prompt es SIEMPRE el objetivo real de la tecla E (misma fuente, cero divergencias).
   937→   937→    const t = logic.interactTarget();
   938→   938→    if (t) {
   939→   939→      const ic = this.content.rules.interact;
   940→   940→      switch (t.kind) {
   941→   941→        case 'door': {
   942→   942→          const door = world.blocks[t.blockId]!;
   943→   943→          if (logic.position().mode === 'inside') return ic.exit;
   944→   944→          return door.access === 'locked'
   945→   945→            ? ic.doorLocked.replace('{block}', this.blockName(door.id))
   946→   946→            : ic.doorEnter.replace('{block}', this.blockName(door.id));
   947→   947→        }
   948→   948→        case 'resident':
   949→   949→          return ic.greet.replace('{name}', this.nameOf(t.residentId));
   950→   950→        case 'mailbox':
   951→   951→          return ic.mailbox.prompt;
   952→   952→        case 'noticeboard':
   953→   953→          return ic.noticeboard.prompt;
   954→   954→        case 'laptop':
   955→   955→          return ic.laptop;
   956→   956→        case 'bed':
   957→   957→          return ic.bed;
   958→   958→      }
   959→   959→    }
   960→   960→    const zone = logic.zone();
   961→   961→    const feat = this.content.blocks[zone.defId]!.features;
   962→   962→    if (feat?.stairs || feat?.elevator) {
   963→   963→      const ic = this.content.rules.interact;
   964→   964→      // Lo que dice el prompt es lo que W/S hace aquí: ascensor operativo manda; si no hay
   965→   965→      // ascensor utilizable pero sí escalera, escalera; núcleo averiado sin escalera = aviso.
   966→   966→      if (feat.elevator && world.blocks[zone.id]!.params.elevator_status === 'ok') return ic.elevator;
   967→   967→      if (feat.stairs) return ic.stairs;
   968→   968→      return ic.elevatorBroken;
   969→   969→    }
   970→   970→    return null;
   971→   971→  }

   973→   973→  /** Viviendas del jugador para el editor de interiores (foto serializable, posiciones ya en %). */
   974→   974→  private homesUi() {
   975→   975→    const world = this.world!;
   976→   976→    const cellH = this.content.rules.view.cellH;
   977→   977→    const mine = this.actorId();
   978→   978→    return Object.values(world.blocks)
   979→   979→      .filter((b) => b.kind === 'dwelling' && b.ownerId === mine)
   980→   980→      .sort((a, b) => a.id.localeCompare(b.id))
   981→   981→      .map((b) => ({
   982→   982→        id: b.id,
   983→   983→        label: blockLabel(b.id),
   984→   984→        rooms: b.rooms.map((room) => {
   985→   985→          const rdef = this.content.rooms[room.defId];
   986→   986→          const slots = (rdef?.furnitureSlots ?? []).map((s) => {
   987→   987→            const itemId = room.furniture[s.id] ?? '';
   988→   988→            const fdef = itemId ? this.content.furniture[itemId] : undefined;
   989→   989→            return {
   990→   990→              id: s.id,
   991→   991→              kind: s.kind,
   992→   992→              left: Math.min(92, Math.max(2, (s.pos[0] / Math.max(0.001, rdef?.size.w ?? 1)) * 100)),
   993→   993→              bottom: Math.min(76, Math.max(0, (s.pos[1] / Math.max(0.001, cellH)) * 100)),
   994→   994→              item: fdef ? { id: fdef.id, name: fdef.name, emoji: fdef.emoji ?? '📦', color: fdef.view.color } : null,
   995→   995→            };
   996→   996→          });
   997→   997→          return { key: room.key, label: rdef?.label ?? room.key, access: room.access, slots };
   998→   998→        }),
   999→   999→      }));
  1000→  1000→  }

  1002→  1002→  private syncHud(): void {
  1003→  1003→    const world = this.world;
  1004→  1004→    if (!world) return;
  1005→  1005→    const player = world.residents[this.actorId()] ?? world.residents[world.playerId];
  1006→  1006→    if (!player) return;
  1007→  1007→    const home = player.home;
  1008→  1008→    const params = visibleParams(world, this.content, home).map((p) => ({ name: p.name, label: p.def.label ?? p.name, value: String(p.value) }));
  1009→  1009→    const missions = world.missions.map((m) => {
  1010→  1010→      const def = this.content.missions[m.id]!;
  1011→  1011→      const step = m.state === 'active' ? def.steps[m.step] : undefined;
  1012→  1012→      return {
  1013→  1013→        id: m.id,
  1014→  1014→        title: def.title,
  1015→  1015→        state: m.state,
  1016→  1016→        hint: step?.hint ?? null,
  1017→  1017→        remaining: m.until !== null ? `quedan ${Math.max(0, m.until - world.tick)} min` : null,
  1018→  1018→      };
  1019→  1019→    });
  1020→  1020→    const travel = travelMinutes(buildCirculation(world, this.content), world.entry, home);
  1021→  1021→    const elevatorBlock = Object.values(world.blocks).find((b) => b.params.elevator_status !== undefined);
  1022→  1022→    useGameStore.getState().syncHud({
  1023→  1023→      clock: { ...world.clock },
  1024→  1024→      weather: world.weather.state,
  1025→  1025→      funds: world.community.funds,
  1026→  1026→      wallet: player.wallet,
  1027→  1027→      reputation: player.reputation,
  1028→  1028→      elevator: String(elevatorBlock?.params.elevator_status ?? 'ok'),
  1029→  1029→      homeLabel: blockLabel(home),
  1030→  1030→      travelText: Number.isFinite(travel) ? `del portal a tu puerta: ${travel} min` : null,
  1031→  1031→      objective: this.objectiveUi(),
  1032→  1032→      prompt: this.promptText(),
  1033→  1033→      params,
  1034→  1034→      missions,
  1035→  1035→      homes: this.homesUi(),
  1036→  1036→      furnitureConfig: { refundRatio: this.content.rules.furniture.refundRatio, roomUnlockPrice: this.content.rules.furniture.roomUnlock.price },
  1037→  1037→    });
  1038→  1038→  }
