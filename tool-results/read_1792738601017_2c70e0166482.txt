    50→  async init(canvas: HTMLCanvasElement): Promise<void> {
    51→    if (this.engine || this.initializing) return;
    52→    this.initializing = true;
    53→    this.canvas = canvas;
    54→
    55→    try {
    56→      // 1) Contenido externo + validación (los errores de referencias se avisan, no abortan).
    57→      const { content, issues } = await loadContent();
    58→      this._content = content;
    59→      if (issues.length > 0) {
    60→        useGameStore.getState().pushToast(`Contenido inválido: ${issues.length} errores de validación`, 'bad');
    61→      }
    62→
    63→      // 2) Motor de render (WebGPU con fallback WebGL2).
    64→      const { engine, backend } = await createEngine(canvas);
    65→      this.engine = engine;
    66→      window.addEventListener('resize', this.onResize);
    67→      await this.restart();
    68→      useGameStore.getState().markReady(backend);
    69→      engine.runRenderLoop(this.frame);
    70→      // Gancho de depuración/pruebas E2E (no lo usa la UI).
    71→      (window as unknown as { __vecinos?: unknown }).__vecinos = { store: useGameStore, controller: this };
    72→    } catch (err) {
    73→      console.error('No se pudo arrancar el juego (contenido o motor de render)', err);
    74→      useGameStore.getState().setEngineError(true);
    75→    }
    76→  }

    84→    this.scene = null;

    89→  onKeyDown(code: string): void {
    90→    if (this.keys.has(code)) return;
    91→    this.keys.add(code);
    92→    const store = useGameStore.getState();
    93→    if (selectModalOpen(store)) return; // los modales gestionan sus propias teclas
    94→    if (code === 'KeyE') {
    95→      this.interact();
    96→      return;
    97→    }
    98→    const cam = CAM_BY_KEY[code];
    99→    if (cam) this.setCamMode(cam);
   100→  }

   102→  onKeyUp(code: string): void {

   106→  private readInput(): Input {
   107→    const held = (...codes: string[]) => codes.some((c) => this.keys.has(c));
   108→    return { left: held('KeyA', 'ArrowLeft'), right: held('KeyD', 'ArrowRight'), up: held('KeyW', 'ArrowUp'), down: held('KeyS', 'ArrowDown') };
   109→  }

   225→  async save(): Promise<void> {
   226→    if (!this.world) return;
   227→    const store = useGameStore.getState();
   228→    store.setSaveState('saving');
   229→    drainEvents(this.world); // la cola de eventos no entra en el guardado
   230→    try {
   231→      const res = await fetch('/api/save', {
   232→        method: 'POST',
   233→        headers: { 'Content-Type': 'application/json' },
   234→        body: JSON.stringify({ slot: 'auto', state: this.world }),
   235→      });
   236→      if (!res.ok) throw new Error(String(res.status));
   237→      store.setSaveState('saved', new Date().toLocaleTimeString('es-ES', { hour: '2-digit', minute: '2-digit' }));
   238→      store.pushToast('Partida guardada. El acta queda en el tablón (y en SQLite).', 'good');
   239→    } catch {
   240→      store.setSaveState('error');
   241→      store.pushToast('No se pudo guardar la partida. ¿El administrador se habrá ido a comer?', 'bad');
   242→    }
   243→  }

   245→  async load(): Promise<void> {
   246→    const store = useGameStore.getState();
   247→    try {
   248→      const res = await fetch('/api/save?slot=auto');
   249→      const json = (await res.json()) as { state?: unknown };
   250→      if (!json.state || typeof json.state !== 'object') {
   251→        store.pushToast('Todavía no hay ninguna partida guardada.', 'info');
   252→        return;
   253→      }
   254→      if (!this._content) {
   255→        store.pushToast('El contenido todavía se está cargando. Espera un momento.', 'info');
   256→        return;
   257→      }
   258→      await this.restart(json.state as World);
   259→      store.pushToast('Partida cargada. El vecindario, exactamente como lo dejaste.', 'good');
   260→    } catch {
   261→      store.pushToast('No se pudo cargar la partida.', 'bad');
   262→    }
   263→  }

   267→  /** (Re)crea el mundo y la escena. Con `state` restaura una partida guardada. */
   268→  private async restart(state?: World): Promise<void> {
   269→    if (!this.engine || !this.canvas || !this._content) return;
   270→    const { Color4, DirectionalLight, HemisphericLight, Scene, UniversalCamera, Vector3 } = B();
   271→    const view = this.content.rules.view;
   272→    useGameStore.getState().resetUi();
   273→
   274→    this.world = state ?? createWorld(this.content, this.content.rules.startScenario);
   275→    // Guardados antiguos sin `entry`: la entrada es la zona común de planta baja.
   276→    if (!this.world.entry) {
   277→      this.world.entry = Object.values(this.world.blocks).find((b) => b.cell.floor === 0 && b.kind !== 'dwelling')?.id ?? Object.keys(this.world.blocks)[0]!;
   278→    }
   279→    // Migración de estancias de guardados v0.1: params y ranuras de muebles según catálogo.
   280→    for (const b of Object.values(this.world.blocks)) migrateRooms(this.content, b.rooms);
   281→    this.logic = new PlayerLogic(this.world, this.content);
   282→
   283→    this.scene?.dispose();
   284→    this.scene = new Scene(this.engine);
   285→    this.scene.clearColor = new Color4(view.sky.r, view.sky.g, view.sky.b, 1);
   286→    new HemisphericLight('ambient', new Vector3(...view.ambient.dir), this.scene).intensity = view.ambient.intensity;
   287→    new DirectionalLight('sun', new Vector3(...view.sun.dir), this.scene).intensity = view.sun.intensity;
   288→
   289→    this.building = createBuilding(this.scene, this.world, this.content);
   290→    this.actors = createActors(this.scene, this.world, this.content);
   291→    const blocks = Object.values(this.world.blocks);
   292→    this.rig = new CameraRig(
   293→      new UniversalCamera('cam', new Vector3(0, 0, -30), this.scene),
   294→      this.engine,
   295→      Math.max(...blocks.map((b) => b.cell.col + b.size.w)),
   296→      Math.max(...blocks.map((b) => b.cell.floor)) + 1,
   297→      view.cellW,
   298→      view.cellH,
   299→    );
   300→    this.rig.mode = useGameStore.getState().camMode;
   301→
   302→    this.clockAcc = 0;
   303→    this.keys.clear();
   304→    this.phoneLines = [];
   305→    this.phoneMission = '';
   306→    this.lastHudSync = 0;
   307→    this.pump();
   308→  }

   314→  // ───────────────────────── Bucle de render ─────────────────────────
   315→
   316→  private frame = (): void => {
   317→    if (!this.engine || !this.scene || !this.world || !this.logic || !this.building || !this.actors || !this.rig) return;
   318→    const dt = Math.min(0.1, this.engine.getDeltaTime() / 1000);
   319→    const store = useGameStore.getState();
   320→    const frozen = selectModalOpen(store) || store.paused;
   321→
   322→    let ticked = false;
   323→    if (!frozen) {
   324→      this.logic.update(dt, this.readInput());
   325→      this.clockAcc += dt * store.speed;
   326→      let guard = 0;
   327→      while (this.clockAcc >= 1 && guard < 60) {
   328→        tick(this.world, this.content);
   329→        this.clockAcc -= 1;
   330→        ticked = true;
   331→        guard += 1;
   332→      }
   333→    }
   334→
   335→    if (this.world.events.length > 0) this.handleEvents();
   336→    if (ticked) this.paintBuilding();
   337→
   338→    const now = performance.now();
   339→    if (now - this.lastHudSync > 200) {
   340→      this.lastHudSync = now;
   341→      this.syncHud();
   342→    }
   343→
   344→    const pos = this.logic.position();
   345→    this.actors.update(dt, pos);
   346→    this.building.update(dt);
   347→    this.rig.update(dt, pos);
   348→    this.scene.render();
   349→  };

   356→    this.syncHud();

   380→  private handleEvents(): void {
   381→    if (!this.world) return;
   382→    const store = useGameStore.getState();
   383→    for (const e of drainEvents(this.world)) {
   384→      const d = e.data as Record<string, unknown>;
   385→      switch (e.type) {
   386→        case 'step_started':
   387→          if (d.type === 'phone_call') this.openPhone(String(d.mission), String(d.dialog));
   388→          break;
   389→        case 'choice_required': {
   390→          const choices = (d.choices as { id: string; label: string; hint?: string | null }[] | undefined) ?? [];
   391→          store.setChoices({ mission: String(d.mission), title: String(d.title), choices: choices.map((c) => ({ id: c.id, label: c.label, hint: c.hint ?? null })) });
   392→          store.pushLog(`⚖️ Decisión pendiente: ${String(d.title)}`, 'info');
   393→          break;
   394→        }
   395→        case 'door_locked': {
   396→          this.building?.shake(String(d.block));
   397→          const price = typeof d.price === 'number' ? d.price : null;
   398→          store.pushToast(`${blockLabel(String(d.block))}: puerta con candado${d.reason === 'for_sale' && price !== null ? ` (a la venta por ${price} €)` : ''}`, 'info');
   399→          break;
   400→        }
   401→        case 'neighbor_arrived':
   402→          store.pushToast(`Nuevo vecino en ${blockLabel(String(d.block))}: ${this.nameOf(d.resident)}`, 'good');
   403→          store.pushLog(`${this.nameOf(d.resident)} se muda a ${blockLabel(String(d.block))}. El edificio ya no es lo que era. Era mejor.`, 'info');
   404→          break;
   405→        case 'mission_activated':
   406→          store.pushLog(`📋 Nueva misión: ${String(d.title)}`, 'info');
   407→          break;
   408→        case 'mission_completed':
   409→          store.pushToast(`Misión completada: ${this.titleOf(d.mission)}`, 'good');
   410→          store.pushLog(`✔ Misión completada: ${this.titleOf(d.mission)}`, 'good');
   411→          break;
   412→        case 'mission_failed':
   413→          store.pushToast(`Misión fallida: ${this.titleOf(d.mission)}`, 'bad');
   414→          store.pushLog(`✘ Misión fallida: ${this.titleOf(d.mission)}`, 'bad');
   415→          break;
   416→        case 'block_unlocked':
   417→          store.pushToast(`${blockLabel(String(d.block))} desbloqueado`, 'good');
   418→          break;
   419→        case 'block_bought':
   420→          store.pushLog(`🔑 ${blockLabel(String(d.block))} comprado por ${String(d.price)} €`, 'good');
   421→          break;
   422→        case 'block_for_sale':
   423→          store.pushLog(`🪧 Cartel de "SE VENDE" en ${blockLabel(String(d.block))} (${String(d.price)} €)`, 'info');
   424→          break;
   425→        case 'noise_complaint': // texto data-driven: lo pinta el uiLog del catálogo (case mantenido por compatibilidad de eventos antiguos)
   426→          if (d.uiLog === true) break;
   427→          store.pushLog(`🔊 Ruido por encima del límite en ${blockLabel(String(d.block))}. Alguien aprieta los dientes.`, 'bad');
   428→          break;
   429→        case 'leak_detected':
   430→          if (d.uiLog === true) break;
   431→          store.pushLog(`💧 Goteo detectado en ${blockLabel(String(d.block))}. El cubo del salón hace ojitos.`, 'bad');
   432→          break;
   433→        case 'furniture_placed':
   434→          store.pushLog(`${String(d.emoji ?? '📦')} ${String(d.name)} colocado en ${blockLabel(String(d.block))} (${String(d.room)}).`, 'info');
   435→          break;
   436→        case 'furniture_removed':
   437→          store.pushLog(`${String(d.emoji ?? '📦')} ${String(d.name)} retirado de ${blockLabel(String(d.block))} (+${String(d.refund)} €).`, 'info');
   438→          break;
   439→        case 'room_renovated':
   440→          store.pushLog(`🧱 Reforma terminada en ${blockLabel(String(d.block))}. Hay estancia nueva que amueblar.`, 'good');
   441→          break;
   442→        case 'weather_changed':
   443→          store.pushLog(`El cielo cambia: ${this.content.rules.labels.weather[String(d.state)] ?? String(d.state)}`, 'info');
   444→          break;
   445→        case 'rent_collected':
   446→          store.pushToast(`+${String(d.amount)} € de alquiler (${blockLabel(String(d.block))})`, 'good');
   447→          break;
   448→        case 'work_done':
   449→          store.pushLog(`💻 Cerrada una "reunión que podía ser un correo": +${String(d.amount)} €`, 'info');
   450→          break;
   451→        case 'activity_started':
   452→          if (d.activity !== 'idle') store.pushLog(`${this.nameOf(d.resident)} ${this.activityLabel(d.resident, String(d.activity))}`, 'info');
   453→          break;
   454→        case 'junta_requested': {
   455→          const agenda = (d.agenda as { id: string; title: string; description: string; cost: number }[] | undefined) ?? [];
   456→          store.setJunta(agenda.map((i) => ({ id: i.id, title: i.title, description: i.description, cost: i.cost })));
   457→          break;
   458→        }
   459→        case 'junta_result':
   460→          store.pushToast(
   461→            d.approved ? `Aprobado: ${String(d.title)} (${String(d.yes)} sí / ${String(d.no)} no)` : `Rechazado: ${String(d.title)} (${String(d.yes)} sí / ${String(d.no)} no)`,
   462→            d.approved ? ('good' as Tone) : ('bad' as Tone),
   463→          );
   464→          break;
   465→        case 'minigame_requested': {
   466→          const def = this.content.minigames[String(d.minigame)];
   467→          if (!def) break;
   468→          store.setMinigame({ ...def, emoji: def.emoji ?? '🎮', instruction: def.instruction ?? null });
   469→          break;
   470→        }
   471→        case 'minigame_finished': {
   472→          const name = this.content.minigames[String(d.minigame)]?.name ?? String(d.minigame);
   473→          store.pushLog(d.success ? `🎮 Minijuego superado: ${name}` : `🎮 Minijuego fallido: ${name}`, d.success ? 'good' : 'bad');
   474→          break;
   475→        }
   476→        case 'ui_toast':
   477→          store.pushToast(String(d.text), (d.tone as Tone | undefined) ?? 'info');
   478→          break;
   479→        case 'finale':
   480→          store.setFinale(String(d.text));
   481→          break;
   482→        case 'funds_changed': {
   483→          const amount = typeof d.amount === 'number' ? d.amount : 0;
   484→          if ((d.to === 'player' || d.to === 'community') && amount !== 0) {
   485→            store.pushLog(amount > 0 ? `+${amount} € (fondos)` : `${amount} € (fondos)`, amount > 0 ? 'good' : 'bad');
   486→          }
   487→          break;
   488→        }
   489→        default: {
   490→          // Umbral de canal con texto del catálogo (odors, wifi y futuros canales): totalmente data-driven.
   491→          if (d.uiLog === true && typeof d.message === 'string') {
   492→            store.pushLog(String(d.message).replace('{block}', blockLabel(String(d.block))), (d.tone as Tone | undefined) ?? 'info');
   493→          }
   494→          break; // resident_moved, param_changed, activity_ended... demasiado ruido para el registro
   495→        }
   496→      }
   497→    }
   498→  }

   502→  private openPhone(mission: string, dialogId: string): void {
   503→    const dlg = this.content.dialogs[dialogId];
   504→    if (!dlg || dlg.lines.length === 0) return;
   505→    this.phoneMission = mission;
   506→    this.phoneLines = dlg.lines;
   507→    this.phoneLine = 0;
   508→    this.showPhoneLine();
   509→  }
   510→
   511→  private showPhoneLine(): void {
   512→    const line = this.phoneLines[this.phoneLine];
   513→    if (!line) return;
   514→    useGameStore.getState().setPhone({ mission: this.phoneMission, speaker: line.speaker, text: line.text, isLast: this.phoneLine === this.phoneLines.length - 1 });
   515→  }

   517→  // ───────────────────────── HUD ─────────────────────────
   518→
   519→  private objectiveUi(): { title: string; text: string } | null {
   520→    if (!this.world) return null;
   521→    const o = activeObjective(this.world, this.content);
   522→    if (!o) return null;
   523→    if (o.state === 'awaiting_choice') return { title: o.title, text: 'El edificio contiene la respiración. Decide qué hacer.' };
   524→    const s = o.step!;
   525→    const left = o.untilTick !== null ? ` · quedan ${Math.max(0, o.untilTick - this.world.tick)} min` : '';
   526→    switch (s.type) {
   527→      case 'phone_call':
   528→        return { title: o.title, text: 'Suena el teléfono. Atiéndelo.' };
   529→      case 'reach':
   530→        return { title: o.title, text: `Llega a la puerta (${blockLabel(s.block ?? '')}).` };
   531→      case 'enter':
   532→        return { title: o.title, text: 'Entra en tu piso con E.' };
   533→      case 'catch_in_act':
   534→        return { title: o.title, text: `Pilla al responsable in fraganti${left}.` };
   535→      default:
   536→        return { title: o.title, text: o.title };
   537→    }
   538→  }
   539→
   540→  private promptText(): string | null {
   541→    const logic = this.logic;
   542→    const world = this.world;
   543→    if (!logic || !world) return null;
   544→    const p = logic.position();
   545→    if (p.mode === 'inside') return 'E · salir a la escalera';
   546→    if (p.climbing) return null;
   547→    const door = logic.doorInFront();
   548→    if (door) return door.access === 'locked' ? `E · ${blockLabel(door.id)}: candado` : `E · entrar en ${blockLabel(door.id)}`;
   549→    const zone = logic.zone();
   550→    if (this.content.blocks[zone.defId]!.features?.stairs) {
   551→      return world.blocks[zone.id]!.params.elevator_status === 'ok' ? 'W/S · ascensor' : 'W/S · escalera (el ascensor no funciona)';
   552→    }
   553→    return null;
   554→  }
   555→
   556→  /** Viviendas del jugador para el editor de interiores (foto serializable, posiciones ya en %). */
   557→  private homesUi() {
   558→    const world = this.world!;
   559→    const cellH = this.content.rules.view.cellH;
   560→    return Object.values(world.blocks)
   561→      .filter((b) => b.kind === 'dwelling' && b.ownerId === world.playerId)
   562→      .sort((a, b) => a.id.localeCompare(b.id))
   563→      .map((b) => ({
   564→        id: b.id,
   565→        label: blockLabel(b.id),
   566→        rooms: b.rooms.map((room) => {
   567→          const rdef = this.content.rooms[room.defId];
   568→          const slots = (rdef?.furnitureSlots ?? []).map((s) => {
   569→            const itemId = room.furniture[s.id] ?? '';
   570→            const fdef = itemId ? this.content.furniture[itemId] : undefined;
   571→            return {
   572→              id: s.id,
   573→              kind: s.kind,
   574→              left: Math.min(92, Math.max(2, (s.pos[0] / Math.max(0.001, rdef?.size.w ?? 1)) * 100)),
   575→              bottom: Math.min(76, Math.max(0, (s.pos[1] / Math.max(0.001, cellH)) * 100)),
   576→              item: fdef ? { id: fdef.id, name: fdef.name, emoji: fdef.emoji ?? '📦', color: fdef.view.color } : null,
   577→            };
   578→          });
   579→          return { key: room.key, label: rdef?.label ?? room.key, access: room.access, slots };
   580→        }),
   581→      }));
   582→  }
   583→
   584→  private syncHud(): void {
   585→    const world = this.world;
   586→    if (!world) return;
   587→    const player = world.residents[world.playerId]!;
   588→    const home = player.home;
   589→    const params = visibleParams(world, this.content, home).map((p) => ({ name: p.name, label: p.def.label ?? p.name, value: String(p.value) }));
   590→    const missions = world.missions.map((m) => {
   591→      const def = this.content.missions[m.id]!;
   592→      const step = m.state === 'active' ? def.steps[m.step] : undefined;
   593→      return {
   594→        id: m.id,
   595→        title: def.title,
   596→        state: m.state,
   597→        hint: step?.hint ?? null,
   598→        remaining: m.until !== null ? `quedan ${Math.max(0, m.until - world.tick)} min` : null,
   599→      };
   600→    });
   601→    const travel = travelMinutes(buildCirculation(world, this.content), world.entry, home);
   602→    const elevatorBlock = Object.values(world.blocks).find((b) => b.params.elevator_status !== undefined);
   603→    useGameStore.getState().syncHud({
   604→      clock: { ...world.clock },
   605→      weather: world.weather.state,
   606→      funds: world.community.funds,
   607→      wallet: player.wallet,
   608→      reputation: player.reputation,
   609→      elevator: String(elevatorBlock?.params.elevator_status ?? 'ok'),
   610→      homeLabel: blockLabel(home),
   611→      travelText: Number.isFinite(travel) ? `del portal a tu puerta: ${travel} min` : null,
   612→      objective: this.objectiveUi(),
   613→      prompt: this.promptText(),
   614→      params,
   615→      missions,
   616→      homes: this.homesUi(),
   617→      furnitureConfig: { refundRatio: this.content.rules.furniture.refundRatio, roomUnlockPrice: this.content.rules.furniture.roomUnlock.price },
   618→    });
   619→  }
   620→}
