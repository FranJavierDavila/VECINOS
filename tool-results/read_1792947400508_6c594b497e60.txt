     1→import type { Color3, Color4, Mesh, Scene, StandardMaterial, TransformNode } from '@babylonjs/core';
     2→import { blockLabel } from '@/game/core/types';
     3→import type { BlockInstance, Content, World } from '@/game/core/types';
     4→import { hash01 } from '@/game/interior';
     5→import { storageCorridorX, storageGeomOf, storageUnitPlan, storageUnitZ } from '@/game/storage';
     6→import { B } from './babylon';
     7→import { instantiateBlockAsset, type BlockMat } from './assets';
     8→
     9→interface BlockView {
    10→  /** Cajas greybox estructurales (se ocultan cuando llega el modelo glTF; panel/candado NO van aquí). */
    11→  greybox: Mesh[];
    12→  /** Materiales que reciben glow de canales y tintado de estado (1 en greybox, N en modelos). */
    13→  mats: BlockMat[];
    14→  /** Emissive base por material (lo que paint() deja sin canales); lo usa el LOD 'glow'. */
    15→  baseEmissive: Color3[];
    16→  /** Color base por material (view.color en greybox; albedo capturado en modelos). */
    17→  baseColor: Color3[];
    18→  /** Fuerza del tintado de estado: 1 = color plano (greybox), 0..1 = mezcla sobre el arte (modelos). */
    19→  tintStrength: number;
    20→  /** Raíz del modelo glTF si este bloque ya lo lleva (null = greybox). */
    21→  model: TransformNode | null;
    22→  front: Mesh | null; // fachada que cierra la vivienda (candado o privacidad)
    23→  /** Puerta y picaporte de la fachada de privacidad (v1.17): la vivienda vacía se ve cerrada. */
    24→  door: Mesh | null;
    25→  handle: Mesh | null;
    26→  lock: Mesh | null;
    27→  lockBase: { x: number; y: number };
    28→  // ── LOD por distancia (rules.view.lod) ──
    29→  /** Centro X del bloque en metros (dx del LOD: |centerX − focus.x|). */
    30→  centerX: number;
    31→  /** Planta del bloque (dy del LOD: |floor − focus.floor| * view.cellH). */
    32→  floor: number;
    33→  /** Pared del fondo de la vivienda (la única malla que puede llevar edgesRendering) o null;
    34→   *  el EdgesRenderer activo solo existe en las paredes destacadas del jugador. */
    35→  edgesMesh: Mesh | null;
    36→  /** true mientras el LOD fuerza baseEmissive (al soltarlo se repinta el cálculo de paint una vez). */
    37→  glowForced: boolean;
    38→  /** true mientras el LOD oculta panel+candado; la visibilidad efectiva es la de paint() && !esto. */
    39→  lodHiddenPanels: boolean;
    40→  /** Segunda pared con edges (v1.19): la pared del fondo se parte si hay puerta a sala trasera. */
    41→  edgesMesh2: Mesh | null;
    42→}
    43→
    44→/** Interior 2D de una vivienda (Fase C): suelo por estancia + tapa de puerta hacia estancia cerrada. */
    45→interface InteriorView {
    46→  /** Suelo tintado por estancia (color del catálogo; lockedColor si está cerrada). */
    47→  floors: { roomKey: string; defId: string; mesh: Mesh; mat: StandardMaterial }[];
    48→  /** Tapas de los huecos de puerta que llevan a una estancia cerrada ("en obras"). */
    49→  covers: { mesh: Mesh; rooms: [string, string] }[];
    50→}
    51→
    52→/**
    53→ * Una caja por bloque (placeholder greybox del futuro prefab) y, si el manifiesto de assets
    54→ * (Content.assets.blocks) declara un modelo glTF para el tipo de bloque, el modelo en su lugar
    55→ * (con la caja oculta). TODA la geometría y los colores salen de content.rules.view +
    56→ * block.view + assets: sustituir el greybox por arte final no toca este código.
    57→ * Las viviendas se dibujan en corte (interior visible) SOLO mientras hay alguien dentro;
    58→ * vacías, la fachada se cierra con su puerta al pasillo (rules.view.privacy) — y las
    59→ * bloqueadas, con panel y candado como siempre.
    60→ */
    61→export function createBuilding(scene: Scene, world: World, content: Content) {
    62→  const { Color3, Color4, MeshBuilder, StandardMaterial, TransformNode } = B();
    63→  const v = content.rules.view;
    64→  const glow = Object.fromEntries(Object.entries(v.channelGlow).map(([k, hex]) => [k, Color3.FromHexString(hex)])) as Record<string, Color3>;
    65→  const views = new Map<string, BlockView>();
    66→  const shakes = new Map<string, number>();
    67→  /** Raíz de escena por bloque (v1.19): colgar aquí TODA malla del bloque permite el
    68→   *  desvanecimiento completo del bloque principal en el apilamiento 3D (swap de bloques). */
    69→  const roots = new Map<string, TransformNode>();
    70→  const attach = (blockId: string, mesh: Mesh): Mesh => {
    71→    const r = roots.get(blockId);
    72→    if (r) mesh.parent = r;
    73→    return mesh;
    74→  };
    75→  /** Interiores 2D (Fase C) por bloque: suelos de estancia y tapas de puertas cerradas. */
    76→  const interiors = new Map<string, InteriorView>();
    77→  /** Foco del LOD (posición del jugador): lo alimenta update() y lo consume paint(). */
    78→  let lastFocus: { x: number; floor: number } | null = null;
    79→  // Plano frontal de las estancias: la cara de la puerta. Con viviendas del doble de calado
    80→  // (v0.6-b), el extra crece hacia el FONDO (+Z) para no envolver el pasillo ni al núcleo.
    81→  // v1.19: los calados del catálogo YA incluyen el +1 módulo (rules.view.depthModule) —
    82→  // depth 4.2 en viviendas y commonDepth 2.8 en comunes son los valores finales.
    83→  const frontZ = v.roomZ - v.commonDepth / 2;
    84→  /** Calado por bloque: viviendas 2× (mobiliario interior); naves del cierre (v1.16) y el
    85→   *  tejado (con aleros) a calado de vivienda; bloques comunes, el de siempre. */
    86→  const depthOf = (b: BlockInstance): number => {
    87→    if (b.kind === 'dwelling') return v.depth;
    88→    if (b.defId === 'trasteros_planta' || b.defId === 'bajocubierta') return v.depth;
    89→    if (b.defId === v.trasteros.blockDefId) return v.depth * v.trasteros.blockDepthFactor;
    90→    if (b.defId === 'tejado') return v.depth + v.roof.overhang * 2;
    91→    return v.commonDepth;
    92→  };
    93→  const mat = (name: string) => new StandardMaterial(name, scene);
    94→  const box = (name: string, w: number, h: number, d: number, x: number, y: number, z: number, m: StandardMaterial) => {
    95→    const mesh = MeshBuilder.CreateBox(name, { width: w, height: h, depth: d }, scene);
    96→    mesh.position.set(x, y, z);
    97→    mesh.material = m;
    98→    return mesh;
    99→  };
   100→
   101→  for (const b of Object.values(world.blocks)) {
   102→    const m = mat(`${b.id}_mat`);
   103→    const cx = (b.cell.col + b.size.w / 2) * v.cellW;
   104→    const bottom = b.cell.floor * v.cellH;
   105→    const w = b.size.w * v.cellW - v.gapW;
   106→    const h = v.cellH - v.gapH;
   107→    const d = depthOf(b);
   108→    const cz = frontZ + d / 2;
   109→    const view: BlockView = {
   110→      greybox: [],
   111→      mats: [m],
   112→      baseEmissive: [m.emissiveColor.clone()],
   113→      baseColor: [Color3.FromHexString(content.blocks[b.defId]!.view?.color ?? '#cccccc')],
   114→      tintStrength: 1,
   115→      model: null,
   116→      front: null,
   117→      door: null,
   118→      handle: null,
   119→      lock: null,
   120→      lockBase: { x: cx, y: bottom + v.cellH / 2 - 0.2 },
   121→      centerX: cx,
   122→      floor: b.cell.floor,
   123→      edgesMesh: null,
   124→      edgesMesh2: null,
   125→      glowForced: false,
   126→      lodHiddenPanels: false,
   127→    };
   128→    const root = new TransformNode(`${b.id}_root`, scene);
   129→    roots.set(b.id, root);
   130→    if (b.layer === 'corridor') {
   131→      // Pasillo exterior (data-driven): tira transitable saliente hacia la calle, con el borde
   132→      // trasero a ras de fachada (frontZ) y barandilla baja en el borde exterior — sitio de
   133→      // sobra para que los personajes se crucen o charlen sin bailar el vacío.
   134→      const d = v.corridorDepth;
   135→      view.greybox.push(box(b.id, b.size.w * v.cellW, v.slab * 2, d, cx, bottom + v.slab, frontZ - d / 2, m));
   136→      if (v.corridorRail > 0) {
   137→        view.greybox.push(
   138→          box(`${b.id}_barandilla`, b.size.w * v.cellW, v.corridorRail, v.wall, cx, bottom + v.slab * 2 + v.corridorRail / 2, frontZ - d, m),
   139→        );
   140→      }
   141→    } else if (b.kind === 'dwelling') {
   142→      const y = bottom + v.cellH / 2;
   143→      // ── Puerta a sala TRASERA (v1.19, apilamiento 3D): la pared del fondo se parte con el
   144→      // hueco del layout (doorway hacia una estancia más profunda que el bloque principal).
   145→      const layoutId0 = content.blocks[b.defId]?.container?.layout;
   146→      const layout0 = layoutId0 ? content.layouts[layoutId0] : undefined;
   147→      const rearKeys = new Set((layout0?.rooms ?? []).filter((r) => r.rect && r.rect.z + r.rect.d > v.depth + 0.01).map((r) => r.key));
   148→      const rearDoor = (layout0?.doorways ?? []).find((dw) => dw.axis === 'x' && rearKeys.has(dw.b));
   149→      const x0w = cx - w / 2;
   150→      const x1w = cx + w / 2;
   151→      if (rearDoor) {
   152→        const gapA = Math.max(x0w, x0w + rearDoor.at - rearDoor.width / 2);
   153→        const gapB = Math.min(x1w, x0w + rearDoor.at + rearDoor.width / 2);
   154→        const doorH = Math.min(h - 0.15, 1.95);
   155→        const lintelH = h - doorH;
   156→        view.edgesMesh = box(`${b.id}_fondo_i`, gapA - x0w, h, v.wall, (x0w + gapA) / 2, y, frontZ + d, m);
   157→        view.edgesMesh2 = box(`${b.id}_fondo_d`, x1w - gapB, h, v.wall, (gapB + x1w) / 2, y, frontZ + d, m);
   158→        view.greybox.push(
   159→          view.edgesMesh,
   160→          view.edgesMesh2,
   161→          box(`${b.id}_fondo_dintel`, gapB - gapA, lintelH, v.wall, (gapA + gapB) / 2, y + h / 2 - lintelH / 2, frontZ + d, m),
   162→          box(`${b.id}_izq`, v.wall, h, d, cx - w / 2, y, cz, m),
   163→          box(`${b.id}_der`, v.wall, h, d, cx + w / 2, y, cz, m),
   164→          box(`${b.id}_suelo`, w, v.wall, d, cx, bottom + v.slab + v.wall / 2, cz, m),
   165→          box(`${b.id}_techo`, w, v.wall, d, cx, bottom + v.cellH - v.slab - v.wall / 2, cz, m),
   166→        );
   167→      } else {
   168→        view.edgesMesh = box(`${b.id}_fondo`, w, h, v.wall, cx, y, frontZ + d, m);
   169→        view.greybox.push(
   170→          view.edgesMesh,
   171→          box(`${b.id}_izq`, v.wall, h, d, cx - w / 2, y, cz, m),
   172→          box(`${b.id}_der`, v.wall, h, d, cx + w / 2, y, cz, m),
   173→          box(`${b.id}_suelo`, w, v.wall, d, cx, bottom + v.slab + v.wall / 2, cz, m),
   174→          box(`${b.id}_techo`, w, v.wall, d, cx, bottom + v.cellH - v.slab - v.wall / 2, cz, m),
   175→        );
   176→      }
   177→      view.front = box(`${b.id}_puerta`, w, h, v.wall, cx, y, frontZ, m);
   178→      view.lock = box(`${b.id}_candado`, 0.3, 0.38, 0.12, cx, view.lockBase.y, frontZ - 0.1, mat(`${b.id}_candado_mat`));
   179→      (view.lock.material as StandardMaterial).diffuseColor = Color3.FromHexString(v.lockColor);
   180→
   181→      // ── Privacidad (v1.17): puerta y picaporte sobre la fachada de la vivienda vacía.
   182→      // Se activan/desactivan cada frame según haya alguien dentro (propietario, visita…
   183→      // o un ladrón); medidas y colores del catálogo (rules.view.privacy).
   184→      const pv = v.privacy;
   185→      view.door = box(
   186→        `${b.id}_puerta_priv`,
   187→        pv.doorW,
   188→        pv.doorH,
   189→        0.06,
   190→        cx,
   191→        bottom + v.slab + v.wall + pv.doorH / 2,
   192→        frontZ - v.wall / 2 - 0.035,
   193→        mat(`${b.id}_puerta_priv_mat`),
   194→      );
   195→      (view.door.material as StandardMaterial).diffuseColor = Color3.FromHexString(pv.doorColor);
   196→      view.handle = box(
   197→        `${b.id}_picaporte`,
   198→        0.07,
   199→        0.07,
   200→        0.05,
   201→        cx + pv.doorW / 2 - 0.14,
   202→        bottom + v.slab + v.wall + pv.doorH * 0.55,
   203→        frontZ - v.wall / 2 - 0.07,
   204→        mat(`${b.id}_picaporte_mat`),
   205→      );
   206→      (view.handle.material as StandardMaterial).diffuseColor = Color3.FromHexString(pv.handleColor);
   207→      view.door.setEnabled(false);
   208→      view.handle.setEnabled(false);
   209→
   210→      // ── Cartel numerado (v1.6): rótulo sobre la puerta con el identificador de la vivienda
   211→      // (blk_p3_c → "3C", blockLabel). Textura dinámica con colores/fuente del catálogo
   212→      // (rules.view.plaque); se añade a greybox porque la vivienda usa modelo OVERLAY (el
   213→      // greybox no se oculta) y no entra en mats: es señalización, no tinte ni glow.
   214→      const pl = v.plaque;
   215→      const plate = MeshBuilder.CreatePlane(`${b.id}_cartel`, { width: pl.w, height: pl.h, sideOrientation: B().Mesh.DOUBLESIDE }, scene);
   216→      const tex = new (B().DynamicTexture)(`${b.id}_cartel_tex`, { width: 256, height: 96 }, scene, true);
   217→      const ctx = tex.getContext() as unknown as CanvasRenderingContext2D;
   218→      ctx.fillStyle = pl.bg;
   219→      ctx.fillRect(0, 0, 256, 96);
   220→      ctx.font = `bold ${pl.fontPx}px Arial, sans-serif`;
   221→      ctx.fillStyle = pl.text;
   222→      ctx.textAlign = 'center';
   223→      ctx.textBaseline = 'middle';
   224→      ctx.fillText(blockLabel(b.id), 128, 52);
   225→      // update(true): el flag es invertY. Con false el canvas se sube SIN voltear y el texto
   226→      // sale CABEZA ABAJO (la "A" se ve como "V"; B/C/D pasan por simétricas y esquivan el ojo).
   227→      tex.update(true);
   228→      const pmat = new StandardMaterial(`${b.id}_cartel_mat`, scene);
   229→      pmat.diffuseTexture = tex;
   230→      pmat.emissiveColor = Color3.FromHexString(pl.bg).scale(0.55); // legible en fachada norte
   231→      pmat.specularColor = new Color3(0, 0, 0);
   232→      plate.material = pmat;
   233→      plate.position.set(cx, bottom + pl.y, frontZ + pl.z);
   234→      view.greybox.push(plate);
   235→
   236→      // ── Interior 2D (Fase C): tabiques con huecos, suelo por estancia y tapas "en obras". ──
   237→      // TODO sale del layout del catálogo (rects + doorways); sin rects no hay interior 2D.
   238→      const layoutId = content.blocks[b.defId]?.container?.layout;
   239→      const layout = layoutId ? content.layouts[layoutId] : undefined;
   240→      // Apilamiento 3D (v1.19): las salas TRASERAS no se pintan en el bloque principal — viven
   241→      // en su propio bloque trasero (más abajo, se activan al entrar).
   242→      const geoRooms = (layout?.rooms ?? []).filter((r) => r.rect !== undefined && r.rect!.z + r.rect!.d <= v.depth + 0.01);
   243→      if (geoRooms.length > 0 && layout) {
   244→        const interior: InteriorView = { floors: [], covers: [] };
   245→        const x0 = b.cell.col * v.cellW;
   246→        const wallH = v.cellH - 2 * v.slab - 2 * v.wall;
   247→        const partMat = mat(`${b.id}_tabique_mat`);
   248→        partMat.diffuseColor = Color3.FromHexString('#d8d2c4');
   249→        // Suelo tintado por estancia (plancha fina sobre el suelo, alpha suave).
   250→        for (const lr of geoRooms) {
   251→          const r = lr.rect!;
   252→          const fmat = mat(`${b.id}_${lr.key}_suelo_mat`);
   253→          fmat.diffuseColor = Color3.FromHexString(content.rooms[lr.defId]?.view?.color ?? '#cccccc');
   254→          fmat.alpha = 0.32;
   255→          const fm = attach(b.id, box(`${b.id}_${lr.key}_suelo`, r.w, 0.02, r.d, x0 + r.x + r.w / 2, bottom + v.slab + v.wall + 0.02, frontZ + r.z + r.d / 2, fmat));
   256→          interior.floors.push({ roomKey: lr.key, defId: lr.defId, mesh: fm, mat: fmat });
   257→        }
   258→        // Tabiques: para cada par de estancias que comparten borde, pared entre ambas.
   259→        // Los doorways del catálogo recortan el hueco (y añaden tapa visible si alguna de las
   260→        // dos estancias está cerrada). Sin hueco declarado → tabique sólido (tampoco se puede
   261→        // pasar: walkable() exige doorway). Tabiques según X a media altura para no tapar la vista.
   262→        const eps = 0.01;
   263→        const seg = (a: number, b: number, along: 'x' | 'z', at: number, height: number, yC: number, name: string) => {
   264→          const len = b - a;
   265→          if (len <= 0.01) return;
   266→          view.greybox.push(
   267→            along === 'z'
   268→              ? box(name, v.wall, height, len, at, yC, (a + b) / 2, partMat)
   269→              : box(name, len, height, v.wall, (a + b) / 2, yC, at, partMat),
   270→          );
   271→        };
   272→        geoRooms.forEach((lr, i) => {
   273→          geoRooms.forEach((other, j) => {
   274→            if (j <= i) return;
   275→            const ra = lr.rect!;
   276→            const rb = other.rect!;
   277→            const door = (layout.doorways ?? []).find(
   278→              (dw) => (dw.a === lr.key && dw.b === other.key) || (dw.a === other.key && dw.b === lr.key),
   279→            );
   280→            // Borde vertical compartido → tabique que corre según Z (se cruza andando en X).
   281→            const touchX = Math.abs(ra.x + ra.w - rb.x) < eps || Math.abs(rb.x + rb.w - ra.x) < eps;
   282→            const overlapZ = Math.min(ra.z + ra.d, rb.z + rb.d) - Math.max(ra.z, rb.z);
   283→            if (touchX && overlapZ > eps) {
   284→              const wx = x0 + Math.max(ra.x, rb.x);
   285→              const z0 = frontZ + Math.max(ra.z, rb.z);
   286→              const z1 = frontZ + Math.min(ra.z + ra.d, rb.z + rb.d);
   287→              const yC = bottom + v.slab + v.wall + wallH / 2;
   288→              if (door && door.axis === 'z') {
   289→                const g0 = Math.max(z0, frontZ + door.at - door.width / 2);
   290→                const g1 = Math.min(z1, frontZ + door.at + door.width / 2);
   291→                seg(z0, g0, 'z', wx, wallH, yC, `${b.id}_tab_${i}_${j}_i`);
   292→                seg(g1, z1, 'z', wx, wallH, yC, `${b.id}_tab_${i}_${j}_d`);
   293→                const cover = attach(b.id, box(`${b.id}_tapa_${i}_${j}`, v.wall * 0.8, wallH, g1 - g0, wx, yC, frontZ + door.at, mat(`${b.id}_tapa_${i}_${j}_mat`)));
   294→                (cover.material as StandardMaterial).diffuseColor = Color3.FromHexString(v.lockedColor);
   295→                cover.setEnabled(false);
   296→                interior.covers.push({ mesh: cover, rooms: [lr.key, other.key] });
   297→              } else {
   298→                seg(z0, z1, 'z', wx, wallH, yC, `${b.id}_tab_${i}_${j}`);
   299→              }
   300→            }
   301→            // Borde horizontal compartido → tabique que corre según X (se cruza andando en Z).
   302→            const touchZ = Math.abs(ra.z + ra.d - rb.z) < eps || Math.abs(rb.z + rb.d - ra.z) < eps;
   303→            const overlapX = Math.min(ra.x + ra.w, rb.x + rb.w) - Math.max(ra.x, rb.x);
   304→            if (touchZ && overlapX > eps) {
   305→              const wz = frontZ + Math.max(ra.z, rb.z);
   306→              const xa = x0 + Math.max(ra.x, rb.x);
   307→              const xb = x0 + Math.min(ra.x + ra.w, rb.x + rb.w);
   308→              const halfH = wallH * 0.45;
   309→              const yHalf = bottom + v.slab + v.wall + halfH / 2;
   310→              if (door && door.axis === 'x') {
   311→                const g0 = Math.max(xa, x0 + door.at - door.width / 2);
   312→                const g1 = Math.min(xb, x0 + door.at + door.width / 2);
   313→                seg(xa, g0, 'x', wz, halfH, yHalf, `${b.id}_tab_${i}_${j}_i`);
   314→                seg(g1, xb, 'x', wz, halfH, yHalf, `${b.id}_tab_${i}_${j}_d`);
   315→                const cover = attach(b.id, box(`${b.id}_tapa_${i}_${j}`, g1 - g0, halfH, v.wall * 0.8, x0 + door.at, yHalf, wz, mat(`${b.id}_tapa_${i}_${j}_mat`)));
   316→                (cover.material as StandardMaterial).diffuseColor = Color3.FromHexString(v.lockedColor);
   317→                cover.setEnabled(false);
   318→                interior.covers.push({ mesh: cover, rooms: [lr.key, other.key] });
   319→              } else {
   320→                seg(xa, xb, 'x', wz, halfH, yHalf, `${b.id}_tab_${i}_${j}`);
   321→              }
   322→            }
   323→          });
   324→        });
   325→        interiors.set(b.id, interior);
   326→      }
   327→    } else if (b.defId === 'trasteros_planta' || b.defId === 'bajocubierta' || b.defId === v.trasteros.blockDefId) {
   328→      // Nave de trasteros / sala de instalaciones (v1.16): forjado + trasera. El arte glTF
   329→      // añade pilares (trasteros) o maquinaria (bajocubierta); las estanterías, trastos y
   330→      // carteles numerados los levanta SIEMPRE el runtime desde rules.view.trasteros.
   331→      // El bloque por COLUMNA (v1.18) comparte estructura: sus compartimentos los levanta
   332→      // el runtime más abajo (pasillo central + celdas cerradas impares/pares).
   333→      view.greybox.push(
   334→        box(`${b.id}_forjado`, w, v.slab * 2, d, cx, bottom + v.slab, frontZ + d / 2, m),

   898→        box(`${blk.id}_cell_suelo_trasto_${j}`, jw, jh, 0.5, x0 + 0.5 + hash01(homeId, 2900 + j) * Math.max(0.1, W - 1.1), baseR + jh / 2, frontZ + 0.5 + hash01(homeId, 3000 + j) * Math.max(0.1, D - 1.2), jm).parent = node;
   899→      }
   900→      const outline = outlineFrame(`${blk.id}_cell`, cxr, W, H, baseR, frontZ - 0.04);
   901→      for (const o of outline) o.parent = node;
   902→      registerRear(blk.id, node, outline, null);
   903→    }
   904→  }

   906→  function color(b: BlockInstance): Color3 {
   907→    const base = Color3.FromHexString(content.blocks[b.defId]!.view?.color ?? '#cccccc');
   908→    if (b.kind === 'dwelling' && b.access === 'locked') return Color3.FromHexString(v.lockedColor);
   909→    if (b.kind === 'dwelling' && b.occupancy !== 'occupied') return Color3.Lerp(base, Color3.White(), v.vacantTint);
   910→    if (b.params.elevator_status !== undefined && b.params.elevator_status !== 'ok') return Color3.FromHexString(v.brokenColor);
   911→    return base;
   912→  }

   914→  /** Color de ESTADO (null = sin estado especial). El greybox lo pinta plano; los modelos lo mezclan. */

   916→    if (b.kind === 'dwelling' && b.access === 'locked') return Color3.FromHexString(v.lockedColor);
   917→    if (b.kind === 'dwelling' && b.occupancy !== 'occupied') return Color3.Lerp(Color3.FromHexString(content.blocks[b.defId]!.view?.color ?? '#cccccc'), Color3.White(), v.vacantTint);
   918→    if (b.params.elevator_status !== undefined && b.params.elevator_status !== 'ok') return Color3.FromHexString(v.brokenColor);
   919→    return null;

   922→  /** Emissive del bloque: canales de influencia → emissive según color del canal en el catálogo. Se acumulan. */
   923→  function applyEmissive(view: BlockView, b: BlockInstance) {
   924→    let e = new Color3(0, 0, 0);
   925→    for (const [channel, value] of Object.entries(world.channels)) {
   926→      const level = value[b.id];
   927→      if (!level) continue;
   928→      const g = glow[channel];
   929→      if (!g) continue;
   930→      e = e.add(g.scale(Math.min(1, level)));
   931→    }
   932→    for (const m of view.mats) m.emissiveColor = e;
   933→  }

   935→  /** Color difuso/albedo por material: greybox plano (tintStrength 1) o mezcla sobre el arte. */
   936→  function applyTint(view: BlockView, b: BlockInstance) {
   937→    const state = color(b);
   938→    const override = stateTint(b);
   939→    for (let i = 0; i < view.mats.length; i++) {
   940→      const m = view.mats[i]!;
   941→      const base = view.baseColor[i]!;
   942→      const c = view.tintStrength >= 1 ? state : override ? Color3.Lerp(base, override, view.tintStrength) : base;
   943→      if (m.diffuseColor) m.diffuseColor = c;
   944→      else if (m.albedoColor) m.albedoColor = c;
   945→    }
   946→  }

   948→  // ── Privacidad de viviendas (v1.17): conjunto de bloques con alguien dentro (el jugador
   949→  //  vía playerInside + cualquier residente presente cuyo location sea el bloque). Se
   950→  //  recalcula por frame en update(): la fachada reacciona al instante cuando un vecino
   951→  //  llega a casa o el jugador entra, sin esperar al siguiente tick.
   952→  let playerInside: string | null = null;
   953→  const privacyOn = v.privacy.enabled;
   954→  const computeOccupancy = (): Set<string> => {
   955→    const occ = new Set<string>();
   956→    if (playerInside) occ.add(playerInside);
   957→    for (const r of Object.values(world.residents)) {
   958→      if (r.present && r.location) occ.add(r.location);
   959→    }
   960→    return occ;
   961→  };

   963→  /** Visibilidad efectiva de fachada/puerta/candado: candado solo si access='locked';
   964→   *  fachada cerrada si está bloqueada O si la privacidad la cierra (nadie dentro).
   965→   *  paint() la fija por el acceso y el LOD puede ocultarlos encima. */
   966→  function applyPanels(view: BlockView, b: BlockInstance, occ?: Set<string>) {
   967→    const locked = b.kind === 'dwelling' && b.access === 'locked';
   968→    let privacyClosed = false;
   969→    if (privacyOn && b.kind === 'dwelling') {
   970→      const set = occ ?? computeOccupancy();
   971→      privacyClosed = !set.has(b.id);
   972→    }
   973→    const wall = (locked || privacyClosed) && !view.lodHiddenPanels;
   974→    view.front?.setEnabled(wall);
   975→    view.door?.setEnabled(wall);
   976→    view.handle?.setEnabled(wall);
   977→    view.lock?.setEnabled(locked && !view.lodHiddenPanels);
   978→  }

   980→  /**
   981→   * LOD por distancia (rules.view.lod). Se ejecuta SIEMPRE al final de update(): con foco degrada
   982→   * cada bloque según su distancia, y sin foco (o con lod.enabled=false) restaura el estado completo.
   983→   * Idempotente y sin allocations por frame: solo aritmética, booleanos y asignaciones.
   984→   *
   985→   * Métrica (Vista · LOD por distancia): dx = |centroX del bloque − focus.x| en metros y
   986→   * dy = |planta del bloque − focus.floor| * view.cellH en metros; dist = Math.hypot(dx, dy).
   987→   * Nivel efectivo = todos los niveles de lod.levels cuyo `dist` ≤ distBloque (el `hide` es
   988→   * acumulado: un flag se oculta si cualquier nivel alcanzado lo lista; sin nivel → todo visible).
   989→   */
   990→  function applyLod(focus: { x: number; floor: number } | null) {
   991→    const lod = v.lod;
   992→    for (const id in world.blocks) {
   993→      const b = world.blocks[id]!;
   994→      const view = views.get(id)!;
   995→      let hideEdges = false;
   996→      let hideGlow = false;
   997→      let hidePanels = false;
   998→      if (lod.enabled && focus) {
   999→        const dx = Math.abs(view.centerX - focus.x);
  1000→        const dy = Math.abs(view.floor - focus.floor) * v.cellH;
  1001→        const dist = Math.hypot(dx, dy);
  1002→        for (let i = 0; i < lod.levels.length; i++) {
  1003→          const level = lod.levels[i];
  1004→          if (level.dist > dist) continue;
  1005→          for (let j = 0; j < level.hide.length; j++) {
  1006→            const flag = level.hide[j];
  1007→            if (flag === 'edges') hideEdges = true;
  1008→            else if (flag === 'glow') hideGlow = true;
  1009→            else if (flag === 'panels') hidePanels = true;
  1010→          }
  1011→        }
  1012→      }
  1013→      // 'edges': interruptor nativo barato — el renderingGroup consulta edgesRenderer.isEnabled cada
  1014→      // frame, así que nunca se destruye ni recrea el EdgesRenderer (disable/enableEdgesRendering sí
  1015→      // lo harían: el primero lo dispone y el segundo recomputaría las aristas). El grosor/color
  1016→      // viven en la malla y sobreviven al apagado. Los modelos glTF no tienen edges (edgesMesh null).
  1017→      if (view.edgesMesh?.edgesRenderer) view.edgesMesh.edgesRenderer.isEnabled = !hideEdges;
  1018→      if (view.edgesMesh2?.edgesRenderer) view.edgesMesh2.edgesRenderer.isEnabled = !hideEdges;
  1019→      // 'glow': pisa el emissive CADA frame (paint() pudo repintar los canales este tick); al soltarlo,
  1020→      // una única llamada repinta el cálculo de paint para no esperar al próximo tick.
  1021→      if (hideGlow) {
  1022→        view.glowForced = true;
  1023→        for (let i = 0; i < view.mats.length; i++) view.mats[i]!.emissiveColor = view.baseEmissive[i]!;
  1024→      } else if (view.glowForced) {
  1025→        view.glowForced = false;
  1026→        applyEmissive(view, b);
  1027→      }
  1028→      // 'panels': solo gira la bandera (la visibilidad la calcula applyPanels); en la transición se
  1029→      // aplica ya para no esperar al próximo paint(), que reproducirá el mismo resultado.
  1030→      if (view.lodHiddenPanels !== hidePanels) {
  1031→        view.lodHiddenPanels = hidePanels;
  1032→        applyPanels(view, b);
  1033→      }
  1034→    }
  1035→  }
  1036→
  1037→  return {
  1038→    paint() {
  1039→      for (const b of Object.values(world.blocks)) {
  1040→        const view = views.get(b.id)!;
  1041→        applyTint(view, b);
  1042→        applyEmissive(view, b);
  1043→        applyPanels(view, b);
  1044→      }
  1045→      // Interiores 2D (Fase C): suelo de cada estancia según su acceso + tapas de puertas cerradas.
  1046→      for (const [blockId, interior] of interiors) {
  1047→        const block = world.blocks[blockId];
  1048→        if (!block) continue;
  1049→        for (const f of interior.floors) {
  1050→          const room = block.rooms.find((r) => r.key === f.roomKey);
  1051→          const locked = room ? room.access !== 'unlocked' : true;
  1052→          const hex = locked ? v.lockedColor : content.rooms[f.defId]?.view?.color ?? '#cccccc';
  1053→          f.mat.diffuseColor = Color3.FromHexString(hex);
  1054→        }
  1055→        for (const c of interior.covers) {
  1056→          const anyLocked = c.rooms.some((key) => block.rooms.find((r) => r.key === key)?.access !== 'unlocked');
  1057→          c.mesh.setEnabled(anyLocked);
  1058→        }
  1059→      }
  1060→      // Muebles del editor de interiores: visibles/escala/color según furniture.view (catálogo).
  1061→      for (const [key, fv] of furnViews) {
  1062→        const [blockId, roomKey, slotId] = key.split(':');
  1063→        const room = world.blocks[blockId ?? '']?.rooms.find((r) => r.key === roomKey);
  1064→        const itemId = room?.furniture[slotId ?? ''] ?? '';
  1065→        const fdef = itemId ? content.furniture[itemId] : undefined;
  1066→        if (!fdef) {
  1067→          fv.mesh.setEnabled(false);
  1068→          continue;
  1069→        }
  1070→        const [iw, ih, idz] = fdef.view.size;
  1071→        fv.mesh.setEnabled(true);
  1072→        fv.mesh.scaling.set(iw, ih, idz);
  1073→        // Con y de anclaje (muebles de pared) cuelga a esa altura; si no, apoyado en el suelo.
  1074→        const yBase = fv.y !== null ? fv.bottom + v.slab + v.wall + fv.y : fv.bottom + v.slab + v.wall;
  1075→        fv.mesh.position.set(fv.x, yBase + ih / 2, fv.z);
  1076→        fv.mat.diffuseColor = Color3.FromHexString(fdef.view.color);
  1077→      }
  1078→    },
  1079→    /** Sacude el candado de una puerta cerrada. */
  1080→    shake(blockId: string) {
  1081→      shakes.set(blockId, 0.4);
  1082→    },
  1083→    /**
  1084→     * Apilamiento 3D (v1.19): fase del bloque trasero de `blockId`.
  1085→     *  · 'idle'   — bloque inactivo (todo como siempre).
  1086→     *  · 'peek'   — el jugador se acerca a la puerta del fondo: el bloque trasero asoma por
  1087→     *               el hueco (transparencia → visible) y la hoja se abre.
  1088→     *  · 'inside' — cruzaste: el bloque principal SE DESVANECE y el trasero queda en primer
  1089→     *               plano con su contorno rojo.
  1090→     */
  1091→    setRearPhase(blockId: string, phase: 'idle' | 'peek' | 'inside'): void {
  1092→      const r = rears.get(blockId);
  1093→      if (r) r.phase = phase;
  1094→    },
  1095→    hasRear(blockId: string): boolean {
  1096→      return rears.has(blockId);
  1097→    },
  1098→    /**
  1099→     * dt en segundos; `focus` es el punto de interés (posición del jugador) para el
  1100→     * LOD por distancia (rules.view.lod): applyLod() lo aplica SIEMPRE al final del frame
  1101→     * (con focus undefined, p. ej. cámara 'edificio', restaura el estado completo).
  1102→     * `inside` es el bloque donde está el jugador ahora (pos.insideBlock): alimenta la
  1103→     * privacidad y re-evalúa las fachadas CADA frame — un vecino que llega a casa abre
  1104→     * su interior al instante, sin esperar al siguiente tick.
  1105→     * `pos` (v1.19) alimenta los carteles guía de trasteros y el apilamiento 3D.
  1106→     */
  1107→    update(dt: number, focus?: { x: number; floor: number }, inside?: string | null, pos?: { x: number; z: number; insideBlock: string | null }) {
  1108→      if (focus) lastFocus = focus;
  1109→      if (inside !== undefined) playerInside = inside;
  1110→      // PRIVACIDAD (v1.17) cada frame: la fachada de una vivienda sin nadie dentro se cierra
  1111→      // con su puerta al pasillo y se abre en cuanto alguien entra (propietario, visita…
  1112→      // o un ladrón). Vive en update() y no en paint() para reaccionar incluso con el
  1113→      // mundo congelado (modal abierto) y sin esperar al siguiente tick.
  1114→      if (privacyOn) {
  1115→        const occ = computeOccupancy();
  1116→        for (const b of Object.values(world.blocks)) {
  1117→          if (b.kind !== 'dwelling') continue;
  1118→          applyPanels(views.get(b.id)!, b, occ);
  1119→        }
  1120→      }
  1121→      // CARTELES GUÍA (v1.19): visibles solo con el jugador dentro del bloque de trasteros;
  1122→      // texto = los trasteros que tienes a cada lado según la profundidad del pasillo.
  1123→      const sgG = colBlocks.length > 0 ? storageGeomOf(v) : null;
  1124→      for (const [blkId, sign] of guideSigns) {
  1125→        const active = !!pos && pos.insideBlock === blkId && !!sgG;
  1126→        const plate = scene.getMeshByName(`${blkId}_guia`);
  1127→        const frame = scene.getMeshByName(`${blkId}_guia_marco`);
  1128→        plate?.setEnabled(active);
  1129→        frame?.setEnabled(active);
  1130→        if (!active || !sgG) continue;
  1131→        const localZ = pos!.z; // en modo 'inside' de trasteros, z es profundidad local
  1132→        const usable = sgG.depth - 2 * sgG.wallT;
  1133→        const perSide = new Map<string, number>();
  1134→        for (const u of sign.plan) perSide.set(u.side, (perSide.get(u.side) ?? 0) + 1);
  1135→        const labelOf = (side: 'left' | 'right'): string => {
  1136→          const count = perSide.get(side) ?? 0;
  1137→          if (count === 0) return tr.emptyLabel;
  1138→          const idx = Math.max(0, Math.min(count - 1, Math.floor((localZ - sgG.wallT) / (usable / count))));
  1139→          const unit = sign.plan.find((u) => u.side === side && u.index === idx);
  1140→          return unit ? blockLabel(`blk_p${unit.floor}_${sign.ownTag}`) : tr.emptyLabel;
  1141→        };
  1142→        const left = labelOf('left');
  1143→        const right = labelOf('right');
  1144→        const key = `${left}|${right}`;
  1145→        if (sign.last !== key) {
  1146→          sign.last = key;
  1147→          const ctx = sign.tex.getContext() as unknown as CanvasRenderingContext2D;
  1148→          ctx.fillStyle = tr.guide.bg;
  1149→          ctx.fillRect(0, 0, 256, 60);
  1150→          ctx.font = 'bold 30px Arial, sans-serif';
  1151→          ctx.textBaseline = 'middle';
  1152→          ctx.fillStyle = tr.guide.text;
  1153→          ctx.textAlign = 'left';
  1154→          ctx.fillText('◀', 10, 32);
  1155→          ctx.textAlign = 'center';
  1156→          const ownLeft = left.includes(sign.ownTag.toUpperCase());
  1157→          const ownRight = right.includes(sign.ownTag.toUpperCase());
  1158→          ctx.fillStyle = ownLeft ? tr.ownBg : tr.guide.text;
  1159→          ctx.fillText((ownLeft ? '★ ' : '') + left, 88, 32);
  1160→          ctx.fillStyle = ownRight ? tr.ownBg : tr.guide.text;
  1161→          ctx.fillText((ownRight ? '★ ' : '') + right, 176, 32);
  1162→          ctx.fillStyle = tr.guide.text;
  1163→          ctx.textAlign = 'right';
  1164→          ctx.fillText('▶', 248, 32);
  1165→          sign.tex.update(true);
  1166→        }
  1167→      }
  1168→      // APILAMIENTO 3D (v1.19): avanzar los fundidos de los bloques traseros.
  1169→      for (const r of rears.values()) {
  1170→        const mainT = r.phase === 'inside' ? 0 : 1;
  1171→        const rearT = r.phase === 'idle' ? 0 : 1;
  1172→        if (r.settled && r.mainAlpha === mainT && r.rearAlpha === rearT) continue;
  1173→        const mRate = dt / Math.max(0.05, rb.fadeSeconds);
  1174→        const rRate = dt / Math.max(0.05, rb.peekSeconds);
  1175→        r.mainAlpha += Math.sign(mainT - r.mainAlpha) * Math.min(Math.abs(mainT - r.mainAlpha), mRate);
  1176→        r.rearAlpha += Math.sign(rearT - r.rearAlpha) * Math.min(Math.abs(rearT - r.rearAlpha), rRate);
  1177→        r.settled = r.mainAlpha === mainT && r.rearAlpha === rearT;
  1178→        const mainKids = r.mainRoot.getChildMeshes();
  1179→        for (const k of mainKids) k.visibility = r.mainAlpha;
  1180→        r.mainRoot.setEnabled(r.mainAlpha > 0.002);
  1181→        const rearKids = r.node.getChildMeshes();
  1182→        for (const k of rearKids) k.visibility = r.rearAlpha;
  1183→        r.node.setEnabled(r.rearAlpha > 0.002);
  1184→        for (const o of r.outline) o.setEnabled(r.mainAlpha < 0.55 && r.rearAlpha > 0.4);
  1185→        if (r.doorLeaf) r.doorLeaf.setEnabled(r.rearAlpha < 0.2);
  1186→      }
  1187→      for (const [id, left] of shakes) {
  1188→        const view = views.get(id)!;
  1189→        const t = left - dt;
  1190→        if (t <= 0) {
  1191→          shakes.delete(id);
  1192→          view.lock?.position.set(view.lockBase.x, view.lockBase.y, view.lock.position.z);
  1193→        } else {
  1194→          shakes.set(id, t);
  1195→          view.lock?.position.set(view.lockBase.x + Math.sin(t * 60) * 0.06, view.lockBase.y, view.lock.position.z);
  1196→        }
  1197→      }
  1198→      // LOD SIEMPRE al final del frame: con foco aplica el nivel efectivo por bloque,
  1199→      // sin foco restaura edges/emissive/paneles al estado completo.
  1200→      applyLod(focus ?? null);
  1201→    },
  1202→  };
