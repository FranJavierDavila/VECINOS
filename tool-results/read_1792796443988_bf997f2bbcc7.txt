    18→// ───────────────────────── Utilidades glTF ─────────────────────────
    19→
    20→type Vec3 = [number, number, number];
    21→type Quat = [number, number, number, number];
    22→
    23→const srgbToLinear = (c: number) => Math.pow(c / 255, 2.2);
    24→const colorLinear = (hex: string): [number, number, number, number] => {
    25→  const h = hex.replace('#', '');
    26→  return [srgbToLinear(parseInt(h.slice(0, 2), 16)), srgbToLinear(parseInt(h.slice(2, 4), 16)), srgbToLinear(parseInt(h.slice(4, 6), 16)), 1];
    27→};
    28→
    29→/** Cuaternión de rotación alrededor de un eje unitario (ángulo en radianes). */
    30→const quatAxis = (ax: number, ay: number, az: number, angle: number): Quat => {
    31→  const s = Math.sin(angle / 2);
    32→  return [ax * s, ay * s, az * s, Math.cos(angle / 2)];
    33→};
    34→
    35→class Gltf {
    36→  private chunks: Uint8Array[] = [];
    37→  private byteLen = 0;
    38→  bufferViews: { buffer: 0; byteOffset: number; byteLength: number; target?: number }[] = [];
    39→  accessors: Record<string, unknown>[] = [];
    40→  nodes: Record<string, unknown>[] = [];
    41→  meshes: Record<string, unknown>[] = [];
    42→  materials: Record<string, unknown>[] = [];
    43→  animations: Record<string, unknown>[] = [];
    44→
    45→  private view(data: ArrayBufferView, target?: number): number {
    46→    // Alineación a 4 bytes (requisito del spec para accessors de float).
    47→    const pad = (4 - (this.byteLen % 4)) % 4;
    48→    if (pad > 0) {
    49→      this.chunks.push(new Uint8Array(pad));
    50→      this.byteLen += pad;
    51→    }
    52→    const bytes = new Uint8Array(data.buffer, data.byteOffset, data.byteLength);
    53→    this.chunks.push(bytes);
    54→    this.bufferViews.push({ buffer: 0, byteOffset: this.byteLen, byteLength: bytes.byteLength, ...(target ? { target } : {}) });
    55→    this.byteLen += bytes.byteLength;
    56→    return this.bufferViews.length - 1;
    57→  }
    58→
    59→  private accessor(viewIdx: number, compType: number, count: number, type: string, minMax?: { min: number[]; max: number[] }): number {
    60→    this.accessors.push({ bufferView: viewIdx, componentType: compType, count, type, ...(minMax ?? {}) });
    61→    return this.accessors.length - 1;
    62→  }
    63→
    64→  times(times: number[]): number {
    65→    return this.accessor(this.view(new Float32Array(times)), 5126, times.length, 'SCALAR', {
    66→      min: [Math.min(...times)],
    67→      max: [Math.max(...times)],
    68→    });
    69→  }
    70→
    71→  vec3(values: number[][]): number {
    72→    const flat = new Float32Array(values.length * 3);
    73→    values.forEach((v, i) => flat.set(v, i * 3));
    74→    const mins = [0, 1, 2].map((k) => Math.min(...values.map((v) => v[k]!)));
    75→    const maxs = [0, 1, 2].map((k) => Math.max(...values.map((v) => v[k]!)));
    76→    return this.accessor(this.view(flat), 5126, values.length, 'VEC3', { min: mins, max: maxs });
    77→  }
    78→
    79→  vec4(values: number[][]): number {
    80→    const flat = new Float32Array(values.length * 4);
    81→    values.forEach((v, i) => flat.set(v, i * 4));
    82→    return this.accessor(this.view(flat), 5126, values.length, 'VEC4', { min: [0, 0, 0, 0], max: [1, 1, 1, 1] });
    83→  }
    84→
    85→  /** Cubo unitario (24 vértices con normales por cara, 36 índices). Se reutiliza con node.scale. */
    86→  private static cubeGeometry(): { pos: Float32Array; nor: Float32Array; idx: Uint16Array } {
    87→    // Cara: normal + 4 esquinas (x,y,z con extensión ±0.5) en orden CCW.
    88→    const faces: { n: Vec3; corners: Vec3[] }[] = [
    89→      { n: [1, 0, 0], corners: [[0.5, -0.5, 0.5], [0.5, -0.5, -0.5], [0.5, 0.5, -0.5], [0.5, 0.5, 0.5]] },
    90→      { n: [-1, 0, 0], corners: [[-0.5, -0.5, -0.5], [-0.5, -0.5, 0.5], [-0.5, 0.5, 0.5], [-0.5, 0.5, -0.5]] },
    91→      { n: [0, 1, 0], corners: [[-0.5, 0.5, 0.5], [0.5, 0.5, 0.5], [0.5, 0.5, -0.5], [-0.5, 0.5, -0.5]] },
    92→      { n: [0, -1, 0], corners: [[-0.5, -0.5, -0.5], [0.5, -0.5, -0.5], [0.5, -0.5, 0.5], [-0.5, -0.5, 0.5]] },
    93→      { n: [0, 0, 1], corners: [[-0.5, -0.5, 0.5], [0.5, -0.5, 0.5], [0.5, 0.5, 0.5], [-0.5, 0.5, 0.5]] },
    94→      { n: [0, 0, -1], corners: [[0.5, -0.5, -0.5], [-0.5, -0.5, -0.5], [-0.5, 0.5, -0.5], [0.5, 0.5, -0.5]] },
    95→    ];
    96→    const pos: number[] = [];
    97→    const nor: number[] = [];
    98→    const idx: number[] = [];
    99→    faces.forEach((f, fi) => {
   100→      const base = fi * 4;
   101→      f.corners.forEach((c) => {
   102→        pos.push(...c);
   103→        nor.push(...f.n);
   104→      });
   105→      idx.push(base, base + 1, base + 2, base, base + 2, base + 3);
   106→    });
   107→    return { pos: new Float32Array(pos), nor: new Float32Array(nor), idx: new Uint16Array(idx) };
   108→  }
   109→
   110→  private cubeAccessors: { pos: number; nor: number; idx: number } | null = null;
   111→
   112→  /** Mesh "cubo" con un material; todas las piezas reutilizan estos accessors vía node.scale. */
   113→  addBoxMesh(name: string, materialIdx: number): number {
   114→    if (!this.cubeAccessors) {
   115→      const g = Gltf.cubeGeometry();
   116→      const pv = this.view(g.pos, 34962);
   117→      const nv = this.view(g.nor, 34962);
   118→      const iv = this.view(g.idx, 34963);
   119→      const posMin = [-0.5, -0.5, -0.5];
   120→      const posMax = [0.5, 0.5, 0.5];
   121→      this.cubeAccessors = {
   122→        pos: this.accessor(pv, 5126, 24, 'VEC3', { min: posMin, max: posMax }),
   123→        nor: this.accessor(nv, 5126, 24, 'VEC3', { min: [-1, -1, -1], max: [1, 1, 1] }),
   124→        idx: this.accessor(iv, 5123, 36, 'SCALAR'),
   125→      };
   126→    }
   127→    this.meshes.push({
   128→      name,
   129→      primitives: [{ attributes: { POSITION: this.cubeAccessors.pos, NORMAL: this.cubeAccessors.nor }, indices: this.cubeAccessors.idx, material: materialIdx }],
   130→    });
   131→    return this.meshes.length - 1;
   132→  }
   133→
   134→  addMaterial(name: string, hex: string, opts: { metallic?: number; roughness?: number } = {}): number {
   135→    this.materials.push({
   136→      name,
   137→      pbrMetallicRoughness: {
   138→        baseColorFactor: colorLinear(hex),
   139→        metallicFactor: opts.metallic ?? 0,
   140→        roughnessFactor: opts.roughness ?? 0.85,
   141→      },
   142→    });
   143→    return this.materials.length - 1;
   144→  }
   145→
   146→  addNode(n: { name: string; mesh?: number; translation?: Vec3; scale?: Vec3; rotation?: Quat; children?: number[] }): number {
   147→    this.nodes.push({ name: n.name, ...(n.mesh !== undefined ? { mesh: n.mesh } : {}), ...(n.translation ? { translation: n.translation } : {}), ...(n.scale ? { scale: n.scale } : {}), ...(n.rotation ? { rotation: n.rotation } : {}), ...(n.children ? { children: n.children } : {}) });
   148→    return this.nodes.length - 1;
   149→  }
   150→
   151→  /** Añade una animación: cada canal es { node, path, times, values } (values: VEC3 o VEC4 ya serializados). */
   152→  addAnimation(name: string, channels: { node: number; path: 'translation' | 'rotation'; times: number[]; values: number[][] }[]): void {
   153→    const samplers: { input: number; output: number; interpolation: 'LINEAR' }[] = [];
   154→    const chs: { sampler: number; target: { node: number; path: string } }[] = [];
   155→    for (const c of channels) {
   156→      const input = this.times(c.times);
   157→      const output = c.path === 'rotation' ? this.vec4(c.values) : this.vec3(c.values);
   158→      samplers.push({ input, output, interpolation: 'LINEAR' });
   159→      chs.push({ sampler: samplers.length - 1, target: { node: c.node, path: c.path } });
   160→    }
   161→    this.animations.push({ name, channels: chs, samplers });
   162→  }
   163→
   164→  finish(generator: string): Record<string, unknown> {
   165→    const total = new Uint8Array(this.byteLen);
   166→    let off = 0;
   167→    for (const c of this.chunks) {
   168→      total.set(c, off);
   169→      off += c.byteLength;
   170→    }
   171→    const uri = `data:application/octet-stream;base64,${Buffer.from(total).toString('base64')}`;
   172→    return {
   173→      asset: { version: '2.0', generator },
   174→      scene: 0,
   175→      scenes: [{ nodes: [0] }],
   176→      nodes: this.nodes,
   177→      meshes: this.meshes,
   178→      materials: this.materials,
   179→      buffers: [{ uri, byteLength: this.byteLen }],
   180→      bufferViews: this.bufferViews,
   181→      accessors: this.accessors,
   182→      ...(this.animations.length ? { animations: this.animations } : {}),
   183→    };
   184→  }
   185→}

   189→function buildKid(): Record<string, unknown> {
   190→  const g = new Gltf();
   191→  const mHoodie = g.addMaterial('hoodie', '#f77f00');
   192→  const mPants = g.addMaterial('pants', '#3a4a5c');
   193→  const mSkin = g.addMaterial('skin', '#f2cc8f');
   194→  const mCap = g.addMaterial('cap', '#e63946');
   195→  const mAccent = g.addMaterial('accent', '#ffd166');
   196→  const mShoe = g.addMaterial('shoe', '#293241');
   197→
   198→  const meshHoodie = g.addBoxMesh('torso', mHoodie);
   199→  const meshArm = g.addBoxMesh('brazo', mHoodie);
   200→  const meshHand = g.addBoxMesh('mano', mSkin);
   201→  const meshPants = g.addBoxMesh('pierna', mPants);
   202→  const meshShoe = g.addBoxMesh('zapato', mShoe);
   203→  const meshHead = g.addBoxMesh('cabeza', mSkin);
   204→  const meshCap = g.addBoxMesh('gorra', mCap);
   205→  const meshVisor = g.addBoxMesh('visor', mCap);
   206→  const meshCan = g.addBoxMesh('spray', mAccent);
   207→
   208→  // Jerarquía: pies en y=0, altura nominal 1.2 m, mira hacia +X.
   209→  // Root → hips → { torso, headPivot → {head, cap, visor}, armL, armR(+spray), legL, legR }
   210→  const root = g.addNode({ name: 'Root', children: [] });
   211→  const hips = g.addNode({ name: 'hips', translation: [0, 0.45, 0], children: [] });
   212→  const torso = g.addNode({ name: 'torso', mesh: meshHoodie, translation: [0, 0.21, 0], scale: [0.34, 0.42, 0.22] });
   213→  const headPivot = g.addNode({ name: 'headPivot', translation: [0, 0.44, 0], children: [] });
   214→  const head = g.addNode({ name: 'head', mesh: meshHead, translation: [0, 0.13, 0], scale: [0.24, 0.26, 0.24] });
   215→  const cap = g.addNode({ name: 'cap', mesh: meshCap, translation: [0.01, 0.27, 0], scale: [0.27, 0.09, 0.27] });
   216→  const visor = g.addNode({ name: 'visor', mesh: meshVisor, translation: [0.17, 0.25, 0], scale: [0.12, 0.035, 0.2] });
   217→  const armL = g.addNode({ name: 'armL', translation: [-0.21, 0.36, 0], children: [] });
   218→  const armLMesh = g.addNode({ name: 'armLMesh', mesh: meshArm, translation: [0, -0.16, 0], scale: [0.09, 0.32, 0.12] });
   219→  const armR = g.addNode({ name: 'armR', translation: [0.21, 0.36, 0], children: [] });
   220→  const armRMesh = g.addNode({ name: 'armRMesh', mesh: meshArm, translation: [0, -0.16, 0], scale: [0.09, 0.32, 0.12] });
   221→  const can = g.addNode({ name: 'spray_can', mesh: meshCan, translation: [0, -0.37, 0], scale: [0.09, 0.13, 0.09] });
   222→  const legL = g.addNode({ name: 'legL', translation: [-0.08, 0.01, 0], children: [] });
   223→  const legLMesh = g.addNode({ name: 'legLMesh', mesh: meshPants, translation: [0, -0.2, 0], scale: [0.11, 0.4, 0.13] });
   224→  const legR = g.addNode({ name: 'legR', translation: [0.08, 0.01, 0], children: [] });
   225→  const legRMesh = g.addNode({ name: 'legRMesh', mesh: meshPants, translation: [0, -0.2, 0], scale: [0.11, 0.4, 0.13] });
   226→  const shoeL = g.addNode({ name: 'shoeL', mesh: meshShoe, translation: [0.02, -0.43, 0], scale: [0.13, 0.06, 0.14] });
   227→  const shoeR = g.addNode({ name: 'shoeR', mesh: meshShoe, translation: [0.02, -0.43, 0], scale: [0.13, 0.06, 0.14] });
   228→
   229→  // Puntas de jerarquía (por índice devuelto, no literal).
   230→  (g.nodes[root] as { children: number[] }).children = [hips];
   231→  (g.nodes[hips] as { children: number[] }).children = [torso, headPivot, armL, armR, legL, legR];
   232→  (g.nodes[headPivot] as { children: number[] }).children = [head, cap, visor];
   233→  (g.nodes[armL] as { children: number[] }).children = [armLMesh];
   234→  (g.nodes[armR] as { children: number[] }).children = [armRMesh, can];
   235→  (g.nodes[legL] as { children: number[] }).children = [legLMesh, shoeL];
   236→  (g.nodes[legR] as { children: number[] }).children = [legRMesh, shoeR];
   237→
   238→  const loopTimes = (T: number, n: number) => Array.from({ length: n + 1 }, (_, i) => (T * i) / n);
   239→  const sinKeys = (T: number, amp: number, base = 0, cycles = 1) => {
   240→    const t = loopTimes(T, 8);
   241→    const v = t.map((x) => base + amp * Math.sin((2 * Math.PI * cycles * x) / T));
   242→    return { t, v };
   243→  };
   244→
   245→  // ── idle: balanceo sutil (2.4 s) ──
   246→  {
   247→    const bob = sinKeys(2.4, 0.012, 0.45);
   248→    const headTilt = sinKeys(2.4, 0.06);
   249→    const armSwayL = sinKeys(2.4, 0.03, 0.06); // brazo izquierdo algo separado del cuerpo
   250→    const armSwayR = sinKeys(2.4, 0.03, -0.06);
   251→    g.addAnimation('idle', [
   252→      { node: hips, path: 'translation', times: bob.t, values: bob.v.map((y) => [0, y, 0]) },
   253→      { node: headPivot, path: 'rotation', times: headTilt.t, values: headTilt.v.map((z) => quatAxis(0, 0, 1, z)) },
   254→      { node: armL, path: 'rotation', times: armSwayL.t, values: armSwayL.v.map((z) => quatAxis(0, 0, 1, z)) },
   255→      { node: armR, path: 'rotation', times: armSwayR.t, values: armSwayR.v.map((z) => quatAxis(0, 0, 1, z)) },
   256→    ]);
   257→  }
   258→
   259→  // ── walk: ciclo de paso (0.8 s) ──
   260→  {
   261→    const t = loopTimes(0.8, 8);
   262→    const swing = t.map((x) => 0.55 * Math.sin((2 * Math.PI * x) / 0.8));
   263→    const bounce = t.map((x) => 0.45 + 0.018 * Math.abs(Math.sin((2 * Math.PI * 2 * x) / 0.8)));
   264→    g.addAnimation('walk', [
   265→      { node: legL, path: 'rotation', times: t, values: swing.map((s) => quatAxis(1, 0, 0, s)) },
   266→      { node: legR, path: 'rotation', times: t, values: swing.map((s) => quatAxis(1, 0, 0, -s)) },
   267→      { node: armL, path: 'rotation', times: t, values: swing.map((s) => quatAxis(1, 0, 0, -s * 0.8)) },
   268→      { node: armR, path: 'rotation', times: t, values: swing.map((s) => quatAxis(1, 0, 0, s * 0.8)) },
   269→      { node: hips, path: 'translation', times: t, values: bounce.map((y) => [0, y, 0]) },
   270→    ]);
   271→  }
   272→
   273→  // ── spray: levanta el brazo derecho con la lata y "pinta" (1.6 s) ──
   274→  {
   275→    const t = [0, 0.25, 0.45, 0.8, 1.15, 1.4, 1.6];
   276→    const raise = [0, 1.9, 2.85, 2.72, 2.98, 2.85, 0]; // 0 → arriba (~163°) con vaivén → vuelve
   277→    const shake = [0, 0.05, -0.06, 0.06, -0.06, 0.05, 0];
   278→    const hipSway = sinKeys(1.6, 0.015, 0.45, 2);
   279→    g.addAnimation('spray', [
   280→      { node: armR, path: 'rotation', times: t, values: raise.map((z) => quatAxis(0, 0, 1, z)) },
   281→      { node: armL, path: 'rotation', times: t, values: t.map(() => quatAxis(0, 0, 1, -0.5)) }, // mano en la cadera
   282→      { node: hips, path: 'rotation', times: t, values: shake.map((y) => quatAxis(0, 1, 0, y)) },
   283→      { node: hips, path: 'translation', times: loopTimes(1.6, 8), values: hipSway.v.map((y) => [0, y, 0]) },
   284→    ]);
   285→  }
   286→
   287→  return g.finish('vecinos-gen-pilot-assets (nino_grafitero v1)');
   288→}

   323→// Convenciones idénticas al niño (verificadas en juego): modelo mira a +X, pivote de brazos/piernas
   324→// con rotación Z = vaivén adelante-atrás, X = lateral; hips bounce en Y. nominalHeight 1.35 m.

   326→interface AdultRig {
   327→  g: Gltf;
   328→  root: number;
   329→  hips: number;
   330→  headPivot: number;
   331→  armL: number;
   332→  armR: number;
   333→  legL: number;
   334→  legR: number;
   335→}
   336→
   337→function buildAdult(opts: {
   338→  torso: string;
   339→  pants: string;
   340→  skin: string;
   341→  shoe: string;
   342→  hipY?: number;
   343→  torsoScale?: [number, number, number];
   344→}): AdultRig {
   345→  const g = new Gltf();
   346→  const mTorso = g.addMaterial('torso', opts.torso);
   347→  const mPants = g.addMaterial('pants', opts.pants);
   348→  const mSkin = g.addMaterial('skin', opts.skin);
   349→  const mShoe = g.addMaterial('shoe', opts.shoe);
   350→
   351→  const meshTorso = g.addBoxMesh('torso', mTorso);
   352→  const meshArm = g.addBoxMesh('brazo', mTorso);
   353→  const meshHand = g.addBoxMesh('mano', mSkin);
   354→  const meshPants = g.addBoxMesh('pierna', mPants);
   355→  const meshShoe = g.addBoxMesh('zapato', mShoe);
   356→  const meshHead = g.addBoxMesh('cabeza', mSkin);
   357→
   358→  const hipY = opts.hipY ?? 0.52;
   359→  const root = g.addNode({ name: 'Root', children: [] });
   360→  const hips = g.addNode({ name: 'hips', translation: [0, hipY, 0], children: [] });
   361→  const torso = g.addNode({ name: 'torso', mesh: meshTorso, translation: [0, 0.26, 0], scale: opts.torsoScale ?? [0.36, 0.5, 0.24] });
   362→  const headPivot = g.addNode({ name: 'headPivot', translation: [0, 0.54, 0], children: [] });
   363→  const head = g.addNode({ name: 'head', mesh: meshHead, translation: [0, 0.15, 0], scale: [0.25, 0.28, 0.25] });
   364→  const armL = g.addNode({ name: 'armL', translation: [-0.23, 0.44, 0], children: [] });
   365→  const armLMesh = g.addNode({ name: 'armLMesh', mesh: meshArm, translation: [0, -0.17, 0], scale: [0.09, 0.34, 0.11] });
   366→  const handL = g.addNode({ name: 'handL', mesh: meshHand, translation: [0, -0.37, 0], scale: [0.08, 0.09, 0.09] });
   367→  const armR = g.addNode({ name: 'armR', translation: [0.23, 0.44, 0], children: [] });
   368→  const armRMesh = g.addNode({ name: 'armRMesh', mesh: meshArm, translation: [0, -0.17, 0], scale: [0.09, 0.34, 0.11] });
   369→  const handR = g.addNode({ name: 'handR', mesh: meshHand, translation: [0, -0.37, 0], scale: [0.08, 0.09, 0.09] });
   370→  const legL = g.addNode({ name: 'legL', translation: [-0.09, 0.02, 0], children: [] });
   371→  const legLMesh = g.addNode({ name: 'legLMesh', mesh: meshPants, translation: [0, -0.23, 0], scale: [0.11, 0.46, 0.12] });
   372→  const shoeL = g.addNode({ name: 'shoeL', mesh: meshShoe, translation: [0.02, -0.48, 0], scale: [0.13, 0.06, 0.14] });
   373→  const legR = g.addNode({ name: 'legR', translation: [0.09, 0.02, 0], children: [] });
   374→  const legRMesh = g.addNode({ name: 'legRMesh', mesh: meshPants, translation: [0, -0.23, 0], scale: [0.11, 0.46, 0.12] });
   375→  const shoeR = g.addNode({ name: 'shoeR', mesh: meshShoe, translation: [0.02, -0.48, 0], scale: [0.13, 0.06, 0.14] });
   376→
   377→  (g.nodes[root] as { children: number[] }).children = [hips];
   378→  (g.nodes[hips] as { children: number[] }).children = [torso, headPivot, armL, armR, legL, legR];
   379→  (g.nodes[headPivot] as { children: number[] }).children = [head];
   380→  (g.nodes[armL] as { children: number[] }).children = [armLMesh, handL];
   381→  (g.nodes[armR] as { children: number[] }).children = [armRMesh, handR];
   382→  (g.nodes[legL] as { children: number[] }).children = [legLMesh, shoeL];
   383→  (g.nodes[legR] as { children: number[] }).children = [legRMesh, shoeR];
   384→
   385→  return { g, root, hips, headPivot, armL, armR, legL, legR };
   386→}
   387→
   388→const loopTimes = (T: number, n: number) => Array.from({ length: n + 1 }, (_, i) => (T * i) / n);
   389→const sinKeys = (T: number, amp: number, base = 0, cycles = 1) => {
   390→  const t = loopTimes(T, 8);
   391→  const v = t.map((x) => base + amp * Math.sin((2 * Math.PI * cycles * x) / T));
   392→  return { t, v };
   393→};

   395→/** idle + walk estándar de adulto sobre el rig; devuelve los nodos para el clip extra. */
   396→function addBaseClips(r: AdultRig, period = 2.6): void {
   397→  const { g, hips, headPivot, armL, armR, legL, legR } = r;
   398→  const bob = sinKeys(period, 0.012, 0.52);
   399→  const headTilt = sinKeys(period, 0.05);
   400→  const swayL = sinKeys(period, 0.03, 0.05);
   401→  const swayR = sinKeys(period, 0.03, -0.05);
   402→  g.addAnimation('idle', [
   403→    { node: hips, path: 'translation', times: bob.t, values: bob.v.map((y) => [0, y, 0]) },
   404→    { node: headPivot, path: 'rotation', times: headTilt.t, values: headTilt.v.map((z) => quatAxis(0, 0, 1, z)) },
   405→    { node: armL, path: 'rotation', times: swayL.t, values: swayL.v.map((z) => quatAxis(0, 0, 1, z)) },
   406→    { node: armR, path: 'rotation', times: swayR.t, values: swayR.v.map((z) => quatAxis(0, 0, 1, z)) },
   407→  ]);
   408→
   409→  const t = loopTimes(0.85, 8);
   410→  const swing = t.map((x) => 0.5 * Math.sin((2 * Math.PI * x) / 0.85));
   411→  const bounce = t.map((x) => 0.52 + 0.016 * Math.abs(Math.sin((2 * Math.PI * 2 * x) / 0.85)));
   412→  g.addAnimation('walk', [
   413→    { node: legL, path: 'rotation', times: t, values: swing.map((s) => quatAxis(1, 0, 0, s)) },
   414→    { node: legR, path: 'rotation', times: t, values: swing.map((s) => quatAxis(1, 0, 0, -s)) },
   415→    { node: armL, path: 'rotation', times: t, values: swing.map((s) => quatAxis(1, 0, 0, -s * 0.75)) },
   416→    { node: armR, path: 'rotation', times: t, values: swing.map((s) => quatAxis(1, 0, 0, s * 0.75)) },
   417→    { node: hips, path: 'translation', times: t, values: bounce.map((y) => [0, y, 0]) },
   418→  ]);
   419→}

   421→/** Dolores, la Cotilla: bata ciruela, rodete y prismáticos de Gala. Clip 'watch' (vigila el pasillo). */
   422→function buildDolores(): Record<string, unknown> {
   423→  const r = buildAdult({ torso: '#b04a7d', pants: '#6d3b56', skin: '#e8b98a', shoe: '#3d2c3a' });
   424→  const { g, hips, headPivot, armL, armR } = r;
   425→  const mHair = g.addMaterial('rodete', '#4a3b45');
   426→  const mGlass = g.addMaterial('prismatico', '#22303a', { metallic: 0.3, roughness: 0.4 });
   427→  const meshHair = g.addBoxMesh('rodete', mHair);
   428→  const meshOpt = g.addBoxMesh('tubo', mGlass);
   429→  const bun = g.addNode({ name: 'bun', mesh: meshHair, translation: [0, 0.34, 0], scale: [0.12, 0.12, 0.12] });
   430→  (g.nodes[headPivot] as { children: number[] }).children.push(bun);
   431→  // Prismáticos colgando del pecho (child de hips, delante): dos tubos + puente.
   432→  const optL = g.addNode({ name: 'optL', mesh: meshOpt, translation: [0.2, 0.18, -0.06], scale: [0.09, 0.12, 0.07] });
   433→  const optR = g.addNode({ name: 'optR', mesh: meshOpt, translation: [0.2, 0.18, 0.06], scale: [0.09, 0.12, 0.07] });
   434→  const optBridge = g.addNode({ name: 'optBridge', mesh: meshOpt, translation: [0.2, 0.2, 0], scale: [0.03, 0.03, 0.08] });
   435→  (g.nodes[hips] as { children: number[] }).children.push(optL, optR, optBridge);
   436→
   437→  addBaseClips(r, 2.4);
   438→  // watch: prismáticos a los ojos (brazos al frente con temblor de escaneo) 2.8 s
   439→  const t = loopTimes(2.8, 10);
   440→  const scan = t.map((x) => 1.35 + 0.09 * Math.sin((2 * Math.PI * 2 * x) / 2.8));
   441→  g.addAnimation('watch', [
   442→    { node: armL, path: 'rotation', times: t, values: scan.map((z) => quatAxis(0, 0, 1, z)) },
   443→    { node: armR, path: 'rotation', times: t, values: scan.map((z) => quatAxis(0, 0, 1, z)) },
   444→    { node: headPivot, path: 'rotation', times: t, values: t.map((x) => quatAxis(0, 1, 0, 0.35 * Math.sin((2 * Math.PI * x) / 2.8))) },
   445→  ]);
   446→  return g.finish('vecinos-gen-pilot-assets (cotilla v1)');
   447→}
