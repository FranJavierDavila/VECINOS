    63→    63→  async init(canvas: HTMLCanvasElement): Promise<void> {
    64→    64→    if (this.engine || this.initializing) return;
    65→    65→    this.initializing = true;
    66→    66→    this.canvas = canvas;
    67→    67→
    68→    68→    try {
    69→    69→      // 1) Contenido externo + validación (los errores de referencias se avisan, no abortan).
    70→    70→      const { content, issues } = await loadContent();
    71→    71→      this._content = content;
    72→    72→      preloadSounds();
    73→    73→      useGameStore.setState({ touchHint: content.rules.touch.hint });
    74→    74→      if (issues.length > 0) {
    75→    75→        useGameStore.getState().pushToast(`Contenido inválido: ${issues.length} errores de validación`, 'bad');
    76→    76→      }
    77→    77→
    78→    78→      // 2) Motor de render (WebGPU con fallback WebGL2).
    79→    79→      const { engine, backend } = await createEngine(canvas);
    80→    80→      this.engine = engine;
    81→    81→      window.addEventListener('resize', this.onResize);
    82→    82→      await this.restart();
    83→    83→      useGameStore.getState().markReady(backend);
    84→    84→      engine.runRenderLoop(this.frame);
    85→    85→      // Gancho de depuración/pruebas E2E (no lo usa la UI).
    86→    86→      (window as unknown as { __vecinos?: unknown }).__vecinos = { store: useGameStore, controller: this };
    87→    87→    } catch (err) {
    88→    88→      console.error('No se pudo arrancar el juego (contenido o motor de render)', err);
    89→    89→      useGameStore.getState().setEngineError(true);
    90→    90→    }

   285→   285→  private connectNet(): void {
   286→   286→    this.net = new NetSession({
   287→   287→      onState: (s) => this.applyRemoteState(s),
   288→   288→      onParticipants: (p) => useGameStore.getState().setMpParticipants(p),
   289→   289→      onJoined: (j) => {
   290→   290→        this.myResidentId = j.yourResidentId;
   291→   291→        this.isHost = j.isHost;
   292→   292→        useGameStore.getState().setMpSession({ code: j.code, isHost: j.isHost, myResidentId: j.yourResidentId });
   293→   293→        useGameStore.getState().setMpParticipants(j.participants);
   294→   294→      },
   295→   295→      onError: (m) => {
   296→   296→        useGameStore.getState().pushToast(m, 'bad');
   297→   297→      },
   298→   298→      onStatus: (st) => {
   299→   299→        if (!this.net) return;
   300→   300→        if (st === 'connected' && useGameStore.getState().mpStatus === 'connecting') return; // el ack decide
   301→   301→        if (st === 'disconnected' && useGameStore.getState().mpStatus === 'connected') {
   302→   302→          useGameStore.getState().setMpStatus('error');
   303→   303→          useGameStore.getState().setMpError('Conexión perdida con el servidor de la comunidad.');
   304→   304→        }
   305→   305→      },
   306→   306→    });
   307→   307→  }

   318→   318→  }

   320→   320→  /** Adopta el estado del servidor (autoridad). Primera vez: reinicia la escena con ese mundo. */
   321→   321→  private applyRemoteState(s: SessionStatePayload): void {
   322→   322→    if (s.seq <= this.appliedSeq) return;
   323→   323→    this.appliedSeq = s.seq;
   324→   324→    if (!this.world || this.world.scenarioId !== s.world.scenarioId) {
   325→   325→      void this.restart(s.world).then(() => {
   326→   326→        if (!this.world) return;
   327→   327→        this.world.events = s.events.slice();
   328→   328→        this.handleEvents();
   329→   329→        this.reconcile();
   330→   330→        this.syncHud();
   331→   331→      });
   332→   332→      return;
   333→   333→    }
   334→   334→    // Aplica en sitio conservando la identidad del objeto (PlayerLogic/building/actors referencian el mismo World).
   335→   335→    Object.assign(this.world, s.world);
   336→   336→    this.world.events = s.events.slice();
   337→   337→    this.handleEvents();
   338→   338→    this.paintBuilding();
   339→   339→    this.reconcile();
   340→   340→    this.syncHud();
   341→   341→  }

   343→   343→  /** Corrige desvíos de posición del avatar local respecto al mundo del servidor (autoridad). */
   344→   344→  private reconcile(): void {
   345→   345→    if (!this.logic || !this.world || !this.net) return;
   346→   346→    const me = this.world.residents[this.actorId()];
   347→   347→    if (!me) return;
   348→   348→    const pos = this.logic.position();
   349→   349→    const locBlock = me.location ? this.world.blocks[me.location] : null;
   350→   350→    if (locBlock) {
   351→   351→      const insideOk = pos.mode === 'inside' && pos.insideBlock === locBlock.id;
   352→   352→      const corridorOk = pos.mode === 'corridor' && !pos.insideBlock;
   353→   353→      if (!(locBlock.kind === 'dwelling' ? insideOk : corridorOk) || Math.abs(pos.floor - locBlock.cell.floor) > 0.01) {
   354→   354→        this.logic.forceLocation(locBlock.id);
   355→   355→      }
   356→   356→    } else if (pos.mode === 'inside') {
   357→   357→      this.logic.forceLocation(null);
   358→   358→    }
   359→   359→  }

   516→   516→      actorId: this.myResidentId ?? undefined,

   519→   519→
   520→   520→    this.scene?.dispose();

   522→   522→    this.scene.clearColor = new Color4(view.sky.r, view.sky.g, view.sky.b, 1);
   523→   523→    new HemisphericLight('ambient', new Vector3(...view.ambient.dir), this.scene).intensity = view.ambient.intensity;
   524→   524→    new DirectionalLight('sun', new Vector3(...view.sun.dir), this.scene).intensity = view.sun.intensity;
   525→   525→
   526→   526→    this.building = createBuilding(this.scene, this.world, this.content);
   527→   527→    this.actors = createActors(this.scene, this.world, this.content);
   528→   528→    const blocks = Object.values(this.world.blocks);
   529→   529→    this.rig = new CameraRig(
   530→   530→      new UniversalCamera('cam', new Vector3(0, 0, -30), this.scene),

   533→   533→      Math.max(...blocks.map((b) => b.cell.floor)) + 1,
   534→   534→      view.cellW,
   535→   535→      view.cellH,
   536→   536→    );
   537→   537→    // Re-sincronización de vista (v0.6-b): la cámara sigue la posición del mundo (restaurado o
   538→   538→    // nuevo), no la preferencia anterior. Dentro de una vivienda → estancia; en pasillo, la
   539→   539→    // preferencia salvo que fuera 'estancia' (ahí estás en el pasillo, no en una estancia).
   540→   540→    const pos = this.logic.position();
   541→   541→    const store = useGameStore.getState();
   542→   542→    const wanted: CamMode = pos.mode === 'inside' ? 'estancia' : store.camMode === 'estancia' ? 'planta' : store.camMode;
   543→   543→    this.rig.mode = wanted;

   548→   548→    this.phoneLines = [];
   549→   549→    this.phoneMission = '';
   550→   550→    this.lastHudSync = 0;

   560→   560→  private frame = (): void => {
   561→   561→    if (!this.engine || !this.scene || !this.world || !this.logic || !this.building || !this.actors || !this.rig) return;
   562→   562→    const dt = Math.min(0.1, this.engine.getDeltaTime() / 1000);
   563→   563→    const store = useGameStore.getState();
   564→   564→    const frozen = selectModalOpen(store) || store.paused;
   565→   565→
   566→   566→    let ticked = false;
   567→   567→    if (!frozen) {
   568→   568→      this.logic.update(dt, this.readInput());
   569→   569→      // En remoto el servidor es la autoridad del reloj: nada de ticks locales.
   570→   570→      if (!this.net) {
   571→   571→        this.clockAcc += dt * store.speed;
   572→   572→        let guard = 0;
   573→   573→        while (this.clockAcc >= 1 && guard < 60) {
   574→   574→          tick(this.world, this.content);
   575→   575→          this.clockAcc -= 1;
   576→   576→          ticked = true;
   577→   577→          guard += 1;
   578→   578→        }
   579→   579→      }
   580→   580→    }
   581→   581→
   582→   582→    if (this.world.events.length > 0) this.handleEvents();
   583→   583→    if (ticked) this.paintBuilding();
   584→   584→
   585→   585→    const now = performance.now();
   586→   586→    if (now - this.lastHudSync > 200) {
   587→   587→      this.lastHudSync = now;
   588→   588→      this.syncHud();
   589→   589→    }
   590→   590→
   591→   591→    const pos = this.logic.position();
   592→   592→    this.actors.update(dt, pos, this.actorId());
   593→   593→    // LOD por distancia: solo con cámaras cercanas (planta/estancia); el plano general (edificio) muestra todo.
   594→   594→    this.building.update(dt, useGameStore.getState().camMode === 'edificio' ? undefined : { x: pos.x, floor: pos.floor });
   595→   595→    this.rig.update(dt, pos);
   596→   596→    this.scene.render();
   597→   597→  };

   793→   793→        return { title: o.title, text: 'Entra en tu piso con E.' };
   794→   794→      case 'catch_in_act':
   795→   795→        return { title: o.title, text: `Pilla al responsable in fraganti${left}.` };
   796→   796→      default:
   797→   797→        return { title: o.title, text: o.title };
   798→   798→    }
   799→   799→  }
   800→   800→
   801→   801→  private promptText(): string | null {
   802→   802→    const logic = this.logic;
   803→   803→    const world = this.world;
   804→   804→    if (!logic || !world) return null;
   805→   805→    if (logic.position().climbing) return null;
   806→   806→    // El prompt es SIEMPRE el objetivo real de la tecla E (misma fuente, cero divergencias).
   807→   807→    const t = logic.interactTarget();
   808→   808→    if (t) {
   809→   809→      const ic = this.content.rules.interact;
   810→   810→      switch (t.kind) {
   811→   811→        case 'door': {
   812→   812→          const door = world.blocks[t.blockId]!;
   813→   813→          if (logic.position().mode === 'inside') return ic.exit;
   814→   814→          return door.access === 'locked'
   815→   815→            ? ic.doorLocked.replace('{block}', blockLabel(door.id))
   816→   816→            : ic.doorEnter.replace('{block}', blockLabel(door.id));
   817→   817→        }
   818→   818→        case 'resident':
   819→   819→          return ic.greet.replace('{name}', this.nameOf(t.residentId));
   820→   820→        case 'mailbox':
   821→   821→          return ic.mailbox.prompt;
   822→   822→        case 'noticeboard':
   823→   823→          return ic.noticeboard.prompt;
   824→   824→        case 'laptop':
   825→   825→          return ic.laptop;
   826→   826→        case 'bed':
   827→   827→          return ic.bed;
   828→   828→      }
   829→   829→    }
   830→   830→    const zone = logic.zone();
   831→   831→    const feat = this.content.blocks[zone.defId]!.features;
   832→   832→    if (feat?.stairs || feat?.elevator) {
   833→   833→      const ic = this.content.rules.interact;
   834→   834→      // Lo que dice el prompt es lo que W/S hace aquí: ascensor operativo manda; si no hay
   835→   835→      // ascensor utilizable pero sí escalera, escalera; núcleo averiado sin escalera = aviso.
   836→   836→      if (feat.elevator && world.blocks[zone.id]!.params.elevator_status === 'ok') return ic.elevator;
   837→   837→      if (feat.stairs) return ic.stairs;
   838→   838→      return ic.elevatorBroken;
   839→   839→    }
   840→   840→    return null;
   841→   841→  }

   843→   843→  /** Viviendas del jugador para el editor de interiores (foto serializable, posiciones ya en %). */
   844→   844→  private homesUi() {
   845→   845→    const world = this.world!;
   846→   846→    const cellH = this.content.rules.view.cellH;
   847→   847→    const mine = this.actorId();
   848→   848→    return Object.values(world.blocks)
   849→   849→      .filter((b) => b.kind === 'dwelling' && b.ownerId === mine)
   850→   850→      .sort((a, b) => a.id.localeCompare(b.id))
   851→   851→      .map((b) => ({
   852→   852→        id: b.id,
   853→   853→        label: blockLabel(b.id),
   854→   854→        rooms: b.rooms.map((room) => {
   855→   855→          const rdef = this.content.rooms[room.defId];
   856→   856→          const slots = (rdef?.furnitureSlots ?? []).map((s) => {
   857→   857→            const itemId = room.furniture[s.id] ?? '';
   858→   858→            const fdef = itemId ? this.content.furniture[itemId] : undefined;
   859→   859→            return {
   860→   860→              id: s.id,
   861→   861→              kind: s.kind,
   862→   862→              left: Math.min(92, Math.max(2, (s.pos[0] / Math.max(0.001, rdef?.size.w ?? 1)) * 100)),
   863→   863→              bottom: Math.min(76, Math.max(0, (s.pos[1] / Math.max(0.001, cellH)) * 100)),
   864→   864→              item: fdef ? { id: fdef.id, name: fdef.name, emoji: fdef.emoji ?? '📦', color: fdef.view.color } : null,
   865→   865→            };
   866→   866→          });
   867→   867→          return { key: room.key, label: rdef?.label ?? room.key, access: room.access, slots };
   868→   868→        }),
   869→   869→      }));
   870→   870→  }

   872→   872→  private syncHud(): void {
   873→   873→    const world = this.world;
   874→   874→    if (!world) return;
   875→   875→    const player = world.residents[this.actorId()] ?? world.residents[world.playerId];
   876→   876→    if (!player) return;
   877→   877→    const home = player.home;
   878→   878→    const params = visibleParams(world, this.content, home).map((p) => ({ name: p.name, label: p.def.label ?? p.name, value: String(p.value) }));
   879→   879→    const missions = world.missions.map((m) => {
   880→   880→      const def = this.content.missions[m.id]!;
   881→   881→      const step = m.state === 'active' ? def.steps[m.step] : undefined;
   882→   882→      return {
   883→   883→        id: m.id,
   884→   884→        title: def.title,
   885→   885→        state: m.state,
   886→   886→        hint: step?.hint ?? null,
   887→   887→        remaining: m.until !== null ? `quedan ${Math.max(0, m.until - world.tick)} min` : null,
   888→   888→      };
   889→   889→    });
   890→   890→    const travel = travelMinutes(buildCirculation(world, this.content), world.entry, home);
   891→   891→    const elevatorBlock = Object.values(world.blocks).find((b) => b.params.elevator_status !== undefined);
   892→   892→    useGameStore.getState().syncHud({
   893→   893→      clock: { ...world.clock },
   894→   894→      weather: world.weather.state,
   895→   895→      funds: world.community.funds,
   896→   896→      wallet: player.wallet,
   897→   897→      reputation: player.reputation,
   898→   898→      elevator: String(elevatorBlock?.params.elevator_status ?? 'ok'),
   899→   899→      homeLabel: blockLabel(home),
   900→   900→      travelText: Number.isFinite(travel) ? `del portal a tu puerta: ${travel} min` : null,
   901→   901→      objective: this.objectiveUi(),
   902→   902→      prompt: this.promptText(),
   903→   903→      params,
   904→   904→      missions,
   905→   905→      homes: this.homesUi(),
   906→   906→      furnitureConfig: { refundRatio: this.content.rules.furniture.refundRatio, roomUnlockPrice: this.content.rules.furniture.roomUnlock.price },
   907→   907→    });
   908→   908→  }
