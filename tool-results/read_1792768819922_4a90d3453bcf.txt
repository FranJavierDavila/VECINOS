    62→  async init(canvas: HTMLCanvasElement): Promise<void> {
    63→    if (this.engine || this.initializing) return;
    64→    this.initializing = true;
    65→    this.canvas = canvas;
    66→
    67→    try {
    68→      // 1) Contenido externo + validación (los errores de referencias se avisan, no abortan).
    69→      const { content, issues } = await loadContent();
    70→      this._content = content;
    71→      preloadSounds();
    72→      useGameStore.setState({ touchHint: content.rules.touch.hint });
    73→      if (issues.length > 0) {
    74→        useGameStore.getState().pushToast(`Contenido inválido: ${issues.length} errores de validación`, 'bad');
    75→      }
    76→
    77→      // 2) Motor de render (WebGPU con fallback WebGL2).
    78→      const { engine, backend } = await createEngine(canvas);
    79→      this.engine = engine;
    80→      window.addEventListener('resize', this.onResize);
    81→      await this.restart();
    82→      useGameStore.getState().markReady(backend);
    83→      engine.runRenderLoop(this.frame);
    84→      // Gancho de depuración/pruebas E2E (no lo usa la UI).
    85→      (window as unknown as { __vecinos?: unknown }).__vecinos = { store: useGameStore, controller: this };
    86→    } catch (err) {
    87→      console.error('No se pudo arrancar el juego (contenido o motor de render)', err);
    88→      useGameStore.getState().setEngineError(true);
    89→    }
    90→  }

   258→  private connectNet(): void {
   259→    this.net = new NetSession({
   260→      onState: (s) => this.applyRemoteState(s),
   261→      onParticipants: (p) => useGameStore.getState().setMpParticipants(p),
   262→      onJoined: (j) => {
   263→        this.myResidentId = j.yourResidentId;
   264→        this.isHost = j.isHost;
   265→        useGameStore.getState().setMpSession({ code: j.code, isHost: j.isHost, myResidentId: j.yourResidentId });
   266→        useGameStore.getState().setMpParticipants(j.participants);
   267→      },
   268→      onError: (m) => {
   269→        useGameStore.getState().pushToast(m, 'bad');
   270→      },
   271→      onStatus: (st) => {
   272→        if (!this.net) return;
   273→        if (st === 'connected' && useGameStore.getState().mpStatus === 'connecting') return; // el ack decide
   274→        if (st === 'disconnected' && useGameStore.getState().mpStatus === 'connected') {
   275→          useGameStore.getState().setMpStatus('error');
   276→          useGameStore.getState().setMpError('Conexión perdida con el servidor de la comunidad.');
   277→        }
   278→      },
   279→    });
   280→  }

   293→  /** Adopta el estado del servidor (autoridad). Primera vez: reinicia la escena con ese mundo. */
   294→  private applyRemoteState(s: SessionStatePayload): void {
   295→    if (s.seq <= this.appliedSeq) return;
   296→    this.appliedSeq = s.seq;
   297→    if (!this.world || this.world.scenarioId !== s.world.scenarioId) {
   298→      void this.restart(s.world).then(() => {
   299→        if (!this.world) return;
   300→        this.world.events = s.events.slice();
   301→        this.handleEvents();
   302→        this.reconcile();
   303→        this.syncHud();
   304→      });
   305→      return;
   306→    }
   307→    // Aplica en sitio conservando la identidad del objeto (PlayerLogic/building/actors referencian el mismo World).
   308→    Object.assign(this.world, s.world);
   309→    this.world.events = s.events.slice();
   310→    this.handleEvents();
   311→    this.paintBuilding();
   312→    this.reconcile();
   313→    this.syncHud();
   314→  }

   316→  /** Corrige desvíos de posición del avatar local respecto al mundo del servidor (autoridad). */
   317→  private reconcile(): void {
   318→    if (!this.logic || !this.world || !this.net) return;
   319→    const me = this.world.residents[this.actorId()];
   320→    if (!me) return;
   321→    const pos = this.logic.position();
   322→    const locBlock = me.location ? this.world.blocks[me.location] : null;
   323→    if (locBlock) {
   324→      const insideOk = pos.mode === 'inside' && pos.insideBlock === locBlock.id;
   325→      const corridorOk = pos.mode === 'corridor' && !pos.insideBlock;
   326→      if (!(locBlock.kind === 'dwelling' ? insideOk : corridorOk) || Math.abs(pos.floor - locBlock.cell.floor) > 0.01) {
   327→        this.logic.forceLocation(locBlock.id);
   328→      }
   329→    } else if (pos.mode === 'inside') {
   330→      this.logic.forceLocation(null);
   331→    }
   332→  }

   414→  async save(): Promise<void> {
   415→    if (!this.world) return;
   416→    const store = useGameStore.getState();
   417→    if (this.net) {
   418→      store.pushToast(this.content.rules.multiplayer.texts.remoteSaveDisabled, 'info');
   419→      return;
   420→    }
   421→    store.setSaveState('saving');
   422→    drainEvents(this.world); // la cola de eventos no entra en el guardado
   423→    try {
   424→      const res = await fetch('/api/save', {
   425→        method: 'POST',
   426→        headers: { 'Content-Type': 'application/json' },
   427→        body: JSON.stringify({ slot: 'auto', state: this.world }),
   428→      });
   429→      if (!res.ok) throw new Error(String(res.status));
   430→      store.setSaveState('saved', new Date().toLocaleTimeString('es-ES', { hour: '2-digit', minute: '2-digit' }));
   431→      store.pushToast('Partida guardada. El acta queda en el tablón (y en SQLite).', 'good');
   432→    } catch {
   433→      store.setSaveState('error');
   434→      store.pushToast('No se pudo guardar la partida. ¿El administrador se habrá ido a comer?', 'bad');
   435→    }
   436→  }

   438→  async load(): Promise<void> {
   439→    const store = useGameStore.getState();
   440→    if (this.net) {
   441→      store.pushToast(this.content.rules.multiplayer.texts.remoteSaveDisabled, 'info');
   442→      return;
   443→    }
   444→    try {
   445→      const res = await fetch('/api/save?slot=auto');
   446→      const json = (await res.json()) as { state?: unknown };
   447→      if (!json.state || typeof json.state !== 'object') {
   448→        store.pushToast('Todavía no hay ninguna partida guardada.', 'info');
   449→        return;
   450→      }
   451→      if (!this._content) {
   452→        store.pushToast('El contenido todavía se está cargando. Espera un momento.', 'info');
   453→        return;
   454→      }
   455→      await this.restart(json.state as World);
   456→      store.pushToast('Partida cargada. El vecindario, exactamente como lo dejaste.', 'good');
   457→    } catch {
   458→      store.pushToast('No se pudo cargar la partida.', 'bad');
   459→    }
   460→  }

   473→  /** (Re)crea el mundo y la escena. Con `state` restaura una partida guardada. */
   474→  private async restart(state?: World): Promise<void> {
   475→    if (!this.engine || !this.canvas || !this._content) return;
   476→    const { Color4, DirectionalLight, HemisphericLight, Scene, UniversalCamera, Vector3 } = B();
   477→    const view = this.content.rules.view;
   478→    useGameStore.getState().resetUi();
   479→
   480→    this.world = state ?? createWorld(this.content, this.activeScenarioId());
   481→    this.scenarioOverrides = this.content.scenarios[this.world.scenarioId ?? '']?.minigameOverrides ?? {};
   482→    // Guardados antiguos sin `entry`: la entrada es la zona común de planta baja.
   483→    if (!this.world.entry) {
   484→      this.world.entry = Object.values(this.world.blocks).find((b) => b.cell.floor === 0 && b.kind !== 'dwelling')?.id ?? Object.keys(this.world.blocks)[0]!;
   485→    }
   486→    // Migración de estancias de guardados v0.1: params y ranuras de muebles según catálogo.
   487→    for (const b of Object.values(this.world.blocks)) migrateRooms(this.content, b.rooms);
   488→    this.logic = new PlayerLogic(this.world, this.content, {
   489→      actorId: this.myResidentId ?? undefined,
   490→      send: (cmd) => this.sendCommand(cmd),
   491→    });
   492→
   493→    this.scene?.dispose();
   494→    this.scene = new Scene(this.engine);
   495→    this.scene.clearColor = new Color4(view.sky.r, view.sky.g, view.sky.b, 1);
   496→    new HemisphericLight('ambient', new Vector3(...view.ambient.dir), this.scene).intensity = view.ambient.intensity;
   497→    new DirectionalLight('sun', new Vector3(...view.sun.dir), this.scene).intensity = view.sun.intensity;
   498→
   499→    this.building = createBuilding(this.scene, this.world, this.content);
   500→    this.actors = createActors(this.scene, this.world, this.content);
   501→    const blocks = Object.values(this.world.blocks);
   502→    this.rig = new CameraRig(
   503→      new UniversalCamera('cam', new Vector3(0, 0, -30), this.scene),
   504→      this.engine,
   505→      Math.max(...blocks.map((b) => b.cell.col + b.size.w)),
   506→      Math.max(...blocks.map((b) => b.cell.floor)) + 1,
   507→      view.cellW,
   508→      view.cellH,
   509→    );
   510→    this.rig.mode = useGameStore.getState().camMode;
   511→
   512→    this.clockAcc = 0;
   513→    this.keys.clear();
   514→    this.phoneLines = [];
   515→    this.phoneMission = '';
   516→    this.lastHudSync = 0;
   517→    this.pump();
   518→  }
   519→
   520→  private onResize = (): void => {
   521→    this.engine?.resize();
   522→  };
   523→
   524→  // ───────────────────────── Bucle de render ─────────────────────────
   525→
   526→  private frame = (): void => {
   527→    if (!this.engine || !this.scene || !this.world || !this.logic || !this.building || !this.actors || !this.rig) return;
   528→    const dt = Math.min(0.1, this.engine.getDeltaTime() / 1000);
   529→    const store = useGameStore.getState();
   530→    const frozen = selectModalOpen(store) || store.paused;
   531→
   532→    let ticked = false;
   533→    if (!frozen) {
   534→      this.logic.update(dt, this.readInput());
   535→      // En remoto el servidor es la autoridad del reloj: nada de ticks locales.
   536→      if (!this.net) {
   537→        this.clockAcc += dt * store.speed;
   538→        let guard = 0;
   539→        while (this.clockAcc >= 1 && guard < 60) {
   540→          tick(this.world, this.content);
   541→          this.clockAcc -= 1;
   542→          ticked = true;
   543→          guard += 1;
   544→        }
   545→      }
   546→    }
   547→
   548→    if (this.world.events.length > 0) this.handleEvents();
   549→    if (ticked) this.paintBuilding();
   550→
   551→    const now = performance.now();
   552→    if (now - this.lastHudSync > 200) {
   553→      this.lastHudSync = now;
   554→      this.syncHud();
   555→    }
   556→
   557→    const pos = this.logic.position();
   558→    this.actors.update(dt, pos, this.actorId());
   559→    // LOD por distancia: solo con cámaras cercanas (planta/estancia); el plano general (edificio) muestra todo.
   560→    this.building.update(dt, useGameStore.getState().camMode === 'edificio' ? undefined : { x: pos.x, floor: pos.floor });
   561→    this.rig.update(dt, pos);
   562→    this.scene.render();
   563→  };
   564→
   565→  /** Bomba de sincronización: reparte eventos pendientes y refresca el HUD. */
   566→  private pump(): void {
   567→    if (!this.world) return;
   568→    this.handleEvents();
   569→    this.paintBuilding();
   570→    this.syncHud();
   571→  }

   574→    this.building?.paint();
   575→  }

   588→  /** Etiqueta visible de una actividad (vive en el arquetipo, no en el código). */

   594→  private handleEvents(): void {
   595→    if (!this.world) return;
   596→    const store = useGameStore.getState();
   597→    for (const e of drainEvents(this.world)) {
   598→      const d = e.data as Record<string, unknown>;
   599→      eventSound(e.type, typeof d.type === 'string' ? d.type : undefined);
   600→      switch (e.type) {
   601→        case 'step_started':
   602→          if (d.type === 'phone_call') this.openPhone(String(d.mission), String(d.dialog));
   603→          break;
   604→        case 'choice_required': {
   605→          const choices = (d.choices as { id: string; label: string; hint?: string | null }[] | undefined) ?? [];
   606→          store.setChoices({ mission: String(d.mission), title: String(d.title), choices: choices.map((c) => ({ id: c.id, label: c.label, hint: c.hint ?? null })) });
   607→          store.pushLog(`⚖️ Decisión pendiente: ${String(d.title)}`, 'info');
   608→          break;
   609→        }
   610→        case 'door_locked': {
   611→          this.building?.shake(String(d.block));
   612→          const price = typeof d.price === 'number' ? d.price : null;
   613→          store.pushToast(`${blockLabel(String(d.block))}: puerta con candado${d.reason === 'for_sale' && price !== null ? ` (a la venta por ${price} €)` : ''}`, 'info');
   614→          break;
   615→        }
   616→        case 'neighbor_arrived':
   617→          store.pushToast(`Nuevo vecino en ${blockLabel(String(d.block))}: ${this.nameOf(d.resident)}`, 'good');
   618→          store.pushLog(`${this.nameOf(d.resident)} se muda a ${blockLabel(String(d.block))}. El edificio ya no es lo que era. Era mejor.`, 'info');
   619→          break;
   620→        case 'mission_activated':
   621→          store.pushLog(`📋 Nueva misión: ${String(d.title)}`, 'info');
   622→          break;
   623→        case 'mission_completed':
   624→          store.pushToast(`Misión completada: ${this.titleOf(d.mission)}`, 'good');
   625→          store.pushLog(`✔ Misión completada: ${this.titleOf(d.mission)}`, 'good');
   626→          break;
   627→        case 'mission_failed':
   628→          store.pushToast(`Misión fallida: ${this.titleOf(d.mission)}`, 'bad');
   629→          store.pushLog(`✘ Misión fallida: ${this.titleOf(d.mission)}`, 'bad');
   630→          break;
   631→        case 'block_unlocked':
   632→          store.pushToast(`${blockLabel(String(d.block))} desbloqueado`, 'good');
   633→          break;
   634→        case 'block_bought':
   635→          store.pushLog(`🔑 ${blockLabel(String(d.block))} comprado por ${String(d.price)} €`, 'good');
   636→          break;
   637→        case 'block_for_sale':
   638→          store.pushLog(`🪧 Cartel de "SE VENDE" en ${blockLabel(String(d.block))} (${String(d.price)} €)`, 'info');
   639→          break;
   640→        case 'noise_complaint': // texto data-driven: lo pinta el uiLog del catálogo (case mantenido por compatibilidad de eventos antiguos)
   641→          if (d.uiLog === true) break;
   642→          store.pushLog(`🔊 Ruido por encima del límite en ${blockLabel(String(d.block))}. Alguien aprieta los dientes.`, 'bad');
   643→          break;
   644→        case 'leak_detected':
   645→          if (d.uiLog === true) break;
   646→          store.pushLog(`💧 Goteo detectado en ${blockLabel(String(d.block))}. El cubo del salón hace ojitos.`, 'bad');
   647→          break;
   648→        case 'furniture_placed':
   649→          store.pushLog(`${String(d.emoji ?? '📦')} ${String(d.name)} colocado en ${blockLabel(String(d.block))} (${String(d.room)}).`, 'info');
   650→          break;
   651→        case 'furniture_removed':
   652→          store.pushLog(`${String(d.emoji ?? '📦')} ${String(d.name)} retirado de ${blockLabel(String(d.block))} (+${String(d.refund)} €).`, 'info');
   653→          break;
   654→        case 'room_renovated':
   655→          store.pushLog(`🧱 Reforma terminada en ${blockLabel(String(d.block))}. Hay estancia nueva que amueblar.`, 'good');
   656→          break;
   657→        case 'weather_changed':
   658→          store.pushLog(`El cielo cambia: ${this.content.rules.labels.weather[String(d.state)] ?? String(d.state)}`, 'info');
   659→          break;
   660→        case 'rent_collected':
   661→          store.pushToast(`+${String(d.amount)} € de alquiler (${blockLabel(String(d.block))})`, 'good');
   662→          break;
   663→        case 'work_done':
   664→          store.pushLog(`💻 Cerrada una "reunión que podía ser un correo": +${String(d.amount)} €`, 'info');
   665→          break;
   666→        case 'activity_started':
   667→          if (d.activity !== 'idle') store.pushLog(`${this.nameOf(d.resident)} ${this.activityLabel(d.resident, String(d.activity))}`, 'info');
   668→          break;
   669→        case 'junta_requested': {
   670→          const agenda = (d.agenda as { id: string; title: string; description: string; cost: number }[] | undefined) ?? [];
   671→          store.setJunta(agenda.map((i) => ({ id: i.id, title: i.title, description: i.description, cost: i.cost })));
   672→          break;
   673→        }
   674→        case 'junta_result':
   675→          store.pushToast(
   676→            d.approved ? `Aprobado: ${String(d.title)} (${String(d.yes)} sí / ${String(d.no)} no)` : `Rechazado: ${String(d.title)} (${String(d.yes)} sí / ${String(d.no)} no)`,
   677→            d.approved ? ('good' as Tone) : ('bad' as Tone),
   678→          );
   679→          break;
   680→        case 'junta_vote_registered':
   681→          store.pushLog(`🗳️ ${this.nameOf(d.resident)} ha depositado su voto. Pendientes: ${String(d.pendingHumans)}`, 'info');
   682→          break;
   683→        case 'junta_closed':
   684→          store.setJunta(null);
   685→          break;
   686→        case 'minigame_requested': {
   687→          const def = this.content.minigames[String(d.minigame)];
   688→          if (!def) break;
   689→          // Overrides numéricos del escenario activo (QA/E2E): solo config, nunca efectos ni textos.
   690→          const ov = this.scenarioOverrides[def.id];
   691→          const effective = ov ? ({ ...def, config: { ...def.config, ...ov } } as MinigameDef) : def;
   692→          store.setMinigame({ ...effective, emoji: effective.emoji ?? '🎮', instruction: effective.instruction ?? null });
   693→          break;
   694→        }
   695→        case 'minigame_finished': {
   696→          const name = this.content.minigames[String(d.minigame)]?.name ?? String(d.minigame);
   697→          store.pushLog(d.success ? `🎮 Minijuego superado: ${name}` : `🎮 Minijuego fallido: ${name}`, d.success ? 'good' : 'bad');
   698→          break;
   699→        }
   700→        case 'ui_toast':
   701→          store.pushToast(String(d.text), (d.tone as Tone | undefined) ?? 'info');
   702→          break;
   703→        case 'finale':
   704→          store.setFinale(String(d.text));
   705→          break;
   706→        case 'funds_changed': {
   707→          const amount = typeof d.amount === 'number' ? d.amount : 0;
   708→          if ((d.to === 'player' || d.to === 'community') && amount !== 0) {
   709→            store.pushLog(amount > 0 ? `+${amount} € (fondos)` : `${amount} € (fondos)`, amount > 0 ? 'good' : 'bad');
   710→          }
   711→          break;
   712→        }
   713→        default: {
   714→          // Umbral de canal con texto del catálogo (odors, wifi y futuros canales): totalmente data-driven.
   715→          if (d.uiLog === true && typeof d.message === 'string') {
   716→            store.pushLog(String(d.message).replace('{block}', blockLabel(String(d.block))), (d.tone as Tone | undefined) ?? 'info');
   717→          }
   718→          break; // resident_moved, param_changed, activity_ended... demasiado ruido para el registro
   719→        }
   720→      }
   721→    }
   722→  }

   743→  private objectiveUi(): { title: string; text: string } | null {
   744→    if (!this.world) return null;
   745→    const o = activeObjective(this.world, this.content);
   746→    if (!o) return null;
   747→    if (o.state === 'awaiting_choice') return { title: o.title, text: 'El edificio contiene la respiración. Decide qué hacer.' };
   748→    const s = o.step!;
   749→    const left = o.untilTick !== null ? ` · quedan ${Math.max(0, o.untilTick - this.world.tick)} min` : '';
   750→    switch (s.type) {
   751→      case 'phone_call':
   752→        return { title: o.title, text: 'Suena el teléfono. Atiéndelo.' };
   753→      case 'reach':
   754→        return { title: o.title, text: `Llega a la puerta (${blockLabel(s.block ?? '')}).` };
   755→      case 'enter':
   756→        return { title: o.title, text: 'Entra en tu piso con E.' };
   757→      case 'catch_in_act':
   758→        return { title: o.title, text: `Pilla al responsable in fraganti${left}.` };
   759→      default:
   760→        return { title: o.title, text: o.title };
   761→    }
   762→  }
   763→
   764→  private promptText(): string | null {
   765→    const logic = this.logic;
   766→    const world = this.world;
   767→    if (!logic || !world) return null;
   768→    const p = logic.position();
   769→    if (p.mode === 'inside') return 'E · salir a la escalera';
   770→    if (p.climbing) return null;
   771→    const door = logic.doorInFront();
   772→    if (door) return door.access === 'locked' ? `E · ${blockLabel(door.id)}: candado` : `E · entrar en ${blockLabel(door.id)}`;
   773→    const zone = logic.zone();
   774→    if (this.content.blocks[zone.defId]!.features?.stairs) {
   775→      return world.blocks[zone.id]!.params.elevator_status === 'ok' ? 'W/S · ascensor' : 'W/S · escalera (el ascensor no funciona)';
   776→    }
   777→    return null;
   778→  }

   780→  /** Viviendas del jugador para el editor de interiores (foto serializable, posiciones ya en %). */
   781→  private homesUi() {
   782→    const world = this.world!;
   783→    const cellH = this.content.rules.view.cellH;
   784→    const mine = this.actorId();
   785→    return Object.values(world.blocks)
   786→      .filter((b) => b.kind === 'dwelling' && b.ownerId === mine)
   787→      .sort((a, b) => a.id.localeCompare(b.id))
   788→      .map((b) => ({
   789→        id: b.id,
   790→        label: blockLabel(b.id),
   791→        rooms: b.rooms.map((room) => {
   792→          const rdef = this.content.rooms[room.defId];
   793→          const slots = (rdef?.furnitureSlots ?? []).map((s) => {
   794→            const itemId = room.furniture[s.id] ?? '';
   795→            const fdef = itemId ? this.content.furniture[itemId] : undefined;
   796→            return {
   797→              id: s.id,
   798→              kind: s.kind,
   799→              left: Math.min(92, Math.max(2, (s.pos[0] / Math.max(0.001, rdef?.size.w ?? 1)) * 100)),
   800→              bottom: Math.min(76, Math.max(0, (s.pos[1] / Math.max(0.001, cellH)) * 100)),
   801→              item: fdef ? { id: fdef.id, name: fdef.name, emoji: fdef.emoji ?? '📦', color: fdef.view.color } : null,
   802→            };
   803→          });
   804→          return { key: room.key, label: rdef?.label ?? room.key, access: room.access, slots };
   805→        }),
   806→      }));
   807→  }
   808→
   809→  private syncHud(): void {
   810→    const world = this.world;
   811→    if (!world) return;
   812→    const player = world.residents[this.actorId()] ?? world.residents[world.playerId];
   813→    if (!player) return;
   814→    const home = player.home;
   815→    const params = visibleParams(world, this.content, home).map((p) => ({ name: p.name, label: p.def.label ?? p.name, value: String(p.value) }));
   816→    const missions = world.missions.map((m) => {
   817→      const def = this.content.missions[m.id]!;
   818→      const step = m.state === 'active' ? def.steps[m.step] : undefined;
   819→      return {
   820→        id: m.id,
   821→        title: def.title,
   822→        state: m.state,
   823→        hint: step?.hint ?? null,
   824→        remaining: m.until !== null ? `quedan ${Math.max(0, m.until - world.tick)} min` : null,
   825→      };
   826→    });
   827→    const travel = travelMinutes(buildCirculation(world, this.content), world.entry, home);
   828→    const elevatorBlock = Object.values(world.blocks).find((b) => b.params.elevator_status !== undefined);
   829→    useGameStore.getState().syncHud({
   830→      clock: { ...world.clock },
   831→      weather: world.weather.state,
   832→      funds: world.community.funds,
   833→      wallet: player.wallet,
   834→      reputation: player.reputation,
   835→      elevator: String(elevatorBlock?.params.elevator_status ?? 'ok'),
   836→      homeLabel: blockLabel(home),
   837→      travelText: Number.isFinite(travel) ? `del portal a tu puerta: ${travel} min` : null,
   838→      objective: this.objectiveUi(),
   839→      prompt: this.promptText(),
   840→      params,
   841→      missions,
   842→      homes: this.homesUi(),
   843→      furnitureConfig: { refundRatio: this.content.rules.furniture.refundRatio, roomUnlockPrice: this.content.rules.furniture.roomUnlock.price },
   844→    });
   845→  }
