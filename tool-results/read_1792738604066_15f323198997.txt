   274→   274→    this.world = state ?? createWorld(this.content, this.content.rules.startScenario);
   275→   275→    // Guardados antiguos sin `entry`: la entrada es la zona común de planta baja.
   276→   276→    if (!this.world.entry) {
   277→   277→      this.world.entry = Object.values(this.world.blocks).find((b) => b.cell.floor === 0 && b.kind !== 'dwelling')?.id ?? Object.keys(this.world.blocks)[0]!;
   278→   278→    }
   279→   279→    // Migración de estancias de guardados v0.1: params y ranuras de muebles según catálogo.
   280→   280→    for (const b of Object.values(this.world.blocks)) migrateRooms(this.content, b.rooms);

   282→   282→
   283→   283→    this.scene?.dispose();
   284→   284→    this.scene = new Scene(this.engine);
   285→   285→    this.scene.clearColor = new Color4(view.sky.r, view.sky.g, view.sky.b, 1);
   286→   286→    new HemisphericLight('ambient', new Vector3(...view.ambient.dir), this.scene).intensity = view.ambient.intensity;
   287→   287→    new DirectionalLight('sun', new Vector3(...view.sun.dir), this.scene).intensity = view.sun.intensity;
   288→   288→
   289→   289→    this.building = createBuilding(this.scene, this.world, this.content);
   290→   290→    this.actors = createActors(this.scene, this.world, this.content);
   291→   291→    const blocks = Object.values(this.world.blocks);
   292→   292→    this.rig = new CameraRig(
   293→   293→      new UniversalCamera('cam', new Vector3(0, 0, -30), this.scene),
   294→   294→      this.engine,
   295→   295→      Math.max(...blocks.map((b) => b.cell.col + b.size.w)),
   296→   296→      Math.max(...blocks.map((b) => b.cell.floor)) + 1,
   297→   297→      view.cellW,
   298→   298→      view.cellH,
   299→   299→    );
   300→   300→    this.rig.mode = useGameStore.getState().camMode;
   301→   301→
   302→   302→    this.clockAcc = 0;
   303→   303→    this.keys.clear();
   304→   304→    this.phoneLines = [];
   305→   305→    this.phoneMission = '';
   306→   306→    this.lastHudSync = 0;

   315→   315→
   316→   316→  private frame = (): void => {
   317→   317→    if (!this.engine || !this.scene || !this.world || !this.logic || !this.building || !this.actors || !this.rig) return;
   318→   318→    const dt = Math.min(0.1, this.engine.getDeltaTime() / 1000);
   319→   319→    const store = useGameStore.getState();
   320→   320→    const frozen = selectModalOpen(store) || store.paused;
   321→   321→
   322→   322→    let ticked = false;
   323→   323→    if (!frozen) {
   324→   324→      this.logic.update(dt, this.readInput());
   325→   325→      this.clockAcc += dt * store.speed;
   326→   326→      let guard = 0;
   327→   327→      while (this.clockAcc >= 1 && guard < 60) {
   328→   328→        tick(this.world, this.content);
   329→   329→        this.clockAcc -= 1;
   330→   330→        ticked = true;
   331→   331→        guard += 1;
   332→   332→      }
   333→   333→    }
   334→   334→
   335→   335→    if (this.world.events.length > 0) this.handleEvents();
   336→   336→    if (ticked) this.paintBuilding();
   337→   337→
   338→   338→    const now = performance.now();
   339→   339→    if (now - this.lastHudSync > 200) {
   340→   340→      this.lastHudSync = now;
   341→   341→      this.syncHud();
   342→   342→    }
   343→   343→
   344→   344→    const pos = this.logic.position();
   345→   345→    this.actors.update(dt, pos);
   346→   346→    this.building.update(dt);
   347→   347→    this.rig.update(dt, pos);
   348→   348→    this.scene.render();
   349→   349→  };

   384→   384→      const d = e.data as Record<string, unknown>;
   385→   385→      switch (e.type) {
   386→   386→        case 'step_started':
   387→   387→          if (d.type === 'phone_call') this.openPhone(String(d.mission), String(d.dialog));
   388→   388→          break;
   389→   389→        case 'choice_required': {
   390→   390→          const choices = (d.choices as { id: string; label: string; hint?: string | null }[] | undefined) ?? [];

   395→   395→        case 'door_locked': {

   446→   446→          store.pushToast(`+${String(d.amount)} € de alquiler (${blockLabel(String(d.block))})`, 'good');
   447→   447→          break;
   448→   448→        case 'work_done':
   449→   449→          store.pushLog(`💻 Cerrada una "reunión que podía ser un correo": +${String(d.amount)} €`, 'info');
   450→   450→          break;
   451→   451→        case 'activity_started':
   452→   452→          if (d.activity !== 'idle') store.pushLog(`${this.nameOf(d.resident)} ${this.activityLabel(d.resident, String(d.activity))}`, 'info');
   453→   453→          break;
   454→   454→        case 'junta_requested': {
   455→   455→          const agenda = (d.agenda as { id: string; title: string; description: string; cost: number }[] | undefined) ?? [];
   456→   456→          store.setJunta(agenda.map((i) => ({ id: i.id, title: i.title, description: i.description, cost: i.cost })));
   457→   457→          break;
   458→   458→        }
   459→   459→        case 'junta_result':
   460→   460→          store.pushToast(
   461→   461→            d.approved ? `Aprobado: ${String(d.title)} (${String(d.yes)} sí / ${String(d.no)} no)` : `Rechazado: ${String(d.title)} (${String(d.yes)} sí / ${String(d.no)} no)`,
   462→   462→            d.approved ? ('good' as Tone) : ('bad' as Tone),
   463→   463→          );
   464→   464→          break;
   465→   465→        case 'minigame_requested': {
   466→   466→          const def = this.content.minigames[String(d.minigame)];
   467→   467→          if (!def) break;
   468→   468→          store.setMinigame({ ...def, emoji: def.emoji ?? '🎮', instruction: def.instruction ?? null });
   469→   469→          break;
   470→   470→        }
   471→   471→        case 'minigame_finished': {
   472→   472→          const name = this.content.minigames[String(d.minigame)]?.name ?? String(d.minigame);
   473→   473→          store.pushLog(d.success ? `🎮 Minijuego superado: ${name}` : `🎮 Minijuego fallido: ${name}`, d.success ? 'good' : 'bad');
   474→   474→          break;
   475→   475→        }
   476→   476→        case 'ui_toast':
   477→   477→          store.pushToast(String(d.text), (d.tone as Tone | undefined) ?? 'info');
   478→   478→          break;
   479→   479→        case 'finale':
   480→   480→          store.setFinale(String(d.text));
   481→   481→          break;
   482→   482→        case 'funds_changed': {
   483→   483→          const amount = typeof d.amount === 'number' ? d.amount : 0;
   484→   484→          if ((d.to === 'player' || d.to === 'community') && amount !== 0) {
   485→   485→            store.pushLog(amount > 0 ? `+${amount} € (fondos)` : `${amount} € (fondos)`, amount > 0 ? 'good' : 'bad');
   486→   486→          }
   487→   487→          break;
   488→   488→        }
   489→   489→        default: {
   490→   490→          // Umbral de canal con texto del catálogo (odors, wifi y futuros canales): totalmente data-driven.
   491→   491→          if (d.uiLog === true && typeof d.message === 'string') {
   492→   492→            store.pushLog(String(d.message).replace('{block}', blockLabel(String(d.block))), (d.tone as Tone | undefined) ?? 'info');
   493→   493→          }
   494→   494→          break; // resident_moved, param_changed, activity_ended... demasiado ruido para el registro
   495→   495→        }
   496→   496→      }
   497→   497→    }
   498→   498→  }

   519→   519→  private objectiveUi(): { title: string; text: string } | null {
   520→   520→    if (!this.world) return null;
   521→   521→    const o = activeObjective(this.world, this.content);
   522→   522→    if (!o) return null;
   523→   523→    if (o.state === 'awaiting_choice') return { title: o.title, text: 'El edificio contiene la respiración. Decide qué hacer.' };
   524→   524→    const s = o.step!;
   525→   525→    const left = o.untilTick !== null ? ` · quedan ${Math.max(0, o.untilTick - this.world.tick)} min` : '';
   526→   526→    switch (s.type) {
   527→   527→      case 'phone_call':
   528→   528→        return { title: o.title, text: 'Suena el teléfono. Atiéndelo.' };
   529→   529→      case 'reach':
   530→   530→        return { title: o.title, text: `Llega a la puerta (${blockLabel(s.block ?? '')}).` };
   531→   531→      case 'enter':
   532→   532→        return { title: o.title, text: 'Entra en tu piso con E.' };
   533→   533→      case 'catch_in_act':
   534→   534→        return { title: o.title, text: `Pilla al responsable in fraganti${left}.` };
   535→   535→      default:
   536→   536→        return { title: o.title, text: o.title };
   537→   537→    }
   538→   538→  }
   539→   539→
   540→   540→  private promptText(): string | null {
   541→   541→    const logic = this.logic;
   542→   542→    const world = this.world;
   543→   543→    if (!logic || !world) return null;
   544→   544→    const p = logic.position();
   545→   545→    if (p.mode === 'inside') return 'E · salir a la escalera';
   546→   546→    if (p.climbing) return null;
   547→   547→    const door = logic.doorInFront();
   548→   548→    if (door) return door.access === 'locked' ? `E · ${blockLabel(door.id)}: candado` : `E · entrar en ${blockLabel(door.id)}`;
   549→   549→    const zone = logic.zone();
   550→   550→    if (this.content.blocks[zone.defId]!.features?.stairs) {
   551→   551→      return world.blocks[zone.id]!.params.elevator_status === 'ok' ? 'W/S · ascensor' : 'W/S · escalera (el ascensor no funciona)';
   552→   552→    }
   553→   553→    return null;
   554→   554→  }

   556→   556→  /** Viviendas del jugador para el editor de interiores (foto serializable, posiciones ya en %). */
   557→   557→  private homesUi() {
   558→   558→    const world = this.world!;
   559→   559→    const cellH = this.content.rules.view.cellH;
   560→   560→    return Object.values(world.blocks)
   561→   561→      .filter((b) => b.kind === 'dwelling' && b.ownerId === world.playerId)
   562→   562→      .sort((a, b) => a.id.localeCompare(b.id))
   563→   563→      .map((b) => ({
   564→   564→        id: b.id,
   565→   565→        label: blockLabel(b.id),
   566→   566→        rooms: b.rooms.map((room) => {
   567→   567→          const rdef = this.content.rooms[room.defId];
   568→   568→          const slots = (rdef?.furnitureSlots ?? []).map((s) => {
   569→   569→            const itemId = room.furniture[s.id] ?? '';
   570→   570→            const fdef = itemId ? this.content.furniture[itemId] : undefined;
   571→   571→            return {
   572→   572→              id: s.id,
   573→   573→              kind: s.kind,
   574→   574→              left: Math.min(92, Math.max(2, (s.pos[0] / Math.max(0.001, rdef?.size.w ?? 1)) * 100)),
   575→   575→              bottom: Math.min(76, Math.max(0, (s.pos[1] / Math.max(0.001, cellH)) * 100)),
   576→   576→              item: fdef ? { id: fdef.id, name: fdef.name, emoji: fdef.emoji ?? '📦', color: fdef.view.color } : null,
   577→   577→            };
   578→   578→          });
   579→   579→          return { key: room.key, label: rdef?.label ?? room.key, access: room.access, slots };
   580→   580→        }),
   581→   581→      }));
   582→   582→  }

   584→   584→  private syncHud(): void {
   585→   585→    const world = this.world;
   586→   586→    if (!world) return;
   587→   587→    const player = world.residents[world.playerId]!;
   588→   588→    const home = player.home;
   589→   589→    const params = visibleParams(world, this.content, home).map((p) => ({ name: p.name, label: p.def.label ?? p.name, value: String(p.value) }));
   590→   590→    const missions = world.missions.map((m) => {
   591→   591→      const def = this.content.missions[m.id]!;
   592→   592→      const step = m.state === 'active' ? def.steps[m.step] : undefined;
   593→   593→      return {
   594→   594→        id: m.id,
   595→   595→        title: def.title,
   596→   596→        state: m.state,
   597→   597→        hint: step?.hint ?? null,
   598→   598→        remaining: m.until !== null ? `quedan ${Math.max(0, m.until - world.tick)} min` : null,
   599→   599→      };
   600→   600→    });
   601→   601→    const travel = travelMinutes(buildCirculation(world, this.content), world.entry, home);
   602→   602→    const elevatorBlock = Object.values(world.blocks).find((b) => b.params.elevator_status !== undefined);
   603→   603→    useGameStore.getState().syncHud({
   604→   604→      clock: { ...world.clock },
   605→   605→      weather: world.weather.state,
   606→   606→      funds: world.community.funds,
   607→   607→      wallet: player.wallet,
   608→   608→      reputation: player.reputation,
   609→   609→      elevator: String(elevatorBlock?.params.elevator_status ?? 'ok'),
   610→   610→      homeLabel: blockLabel(home),
   611→   611→      travelText: Number.isFinite(travel) ? `del portal a tu puerta: ${travel} min` : null,
   612→   612→      objective: this.objectiveUi(),
   613→   613→      prompt: this.promptText(),
   614→   614→      params,
   615→   615→      missions,
   616→   616→      homes: this.homesUi(),
   617→   617→      furnitureConfig: { refundRatio: this.content.rules.furniture.refundRatio, roomUnlockPrice: this.content.rules.furniture.roomUnlock.price },
   618→   618→    });
   619→   619→  }
