    63→  async init(canvas: HTMLCanvasElement): Promise<void> {
    64→    if (this.engine || this.initializing) return;
    65→    this.initializing = true;
    66→    this.canvas = canvas;
    67→
    68→    try {
    69→      // 1) Contenido externo + validación (los errores de referencias se avisan, no abortan).
    70→      const { content, issues } = await loadContent();
    71→      this._content = content;
    72→      preloadSounds();
    73→      useGameStore.setState({ touchHint: content.rules.touch.hint });
    74→      if (issues.length > 0) {
    75→        useGameStore.getState().pushToast(`Contenido inválido: ${issues.length} errores de validación`, 'bad');
    76→      }
    77→
    78→      // 2) Motor de render (WebGPU con fallback WebGL2).
    79→      const { engine, backend } = await createEngine(canvas);
    80→      this.engine = engine;
    81→      window.addEventListener('resize', this.onResize);
    82→      await this.restart();
    83→      useGameStore.getState().markReady(backend);
    84→      engine.runRenderLoop(this.frame);
    85→      // Gancho de depuración/pruebas E2E (no lo usa la UI).
    86→      (window as unknown as { __vecinos?: unknown }).__vecinos = { store: useGameStore, controller: this };
    87→    } catch (err) {
    88→      console.error('No se pudo arrancar el juego (contenido o motor de render)', err);
    89→      useGameStore.getState().setEngineError(true);
    90→    }
    91→  }

   285→  private connectNet(): void {
   286→    this.net = new NetSession({
   287→      onState: (s) => this.applyRemoteState(s),
   288→      onParticipants: (p) => useGameStore.getState().setMpParticipants(p),
   289→      onJoined: (j) => {
   290→        this.myResidentId = j.yourResidentId;
   291→        this.isHost = j.isHost;
   292→        useGameStore.getState().setMpSession({ code: j.code, isHost: j.isHost, myResidentId: j.yourResidentId });
   293→        useGameStore.getState().setMpParticipants(j.participants);
   294→      },
   295→      onError: (m) => {
   296→        useGameStore.getState().pushToast(m, 'bad');
   297→      },
   298→      onStatus: (st) => {
   299→        if (!this.net) return;
   300→        if (st === 'connected' && useGameStore.getState().mpStatus === 'connecting') return; // el ack decide
   301→        if (st === 'disconnected' && useGameStore.getState().mpStatus === 'connected') {
   302→          useGameStore.getState().setMpStatus('error');
   303→          useGameStore.getState().setMpError('Conexión perdida con el servidor de la comunidad.');
   304→        }
   305→      },
   306→    });
   307→  }

   320→  /** Adopta el estado del servidor (autoridad). Primera vez: reinicia la escena con ese mundo. */
   321→  private applyRemoteState(s: SessionStatePayload): void {
   322→    if (s.seq <= this.appliedSeq) return;
   323→    this.appliedSeq = s.seq;
   324→    if (!this.world || this.world.scenarioId !== s.world.scenarioId) {
   325→      void this.restart(s.world).then(() => {
   326→        if (!this.world) return;
   327→        this.world.events = s.events.slice();
   328→        this.handleEvents();
   329→        this.reconcile();
   330→        this.syncHud();
   331→      });
   332→      return;
   333→    }
   334→    // Aplica en sitio conservando la identidad del objeto (PlayerLogic/building/actors referencian el mismo World).
   335→    Object.assign(this.world, s.world);
   336→    this.world.events = s.events.slice();
   337→    this.handleEvents();
   338→    this.paintBuilding();
   339→    this.reconcile();
   340→    this.syncHud();
   341→  }

   343→  /** Corrige desvíos de posición del avatar local respecto al mundo del servidor (autoridad). */
   344→  private reconcile(): void {
   345→    if (!this.logic || !this.world || !this.net) return;
   346→    const me = this.world.residents[this.actorId()];
   347→    if (!me) return;
   348→    const pos = this.logic.position();
   349→    const locBlock = me.location ? this.world.blocks[me.location] : null;
   350→    if (locBlock) {
   351→      const insideOk = pos.mode === 'inside' && pos.insideBlock === locBlock.id;
   352→      const corridorOk = pos.mode === 'corridor' && !pos.insideBlock;
   353→      if (!(locBlock.kind === 'dwelling' ? insideOk : corridorOk) || Math.abs(pos.floor - locBlock.cell.floor) > 0.01) {
   354→        this.logic.forceLocation(locBlock.id);
   355→      }
   356→    } else if (pos.mode === 'inside') {
   357→      this.logic.forceLocation(null);
   358→    }
   359→  }

   441→  async save(): Promise<void> {
   442→    if (!this.world) return;
   443→    const store = useGameStore.getState();
   444→    if (this.net) {
   445→      store.pushToast(this.content.rules.multiplayer.texts.remoteSaveDisabled, 'info');
   446→      return;
   447→    }
   448→    store.setSaveState('saving');
   449→    drainEvents(this.world); // la cola de eventos no entra en el guardado
   450→    try {
   451→      const res = await fetch('/api/save', {
   452→        method: 'POST',
   453→        headers: { 'Content-Type': 'application/json' },
   454→        body: JSON.stringify({ slot: 'auto', state: this.world }),
   455→      });
   456→      if (!res.ok) throw new Error(String(res.status));
   457→      store.setSaveState('saved', new Date().toLocaleTimeString('es-ES', { hour: '2-digit', minute: '2-digit' }));
   458→      store.pushToast('Partida guardada. El acta queda en el tablón (y en SQLite).', 'good');
   459→    } catch {
   460→      store.setSaveState('error');
   461→      store.pushToast('No se pudo guardar la partida. ¿El administrador se habrá ido a comer?', 'bad');
   462→    }
   463→  }

   465→  async load(): Promise<void> {
   466→    const store = useGameStore.getState();
   467→    if (this.net) {
   468→      store.pushToast(this.content.rules.multiplayer.texts.remoteSaveDisabled, 'info');
   469→      return;
   470→    }
   471→    try {
   472→      const res = await fetch('/api/save?slot=auto');
   473→      const json = (await res.json()) as { state?: unknown };
   474→      if (!json.state || typeof json.state !== 'object') {
   475→        store.pushToast('Todavía no hay ninguna partida guardada.', 'info');
   476→        return;
   477→      }
   478→      if (!this._content) {
   479→        store.pushToast('El contenido todavía se está cargando. Espera un momento.', 'info');
   480→        return;
   481→      }
   482→      await this.restart(json.state as World);
   483→      store.pushToast('Partida cargada. El vecindario, exactamente como lo dejaste.', 'good');
   484→    } catch {
   485→      store.pushToast('No se pudo cargar la partida.', 'bad');
   486→    }
   487→  }

   500→  /** (Re)crea el mundo y la escena. Con `state` restaura una partida guardada. */
   501→  private async restart(state?: World): Promise<void> {
   502→    if (!this.engine || !this.canvas || !this._content) return;
   503→    const { Color4, DirectionalLight, HemisphericLight, Scene, UniversalCamera, Vector3 } = B();
   504→    const view = this.content.rules.view;
   505→    useGameStore.getState().resetUi();
   506→
   507→    this.world = state ?? createWorld(this.content, this.activeScenarioId());
   508→    this.scenarioOverrides = this.content.scenarios[this.world.scenarioId ?? '']?.minigameOverrides ?? {};
   509→    // Guardados antiguos sin `entry`: la entrada es la zona común de planta baja.
   510→    if (!this.world.entry) {
   511→      this.world.entry = Object.values(this.world.blocks).find((b) => b.cell.floor === 0 && b.kind !== 'dwelling')?.id ?? Object.keys(this.world.blocks)[0]!;
   512→    }
   513→    // Migración de estancias de guardados v0.1: params y ranuras de muebles según catálogo.
   514→    for (const b of Object.values(this.world.blocks)) migrateRooms(this.content, b.rooms);
   515→    this.logic = new PlayerLogic(this.world, this.content, {
   516→      actorId: this.myResidentId ?? undefined,
   517→      send: (cmd) => this.sendCommand(cmd),
   518→    });
   519→
   520→    this.scene?.dispose();
   521→    this.scene = new Scene(this.engine);
   522→    this.scene.clearColor = new Color4(view.sky.r, view.sky.g, view.sky.b, 1);
   523→    new HemisphericLight('ambient', new Vector3(...view.ambient.dir), this.scene).intensity = view.ambient.intensity;
   524→    new DirectionalLight('sun', new Vector3(...view.sun.dir), this.scene).intensity = view.sun.intensity;
   525→
   526→    this.building = createBuilding(this.scene, this.world, this.content);
   527→    this.actors = createActors(this.scene, this.world, this.content);
   528→    const blocks = Object.values(this.world.blocks);
   529→    this.rig = new CameraRig(
   530→      new UniversalCamera('cam', new Vector3(0, 0, -30), this.scene),
   531→      this.engine,
   532→      Math.max(...blocks.map((b) => b.cell.col + b.size.w)),
   533→      Math.max(...blocks.map((b) => b.cell.floor)) + 1,
   534→      view.cellW,
   535→      view.cellH,
   536→    );
   537→    // Re-sincronización de vista (v0.6-b): la cámara sigue la posición del mundo (restaurado o
   538→    // nuevo), no la preferencia anterior. Dentro de una vivienda → estancia; en pasillo, la
   539→    // preferencia salvo que fuera 'estancia' (ahí estás en el pasillo, no en una estancia).
   540→    const pos = this.logic.position();
   541→    const store = useGameStore.getState();
   542→    const wanted: CamMode = pos.mode === 'inside' ? 'estancia' : store.camMode === 'estancia' ? 'planta' : store.camMode;
   543→    this.rig.mode = wanted;
   544→    store.setCamMode(wanted);
   545→
   546→    this.clockAcc = 0;
   547→    this.keys.clear();
   548→    this.phoneLines = [];
   549→    this.phoneMission = '';
   550→    this.lastHudSync = 0;
   551→    this.pump();
   552→  }

   558→  // ───────────────────────── Bucle de render ─────────────────────────
   559→
   560→  private frame = (): void => {
   561→    if (!this.engine || !this.scene || !this.world || !this.logic || !this.building || !this.actors || !this.rig) return;
   562→    const dt = Math.min(0.1, this.engine.getDeltaTime() / 1000);
   563→    const store = useGameStore.getState();
   564→    const frozen = selectModalOpen(store) || store.paused;
   565→
   566→    let ticked = false;
   567→    if (!frozen) {
   568→      this.logic.update(dt, this.readInput());
   569→      // En remoto el servidor es la autoridad del reloj: nada de ticks locales.
   570→      if (!this.net) {
   571→        this.clockAcc += dt * store.speed;
   572→        let guard = 0;
   573→        while (this.clockAcc >= 1 && guard < 60) {
   574→          tick(this.world, this.content);
   575→          this.clockAcc -= 1;
   576→          ticked = true;
   577→          guard += 1;
   578→        }
   579→      }
   580→    }
   581→
   582→    if (this.world.events.length > 0) this.handleEvents();
   583→    if (ticked) this.paintBuilding();
   584→
   585→    const now = performance.now();
   586→    if (now - this.lastHudSync > 200) {
   587→      this.lastHudSync = now;
   588→      this.syncHud();
   589→    }
   590→
   591→    const pos = this.logic.position();
   592→    this.actors.update(dt, pos, this.actorId());
   593→    // LOD por distancia: solo con cámaras cercanas (planta/estancia); el plano general (edificio) muestra todo.
   594→    this.building.update(dt, useGameStore.getState().camMode === 'edificio' ? undefined : { x: pos.x, floor: pos.floor });
   595→    this.rig.update(dt, pos);
   596→    this.scene.render();
   597→  };

   628→  private handleEvents(): void {
   629→    if (!this.world) return;
   630→    const store = useGameStore.getState();
   631→    for (const e of drainEvents(this.world)) {
   632→      const d = e.data as Record<string, unknown>;
   633→      eventSound(e.type, typeof d.type === 'string' ? d.type : undefined);
   634→      switch (e.type) {
   635→        case 'step_started':
   636→          if (d.type === 'phone_call') this.openPhone(String(d.mission), String(d.dialog));
   637→          break;
   638→        case 'choice_required': {
   639→          const choices = (d.choices as { id: string; label: string; hint?: string | null }[] | undefined) ?? [];
   640→          store.setChoices({ mission: String(d.mission), title: String(d.title), choices: choices.map((c) => ({ id: c.id, label: c.label, hint: c.hint ?? null })) });
   641→          store.pushLog(`⚖️ Decisión pendiente: ${String(d.title)}`, 'info');
   642→          break;
   643→        }
   644→        case 'door_locked': {
   645→          this.building?.shake(String(d.block));
   646→          const price = typeof d.price === 'number' ? d.price : null;
   647→          store.pushToast(`${blockLabel(String(d.block))}: puerta con candado${d.reason === 'for_sale' && price !== null ? ` (a la venta por ${price} €)` : ''}`, 'info');
   648→          break;
   649→        }
   650→        case 'neighbor_arrived':
   651→          store.pushToast(`Nuevo vecino en ${blockLabel(String(d.block))}: ${this.nameOf(d.resident)}`, 'good');
   652→          store.pushLog(`${this.nameOf(d.resident)} se muda a ${blockLabel(String(d.block))}. El edificio ya no es lo que era. Era mejor.`, 'info');
   653→          break;
   654→        case 'mission_activated':
   655→          store.pushLog(`📋 Nueva misión: ${String(d.title)}`, 'info');
   656→          break;
   657→        case 'mission_completed':
   658→          store.pushToast(`Misión completada: ${this.titleOf(d.mission)}`, 'good');
   659→          store.pushLog(`✔ Misión completada: ${this.titleOf(d.mission)}`, 'good');
   660→          break;
   661→        case 'mission_failed':
   662→          store.pushToast(`Misión fallida: ${this.titleOf(d.mission)}`, 'bad');
   663→          store.pushLog(`✘ Misión fallida: ${this.titleOf(d.mission)}`, 'bad');
   664→          break;
   665→        case 'block_unlocked':
   666→          store.pushToast(`${blockLabel(String(d.block))} desbloqueado`, 'good');
   667→          break;
   668→        case 'block_bought':
   669→          store.pushLog(`🔑 ${blockLabel(String(d.block))} comprado por ${String(d.price)} €`, 'good');
   670→          break;
   671→        case 'block_for_sale':
   672→          store.pushLog(`🪧 Cartel de "SE VENDE" en ${blockLabel(String(d.block))} (${String(d.price)} €)`, 'info');
   673→          break;
   674→        case 'knock_result':
   675→          store.pushToast(`🗣️ ${String(d.name)}: «${String(d.line)}»`, d.asleep ? 'info' : 'good');
   676→          break;
   677→        case 'noise_complaint': // texto data-driven: lo pinta el uiLog del catálogo (case mantenido por compatibilidad de eventos antiguos)
   678→          if (d.uiLog === true) break;
   679→          store.pushLog(`🔊 Ruido por encima del límite en ${blockLabel(String(d.block))}. Alguien aprieta los dientes.`, 'bad');
   680→          break;
   681→        case 'leak_detected':
   682→          if (d.uiLog === true) break;
   683→          store.pushLog(`💧 Goteo detectado en ${blockLabel(String(d.block))}. El cubo del salón hace ojitos.`, 'bad');
   684→          break;
   685→        case 'furniture_placed':
   686→          store.pushLog(`${String(d.emoji ?? '📦')} ${String(d.name)} colocado en ${blockLabel(String(d.block))} (${String(d.room)}).`, 'info');
   687→          break;
   688→        case 'furniture_removed':
   689→          store.pushLog(`${String(d.emoji ?? '📦')} ${String(d.name)} retirado de ${blockLabel(String(d.block))} (+${String(d.refund)} €).`, 'info');
   690→          break;
   691→        case 'room_renovated':
   692→          store.pushLog(`🧱 Reforma terminada en ${blockLabel(String(d.block))}. Hay estancia nueva que amueblar.`, 'good');
   693→          break;
   694→        case 'weather_changed':
   695→          store.pushLog(`El cielo cambia: ${this.content.rules.labels.weather[String(d.state)] ?? String(d.state)}`, 'info');
   696→          break;
   697→        case 'rent_collected':
   698→          store.pushToast(`+${String(d.amount)} € de alquiler (${blockLabel(String(d.block))})`, 'good');
   699→          break;
   700→        case 'work_done':
   701→          store.pushLog(`💻 Cerrada una "reunión que podía ser un correo": +${String(d.amount)} €`, 'info');
   702→          break;
   703→        case 'activity_started':
   704→          if (d.activity !== 'idle') store.pushLog(`${this.nameOf(d.resident)} ${this.activityLabel(d.resident, String(d.activity))}`, 'info');
   705→          break;
   706→        case 'junta_requested': {
   707→          const agenda = (d.agenda as { id: string; title: string; description: string; cost: number }[] | undefined) ?? [];
   708→          store.setJunta(agenda.map((i) => ({ id: i.id, title: i.title, description: i.description, cost: i.cost })));
   709→          break;
   710→        }
   711→        case 'junta_result':
   712→          store.pushToast(
   713→            d.approved ? `Aprobado: ${String(d.title)} (${String(d.yes)} sí / ${String(d.no)} no)` : `Rechazado: ${String(d.title)} (${String(d.yes)} sí / ${String(d.no)} no)`,
   714→            d.approved ? ('good' as Tone) : ('bad' as Tone),
   715→          );
   716→          break;
   717→        case 'junta_vote_registered':
   718→          store.pushLog(`🗳️ ${this.nameOf(d.resident)} ha depositado su voto. Pendientes: ${String(d.pendingHumans)}`, 'info');
   719→          break;
   720→        case 'junta_closed':
   721→          store.setJunta(null);
   722→          break;
   723→        case 'minigame_requested': {
   724→          const def = this.content.minigames[String(d.minigame)];
   725→          if (!def) break;
   726→          // Overrides numéricos del escenario activo (QA/E2E): solo config, nunca efectos ni textos.
   727→          const ov = this.scenarioOverrides[def.id];
   728→          const effective = ov ? ({ ...def, config: { ...def.config, ...ov } } as MinigameDef) : def;
   729→          store.setMinigame({ ...effective, emoji: effective.emoji ?? '🎮', instruction: effective.instruction ?? null });
   730→          break;
   731→        }
   732→        case 'minigame_finished': {
   733→          const name = this.content.minigames[String(d.minigame)]?.name ?? String(d.minigame);
   734→          store.pushLog(d.success ? `🎮 Minijuego superado: ${name}` : `🎮 Minijuego fallido: ${name}`, d.success ? 'good' : 'bad');
   735→          break;
   736→        }
   737→        case 'ui_toast':
   738→          store.pushToast(String(d.text), (d.tone as Tone | undefined) ?? 'info');
   739→          break;
   740→        case 'finale':
   741→          store.setFinale(String(d.text));
   742→          break;
   743→        case 'funds_changed': {
   744→          const amount = typeof d.amount === 'number' ? d.amount : 0;
   745→          if ((d.to === 'player' || d.to === 'community') && amount !== 0) {
   746→            store.pushLog(amount > 0 ? `+${amount} € (fondos)` : `${amount} € (fondos)`, amount > 0 ? 'good' : 'bad');
   747→          }
   748→          break;
   749→        }
   750→        default: {
   751→          // Umbral de canal con texto del catálogo (odors, wifi y futuros canales): totalmente data-driven.
   752→          if (d.uiLog === true && typeof d.message === 'string') {
   753→            store.pushLog(String(d.message).replace('{block}', blockLabel(String(d.block))), (d.tone as Tone | undefined) ?? 'info');
   754→          }
   755→          break; // resident_moved, param_changed, activity_ended... demasiado ruido para el registro
   756→        }
   757→      }
   758→    }
   759→  }

   780→  private objectiveUi(): { title: string; text: string } | null {
   781→    if (!this.world) return null;
   782→    const o = activeObjective(this.world, this.content);
   783→    if (!o) return null;
   784→    if (o.state === 'awaiting_choice') return { title: o.title, text: 'El edificio contiene la respiración. Decide qué hacer.' };
   785→    const s = o.step!;
   786→    const left = o.untilTick !== null ? ` · quedan ${Math.max(0, o.untilTick - this.world.tick)} min` : '';
   787→    switch (s.type) {
   788→      case 'phone_call':
   789→        return { title: o.title, text: 'Suena el teléfono. Atiéndelo.' };
   790→      case 'reach':
   791→        return { title: o.title, text: `Llega a la puerta (${blockLabel(s.block ?? '')}).` };
   792→      case 'enter':
   793→        return { title: o.title, text: 'Entra en tu piso con E.' };
   794→      case 'catch_in_act':
   795→        return { title: o.title, text: `Pilla al responsable in fraganti${left}.` };
   796→      default:
   797→        return { title: o.title, text: o.title };
   798→    }
   799→  }

   801→  private promptText(): string | null {
   802→    const logic = this.logic;
   803→    const world = this.world;
   804→    if (!logic || !world) return null;
   805→    if (logic.position().climbing) return null;
   806→    // El prompt es SIEMPRE el objetivo real de la tecla E (misma fuente, cero divergencias).
   807→    const t = logic.interactTarget();
   808→    if (t) {
   809→      const ic = this.content.rules.interact;
   810→      switch (t.kind) {
   811→        case 'door': {
   812→          const door = world.blocks[t.blockId]!;
   813→          if (logic.position().mode === 'inside') return ic.exit;
   814→          return door.access === 'locked'
   815→            ? ic.doorLocked.replace('{block}', blockLabel(door.id))
   816→            : ic.doorEnter.replace('{block}', blockLabel(door.id));
   817→        }
   818→        case 'resident':
   819→          return ic.greet.replace('{name}', this.nameOf(t.residentId));
   820→        case 'mailbox':
   821→          return ic.mailbox.prompt;
   822→        case 'noticeboard':
   823→          return ic.noticeboard.prompt;
   824→        case 'laptop':
   825→          return ic.laptop;
   826→        case 'bed':
   827→          return ic.bed;
   828→      }
   829→    }
   830→    const zone = logic.zone();
   831→    const feat = this.content.blocks[zone.defId]!.features;
   832→    if (feat?.stairs || feat?.elevator) {
   833→      const ic = this.content.rules.interact;
   834→      // Lo que dice el prompt es lo que W/S hace aquí: ascensor operativo manda; si no hay
   835→      // ascensor utilizable pero sí escalera, escalera; núcleo averiado sin escalera = aviso.
   836→      if (feat.elevator && world.blocks[zone.id]!.params.elevator_status === 'ok') return ic.elevator;
   837→      if (feat.stairs) return ic.stairs;
   838→      return ic.elevatorBroken;
   839→    }
   840→    return null;
   841→  }

   843→  /** Viviendas del jugador para el editor de interiores (foto serializable, posiciones ya en %). */
   844→  private homesUi() {
   845→    const world = this.world!;
   846→    const cellH = this.content.rules.view.cellH;
   847→    const mine = this.actorId();
   848→    return Object.values(world.blocks)
   849→      .filter((b) => b.kind === 'dwelling' && b.ownerId === mine)
   850→      .sort((a, b) => a.id.localeCompare(b.id))
   851→      .map((b) => ({
   852→        id: b.id,
   853→        label: blockLabel(b.id),
   854→        rooms: b.rooms.map((room) => {
   855→          const rdef = this.content.rooms[room.defId];
   856→          const slots = (rdef?.furnitureSlots ?? []).map((s) => {
   857→            const itemId = room.furniture[s.id] ?? '';
   858→            const fdef = itemId ? this.content.furniture[itemId] : undefined;
   859→            return {
   860→              id: s.id,
   861→              kind: s.kind,
   862→              left: Math.min(92, Math.max(2, (s.pos[0] / Math.max(0.001, rdef?.size.w ?? 1)) * 100)),
   863→              bottom: Math.min(76, Math.max(0, (s.pos[1] / Math.max(0.001, cellH)) * 100)),
   864→              item: fdef ? { id: fdef.id, name: fdef.name, emoji: fdef.emoji ?? '📦', color: fdef.view.color } : null,
   865→            };
   866→          });
   867→          return { key: room.key, label: rdef?.label ?? room.key, access: room.access, slots };
   868→        }),
   869→      }));
   870→  }

   872→  private syncHud(): void {
   873→    const world = this.world;
   874→    if (!world) return;
   875→    const player = world.residents[this.actorId()] ?? world.residents[world.playerId];
   876→    if (!player) return;
   877→    const home = player.home;
   878→    const params = visibleParams(world, this.content, home).map((p) => ({ name: p.name, label: p.def.label ?? p.name, value: String(p.value) }));
   879→    const missions = world.missions.map((m) => {
   880→      const def = this.content.missions[m.id]!;
   881→      const step = m.state === 'active' ? def.steps[m.step] : undefined;
   882→      return {
   883→        id: m.id,
   884→        title: def.title,
   885→        state: m.state,
   886→        hint: step?.hint ?? null,
   887→        remaining: m.until !== null ? `quedan ${Math.max(0, m.until - world.tick)} min` : null,
   888→      };
   889→    });
   890→    const travel = travelMinutes(buildCirculation(world, this.content), world.entry, home);
   891→    const elevatorBlock = Object.values(world.blocks).find((b) => b.params.elevator_status !== undefined);
   892→    useGameStore.getState().syncHud({
   893→      clock: { ...world.clock },
   894→      weather: world.weather.state,
   895→      funds: world.community.funds,
   896→      wallet: player.wallet,
   897→      reputation: player.reputation,
   898→      elevator: String(elevatorBlock?.params.elevator_status ?? 'ok'),
   899→      homeLabel: blockLabel(home),
   900→      travelText: Number.isFinite(travel) ? `del portal a tu puerta: ${travel} min` : null,
   901→      objective: this.objectiveUi(),
   902→      prompt: this.promptText(),
   903→      params,
   904→      missions,
   905→      homes: this.homesUi(),
   906→      furnitureConfig: { refundRatio: this.content.rules.furniture.refundRatio, roomUnlockPrice: this.content.rules.furniture.roomUnlock.price },
   907→    });
   908→  }
