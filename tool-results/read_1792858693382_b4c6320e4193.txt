    76→  async init(canvas: HTMLCanvasElement): Promise<void> {
    77→    if (this.engine || this.initializing) return;
    78→    this.initializing = true;
    79→    this.canvas = canvas;
    80→
    81→    try {
    82→      // 1) Contenido externo + validación (los errores de referencias se avisan, no abortan).
    83→      const { content, issues } = await loadContent();
    84→      this._content = content;
    85→      this.cutscenePlayer = new CutscenePlayer(content);
    86→      preloadSounds();
    87→      useGameStore.setState({ touchHint: content.rules.touch.hint });
    88→      if (issues.length > 0) {
    89→        useGameStore.getState().pushToast(`Contenido inválido: ${issues.length} errores de validación`, 'bad');
    90→      }
    91→
    92→      // 2) Motor de render (WebGPU con fallback WebGL2). La calidad la elige el jugador en Opciones.
    93→      const { engine, backend } = await createEngine(canvas);
    94→      this.engine = engine;
    95→      engine.setHardwareScalingLevel(QUALITY_SCALE[loadOptions().quality]);
    96→      window.addEventListener('resize', this.onResize);
    97→      await this.restart();
    98→      useGameStore.getState().markReady(backend);
    99→      startGamepadPolling();
   100→      engine.runRenderLoop(this.frame);
   101→      // Gancho de depuración/pruebas E2E (no lo usa la UI).
   102→      (window as unknown as { __vecinos?: unknown }).__vecinos = { store: useGameStore, controller: this };
   103→    } catch (err) {
   104→      console.error('No se pudo arrancar el juego (contenido o motor de render)', err);
   105→      useGameStore.getState().setEngineError(true);
   106→    }
   107→  }

   122→  onKeyDown(code: string): void {
   123→    // Cutscene en curso: E/Espacio/Enter/Escape saltan el espectáculo; nada más pasa.
   124→    if (this.cutscenePlayer?.active) {
   125→      if (code === 'KeyE' || code === 'Space' || code === 'Enter' || code === 'Escape') this.cutsceneSkip();
   126→      return;
   127→    }
   128→    if (this.keys.has(code)) return;
   129→    this.keys.add(code);
   130→    const store = useGameStore.getState();
   131→    if (selectModalOpen(store)) return; // los modales gestionan sus propias teclas
   132→    if (code === 'KeyE') {
   133→      this.interact();
   134→      return;
   135→    }
   136→    const cam = CAM_BY_KEY[code];
   137→    if (cam) this.setCamMode(cam);
   138→  }

   330→  private connectNet(): void {
   331→    this.net = new NetSession({
   332→      onState: (s) => this.applyRemoteState(s),
   333→      onParticipants: (p) => useGameStore.getState().setMpParticipants(p),
   334→      onJoined: (j) => {
   335→        this.myResidentId = j.yourResidentId;
   336→        this.isHost = j.isHost;
   337→        useGameStore.getState().setMpSession({ code: j.code, isHost: j.isHost, myResidentId: j.yourResidentId });
   338→        useGameStore.getState().setMpParticipants(j.participants);
   339→      },
   340→      onError: (m) => {
   341→        useGameStore.getState().pushToast(m, 'bad');
   342→      },
   343→      onStatus: (st) => {
   344→        if (!this.net) return;
   345→        if (st === 'connected' && useGameStore.getState().mpStatus === 'connecting') return; // el ack decide
   346→        if (st === 'disconnected' && useGameStore.getState().mpStatus === 'connected') {
   347→          useGameStore.getState().setMpStatus('error');
   348→          useGameStore.getState().setMpError('Conexión perdida con el servidor de la comunidad.');
   349→        }
   350→      },
   351→    });
   352→  }

   365→  /** Adopta el estado del servidor (autoridad). Primera vez: reinicia la escena con ese mundo. */
   366→  private applyRemoteState(s: SessionStatePayload): void {
   367→    if (s.seq <= this.appliedSeq) return;
   368→    this.appliedSeq = s.seq;
   369→    if (!this.world || this.world.scenarioId !== s.world.scenarioId) {
   370→      void this.restart(s.world).then(() => {
   371→        if (!this.world) return;
   372→        this.world.events = s.events.slice();
   373→        this.handleEvents();
   374→        this.reconcile();
   375→        this.syncHud();
   376→      });
   377→      return;
   378→    }
   379→    // Aplica en sitio conservando la identidad del objeto (PlayerLogic/building/actors referencian el mismo World).
   380→    Object.assign(this.world, s.world);
   381→    this.world.events = s.events.slice();
   382→    this.handleEvents();
   383→    this.paintBuilding();
   384→    this.reconcile();
   385→    this.syncHud();
   386→  }

   388→  /** Corrige desvíos de posición del avatar local respecto al mundo del servidor (autoridad). */
   389→  private reconcile(): void {
   390→    if (!this.logic || !this.world || !this.net) return;
   391→    const me = this.world.residents[this.actorId()];
   392→    if (!me) return;
   393→    const pos = this.logic.position();
   394→    const locBlock = me.location ? this.world.blocks[me.location] : null;
   395→    if (locBlock) {
   396→      const insideOk = pos.mode === 'inside' && pos.insideBlock === locBlock.id;
   397→      const corridorOk = pos.mode === 'corridor' && !pos.insideBlock;
   398→      if (!(locBlock.kind === 'dwelling' ? insideOk : corridorOk) || Math.abs(pos.floor - locBlock.cell.floor) > 0.01) {
   399→        this.logic.forceLocation(locBlock.id);
   400→      }
   401→    } else if (pos.mode === 'inside') {
   402→      this.logic.forceLocation(null);
   403→    }
   404→  }

   406→  /** Siguiente frase del teléfono; al colgar responde la llamada en el sim y saca la siguiente de la cola. */
   407→  phoneNext(): void {
   408→    if (this.phoneLine < this.phoneLines.length - 1) {
   409→      this.phoneLine += 1;
   410→      this.showPhoneLine();
   411→      return;
   412→    }
   413→    const mission = this.phoneMission;
   414→    this.phoneLines = [];
   415→    this.phoneMission = '';
   416→    useGameStore.getState().setPhone(null);
   417→    this.actors?.director.setGesture(this.actorId(), null); // cuelgas: se suelta el teléfono
   418→    if (this.world && mission) {
   419→      this.sendCommand({ type: 'answer_call', mission });
   420→    }
   421→    const next = this.phoneQueue.shift();
   422→    if (next) this.openPhone(next.mission, next.dialogId);
   423→  }

   550→  /** (Re)crea el mundo y la escena. Con `state` restaura una partida guardada. */
   551→  private async restart(state?: World): Promise<void> {
   552→    if (!this.engine || !this.canvas || !this._content) return;
   553→    // Ninguna cutscene sobrevive a un cambio de mundo: cola fuera y teatro desmontado.
   554→    this.cutsceneQueue = [];
   555→    this.cutscenePlayer?.abort();
   556→    const { Color4, DirectionalLight, HemisphericLight, Scene, UniversalCamera, Vector3 } = B();
   557→    const view = this.content.rules.view;
   558→    useGameStore.getState().resetUi();
   559→
   560→    this.world = state ?? createWorld(this.content, this.activeScenarioId());
   561→    this.scenarioOverrides = this.content.scenarios[this.world.scenarioId ?? '']?.minigameOverrides ?? {};
   562→    // Guardados antiguos sin `entry`: la entrada es la zona común de planta baja.
   563→    if (!this.world.entry) {
   564→      this.world.entry = Object.values(this.world.blocks).find((b) => b.cell.floor === 0 && b.kind !== 'dwelling')?.id ?? Object.keys(this.world.blocks)[0]!;
   565→    }
   566→    // Migración de estancias de guardados v0.1: params y ranuras de muebles según catálogo.
   567→    for (const b of Object.values(this.world.blocks)) migrateRooms(this.content, b.rooms);
   568→    this.logic = new PlayerLogic(this.world, this.content, {
   569→      actorId: this.myResidentId ?? undefined,
   570→      send: (cmd) => this.sendCommand(cmd),
   571→    });
   572→
   573→    this.scene?.dispose();
   574→    this.scene = new Scene(this.engine);
   575→    this.scene.clearColor = new Color4(view.sky.r, view.sky.g, view.sky.b, 1);
   576→    new HemisphericLight('ambient', new Vector3(...view.ambient.dir), this.scene).intensity = view.ambient.intensity;
   577→    new DirectionalLight('sun', new Vector3(...view.sun.dir), this.scene).intensity = view.sun.intensity;
   578→
   579→    this.building = createBuilding(this.scene, this.world, this.content);
   580→    this.elevators = createElevators(this.scene, this.world, this.content);
   581→    this.actors = createActors(this.scene, this.world, this.content, this.elevators);
   582→    const blocks = Object.values(this.world.blocks);
   583→    // El cubo de escalera (v1.4) sobresale spanW−1 celdas por la derecha del edificio:
   584→    // el encuadre lo incluye para que la torre no se corte en cámara.
   585→    const hasStairs = blocks.some((b) => this.content.blocks[b.defId]?.features?.stairs === true);
   586→    const cols = Math.max(...blocks.map((b) => b.cell.col + b.size.w)) + (hasStairs ? this.content.rules.view.stairs.spanW - 1 : 0);
   587→    this.rig = new CameraRig(
   588→      new UniversalCamera('cam', new Vector3(0, 0, -30), this.scene),
   589→      this.engine,
   590→      cols,
   591→      Math.max(...blocks.map((b) => b.cell.floor)) + 1,
   592→      view.cellW,
   593→      view.cellH,
   594→    );
   595→    // Re-sincronización de vista (v0.6-b): la cámara sigue la posición del mundo (restaurado o
   596→    // nuevo), no la preferencia anterior. Dentro de una vivienda → estancia; en pasillo, la
   597→    // preferencia salvo que fuera 'estancia' (ahí estás en el pasillo, no en una estancia).
   598→    const pos = this.logic.position();
   599→    const store = useGameStore.getState();
   600→    const wanted: CamMode = pos.mode === 'inside' ? 'estancia' : store.camMode === 'estancia' ? 'planta' : store.camMode;
   601→    this.rig.mode = wanted;
   602→    store.setCamMode(wanted);
   603→
   604→    this.clockAcc = 0;
   605→    this.keys.clear();
   606→    this.phoneLines = [];
   607→    this.phoneMission = '';
   608→    this.phoneQueue = [];
   609→    this.lastHudSync = 0;
   610→
   611→    // Re-llamada (Fase F): si la partida montada tenía pasos de teléfono pendientes, vuelven a
   612→    // sonar. La cola de openPhone ordena el resto sin que ninguna llamada se pierda.
   613→    if (this.world) {
   614→      for (const m of this.world.missions) {
   615→        if (m.state !== 'active') continue;
   616→        const step = this.content.missions[m.id]?.steps[m.step];
   617→        if (step?.type === 'phone_call') this.openPhone(m.id, step.dialog);
   618→      }
   619→    }
   620→
   621→    this.pump();
   622→  }

   630→  private frame = (): void => {
   631→    if (!this.engine || !this.scene || !this.world || !this.logic || !this.building || !this.actors || !this.rig || !this.elevators) return;
   632→    const dt = Math.min(0.1, this.engine.getDeltaTime() / 1000);
   633→    const store = useGameStore.getState();
   634→    const frozen = selectModalOpen(store) || store.paused;
   635→
   636→    let ticked = false;
   637→    if (!frozen) {
   638→      this.logic.update(dt, this.readInput());
   639→      // Ascensor (v1.5): el director mueve cabinas y puertas; el jugador llama con W/S en el
   640→      // núcleo y su escalada EXTERNA la gobierna la cabina (espera → embarco → viaje → salida).
   641→      this.elevators.update(dt);
   642→      const climb = this.logic.climbInfo();
   643→      if (climb) {
   644→        if (!this.logic.externalRequested) {
   645→          const zone = this.logic.zone();
   646→          this.elevators.requestRide(this.actorId(), zone.cell.col, climb.from, climb.to);
   647→          this.logic.externalRequested = true;
   648→        }
   649→        const st = this.elevators.status(this.actorId());
   650→        if (!st) {
   651→          this.logic.finishClimb(); // fuera de servicio: la escalera manda
   652→          this.logic.externalRequested = false;
   653→          this.playerBoardT = 0;
   654→        } else {
   655→          this.logic.setClimbProgress(st.phase === 'riding' ? st.progress : 0);
   656→          const inside = st.phase === 'boarding' || st.phase === 'riding';
   657→          this.logic.setClimbZ(inside ? this.content.rules.view.roomZ - this.content.rules.view.commonDepth / 2 + this.content.rules.view.elevator.rideZ : null);
   658→          // El jugador "da un paso" dentro de la cabina un instante tras abrirse la puerta
   659→          // (el zOverride ya lo acerca); avisar al director para que cierre y viaje.
   660→          if (st.phase === 'boarding') {
   661→            this.playerBoardT += dt;
   662→            if (this.playerBoardT > 1.0) this.elevators.notifyBoarded(this.actorId());
   663→          } else {
   664→            this.playerBoardT = 0;
   665→          }
   666→          if (st.phase === 'exiting') {
   667→            this.logic.finishClimb();
   668→            this.logic.externalRequested = false;
   669→          }
   670→        }
   671→      } else if (this.logic.externalRequested) {
   672→        this.logic.externalRequested = false;
   673→      }
   674→      // En remoto el servidor es la autoridad del reloj: nada de ticks locales.
   675→      if (!this.net) {
   676→        this.clockAcc += dt * store.speed;
   677→        let guard = 0;
   678→        while (this.clockAcc >= 1 && guard < 60) {
   679→          tick(this.world, this.content);
   680→          this.clockAcc -= 1;
   681→          ticked = true;
   682→          guard += 1;
   683→        }
   684→      }
   685→    }
   686→
   687→    if (this.world.events.length > 0) this.handleEvents();
   688→    if (ticked) this.paintBuilding();
   689→
   690→    const now = performance.now();
   691→    if (now - this.lastHudSync > 200) {
   692→      this.lastHudSync = now;
   693→      useGameStore.getState().setFps(Math.round(this.engine.getFps()));
   694→      this.syncHud();
   695→    }
   696→
   697→    const pos = this.logic.position();
   698→    this.actors.update(dt, pos, this.actorId());
   699→    // Cutscene (Fase D): sus beats mueven cámara/actores DESPUÉS de los eventos y ANTES del render.
   700→    this.cutscenePlayer?.update(dt);
   701→    // Si la cutscene acaba y queda cola, encadena la siguiente aquí: no dependemos de que el sim
   702→    // genere otro evento para descolgar la siguiente (durante el teatro el sim está congelado).
   703→    if (this.cutsceneQueue.length > 0 && !this.cutscenePlayer?.active) this.pumpCutscene();
   704→    // LOD por distancia: solo con cámaras cercanas (planta/estancia); el plano general (edificio) muestra todo.
   705→    this.building.update(dt, useGameStore.getState().camMode === 'edificio' ? undefined : { x: pos.x, floor: pos.floor });
   706→    this.rig.update(dt, pos);
   707→    this.scene.render();
   708→  };

   868→        }
   869→        default: {

   871→          if (d.uiLog === true && typeof d.message === 'string') {

   875→        }
   876→      }
   877→      // Cutscenes (Fase D): después del reparto, ¿alguna quiere este evento como disparador?
   878→      this.maybeCutscene(e);
   879→    }
   880→    this.pumpCutscene();
   881→  }

   885→  /** Si el evento encaja con el trigger de alguna cutscene del catálogo, se encola (máx. 3). */
   886→  private maybeCutscene(e: GameEvent): void {
   887→    const defs = Object.values(this.content.cutscenes);
   888→    if (defs.length === 0) return;
   889→    const d = e.data as Record<string, unknown>;
   890→    const hit = defs.find((c) => {
   891→      const t = c.trigger;
   892→      if (t.on !== e.type) return false;
   893→      if (t.mission !== undefined && t.mission !== String(d.mission ?? '')) return false;
   894→      if (t.step !== undefined && t.step !== String(d.step ?? '')) return false;
   895→      return true;
   896→    });
   897→    if (!hit) return;
   898→    if (this.cutsceneQueue.length >= 3) return; // nunca más de 3 esperando: el teatro no atravaca la partida
   899→    this.cutsceneQueue.push({
   900→      def: hit,
   901→      ctx: {
   902→        mission: typeof d.mission === 'string' ? d.mission : null,
   903→        resident: typeof d.resident === 'string' ? d.resident : null,
   904→        block: typeof d.block === 'string' ? d.block : null,
   905→      },
   906→    });
   907→  }

   909→  /** Lanza la siguiente cutscene en cola si no hay nada en pantalla (también un modal la pospone). */
   910→  private pumpCutscene(): void {
   911→    if (!this.cutscenePlayer || !this.world || !this.rig || !this.actors || !this.engine || !this.logic) return;
   912→    if (this.cutscenePlayer.active || this.cutsceneQueue.length === 0) return;
   913→    if (selectModalOpen(useGameStore.getState())) return; // no se roba la pantalla a un modal
   914→    const next = this.cutsceneQueue.shift()!;
   915→    this.cutscenePlayer.start(next.def, next.ctx, {
   916→      engine: this.engine,
   917→      rig: this.rig,
   918→      actors: this.actors.director,
   919→      world: this.world,
   920→      playerPos: () => this.logic!.position(),
   921→    });
   922→  }

   926→  private openPhone(mission: string, dialogId: string): void {
   927→    const dlg = this.content.dialogs[dialogId];
   928→    if (!dlg || dlg.lines.length === 0) return;
   929→    // Si ya hay una llamada en pantalla, la nueva espera en cola (dedupe por misión).
   930→    if (this.phoneMission) {
   931→      if (this.phoneMission !== mission && !this.phoneQueue.some((q) => q.mission === mission)) {
   932→        this.phoneQueue.push({ mission, dialogId });
   933→      }
   934→      return;
   935→    }
   936→    this.phoneMission = mission;
   937→    this.phoneLines = dlg.lines;
   938→    this.phoneLine = 0;
   939→    this.actors?.director.setGesture(this.actorId(), 'phone'); // el jugador atiende con el móvil a la oreja
   940→    this.showPhoneLine();
   941→  }

   958→  private objectiveUi(): { title: string; text: string } | null {
   959→    if (!this.world) return null;
   960→    const o = activeObjective(this.world, this.content);
   961→    if (!o) return null;
   962→    if (o.state === 'awaiting_choice') return { title: o.title, text: 'El edificio contiene la respiración. Decide qué hacer.' };
   963→    const s = o.step!;
   964→    const left = o.untilTick !== null ? ` · quedan ${Math.max(0, o.untilTick - this.world.tick)} min` : '';
   965→    switch (s.type) {
   966→      case 'phone_call':
   967→        return { title: o.title, text: 'Suena el teléfono. Atiéndelo.' };
   968→      case 'reach':
   969→        return { title: o.title, text: `Llega a la puerta (${this.blockName(s.block ?? '')}).` };
   970→      case 'enter':
   971→        return { title: o.title, text: 'Entra en tu piso con E.' };
   972→      case 'catch_in_act':
   973→        return { title: o.title, text: `Pilla al responsable in fraganti${left}.` };
   974→      default:
   975→        return { title: o.title, text: o.title };
   976→    }
   977→  }
   978→
   979→  private promptText(): string | null {
   980→    const logic = this.logic;
   981→    const world = this.world;
   982→    if (!logic || !world) return null;
   983→    if (logic.position().climbing) return null;
   984→    // El prompt es SIEMPRE el objetivo real de la tecla E (misma fuente, cero divergencias).
   985→    const t = logic.interactTarget();
   986→    if (t) {
   987→      const ic = this.content.rules.interact;
   988→      switch (t.kind) {
   989→        case 'door': {
   990→          const door = world.blocks[t.blockId]!;
   991→          if (logic.position().mode === 'inside') return ic.exit;
   992→          return door.access === 'locked'
   993→            ? ic.doorLocked.replace('{block}', this.blockName(door.id))
   994→            : ic.doorEnter.replace('{block}', this.blockName(door.id));
   995→        }
   996→        case 'resident':
   997→          return ic.greet.replace('{name}', this.nameOf(t.residentId));
   998→        case 'mailbox':
   999→          return ic.mailbox.prompt;
  1000→        case 'noticeboard':
  1001→          return ic.noticeboard.prompt;
  1002→        case 'laptop':
  1003→          return ic.laptop;
  1004→        case 'bed':
  1005→          return ic.bed;
  1006→      }
  1007→    }
  1008→    const zone = logic.zone();
  1009→    const feat = this.content.blocks[zone.defId]!.features;
  1010→    if (feat?.stairs || feat?.elevator) {
  1011→      const ic = this.content.rules.interact;
  1012→      // Lo que dice el prompt es lo que W/S hace aquí: ascensor operativo manda; si no hay
  1013→      // ascensor utilizable pero sí escalera, escalera; núcleo averiado sin escalera = aviso.
  1014→      if (feat.elevator && world.blocks[zone.id]!.params.elevator_status === 'ok') return ic.elevator;
  1015→      if (feat.stairs) return ic.stairs;
  1016→      return ic.elevatorBroken;
  1017→    }
  1018→    return null;
  1019→  }

  1021→  /** Viviendas del jugador para el editor de interiores (foto serializable, posiciones ya en %). */
  1022→  private homesUi() {
  1023→    const world = this.world!;
  1024→    const cellH = this.content.rules.view.cellH;
  1025→    const mine = this.actorId();
  1026→    return Object.values(world.blocks)
  1027→      .filter((b) => b.kind === 'dwelling' && b.ownerId === mine)
  1028→      .sort((a, b) => a.id.localeCompare(b.id))
  1029→      .map((b) => ({
  1030→        id: b.id,
  1031→        label: blockLabel(b.id),
  1032→        rooms: b.rooms.map((room) => {
  1033→          const rdef = this.content.rooms[room.defId];
  1034→          const slots = (rdef?.furnitureSlots ?? []).map((s) => {
  1035→            const itemId = room.furniture[s.id] ?? '';
  1036→            const fdef = itemId ? this.content.furniture[itemId] : undefined;
  1037→            return {
  1038→              id: s.id,
  1039→              kind: s.kind,
  1040→              left: Math.min(92, Math.max(2, (s.pos[0] / Math.max(0.001, rdef?.size.w ?? 1)) * 100)),
  1041→              bottom: Math.min(76, Math.max(0, (s.pos[1] / Math.max(0.001, cellH)) * 100)),
  1042→              item: fdef ? { id: fdef.id, name: fdef.name, emoji: fdef.emoji ?? '📦', color: fdef.view.color } : null,
  1043→            };
  1044→          });
  1045→          return { key: room.key, label: rdef?.label ?? room.key, access: room.access, slots };
  1046→        }),
  1047→      }));
  1048→  }

  1050→  private syncHud(): void {
  1051→    const world = this.world;
  1052→    if (!world) return;
  1053→    const player = world.residents[this.actorId()] ?? world.residents[world.playerId];
  1054→    if (!player) return;
  1055→    const home = player.home;
  1056→    const params = visibleParams(world, this.content, home).map((p) => ({ name: p.name, label: p.def.label ?? p.name, value: String(p.value) }));
  1057→    const missions = world.missions.map((m) => {
  1058→      const def = this.content.missions[m.id]!;
  1059→      const step = m.state === 'active' ? def.steps[m.step] : undefined;
  1060→      return {
  1061→        id: m.id,
  1062→        title: def.title,
  1063→        state: m.state,
  1064→        hint: step?.hint ?? null,
  1065→        remaining: m.until !== null ? `quedan ${Math.max(0, m.until - world.tick)} min` : null,
  1066→      };
  1067→    });
  1068→    const travel = travelMinutes(buildCirculation(world, this.content), world.entry, home);
  1069→    const elevatorBlock = Object.values(world.blocks).find((b) => b.params.elevator_status !== undefined);
  1070→    useGameStore.getState().syncHud({
  1071→      clock: { ...world.clock },
  1072→      weather: world.weather.state,
  1073→      funds: world.community.funds,
  1074→      wallet: player.wallet,
  1075→      reputation: player.reputation,
  1076→      elevator: String(elevatorBlock?.params.elevator_status ?? 'ok'),
  1077→      homeLabel: blockLabel(home),
  1078→      travelText: Number.isFinite(travel) ? `del portal a tu puerta: ${travel} min` : null,
  1079→      objective: this.objectiveUi(),
  1080→      prompt: this.promptText(),
  1081→      params,
  1082→      missions,
  1083→      homes: this.homesUi(),
  1084→      furnitureConfig: { refundRatio: this.content.rules.furniture.refundRatio, roomUnlockPrice: this.content.rules.furniture.roomUnlock.price },
  1085→    });
  1086→  }
