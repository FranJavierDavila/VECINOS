    33→  /** Pared del fondo de la vivienda (la única malla que puede llevar edgesRendering) o null;
    34→   *  el EdgesRenderer activo solo existe en las paredes destacadas del jugador. */
    35→  edgesMesh: Mesh | null;
    36→  /** true mientras el LOD fuerza baseEmissive (al soltarlo se repinta el cálculo de paint una vez). */
    37→  glowForced: boolean;
    38→  /** true mientras el LOD oculta panel+candado; la visibilidad efectiva es la de paint() && !esto. */
    39→  lodHiddenPanels: boolean;
    40→  /** Segunda pared con edges (v1.19): la pared del fondo se parte si hay puerta a sala trasera. */
    41→  edgesMesh2: Mesh | null;
    42→}
    43→
    44→/** Interior 2D de una vivienda (Fase C): suelo por estancia + tapa de puerta hacia estancia cerrada. */
    45→interface InteriorView {
    46→  /** Suelo tintado por estancia (color del catálogo; lockedColor si está cerrada). */
    47→  floors: { roomKey: string; defId: string; mesh: Mesh; mat: StandardMaterial }[];
    48→  /** Tapas de los huecos de puerta que llevan a una estancia cerrada ("en obras"). */
    49→  covers: { mesh: Mesh; rooms: [string, string] }[];
    50→}
    51→
    52→/**
    53→ * Una caja por bloque (placeholder greybox del futuro prefab) y, si el manifiesto de assets
    54→ * (Content.assets.blocks) declara un modelo glTF para el tipo de bloque, el modelo en su lugar
    55→ * (con la caja oculta). TODA la geometría y los colores salen de content.rules.view +
    56→ * block.view + assets: sustituir el greybox por arte final no toca este código.
    57→ * Las viviendas se dibujan en corte (interior visible) SOLO mientras hay alguien dentro;
    58→ * vacías, la fachada se cierra con su puerta al pasillo (rules.view.privacy) — y las
    59→ * bloqueadas, con panel y candado como siempre.
    60→ */
    61→export function createBuilding(scene: Scene, world: World, content: Content) {
    62→  const { Color3, Color4, MeshBuilder, StandardMaterial, TransformNode } = B();
    63→  const v = content.rules.view;
    64→  const glow = Object.fromEntries(Object.entries(v.channelGlow).map(([k, hex]) => [k, Color3.FromHexString(hex)])) as Record<string, Color3>;
    65→  const views = new Map<string, BlockView>();
    66→  const shakes = new Map<string, number>();
    67→  /** Raíz de escena por bloque (v1.19): colgar aquí TODA malla del bloque permite el
    68→   *  desvanecimiento completo del bloque principal en el apilamiento 3D (swap de bloques). */
    69→  const roots = new Map<string, TransformNode>();
    70→  const attach = (blockId: string, mesh: Mesh): Mesh => {
    71→    const r = roots.get(blockId);
    72→    if (r) mesh.parent = r;
    73→    return mesh;
    74→  };
    75→  /** Interiores 2D (Fase C) por bloque: suelos de estancia y tapas de puertas cerradas. */
    76→  const interiors = new Map<string, InteriorView>();
    77→  /** Foco del LOD (posición del jugador): lo alimenta update() y lo consume paint(). */
    78→  let lastFocus: { x: number; floor: number } | null = null;
    79→  // Plano frontal de las estancias: la cara de la puerta. Con viviendas del doble de calado
    80→  // (v0.6-b), el extra crece hacia el FONDO (+Z) para no envolver el pasillo ni al núcleo.
    81→  // v1.19: los calados del catálogo YA incluyen el +1 módulo (rules.view.depthModule) —
    82→  // depth 4.2 en viviendas y commonDepth 2.8 en comunes son los valores finales.
    83→  const frontZ = v.roomZ - v.commonDepth / 2;
    84→  /** Calado por bloque: viviendas 2× (mobiliario interior); naves del cierre (v1.16) y el
    85→   *  tejado (con aleros) a calado de vivienda; bloques comunes, el de siempre. */
    86→  const depthOf = (b: BlockInstance): number => {
    87→    if (b.kind === 'dwelling') return v.depth;
    88→    if (b.defId === 'trasteros_planta' || b.defId === 'bajocubierta') return v.depth;
    89→    if (b.defId === v.trasteros.blockDefId) return v.depth * v.trasteros.blockDepthFactor;
    90→    if (b.defId === 'tejado') return v.depth + v.roof.overhang * 2;
    91→    return v.commonDepth;
    92→  };
    93→  const mat = (name: string) => new StandardMaterial(name, scene);
    94→  const box = (name: string, w: number, h: number, d: number, x: number, y: number, z: number, m: StandardMaterial) => {
    95→    const mesh = MeshBuilder.CreateBox(name, { width: w, height: h, depth: d }, scene);
    96→    mesh.position.set(x, y, z);
    97→    mesh.material = m;
    98→    return mesh;
    99→  };
   100→
   101→  for (const b of Object.values(world.blocks)) {
   102→    const m = mat(`${b.id}_mat`);
   103→    const cx = (b.cell.col + b.size.w / 2) * v.cellW;
   104→    const bottom = b.cell.floor * v.cellH;
   105→    const w = b.size.w * v.cellW - v.gapW;
   106→    const h = v.cellH - v.gapH;
   107→    const d = depthOf(b);
   108→    const cz = frontZ + d / 2;
   109→    const view: BlockView = {
   110→      greybox: [],
   111→      mats: [m],
   112→      baseEmissive: [m.emissiveColor.clone()],
   113→      baseColor: [Color3.FromHexString(content.blocks[b.defId]!.view?.color ?? '#cccccc')],
   114→      tintStrength: 1,
   115→      model: null,
   116→      front: null,
   117→      door: null,
   118→      handle: null,
   119→      lock: null,
   120→      lockBase: { x: cx, y: bottom + v.cellH / 2 - 0.2 },
   121→      centerX: cx,
   122→      floor: b.cell.floor,
   123→      edgesMesh: null,
   124→      edgesMesh2: null,
   125→      glowForced: false,
   126→      lodHiddenPanels: false,
   127→    };
   128→    const root = new TransformNode(`${b.id}_root`, scene);
   129→    roots.set(b.id, root);
   130→    if (b.layer === 'corridor') {
   131→      // Pasillo exterior (data-driven): tira transitable saliente hacia la calle, con el borde
   132→      // trasero a ras de fachada (frontZ) y barandilla baja en el borde exterior — sitio de
   133→      // sobra para que los personajes se crucen o charlen sin bailar el vacío.
   134→      const d = v.corridorDepth;
   135→      view.greybox.push(box(b.id, b.size.w * v.cellW, v.slab * 2, d, cx, bottom + v.slab, frontZ - d / 2, m));
   136→      if (v.corridorRail > 0) {
   137→        view.greybox.push(
   138→          box(`${b.id}_barandilla`, b.size.w * v.cellW, v.corridorRail, v.wall, cx, bottom + v.slab * 2 + v.corridorRail / 2, frontZ - d, m),
   139→        );
   140→      }
   141→    } else if (b.kind === 'dwelling') {
   142→      const y = bottom + v.cellH / 2;
   143→      // ── Puerta a sala TRASERA (v1.19, apilamiento 3D): la pared del fondo se parte con el
   144→      // hueco del layout (doorway hacia una estancia más profunda que el bloque principal).
   145→      const layoutId0 = content.blocks[b.defId]?.container?.layout;
   146→      const layout0 = layoutId0 ? content.layouts[layoutId0] : undefined;
   147→      const rearKeys = new Set((layout0?.rooms ?? []).filter((r) => r.rect && r.rect.z + r.rect.d > v.depth + 0.01).map((r) => r.key));
   148→      const rearDoor = (layout0?.doorways ?? []).find((dw) => dw.axis === 'x' && rearKeys.has(dw.b));
   149→      const x0w = cx - w / 2;
   150→      const x1w = cx + w / 2;
   151→      if (rearDoor) {
   152→        const gapA = Math.max(x0w, x0w + rearDoor.at - rearDoor.width / 2);
   153→        const gapB = Math.min(x1w, x0w + rearDoor.at + rearDoor.width / 2);
   154→        const doorH = Math.min(h - 0.15, 1.95);
   155→        const lintelH = h - doorH;
   156→        view.edgesMesh = box(`${b.id}_fondo_i`, gapA - x0w, h, v.wall, (x0w + gapA) / 2, y, frontZ + d, m);
   157→        view.edgesMesh2 = box(`${b.id}_fondo_d`, x1w - gapB, h, v.wall, (gapB + x1w) / 2, y, frontZ + d, m);
   158→        view.greybox.push(
   159→          view.edgesMesh,
   160→          view.edgesMesh2,
   161→          box(`${b.id}_fondo_dintel`, gapB - gapA, lintelH, v.wall, (gapA + gapB) / 2, y + h / 2 - lintelH / 2, frontZ + d, m),
   162→          box(`${b.id}_izq`, v.wall, h, d, cx - w / 2, y, cz, m),
   163→          box(`${b.id}_der`, v.wall, h, d, cx + w / 2, y, cz, m),
   164→          box(`${b.id}_suelo`, w, v.wall, d, cx, bottom + v.slab + v.wall / 2, cz, m),
   165→          box(`${b.id}_techo`, w, v.wall, d, cx, bottom + v.cellH - v.slab - v.wall / 2, cz, m),
   166→        );
   167→      } else {
   168→        view.edgesMesh = box(`${b.id}_fondo`, w, h, v.wall, cx, y, frontZ + d, m);
   169→        view.greybox.push(
   170→          view.edgesMesh,
   171→          box(`${b.id}_izq`, v.wall, h, d, cx - w / 2, y, cz, m),
   172→          box(`${b.id}_der`, v.wall, h, d, cx + w / 2, y, cz, m),
   173→          box(`${b.id}_suelo`, w, v.wall, d, cx, bottom + v.slab + v.wall / 2, cz, m),
   174→          box(`${b.id}_techo`, w, v.wall, d, cx, bottom + v.cellH - v.slab - v.wall / 2, cz, m),
   175→        );
   176→      }
   177→      view.front = box(`${b.id}_puerta`, w, h, v.wall, cx, y, frontZ, m);
   178→      view.lock = box(`${b.id}_candado`, 0.3, 0.38, 0.12, cx, view.lockBase.y, frontZ - 0.1, mat(`${b.id}_candado_mat`));
   179→      (view.lock.material as StandardMaterial).diffuseColor = Color3.FromHexString(v.lockColor);
   180→
   181→      // ── Privacidad (v1.17): puerta y picaporte sobre la fachada de la vivienda vacía.
   182→      // Se activan/desactivan cada frame según haya alguien dentro (propietario, visita…
   183→      // o un ladrón); medidas y colores del catálogo (rules.view.privacy).
   184→      const pv = v.privacy;
   185→      view.door = box(
   186→        `${b.id}_puerta_priv`,
   187→        pv.doorW,
   188→        pv.doorH,
   189→        0.06,
   190→        cx,
   191→        bottom + v.slab + v.wall + pv.doorH / 2,
   192→        frontZ - v.wall / 2 - 0.035,
   193→        mat(`${b.id}_puerta_priv_mat`),
   194→      );
   195→      (view.door.material as StandardMaterial).diffuseColor = Color3.FromHexString(pv.doorColor);
   196→      view.handle = box(
   197→        `${b.id}_picaporte`,
   198→        0.07,
   199→        0.07,
   200→        0.05,
   201→        cx + pv.doorW / 2 - 0.14,
   202→        bottom + v.slab + v.wall + pv.doorH * 0.55,
   203→        frontZ - v.wall / 2 - 0.07,
   204→        mat(`${b.id}_picaporte_mat`),
   205→      );
   206→      (view.handle.material as StandardMaterial).diffuseColor = Color3.FromHexString(pv.handleColor);
   207→      view.door.setEnabled(false);
   208→      view.handle.setEnabled(false);
   209→
   210→      // ── Cartel numerado (v1.6): rótulo sobre la puerta con el identificador de la vivienda
   211→      // (blk_p3_c → "3C", blockLabel). Textura dinámica con colores/fuente del catálogo
   212→      // (rules.view.plaque); se añade a greybox porque la vivienda usa modelo OVERLAY (el
   213→      // greybox no se oculta) y no entra en mats: es señalización, no tinte ni glow.
   214→      const pl = v.plaque;
   215→      const plate = MeshBuilder.CreatePlane(`${b.id}_cartel`, { width: pl.w, height: pl.h, sideOrientation: B().Mesh.DOUBLESIDE }, scene);
   216→      const tex = new (B().DynamicTexture)(`${b.id}_cartel_tex`, { width: 256, height: 96 }, scene, true);
   217→      const ctx = tex.getContext() as unknown as CanvasRenderingContext2D;
   218→      ctx.fillStyle = pl.bg;
   219→      ctx.fillRect(0, 0, 256, 96);
   220→      ctx.font = `bold ${pl.fontPx}px Arial, sans-serif`;
   221→      ctx.fillStyle = pl.text;
   222→      ctx.textAlign = 'center';
   223→      ctx.textBaseline = 'middle';
   224→      ctx.fillText(blockLabel(b.id), 128, 52);
   225→      // update(true): el flag es invertY. Con false el canvas se sube SIN voltear y el texto
   226→      // sale CABEZA ABAJO (la "A" se ve como "V"; B/C/D pasan por simétricas y esquivan el ojo).
   227→      tex.update(true);
   228→      const pmat = new StandardMaterial(`${b.id}_cartel_mat`, scene);
   229→      pmat.diffuseTexture = tex;
   230→      pmat.emissiveColor = Color3.FromHexString(pl.bg).scale(0.55); // legible en fachada norte
   231→      pmat.specularColor = new Color3(0, 0, 0);
   232→      plate.material = pmat;
   233→      plate.position.set(cx, bottom + pl.y, frontZ + pl.z);
   234→      view.greybox.push(plate);
   235→
   236→      // ── Interior 2D (Fase C): tabiques con huecos, suelo por estancia y tapas "en obras". ──
   237→      // TODO sale del layout del catálogo (rects + doorways); sin rects no hay interior 2D.
   238→      const layoutId = content.blocks[b.defId]?.container?.layout;
   239→      const layout = layoutId ? content.layouts[layoutId] : undefined;
   240→      // Apilamiento 3D (v1.19): las salas TRASERAS no se pintan en el bloque principal — viven
   241→      // en su propio bloque trasero (más abajo, se activan al entrar).
   242→      const geoRooms = (layout?.rooms ?? []).filter((r) => r.rect !== undefined && r.rect!.z + r.rect!.d <= v.depth + 0.01);
   243→      if (geoRooms.length > 0 && layout) {
   244→        const interior: InteriorView = { floors: [], covers: [] };
   245→        const x0 = b.cell.col * v.cellW;
   246→        const wallH = v.cellH - 2 * v.slab - 2 * v.wall;
   247→        const partMat = mat(`${b.id}_tabique_mat`);
   248→        partMat.diffuseColor = Color3.FromHexString('#d8d2c4');
   249→        // Suelo tintado por estancia (plancha fina sobre el suelo, alpha suave).
   250→        for (const lr of geoRooms) {
   251→          const r = lr.rect!;
   252→          const fmat = mat(`${b.id}_${lr.key}_suelo_mat`);
   253→          fmat.diffuseColor = Color3.FromHexString(content.rooms[lr.defId]?.view?.color ?? '#cccccc');
   254→          fmat.alpha = 0.32;
   255→          const fm = attach(b.id, box(`${b.id}_${lr.key}_suelo`, r.w, 0.02, r.d, x0 + r.x + r.w / 2, bottom + v.slab + v.wall + 0.02, frontZ + r.z + r.d / 2, fmat));
   256→          interior.floors.push({ roomKey: lr.key, defId: lr.defId, mesh: fm, mat: fmat });
   257→        }
   258→        // Tabiques: para cada par de estancias que comparten borde, pared entre ambas.
   259→        // Los doorways del catálogo recortan el hueco (y añaden tapa visible si alguna de las
   260→        // dos estancias está cerrada). Sin hueco declarado → tabique sólido (tampoco se puede
   261→        // pasar: walkable() exige doorway). Tabiques según X a media altura para no tapar la vista.
   262→        const eps = 0.01;
   263→        const seg = (a: number, b: number, along: 'x' | 'z', at: number, height: number, yC: number, name: string) => {
   264→          const len = b - a;
   265→          if (len <= 0.01) return;
   266→          view.greybox.push(
   267→            along === 'z'
   268→              ? box(name, v.wall, height, len, at, yC, (a + b) / 2, partMat)
   269→              : box(name, len, height, v.wall, (a + b) / 2, yC, at, partMat),
   270→          );
   271→        };
   272→        geoRooms.forEach((lr, i) => {
   273→          geoRooms.forEach((other, j) => {
   274→            if (j <= i) return;
   275→            const ra = lr.rect!;
   276→            const rb = other.rect!;
   277→            const door = (layout.doorways ?? []).find(
   278→              (dw) => (dw.a === lr.key && dw.b === other.key) || (dw.a === other.key && dw.b === lr.key),
   279→            );
   280→            // Borde vertical compartido → tabique que corre según Z (se cruza andando en X).
   281→            const touchX = Math.abs(ra.x + ra.w - rb.x) < eps || Math.abs(rb.x + rb.w - ra.x) < eps;
   282→            const overlapZ = Math.min(ra.z + ra.d, rb.z + rb.d) - Math.max(ra.z, rb.z);
   283→            if (touchX && overlapZ > eps) {
   284→              const wx = x0 + Math.max(ra.x, rb.x);
   285→              const z0 = frontZ + Math.max(ra.z, rb.z);
   286→              const z1 = frontZ + Math.min(ra.z + ra.d, rb.z + rb.d);
   287→              const yC = bottom + v.slab + v.wall + wallH / 2;
   288→              if (door && door.axis === 'z') {
   289→                const g0 = Math.max(z0, frontZ + door.at - door.width / 2);
   290→                const g1 = Math.min(z1, frontZ + door.at + door.width / 2);
   291→                seg(z0, g0, 'z', wx, wallH, yC, `${b.id}_tab_${i}_${j}_i`);
   292→                seg(g1, z1, 'z', wx, wallH, yC, `${b.id}_tab_${i}_${j}_d`);
   293→                const cover = attach(b.id, box(`${b.id}_tapa_${i}_${j}`, v.wall * 0.8, wallH, g1 - g0, wx, yC, frontZ + door.at, mat(`${b.id}_tapa_${i}_${j}_mat`)));
   294→                (cover.material as StandardMaterial).diffuseColor = Color3.FromHexString(v.lockedColor);
   295→                cover.setEnabled(false);
   296→                interior.covers.push({ mesh: cover, rooms: [lr.key, other.key] });
   297→              } else {
   298→                seg(z0, z1, 'z', wx, wallH, yC, `${b.id}_tab_${i}_${j}`);
   299→              }
   300→            }
   301→            // Borde horizontal compartido → tabique que corre según X (se cruza andando en Z).
   302→            const touchZ = Math.abs(ra.z + ra.d - rb.z) < eps || Math.abs(rb.z + rb.d - ra.z) < eps;
   303→            const overlapX = Math.min(ra.x + ra.w, rb.x + rb.w) - Math.max(ra.x, rb.x);
   304→            if (touchZ && overlapX > eps) {
   305→              const wz = frontZ + Math.max(ra.z, rb.z);
   306→              const xa = x0 + Math.max(ra.x, rb.x);
   307→              const xb = x0 + Math.min(ra.x + ra.w, rb.x + rb.w);
   308→              const halfH = wallH * 0.45;
   309→              const yHalf = bottom + v.slab + v.wall + halfH / 2;
   310→              if (door && door.axis === 'x') {
   311→                const g0 = Math.max(xa, x0 + door.at - door.width / 2);
   312→                const g1 = Math.min(xb, x0 + door.at + door.width / 2);
   313→                seg(xa, g0, 'x', wz, halfH, yHalf, `${b.id}_tab_${i}_${j}_i`);
   314→                seg(g1, xb, 'x', wz, halfH, yHalf, `${b.id}_tab_${i}_${j}_d`);
   315→                const cover = attach(b.id, box(`${b.id}_tapa_${i}_${j}`, g1 - g0, halfH, v.wall * 0.8, x0 + door.at, yHalf, wz, mat(`${b.id}_tapa_${i}_${j}_mat`)));
   316→                (cover.material as StandardMaterial).diffuseColor = Color3.FromHexString(v.lockedColor);
   317→                cover.setEnabled(false);
   318→                interior.covers.push({ mesh: cover, rooms: [lr.key, other.key] });
   319→              } else {
   320→                seg(xa, xb, 'x', wz, halfH, yHalf, `${b.id}_tab_${i}_${j}`);
   321→              }
   322→            }
   323→          });
   324→        });
   325→        interiors.set(b.id, interior);
   326→      }
   327→    } else if (b.defId === 'trasteros_planta' || b.defId === 'bajocubierta' || b.defId === v.trasteros.blockDefId) {
   328→      // Nave de trasteros / sala de instalaciones (v1.16): forjado + trasera. El arte glTF
   329→      // añade pilares (trasteros) o maquinaria (bajocubierta); las estanterías, trastos y
   330→      // carteles numerados los levanta SIEMPRE el runtime desde rules.view.trasteros.
   331→      // El bloque por COLUMNA (v1.18) comparte estructura: sus compartimentos los levanta
   332→      // el runtime más abajo (pasillo central + celdas cerradas impares/pares).
   333→      view.greybox.push(
   334→        box(`${b.id}_forjado`, w, v.slab * 2, d, cx, bottom + v.slab, frontZ + d / 2, m),
   335→        box(`${b.id}_trasera`, w, h, v.wall, cx, bottom + v.cellH / 2, frontZ + d - v.wall / 2, m),
   336→      );
   337→    } else if (b.defId === 'tejado') {
   338→      // Tejadillo (v1.16): fallback = canto de cerramiento (el glTF trae faldones, cumbrera,
   339→      // chimenea y antena).
   340→      view.greybox.push(box(`${b.id}_canto`, w, v.slab * 2, d, cx, bottom + v.slab, frontZ + d / 2, m));
   341→    } else {
   342→      view.greybox.push(box(b.id, w, h, d, cx, bottom + v.cellH / 2, cz, m));
   343→    }
   344→    // v1.19: toda malla estructural del bloque cuelga de su raíz (swap de bloques).
   345→    for (const mesh of view.greybox) mesh.parent = root;
   346→    if (view.front) view.front.parent = root;
   347→    if (view.lock) view.lock.parent = root;
   348→    if (view.door) view.door.parent = root;
   349→    if (view.handle) view.handle.parent = root;
   350→    views.set(b.id, view);
   351→  }
   352→
   353→  // ── Forjado interior CONTINUO del cubo de escalera (v1.18): a nivel de entrada de cada
   354→  //  planta, una losa que cubre la huella completa del cubo (spanW celdas). La authoría glTF
   355→  //  solo trae el forjado de la puerta; sin esta losa se camina (y se ve el vacío) entre la
   356→  //  puerta y los tramos. Malla propia que SOBREVIVE a la carga del modelo (no está en el
   357→  //  greybox, que se oculta) — 5 mm por debajo del forjado del arte: cero z-fighting.
   358→  const stCfg = v.stairs.floorSlab;
   359→  if (stCfg.enabled) {
   360→    const stMat = mat('escalera_forjado_mat');
   361→    stMat.diffuseColor = Color3.FromHexString(stCfg.color);
   362→    for (const b of Object.values(world.blocks)) {
