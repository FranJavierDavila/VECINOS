    90→  async init(canvas: HTMLCanvasElement): Promise<void> {
    91→    if (this.engine || this.initializing) return;
    92→    this.initializing = true;
    93→    this.canvas = canvas;
    94→
    95→    try {
    96→      // 1) Contenido externo + validación (los errores de referencias se avisan, no abortan).
    97→      const { content, issues } = await loadContent();
    98→      this._content = content;
    99→      this.cutscenePlayer = new CutscenePlayer(content);
   100→      preloadSounds();
   101→      useGameStore.setState({ touchHint: content.rules.touch.hint });
   102→      if (issues.length > 0) {
   103→        useGameStore.getState().pushToast(`Contenido inválido: ${issues.length} errores de validación`, 'bad');
   104→      }
   105→
   106→      // 2) Motor de render (WebGPU con fallback WebGL2). La calidad la elige el jugador en Opciones.
   107→      const { engine, backend } = await createEngine(canvas);
   108→      this.engine = engine;
   109→      engine.setHardwareScalingLevel(QUALITY_SCALE[loadOptions().quality]);
   110→      window.addEventListener('resize', this.onResize);
   111→      await this.restart();
   112→      useGameStore.getState().markReady(backend);
   113→      startGamepadPolling();
   114→      // Logros (v1.17): si el jugador cambia el idioma de la interfaz, los textos ya
   115→      // resueltos de los logros se re-pintan con el catálogo (los desbloqueos se quedan).
   116→      useGameStore.subscribe((s, prev) => {
   117→        if (s.lang !== prev.lang) this.rebuildAchievementTexts();
   118→      });
   119→      engine.runRenderLoop(this.frame);
   120→      // Gancho de depuración/pruebas E2E (no lo usa la UI).
   121→      (window as unknown as { __vecinos?: unknown }).__vecinos = { store: useGameStore, controller: this };
   122→    } catch (err) {
   123→      console.error('No se pudo arrancar el juego (contenido o motor de render)', err);
   124→      useGameStore.getState().setEngineError(true);
   125→    }
   126→  }

   145→  onKeyDown(code: string): void {
   146→    // Cutscene en curso: E/Espacio/Enter/Escape saltan el espectáculo; nada más pasa.
   147→    if (this.cutscenePlayer?.active) {
   148→      if (code === 'KeyE' || code === 'Space' || code === 'Enter' || code === 'Escape') this.cutsceneSkip();
   149→      return;
   150→    }
   151→    if (this.keys.has(code)) return;
   152→    this.keys.add(code);
   153→    const store = useGameStore.getState();
   154→    if (selectModalOpen(store)) return; // los modales gestionan sus propias teclas
   155→    if (code === 'KeyE') {
   156→      this.interact();
   157→      return;
   158→    }
   159→    if (code === 'KeyJ') {
   160→      // Presidencia (v1.18): J junto al tablón convoca una junta con la agenda del catálogo.
   161→      if (this.logic?.interactTarget()?.kind === 'noticeboard') this.convocaJunta();
   162→      return;
   163→    }
   164→    const cam = CAM_BY_KEY[code];
   165→    if (cam) this.setCamMode(cam);
   166→  }

   244→  /** Interacción (tecla E o botón virtual ✋): usar el objeto cercano, entrar o salir. */
   245→  interact(): void {
   246→    // El ✋ táctil también salta la cutscene (es la única acción habilitada mientras dura).
   247→    if (this.cutscenePlayer?.active) {
   248→      this.cutsceneSkip();
   249→      return;
   250→    }
   251→    if (!this.logic) return;
   252→    // Ascensor (v1.9): si el prompt anuncia ascensor (sin objetivo de E delante), la E abre el
   253→    // panel de destino. Misma condición que promptText → lo que ves es lo que la E ejecuta.
   254→    if (!this.logic.interactTarget()) {
   255→      const panel = this.logic.elevatorPanelTarget();
   256→      if (panel) {
   257→        useGameStore.getState().setElevatorPanel(panel);
   258→        return;
   259→      }
   260→    }
   261→    const r = this.logic.interact();
   262→    if (r.type === 'rearEnter') {
   263→      // Apilamiento 3D (v1.19): el swap lo ejecuta building.update por fases; aquí solo
   264→      // encuadra la cámara dentro del bloque trasero.
   265→      this.setCamMode('estancia');
   266→      this.pump();
   267→      return;
   268→    }
   269→    if (r.type === 'rearExit') {
   270→      this.setCamMode('planta');
   271→      this.pump();
   272→      return;
   273→    }
   274→    if (r.type === 'entered') this.setCamMode('estancia');
   275→    else if (r.type === 'exited') this.setCamMode('planta');
   276→    else if (r.type === 'used') this.presentInteraction(r.kind);
   277→    this.pump();
   278→  }

   306→  private presentInteraction(kind: 'laptop' | 'bed' | 'mailbox' | 'noticeboard' | 'resident'): void {
   307→    if (!this.world) return;
   308→    const ic = this.content.rules.interact;
   309→    if (kind === 'mailbox') {
   310→      const lines = ic.mailbox.lines;
   311→      useGameStore.getState().pushToast(lines[this.world.tick % lines.length]!, 'info');
   312→    } else if (kind === 'noticeboard') {
   313→      const elevatorBlock = Object.values(this.world.blocks).find((b) => b.params.elevator_status !== undefined);
   314→      const status = String(elevatorBlock?.params.elevator_status ?? 'ok');
   315→      let text = ic.noticeboard.template
   316→        .replace('{day}', String(this.world.clock.day))
   317→        .replace('{funds}', String(this.world.community.funds))
   318→        .replace('{elevator}', this.content.rules.labels.elevator[status] ?? status)
   319→        .replace('{weather}', this.content.rules.labels.weather[this.world.weather.state] ?? this.world.weather.state);
   320→      // Presidencia (v1.18): con el mandato activo, el tablón también convoca.
   321→      const cfg = this.content.rules.sim.campaign;
   322→      if (cfg && useGameStore.getState().campaign?.active) text += `\n🗳️ [J] Convocar junta de vecinos (${cfg.presidentLabel.toLowerCase()})`;
   323→      useGameStore.getState().pushToast(text, 'info');
   324→    }
   325→  }

   327→  /**
   328→   * Junta extraordinaria convocada por la presidencia desde el tablón (v1.18): la agenda
   329→   * sale del catálogo (rules.sim.campaign.presetAgenda) y la votación la resuelve el sim
   330→   * (mismo flujo que las juntas de misión). Cooldown del catálogo; solo con mandato activo.
   331→   */
   332→  private convocaJunta(): void {
   333→    const world = this.world;
   334→    const cfg = this.content.rules.sim.campaign;
   335→    if (!world || !cfg || world.junta) return;
   336→    if (!useGameStore.getState().campaign?.active) {
   337→      useGameStore.getState().pushToast('Solo la presidencia convoca juntas. De momento, eres un vecino más (con suerte).', 'info');
   338→      return;
   339→    }
   340→    if (world.tick - this.lastConvocaTick < cfg.convocationCooldownMinutes) {
   341→      const left = Math.ceil(cfg.convocationCooldownMinutes - (world.tick - this.lastConvocaTick));
   342→      useGameStore.getState().pushToast(`Juana dice que la última junta fue hace poco. Próxima disponible en ${left} min. La democracia necesita descansos.`, 'info');
   343→      return;
   344→    }
   345→    this.lastConvocaTick = world.tick;
   346→    world.junta = { agenda: cfg.presetAgenda.map((i) => ({ ...i })), votes: {}, playerVoted: false };
   347→    useGameStore.getState().setJunta(cfg.presetAgenda.map((i) => ({ id: i.id, title: i.title, description: i.description, cost: i.cost })));
   348→    useGameStore.getState().pushLog('🗳️ Junta extraordinaria convocada desde el tablón. Los vecinos, en camino.', 'info');
   349→  }

   422→  private connectNet(): void {
   423→    this.net = new NetSession({
   424→      onState: (s) => this.applyRemoteState(s),
   425→      onParticipants: (p) => useGameStore.getState().setMpParticipants(p),
   426→      onJoined: (j) => {
   427→        this.myResidentId = j.yourResidentId;
   428→        this.isHost = j.isHost;
   429→        useGameStore.getState().setMpSession({ code: j.code, isHost: j.isHost, myResidentId: j.yourResidentId });
   430→        useGameStore.getState().setMpParticipants(j.participants);
   431→      },
   432→      onError: (m) => {
   433→        useGameStore.getState().pushToast(m, 'bad');
   434→      },
   435→      onStatus: (st) => {
   436→        if (!this.net) return;
   437→        if (st === 'connected' && useGameStore.getState().mpStatus === 'connecting') return; // el ack decide
   438→        if (st === 'disconnected' && useGameStore.getState().mpStatus === 'connected') {
   439→          useGameStore.getState().setMpStatus('error');
   440→          useGameStore.getState().setMpError('Conexión perdida con el servidor de la comunidad.');
   441→        }
   442→      },
   443→    });
   444→  }

   446→  private onHostJoinResult(res: { ok: boolean; code?: string; isHost?: boolean; yourResidentId?: string; error?: string }): void {
   447→    const store = useGameStore.getState();
   448→    if (!res.ok) {
   449→      store.setMpStatus('error');
   450→      store.setMpError(res.error === 'timeout' ? 'El servidor no responde. ¿Se habrá ido a comer?' : res.error ?? 'No se pudo abrir la sesión.');
   451→      return;
   452→    }
   453→    store.setMpStatus('connected');
   454→    // onJoined ya habrá fijado myResidentId/isHost/código con session:joined.
   455→  }

   457→  /** Adopta el estado del servidor (autoridad). Primera vez: reinicia la escena con ese mundo. */
   458→  private applyRemoteState(s: SessionStatePayload): void {
   459→    if (s.seq <= this.appliedSeq) return;
   460→    this.appliedSeq = s.seq;
   461→    if (!this.world || this.world.scenarioId !== s.world.scenarioId) {
   462→      void this.restart(s.world).then(() => {
   463→        if (!this.world) return;
   464→        this.world.events = s.events.slice();
   465→        this.handleEvents();
   466→        this.reconcile();
   467→        this.syncHud();
   468→      });
   469→      return;
   470→    }
   471→    // Aplica en sitio conservando la identidad del objeto (PlayerLogic/building/actors referencian el mismo World).
   472→    Object.assign(this.world, s.world);
   473→    this.world.events = s.events.slice();
   474→    this.handleEvents();
   475→    this.paintBuilding();
   476→    this.reconcile();
   477→    this.syncHud();
   478→  }

   642→  /** (Re)crea el mundo y la escena. Con `state` restaura una partida guardada. */
   643→  private async restart(state?: World): Promise<void> {
   644→    if (!this.engine || !this.canvas || !this._content) return;
   645→    // Ninguna cutscene sobrevive a un cambio de mundo: cola fuera y teatro desmontado.
   646→    this.cutsceneQueue = [];
   647→    this.cutscenePlayer?.abort();
   648→    const { Color4, DirectionalLight, HemisphericLight, Scene, UniversalCamera, Vector3 } = B();
   649→    const view = this.content.rules.view;
   650→    useGameStore.getState().resetUi();
   651→
   652→    this.world = state ?? createWorld(this.content, this.activeScenarioId());
   653→    this.scenarioOverrides = this.content.scenarios[this.world.scenarioId ?? '']?.minigameOverrides ?? {};
   654→    // Guardados antiguos sin `entry`: la entrada es la zona común de planta baja.
   655→    if (!this.world.entry) {
   656→      this.world.entry = Object.values(this.world.blocks).find((b) => b.cell.floor === 0 && b.kind !== 'dwelling')?.id ?? Object.keys(this.world.blocks)[0]!;
   657→    }
   658→    // Migración de estancias de guardados v0.1: params y ranuras de muebles según catálogo.
   659→    for (const b of Object.values(this.world.blocks)) migrateRooms(this.content, b.rooms);
   660→    this.logic = new PlayerLogic(this.world, this.content, {
   661→      actorId: this.myResidentId ?? undefined,
   662→      send: (cmd) => this.sendCommand(cmd),
   663→    });
   664→
   665→    this.scene?.dispose();
   666→    this.scene = new Scene(this.engine);
   667→    this.scene.clearColor = new Color4(view.sky.r, view.sky.g, view.sky.b, 1);
   668→    new HemisphericLight('ambient', new Vector3(...view.ambient.dir), this.scene).intensity = view.ambient.intensity;
   669→    new DirectionalLight('sun', new Vector3(...view.sun.dir), this.scene).intensity = view.sun.intensity;
   670→
   671→    this.building = createBuilding(this.scene, this.world, this.content);
   672→    this.elevators = createElevators(this.scene, this.world, this.content);
   673→    this.actors = createActors(this.scene, this.world, this.content, this.elevators);
   674→    // Ecosistema del edificio (v1.14): cerebro data-driven + teatro de extras. Se recrea con
   675→    // cada mundo (referencia al World vivo) y sigue la posición del jugador para las luces.
   676→    this.ecosystem?.dispose();
   677→    this.ecosystem = createEcosystem(this.scene, this.world, this.content, () => this.logic!.position());
   678→    // Sonda de QA (SOLO dev): escena y controlador accesibles para el E2E automatizado.
   679→    // En producción no se expone nada (la partida queda encapsulada igual que siempre).
   680→    if (process.env.NODE_ENV !== 'production') {
   681→      (window as unknown as { __vecinosQA?: { scene: unknown; controller: unknown; elevators: unknown; ecosystem: unknown } }).__vecinosQA = {
   682→        scene: this.scene,
   683→        controller: this,
   684→        elevators: this.elevators,
   685→        ecosystem: this.ecosystem,
   686→      };
   687→    }
   688→    // Bocadillos de expresión (v1.10): capa DOM sobre el canvas, recreada con el mundo
   689→    // (referencia al World vivo). Sin emotes.json en el catálogo, director nulo sin capa.
   690→    this.bubbles?.dispose();
   691→    const stage = this.canvas?.parentElement;
   692→    this.bubbles = stage && this.actors ? createBubbles(stage, this.scene, this.engine, this.world, this.content, this.actors.director) : null;
   693→    const blocks = Object.values(this.world.blocks);
   694→    // El cubo de escalera (v1.4) sobresale spanW−1 celdas por la derecha del edificio:
   695→    // el encuadre lo incluye para que la torre no se corte en cámara.
   696→    const hasStairs = blocks.some((b) => this.content.blocks[b.defId]?.features?.stairs === true);
   697→    const cols = Math.max(...blocks.map((b) => b.cell.col + b.size.w)) + (hasStairs ? this.content.rules.view.stairs.spanW - 1 : 0);
   698→    this.rig = new CameraRig(
   699→      new UniversalCamera('cam', new Vector3(0, 0, -30), this.scene),
   700→      this.engine,
   701→      cols,
   702→      Math.max(...blocks.map((b) => b.cell.floor)) + 1,
   703→      view.cellW,
   704→      view.cellH,
   705→    );
   706→    // Re-sincronización de vista (v0.6-b): la cámara sigue la posición del mundo (restaurado o
   707→    // nuevo), no la preferencia anterior. Dentro de una vivienda → estancia; en pasillo, la
   708→    // preferencia salvo que fuera 'estancia' (ahí estás en el pasillo, no en una estancia).
   709→    const pos = this.logic.position();
   710→    const store = useGameStore.getState();
   711→    const wanted: CamMode = pos.mode === 'inside' ? 'estancia' : store.camMode === 'estancia' ? 'planta' : store.camMode;
   712→    this.rig.mode = wanted;
   713→    store.setCamMode(wanted);
   714→
   715→    this.clockAcc = 0;
   716→    this.keys.clear();
   717→    this.phoneLines = [];
   718→    this.phoneMission = '';
   719→    this.phoneQueue = [];
   720→    this.lastHudSync = 0;
   721→
   722→    // Re-llamada (Fase F): si la partida montada tenía pasos de teléfono pendientes, vuelven a
   723→    // sonar. La cola de openPhone ordena el resto sin que ninguna llamada se pierda.
   724→    if (this.world) {
   725→      for (const m of this.world.missions) {
   726→        if (m.state !== 'active') continue;
   727→        const step = this.content.missions[m.id]?.steps[m.step];
   728→        if (step?.type === 'phone_call') this.openPhone(m.id, step.dialog);
   729→      }
   730→    }
   731→
   732→    this.pump();
   733→    // Logros (v1.17): primera foto del catálogo para el panel (y desbloqueos de la partida restaurada).
   734→    this.syncAchievements();
   735→    // Campaña (v1.18): primer estado del mandato (sobrevive a Cargar vía localStorage).
   736→    syncCampaign(this.world!, this.content);
   737→  }

   743→  // ───────────────────────── Bucle de render ─────────────────────────
   744→
   745→  private frame = (): void => {
   746→    if (!this.engine || !this.scene || !this.world || !this.logic || !this.building || !this.actors || !this.rig || !this.elevators) return;
   747→    const dt = Math.min(0.1, this.engine.getDeltaTime() / 1000);
   748→    const store = useGameStore.getState();
   749→    const frozen = selectModalOpen(store) || store.paused;
   750→
   751→    let ticked = false;
   752→    if (!frozen) {
   753→      this.logic.update(dt, this.readInput());
   754→      // Ascensor (v1.5): el director mueve cabinas y puertas; el jugador llama con W/S en el
   755→      // núcleo y su escalada EXTERNA la gobierna la cabina (espera → embarco → viaje → salida).
   756→      this.elevators.update(dt);
   757→      const climb = this.logic.climbInfo();
   758→      if (climb) {
   759→        if (!this.logic.externalRequested) {
   760→          const zone = this.logic.zone();
   761→          this.elevators.requestRide(this.actorId(), zone.cell.col, climb.from, climb.to);
   762→          this.logic.externalRequested = true;
   763→        }
   764→        const st = this.elevators.status(this.actorId());
   765→        if (!st) {
   766→          this.logic.finishClimbAtNearest(); // fuera de servicio: aterriza en la planta más cercana (sin teletransporte)
   767→          this.logic.externalRequested = false;
   768→          this.playerBoardT = 0;
   769→        } else {
   770→          this.logic.setClimbProgress(st.phase === 'riding' ? st.progress : 0);
   771→          const inside = st.phase === 'boarding' || st.phase === 'riding';
   772→          this.logic.setClimbZ(inside ? this.content.rules.view.roomZ - this.content.rules.view.commonDepth / 2 + this.content.rules.view.elevator.rideZ : null);
   773→          // El jugador "da un paso" dentro de la cabina un instante tras abrirse la puerta
   774→          // (el zOverride ya lo acerca); avisar al director para que cierre y viaje.
   775→          if (st.phase === 'boarding') {
   776→            this.playerBoardT += dt;
   777→            if (this.playerBoardT > 1.0) this.elevators.notifyBoarded(this.actorId());
   778→          } else {
   779→            this.playerBoardT = 0;
   780→          }
   781→          if (st.phase === 'exiting') {
   782→            this.logic.finishClimb();
   783→            this.logic.externalRequested = false;
   784→          }
   785→        }
   786→      } else if (this.logic.externalRequested) {
   787→        this.logic.externalRequested = false;
   788→      }
   789→      // En remoto el servidor es la autoridad del reloj: nada de ticks locales.
   790→      if (!this.net) {
   791→        this.clockAcc += dt * store.speed;
   792→        let guard = 0;
   793→        while (this.clockAcc >= 1 && guard < 60) {
   794→          tick(this.world, this.content);
   795→          this.clockAcc -= 1;
   796→          ticked = true;
   797→          guard += 1;
   798→        }
   799→      }
   800→    }
   801→
   802→    if (this.world.events.length > 0) this.handleEvents();
   803→    if (ticked) this.paintBuilding();
   804→
   805→    const now = performance.now();
   806→    if (now - this.lastHudSync > 200) {
   807→      this.lastHudSync = now;
   808→      useGameStore.getState().setFps(Math.round(this.engine.getFps()));
   809→      this.syncHud();
   810→    }
   811→
   812→    // Logros (v1.17): evaluación perezosa a cadencia (~cada 2 s de tiempo real).
   813→    // El evaluador es puro y solo lee el mundo, así que da igual si hay modal o pausa.
   814→    this.achAcc += dt;
   815→    if (this.achAcc >= 2) {
   816→      this.achAcc = 0;
   817→      this.syncAchievements();
   818→      // Campaña (v1.18): el día de mandato avanza con el reloj — misma cadencia.
   819→      syncCampaign(this.world!, this.content);
   820→    }
   821→
   822→    const pos = this.logic.position();
   823→    this.actors.update(dt, pos, this.actorId());
   824→    // Cutscene (Fase D): sus beats mueven cámara/actores DESPUÉS de los eventos y ANTES del render.
   825→    this.cutscenePlayer?.update(dt);
   826→    // Bocadillos (v1.10): después de actores y cámara — proyectan la posición FINAL del frame.
   827→    this.bubbles?.update(dt, this.actorId());
   828→    // Ecosistema (v1.14): cerebro a cadencia + extras + luces + deterioro. En multijugador
   829→    // queda en pausa (el servidor es la autoridad; mismo criterio que partyDemo).
   830→    if (!this.net) this.ecosystem?.update(dt);
   831→    // Si la cutscene acaba y queda cola, encadena la siguiente aquí: no dependemos de que el sim
   832→    // genere otro evento para descolgar la siguiente (durante el teatro el sim está congelado).
   833→    if (this.cutsceneQueue.length > 0 && !this.cutscenePlayer?.active) this.pumpCutscene();
   834→    // LOD por distancia: solo con cámaras cercanas (planta/estancia); el plano general (edificio) muestra todo.
   835→    // Privacidad (v1.17): el tercer argumento es el bloque donde está el jugador — las fachadas
   836→    // de las viviendas reaccionan al instante cuando entra o sale.
   837→    // v1.19: el cuarto argumento alimenta carteles guía de trasteros y el apilamiento 3D.
   838→    this.syncRear(pos);
   839→    this.building.update(
   840→      dt,
   841→      useGameStore.getState().camMode === 'edificio' ? undefined : { x: pos.x, floor: pos.floor },
   842→      pos.insideBlock,
   843→      { x: pos.x, z: pos.z, insideBlock: pos.insideBlock },
   844→    );
   845→    this.rig.update(dt, pos);
   846→    this.scene.render();
   847→  };
   848→
   849→  /**
   850→   * Apilamiento 3D (v1.19): cada frame decide la fase del bloque trasero activo (y apaga el
   851→   * resto). Trastero propio: E manda ('inside' mientras rearCell). Cuarto trasero de la
   852→   * vivienda: asoma al acercarte a la puerta del fondo ('peek') y se activa al cruzar ('inside').
   853→   */
   854→  private syncRear(pos: ReturnType<PlayerLogic['position']>): void {
   855→    if (!this.building || !this.world || !this.logic) return;
   856→    let activeId: string | null = null;
   857→    let phase: 'idle' | 'peek' | 'inside' = 'idle';
   858→    if (this.logic.rearCell) {
   859→      activeId = this.logic.rearCell.block;
   860→      phase = 'inside';
   861→    } else if (pos.mode === 'inside' && pos.insideBlock) {
   862→      const b = this.world.blocks[pos.insideBlock];
   863→      if (b && b.kind === 'dwelling') {
   864→        const geo = layoutGeometryOf(this.content, b);
   865→        const depth = this.content.rules.view.depth;
   866→        const rear = geo?.rooms.find((r) => r.rect.z + r.rect.d > depth + 0.01);
   867→        if (geo && rear) {
   868→          const localX = pos.x - b.cell.col * this.content.rules.view.cellW;
   869→          const room = roomAt(geo, localX, pos.z);
   870→          const rearDoor = geo.doorways.find((d) => d.axis === 'x' && d.b === rear.key);
   871→          const near =
   872→            rearDoor &&
   873→            pos.z > depth - this.content.rules.view.rearBlock.peekDistance &&
   874→            Math.abs(localX - rearDoor.at) < rearDoor.width / 2 + 0.3;
   875→          if (room.key === rear.key) {
   876→            activeId = b.id;
   877→            phase = 'inside';
   878→          } else if (near) {
   879→            activeId = b.id;
   880→            phase = 'peek';
   881→          }
   882→        }
   883→      }
   884→    }
   885→    for (const b of Object.values(this.world.blocks)) {
   886→      if (!this.building.hasRear(b.id)) continue;
   887→      this.building.setRearPhase(b.id, b.id === activeId ? phase : 'idle');
   888→    }
   889→  }

  1070→  /** Si el evento encaja con el trigger de alguna cutscene del catálogo, se encola (máx. 3). */
  1071→  private maybeCutscene(e: GameEvent): void {
  1072→    const defs = Object.values(this.content.cutscenes);
  1073→    if (defs.length === 0) return;
  1074→    const d = e.data as Record<string, unknown>;
  1075→    const hit = defs.find((c) => {
  1076→      const t = c.trigger;
  1077→      if (t.on !== e.type) return false;
  1078→      if (t.mission !== undefined && t.mission !== String(d.mission ?? '')) return false;
  1079→      if (t.step !== undefined && t.step !== String(d.step ?? '')) return false;
  1080→      return true;
  1081→    });
  1082→    if (!hit) return;
  1083→    if (this.cutsceneQueue.length >= 3) return; // nunca más de 3 esperando: el teatro no atravaca la partida
  1084→    this.cutsceneQueue.push({
  1085→      def: hit,
  1086→      ctx: {
  1087→        mission: typeof d.mission === 'string' ? d.mission : null,
  1088→        resident: typeof d.resident === 'string' ? d.resident : null,
  1089→        block: typeof d.block === 'string' ? d.block : null,
  1090→      },
  1091→    });
  1092→  }

  1164→  private promptText(): string | null {
  1165→    const logic = this.logic;
  1166→    const world = this.world;
  1167→    if (!logic || !world) return null;
  1168→    if (logic.position().climbing) return null;

  1170→    const t = logic.interactTarget();
  1171→    if (t) {
  1172→      const ic = this.content.rules.interact;
  1173→      switch (t.kind) {
  1174→        case 'door': {
  1175→          const door = world.blocks[t.blockId]!;

  1178→            ? ic.doorLocked.replace('{block}', this.blockName(door.id))
  1179→            : ic.doorEnter.replace('{block}', this.blockName(door.id));
  1180→        }
  1181→        case 'resident':
  1182→          return ic.greet.replace('{name}', this.nameOf(t.residentId));
  1183→        case 'mailbox':
  1184→          return ic.mailbox.prompt;
  1185→        case 'noticeboard':
  1186→          return ic.noticeboard.prompt;
  1187→        case 'laptop':
  1188→          return ic.laptop;
  1189→        case 'bed':
  1190→          return ic.bed;
  1191→        case 'storageCell':
  1192→          return ic.storageEnter;
  1193→        case 'rearExit':
  1194→          return ic.rearExit;
  1195→      }
  1196→    }
  1197→    const zone = logic.zone();
  1198→    const feat = this.content.blocks[zone.defId]!.features;
  1199→    if (feat?.stairs || feat?.elevator) {
  1200→      const ic = this.content.rules.interact;
  1201→      // Lo que dice el prompt es lo que W/S hace aquí: ascensor operativo manda; si no hay
  1202→      // ascensor utilizable pero sí escalera, escalera; núcleo averiado sin escalera = aviso.
  1203→      if (feat.elevator && world.blocks[zone.id]!.params.elevator_status === 'ok') return ic.elevator;
  1204→      if (feat.stairs) return ic.stairs;
  1205→      return ic.elevatorBroken;
  1206→    }
  1207→    return null;

  1309→   * desbloqueos en el perfil (localStorage), se anuncian en el registro y se encola el
  1310→   * popup estilo Steam (uno en pantalla; el resto queda `newly` para AchievementPopup).
  1311→   * Sin catálogo (achievements.json ausente) no hay logros y no se toca nada.
  1312→   */
  1313→  private syncAchievements(): void {
  1314→    const catalog = this._content?.achievements;
  1315→    const world = this.world;
  1316→    if (!catalog || !world) return;
  1317→    const player = world.residents[this.actorId()] ?? world.residents[world.playerId];
  1318→    if (!player) return;
  1319→    const store = useGameStore.getState();
  1320→
  1321→    // Snapshot con las mismas fuentes que el HUD (día del reloj, funds, ascensor del núcleo…).
  1322→    const elevatorBlock = Object.values(world.blocks).find((b) => b.params.elevator_status !== undefined);
  1323→    let bestAffinity = 0;
  1324→    for (const rel of player.relations) bestAffinity = Math.max(bestAffinity, rel.affinity);
  1325→    const snap: AchievementSnapshot = {
  1326→      day: world.clock.day,
  1327→      tick: world.tick,
  1328→      wallet: player.wallet,
  1329→      funds: world.community.funds,
  1330→      reputation: player.reputation,
  1331→      elevatorStatus: String(elevatorBlock?.params.elevator_status ?? 'ok'),
  1332→      missionsDone: world.missions.filter((m) => m.state === 'done').map((m) => m.id),
  1333→      bestAffinity,
  1334→    };
  1335→
  1336→    const fresh = evaluateAchievements(catalog, snap, this.unlockedAchievements);
  1337→    if (fresh.length === 0 && store.achievements.length > 0) return; // nada nuevo y lista ya pintada
  1338→
  1339→    if (fresh.length > 0) {
  1340→      for (const id of fresh) this.unlockedAchievements.add(id);
  1341→      saveUnlocked([...this.unlockedAchievements]);
  1342→    }
  1343→    store.setAchievements(
  1344→      catalog.achievements.map((def) => this.achievementUi(def, this.unlockedAchievements.has(def.id), fresh.includes(def.id))),
  1345→    );
  1346→
  1347→    const lang = useGameStore.getState().lang;
  1348→    for (const id of fresh) {
  1349→      const def = catalog.achievements.find((a) => a.id === id);
  1350→      if (def) store.pushLog(`🏆 Logro desbloqueado: ${def.name[lang] ?? def.name.es}`, 'good');
  1351→    }
  1352→    if (fresh.length > 0 && useGameStore.getState().achievementPopup === null) {
  1353→      const first = catalog.achievements.find((a) => a.id === fresh[0]);
  1354→      if (first) useGameStore.getState().setAchievementPopup(this.achievementUi(first, true));
  1355→    }
  1356→  }
  1357→}

  1363→/** Crea (o reutiliza) el controlador y lo arranca sobre el canvas dado. */
  1364→export function createGameController(canvas: HTMLCanvasElement): GameController {
  1365→  if (!instance) {
  1366→    instance = new GameController();
  1367→    setController(instance); // la UI lo usa desde ya; sus acciones toleran mundo nulo
  1368→    void instance.init(canvas);
  1369→    // Gancho de depuración/E2E (solo lectura + teleport QoL de pruebas).
  1370→    const dbg = instance as unknown as { logic?: PlayerLogic; world?: World } | null;
  1371→    (globalThis as Record<string, unknown>).__vecinosDebug = () => ({
  1372→      player: dbg?.logic
  1373→        ? {
  1374→            x: +dbg.logic.x.toFixed(2),
  1375→            z: +dbg.logic.z.toFixed(2),
  1376→            floor: dbg.logic.floor,
  1377→            mode: dbg.logic.mode,
  1378→            insideBlock: dbg.logic.insideBlock,
  1379→            climbing: !!dbg.logic.position().climbing,
  1380→          }
  1381→        : null,
  1382→      clock: dbg?.world ? { ...dbg.world.clock } : null,
  1383→      camMode: useGameStore.getState().camMode,
  1384→      paused: useGameStore.getState().paused,
  1385→      simLocation: dbg?.world?.residents[dbg.world.playerId]?.location ?? null,
  1386→      missions: dbg?.world?.missions.map((m) => `${m.id}:${m.state}${m.state === 'active' ? `#${m.step}` : ''}`) ?? [],
  1387→      /** QoL de pruebas E2E: colocar al jugador ante la puerta de un bloque (recorrido legal del grafo). */
  1388→      tp: (blockId: string) => !!dbg?.logic?.teleportTo(blockId),
  1389→      /** QoL de pruebas E2E: comando del sim directo (diagnóstico de pasos). */
  1390→      cmd: (c: unknown) => {
  1391→        const w = dbg?.world;
  1392→        const ct = (dbg as unknown as { _content?: Content } | null)?._content;
  1393→        return w && ct ? JSON.stringify(dispatch(w, ct, c as Command)) : 'no-world';
  1394→      },
  1395→      /** QoL de pruebas E2E (pestaña en segundo plano: rAF suspendido): avanza n ticks del sim. */
  1396→      tickN: (n: number) => {
  1397→        const w = dbg?.world;
  1398→        const ct = (dbg as unknown as { _content?: Content } | null)?._content;
  1399→        if (!w || !ct) return 'no-world';
  1400→        for (let i = 0; i < n; i++) tick(w, ct);
  1401→        (dbg as unknown as { handleEvents?: () => void }).handleEvents?.();
  1402→        return JSON.stringify({ ...w.clock });
  1403→      },
  1404→      /** QoL de pruebas E2E: coloca la X del jugador (presentación) para apuntar interacciones fijas. */
  1405→      setX: (x: number) => {
  1406→        const lg = dbg?.logic;
  1407→        if (!lg) return 'no-logic';
  1408→        lg.x = x;
  1409→        return JSON.stringify({ x: lg.x });
  1410→      },
  1411→    });
  1412→  }
  1413→  return instance;
  1414→}
