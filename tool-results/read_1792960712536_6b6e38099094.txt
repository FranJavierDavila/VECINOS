    17→/**

    24→/**
    25→ * Lógica del personaje sin gráficos: dónde está, qué puerta tiene delante, escaleras y ascensor.
    26→ * No decide nada por su cuenta (cada cambio de bloque pasa por el sim) y NO contiene constantes:
    27→ * geometría y velocidades vienen de content.rules.view (catálogo greybox sustituible por arte final).
    28→ *
    29→ * Multi-actor: `actorId` identifica al residente controlado (por defecto world.playerId, igual que
    30→ * siempre). Con `send`, cada comando sale por esa vía (multijugador) en lugar de hacer dispatch
    31→ * local; sin `send` el comportamiento es idéntico al clásico.
    32→ */
    33→export class PlayerLogic {
    34→  x: number;
    35→  /** Profundidad en metros dentro de la vivienda (0 = plano frontal de la puerta). */
    36→  z: number;
    37→  floor: number;

    39→  insideBlock: string | null = null;
    40→  private climb: { from: number; to: number; t: number; duration: number; external?: boolean; progress?: number; zOverride?: number | null } | null = null;
    41→  /** Escalada REAL por el cubo (v1.4): el jugador recorre los peldaños del contrato
    42→   *  (rules.view.stairs vía src/game/stairs.ts) — tramos con descansillo, X/Z variables. */
    43→  private stairPath: { pts: { x: number; y: number; z: number }[]; i: number; to: number } | null = null;
    44→  private climbPos = { x: 0, y: 0, z: 0 };
    45→  private approached: string | null = null;
    46→  /** Apilamiento 3D (v1.19): celda de trastero propia activa como bloque trasero.
    47→   *  El sim no cambia (sigues "en el bloque de trasteros"); la presentación hace el swap. */
    48→  rearCell: { block: string; side: 'left' | 'right'; floor: number } | null = null;
    49→  /** Congela la locomoción del sim mientras el cuerpo "está" dentro del bloque trasero. */
    50→  rearHold = false;
    51→  private readonly width: number;
    52→  private readonly topFloor: number;
    53→  private readonly view;
    54→  /** Residente controlado por esta lógica (world.playerId en solitario). */
    55→  private readonly actorId: string;
    56→  /** Vía de salida de comandos; si falta, el comando se despacha localmente (clásico). */
    57→  private readonly send: SendCommand | null;
    58→  /**
    59→   * Anti-duplicado de move en remoto: mientras el servidor no haya confirmado el destino
    60→   * (location !== z.id), no se reenvía el mismo move en cada frame.
    61→   */
    62→  private lastMoveTarget: string | null = null;
    63→  /** Llamada de ascensor del jugador ya registrada (evita pedir la misma cada frame). */
    64→  externalRequested = false;

    66→  constructor(
    67→    private readonly world: World,
    68→    private readonly content: Content,
    69→    opts?: { actorId?: string; send?: SendCommand },
    70→  ) {
    71→    this.actorId = opts?.actorId ?? world.playerId;
    72→    this.send = opts?.send ?? null;
    73→    this.view = content.rules.view;
    74→    const blocks = Object.values(world.blocks);
    75→    // El ancho transitable lo marcan los bloques CON SUELO (v1.18): el tejado (traversable
    76→    // false, w 11) NO puede abrir el clamp — con él, el jugador caminaba PASADA la pared
    77→    // del edificio (x>24) sobre el vuelo del tejado: el "tramo que no muestra suelo".
    78→    this.width = Math.max(
    79→      ...blocks.filter((b) => this.content.blocks[b.defId]!.traversable !== false).map((b) => (b.cell.col + b.size.w) * this.view.cellW),
    80→    );
    81→    this.topFloor = Math.max(
    82→      // Solo plantas con suelo transitable (v1.16): el tejado cierra el edificio pero no
    83→      // tiene forjado — nunca es topFloor aunque sea la planta más alta de la plantilla.
    84→      ...blocks.filter((b) => this.content.blocks[b.defId]!.traversable !== false).map((b) => b.cell.floor),
    85→    );
    86→    // Si el actor aún no existe en el mundo (estado del servidor en camino), se cae al jugador clásico.
    87→    const me = world.residents[this.actorId] ?? world.residents[world.playerId]!;
    88→    const start = world.blocks[me.location ?? me.home] ?? blocks.find((b) => b.kind !== 'dwelling')!;
    89→    this.floor = start.cell.floor;
    90→    this.x = this.centerOf(start);
    91→    this.z = this.view.corridorZ; // en pasillo la Z es ABSOLUTA (v1.19: dos carriles)
    92→    // Guardado (o sesión remota) con el residente DENTRO de una vivienda: aterriza en el interior
    93→    // (sin esto, el modo 'corridor' del constructor sincronizaría un move al pasillo en el
    94→    // primer frame y el jugador "saldría de casa" solo al cargar la partida).
    95→    if (start.kind === 'dwelling') this.landInside(start);
    96→  }
    97→
    98→  private rangeOf(b: BlockInstance): [number, number] {
    99→    return [b.cell.col * this.view.cellW, (b.cell.col + b.size.w) * this.view.cellW];
   100→  }
   101→  private centerOf(b: BlockInstance): number {
   102→    const [a, z] = this.rangeOf(b);
   103→    return (a + z) / 2;
   104→  }
   105→  private within(x: number, b: BlockInstance): boolean {
   106→    const [a, z] = this.rangeOf(b);
   107→    return x >= a && x < z;
   108→  }
   109→
   110→  private get location(): string | null {
   111→    return this.world.residents[this.actorId]!.location;
   112→  }
   113→
   114→  private onFloor(f: number) {
   115→    return Object.values(this.world.blocks).filter((b) => b.cell.floor === f);
   116→  }
   117→
   118→  private corridor(f = this.floor): BlockInstance {
   119→    return this.onFloor(f).find((b) => b.layer === 'corridor')!;
   120→  }
   121→
   122→  /** ¿Carril EXTERIOR del pasillo? (v1.19): sin avisos de elementos de pared. */
   123→  onOuterLane(): boolean {
   124→    return this.mode === 'corridor' && this.z < corridorLanes(this.view).threshold;
   125→  }
   126→
   127→  /** Plano frontal de los bloques (compartido con building/storage). */
   128→  private get frontZ(): number {
   129→    return this.view.roomZ - this.view.commonDepth / 2;
   130→  }
   131→
   132→  private elevatorBlock(f: number): BlockInstance | undefined {
   133→    return this.onFloor(f).find((b) => this.content.blocks[b.defId]!.features?.elevator);
   134→  }

   136→  // ── Fase C: geometría interior 2D — compartida con la vida interior de los vecinos ──
   137→  // (src/game/interior.ts: rects, tabiques, huecos, rutas y anclas; UNA fuente de verdad
   138→  // para el jugador y los NPCs, siempre desde el catálogo layouts/rooms + rules.view).
   139→
   140→  /** Geometría 2D del layout de una vivienda; null si el layout no declara rects (catálogo antiguo). */
   141→  private geometryOf(block: BlockInstance): LayoutGeometry | null {
   142→    return layoutGeometryOf(this.content, block);
   143→  }
   144→
   145→  /** Aterriza dentro de una vivienda: plano frontal, y si cae en estancia cerrada, a la abierta más al frente. */
   146→  private landInside(block: BlockInstance): void {
   147→    this.mode = 'inside';
   148→    this.insideBlock = block.id;
   149→    this.z = 0.45;
   150→    const geo = this.geometryOf(block);
   151→    if (geo) {
   152→      const room = roomAt(geo, this.x - block.cell.col * this.view.cellW, this.z);
   153→      if (room.locked) {
   154→        const open = geo.rooms.find((r) => !r.locked);
   155→        if (open) this.x = block.cell.col * this.view.cellW + open.rect.x + open.rect.w / 2;
   156→      }
   157→    }
   158→  }
   159→
   160→  /** Bloque bajo los pies del jugador (zona común) o, si no, el pasillo.
   161→   *  v1.17: el cubo de escalera y el núcleo (con features) ganan al pasillo cuando se
   162→   *  solapan — los pasillos extendidos (v1.16) comparten columna con el pozo y el orden
   163→   *  del mapa dejaba la zona en el pasillo, dejando al jugador sin escalera en esas plantas.
   164→   *  v1.18: los bloques SIN SUELO (tejado, traversable false) jamás son zona — encima de
   165→   *  ellos no se camina y no deben robar el pozo al pasillo real. */
   166→  zone(): BlockInstance {
   167→    const common = this.onFloor(this.floor).filter(
   168→      (b) => b.layer === 'main' && b.kind !== 'dwelling' && this.content.blocks[b.defId]!.traversable !== false && this.within(this.x, b),
   169→    );
   170→    const withShaft = common.find((b) => {
   171→      const f = this.content.blocks[b.defId]!.features;
   172→      return !!f && (f.stairs === true || f.elevator === true);
   173→    });
   174→    return withShaft ?? common[0] ?? this.corridor();
   175→  }
   176→
   177→  /** Vivienda o local justo delante (la puerta que puede abrir), si la hay.
   178→   *  v1.19: desde el carril EXTERIOR del pasillo no hay puertas delante — es la vía de
   179→   *  adelantamiento: ni avisos ni approach; los elementos de pared no se activan. */
   180→  doorInFront(): BlockInstance | null {
   181→    if (this.mode !== 'corridor') return null;
   182→    if (this.onOuterLane()) return null;
   183→    return this.onFloor(this.floor).find((b) => this.isDoorTarget(b) && this.within(this.x, b)) ?? null;
   184→  }
   185→
   186→  /** Objetivos de puerta: viviendas siempre; locales comunes transitables sin features (bar,
   187→   *  lavandería…); bloques con PASILLO INTERIOR (v1.18, trasteros por columna) también; y los
   188→   *  bloques de mobiliario común (v1.18: portal con buzones/tablón/banco) AVISAN de su puerta
   189→   *  (approach) — el tablón vive ahí y las misiones pueden pedirte llegar hasta él. Los pozos
   190→   *  (escalera/ascensor) siguen sin ser puertas: son zonas de tránsito vertical. */
   191→  private isDoorTarget(b: BlockInstance): boolean {
   192→    if (b.kind === 'dwelling') return true;
   193→    if (b.layer !== 'main') return false;
   194→    const def = this.content.blocks[b.defId]!;
   195→    if (!def.traversable) return false;
   196→    const f = def.features;
   197→    if (f?.innerCorridor) return true;
   198→    if (!f) return true;
   199→    if (f.stairs || f.elevator) return false;
   200→    return true;
   201→  }
   202→
   203→  position(): PlayerPosition {
   204→    // En el cubo de escalera (v1.4) la posición la marca el recorrido: X y Z absolutas del
   205→    // tramo actual y planta fraccionaria derivada de la altura de pies (nodo = floor·cellH + canto).
   206→    if (this.stairPath) {
   207→      return {
   208→        x: this.climbPos.x,
   209→        z: 0,
   210→        floor: (this.climbPos.y - this.view.slab * 2) / this.view.cellH,
   211→        mode: 'corridor',
   212→        insideBlock: null,
   213→        climbing: true,
   214→        pathZ: this.climbPos.z,
   215→        rear: null,
   216→      };
   217→    }
   218→    let floor = this.floor;
   219→    if (this.climb) {
   220→      if (this.climb.external) {
   221→        // Viaje en ascensor (v1.12): el progreso lo marca el director de la cabina y YA trae la
   222→        // física del viaje (trapezoide aceleración–crucero–frenado proporcional a la distancia).
   223→        // NO se re-suaviza aquí: la curva del jugador ha de ser EXACTAMENTE la de la cabina.
   224→        const k = Math.min(1, Math.max(0, this.climb.progress ?? 0));
   225→        floor = this.climb.from + (this.climb.to - this.climb.from) * k;
   226→        return { x: this.x, z: this.z, floor, mode: this.mode, insideBlock: this.insideBlock, climbing: true, pathZ: this.climb.zOverride ?? null, rear: null };
   227→      }
   228→      const k = Math.min(1, this.climb.t / this.climb.duration);
   229→      floor = this.climb.from + (this.climb.to - this.climb.from) * (k * k * (3 - 2 * k));
   230→    }
   231→    // Apilamiento 3D (v1.19): dentro del trastero propio el cuerpo vive en el bloque trasero.
   232→    const rear = this.rearCell ? this.rearSeat() : null;
   233→    return { x: this.x, z: this.z, floor, mode: this.mode, insideBlock: this.insideBlock, climbing: this.climb !== null, pathZ: null, rear };
   234→  }
   235→
   236→  /** Posición (mundo) del cuerpo dentro del interior presentable del trastero propio. */
   237→  private rearSeat(): { x: number; z: number } {
   238→    const b = this.world.blocks[this.rearCell!.block]!;
   239→    const cx = (b.cell.col + b.size.w / 2) * this.view.cellW;
   240→    const front = this.view.roomZ - this.view.commonDepth / 2;
   241→    // La sala presentable vive en el frente del bloque (mismo plano que el resto del edificio).
   242→    const z = front + this.view.trasteros.cellRoomDepth / 2 - 0.3;
   243→    return { x: cx, z };
   244→  }

   246→  /** Info de la escalada en curso para el cableado del ascensor (null si no hay). */
   247→  climbInfo(): { from: number; to: number } | null {
   248→    return this.climb?.external ? { from: this.climb.from, to: this.climb.to } : null;
   249→  }

   251→  /** Progreso del viaje de cabina (0..1, lo pinta el director del ascensor). */
   252→  setClimbProgress(k: number): void {
   253→    if (this.climb?.external) this.climb.progress = Math.min(1, Math.max(0, k));
   254→  }
   255→
   256→  /** Z absoluta mientras el jugador "está dentro" de la cabina (null = pasillo de siempre). */
   257→  setClimbZ(z: number | null): void {
   258→    if (this.climb?.external) this.climb.zOverride = z;
   259→  }

   261→  /** Cierra el viaje externo: planta definitiva y sincronización con el sim. */
   262→  finishClimb(): void {
   263→    if (!this.climb?.external) return;
   264→    this.finishRide(this.climb.to);
   265→  }
   266→
   267→  /**
   268→   * Fin de viaje externo por AVERÍA a mitad de ruta (el director canceló la cabina): el jugador
   269→   * aterriza en la planta MÁS CERCANA al progreso alcanzado — nunca un teletransporte a destino.
   270→   */
   271→  finishClimbAtNearest(): void {
   272→    if (!this.climb?.external) return;
   273→    const { from, to, progress } = this.climb;
   274→    const nearest = Math.round(from + (to - from) * (progress ?? 0));
   275→    this.finishRide(nearest);
   276→  }
   277→
   278→  /** Aterriza el viaje externo en `landing` y sincroniza el desplazamiento vertical con el sim. */
   279→  private finishRide(landing: number): void {
   280→    const floor = Math.max(0, Math.min(this.topFloor, landing));
   281→    this.climb = null;
   282→    this.floor = floor;
   283→    this.z = this.view.corridorZ; // salida de cabina: carril interior del pasillo
   284→    this.syncRideFloors(floor);
   285→  }
   286→
   287→  /**
   288→   * Sincroniza la location del sim tras un viaje multialtura (v1.11): el comando `move` solo
   289→   * admite saltos contiguos, así que un aterrizaje a N plantas de golpe se traduce en una
   290→   * CADENA de moves calculada con el router del propio sim (nextStepTowards): por el pozo si
   291→   * el ascensor está operativo, por la escalera si no — siempre saltos legales de 1 en 1.
   292→   * Sin esto, la location del sim quedaba anclada a la planta de origen (desync permanente
   293→   * tras cualquier viaje multialtura). Si un eslabón no procede, la cadena se detiene y el
   294→   * sync normal de caminar seguirá acercando la location desde la última planta confirmada.
   295→   */
   296→  private syncRideFloors(landed: number): void {
   297→    const col = this.zone().cell.col;
   298→    const target = this.onFloor(landed).find((b) => b.layer === 'main' && b.cell.col === col && this.content.blocks[b.defId]!.features?.elevator);
   299→    if (!target) return;
   300→    this.lastMoveTarget = null;
   301→    for (let guard = 0; guard < 32; guard++) {
   302→      const loc = this.location;
   303→      if (!loc || loc === target.id) return;
   304→      const next = nextStepTowards(this.world, this.content, loc, target.id);
   305→      if (!next) return;
   306→      const r = this.exec({ type: 'move', resident: this.actorId, to: next });
   307→      if (r && !r.ok) return;
   308→    }
   309→  }

   313→  private elevatorWorks(a: number, b: number) {
   314→    const ok = (f: number) => this.elevatorBlock(f)?.params.elevator_status === 'ok';
   315→    return ok(a) && ok(b);
   316→  }
   317→
   318→  /**
   319→   * Panel de destino del ascensor (v1.9): datos del pozo donde está el jugador para pintar el
   320→   * selector de plantas (solo paradas OPERATIVAS en la misma columna). null si no está delante
   321→   * de un núcleo operativo — condición GEMELA del prompt de la E: lo que el prompt anuncia es
   322→   * exactamente lo que la E abre.
   323→   */
   324→  elevatorPanelTarget(): { col: number; current: number; floors: number[] } | null {
   325→    if (this.climb || this.stairPath || this.mode !== 'corridor') return null;
   326→    const z = this.zone();
   327→    const feat = this.content.blocks[z.defId]!.features;
   328→    if (feat?.elevator !== true || z.params.elevator_status !== 'ok') return null;
   329→    const col = z.cell.col;
   330→    const floors: number[] = [];
   331→    for (let f = 0; f <= this.topFloor; f++) {
   332→      const b = this.onFloor(f).find((x) => x.cell.col === col && x.layer === 'main' && this.content.blocks[x.defId]!.features?.elevator);
   333→      if (b && b.params.elevator_status === 'ok') floors.push(f);
   334→    }
   335→    if (floors.length < 2) return null;
   336→    return { col, current: this.floor, floors };
   337→  }
   338→
   339→  /**
   340→   * Llamada de ascensor con destino EXPLÍCITO (panel v1.9): mismo viaje externo que W/S pero a
   341→   * cualquier planta del pozo. La autoridad del viaje sigue siendo del sim (move al desembarcar);
   342→   * aquí solo se pide cabina. false si la llamada no procede (avería, planta sin parada…).
   343→   */
   344→  elevatorCall(to: number): boolean {
   345→    if (this.climb || this.stairPath || this.mode !== 'corridor') return false;
   346→    if (to < 0 || to > this.topFloor || to === this.floor) return false;
   347→    const z = this.zone();
   348→    const feat = this.content.blocks[z.defId]!.features;
   349→    if (feat?.elevator !== true || z.params.elevator_status !== 'ok') return false;
   350→    if (!this.elevatorWorks(this.floor, to)) return false;
   351→    const [a, b] = this.rangeOf(z);
   352→    this.x = (a + b) / 2;
   353→    this.climb = { from: this.floor, to, t: 0, duration: 0, external: true, progress: 0, zOverride: null };
   354→    return true;
   355→  }
   356→
   357→  /**
   358→   * Vía única de comandos: con `send` sale por ahí (multijugador; resultado null, el servidor
   359→   * decide y confirma por session:state); sin `send`, dispatch local devolviendo su resultado
   360→   * exactamente como siempre (clásico).
   361→   */
   362→  private exec(cmd: Command): CommandResult | null {
   363→    const r = this.send ? this.send(cmd) : dispatch(this.world, this.content, cmd);
   364→    return r ?? null;
   365→  }
   366→
   367→  private sync() {
   368→    const z = this.zone();
   369→    const loc = this.location;
   370→    if (loc === z.id) {
   371→      this.lastMoveTarget = null; // el servidor (o el sim local) ya confirmó este destino
   372→      return;
   373→    }
   374→    if (z.id === this.lastMoveTarget) return; // en vuelo: no reenviar el mismo move cada frame
   375→    this.lastMoveTarget = z.id;
   376→    this.exec({ type: 'move', resident: this.actorId, to: z.id });
   377→  }
   378→
   379→  /**
   380→   * Subir/bajar depende del pozo en el que estés. Escalera (v1.4): recorrido REAL por el cubo
   381→   * (tramo A, descansillo, tramo B en dirección contraria) con el ritmo del catálogo
   382→   * (stairs.climbSpeed). Ascensor operativo: viaje elevatorSeconds (cabina real en Task 30);
   383→   * núcleo averiado no da gesto: hay que ir a la escalera.
   384→   * v1.19: devuelve true SOLO si arrancó una escalada; si no, el llamador usa W/S para
   385→   * cambiar de CARRIL en el pasillo (dos vías: interior junto a puertas, exterior junto a
   386→   * la barandilla — desde el exterior no saltan avisos de pared).
   387→   */
   388→  private startClimb(delta: 1 | -1): boolean {
   389→    const to = this.floor + delta;
   390→    if (to < 0 || to > this.topFloor) return false; // planta 0 no baja; última no sube
   391→    const z = this.zone();
   392→    const feat = this.content.blocks[z.defId]!.features;
   393→    if (!feat) return false;
   394→    const ride = feat.elevator === true && this.elevatorWorks(this.floor, to);
   395→    if (!ride && feat.stairs !== true) return false;
   396→    const [a, zz] = this.rangeOf(z);
   397→    this.x = (a + zz) / 2;
   398→    const geom = !ride && feat.stairs === true ? stairGeomOf(this.view) : null;
   399→    if (geom) {
   400→      const y0 = this.floor * this.view.cellH + this.view.slab * 2;
   401→      const y1 = to * this.view.cellH + this.view.slab * 2;
   402→      const pts = stairPathPoints(this.view, this.x, y0, y1);
   403→      if (pts.length > 1) {
   404→        this.stairPath = { pts, i: 1, to };
   405→        this.climbPos = { x: pts[0]!.x, y: pts[0]!.y, z: pts[0]!.z };
   406→        return true;
   407→      }
   408→    }
   409→    this.climb = ride
   410→      ? { from: this.floor, to, t: 0, duration: 0, external: true, progress: 0, zOverride: null }
   411→      : { from: this.floor, to, t: 0, duration: this.view.elevatorSeconds };
   412→    return true;
   413→  }

   415→  update(dt: number, input: Input) {
   416→    // Apilamiento 3D (v1.19): el cuerpo está dentro del bloque trasero — la locomoción del
   417→    // sim queda congelada (la salida es por la puerta: tecla E, que gestiona el controlador).
   418→    if (this.rearHold) return;
   419→    // Escalada real por el cubo (v1.4): avanzar por los peldaños al ritmo del catálogo.
   420→    if (this.stairPath) {
   421→      const sp = this.stairPath;
   422→      let budget = this.view.stairs.climbSpeed * dt;
   423→      while (budget > 0 && sp.i < sp.pts.length) {
   424→        const b = sp.pts[sp.i]!;
   425→        const d = Math.hypot(b.x - this.climbPos.x, b.y - this.climbPos.y, b.z - this.climbPos.z);
   426→        if (d <= budget + 1e-6) {
   427→          this.climbPos = { x: b.x, y: b.y, z: b.z };
   428→          budget -= d;
   429→          sp.i += 1;
   430→        } else {
   431→          const t = budget / d;
   432→          this.climbPos = { x: this.climbPos.x + (b.x - this.climbPos.x) * t, y: this.climbPos.y + (b.y - this.climbPos.y) * t, z: this.climbPos.z + (b.z - this.climbPos.z) * t };
   433→          budget = 0;
   434→        }
   435→      }
   436→      if (sp.i >= sp.pts.length) {
   437→        this.floor = sp.to;
   438→        this.stairPath = null;
   439→        this.z = this.view.corridorZ; // desembarco: al carril interior del pasillo
   440→        this.sync();
   441→      }
   442→      return;
   443→    }
   444→    if (this.climb) {
   445→      if (this.climb.external) return; // el director del ascensor marca el progreso
   446→      this.climb.t += dt;
   447→      if (this.climb.t >= this.climb.duration) {
   448→        this.floor = this.climb.to;
   449→        this.climb = null;
   450→        this.sync();
   451→      }
   452→      return;
   453→    }
   454→    const dir = (input.right ? 1 : 0) - (input.left ? 1 : 0);
   455→    // Correr (v1.2): multiplicador del catálogo (rules.view.runMultiplier); andar es el default.
   456→    const speedMul = input.run ? this.view.runMultiplier : 1;
   457→    if (this.mode === 'inside') {
   458→      const home = this.world.blocks[this.insideBlock!]!;
   459→      const dx = dir * this.view.roomSpeed * speedMul * dt;
   460→      const dz = ((input.up ? 1 : 0) - (input.down ? 1 : 0)) * this.view.roomSpeed * speedMul * dt;
   461→      // ── Pasillo interior (v1.18, trasteros por columna): banda central transitable en X,
   462→      //  fondo libre en Z hasta el tabique trasero. TODO sale del catálogo (storage.ts).
   463→      if (this.content.blocks[home.defId]!.features?.innerCorridor) {
   464→        const sg = storageGeomOf(this.view);
   465→        const [ca, cb] = storageCorridorX(home.cell.col * this.view.cellW, sg);
   466→        const m = this.view.margin;
   467→        this.x = Math.min(cb - m, Math.max(ca + m, this.x + dx));
   468→        this.z = Math.min(sg.depth - m, Math.max(m, this.z + dz));
   469→        this.sync();
   470→        return;
   471→      }
   472→      const geo = this.geometryOf(home);
   473→      if (!geo) {
   474→        // Layout sin geometría 2D (catálogo antiguo): movimiento 1D de siempre.
   475→        const [a, b] = this.rangeOf(home);
   476→        this.x = Math.min(b - this.view.margin, Math.max(a + this.view.margin, this.x + dx));
   477→        return;
   478→      }
   479→      // Movimiento 2D con deslizamiento: cada eje se prueba por separado contra tabiques,
   480→      // huecos de puerta y estancias cerradas. TODO sale del catálogo (layout + rules.view).
   481→      // Los rects del layout son LOCALES a la vivienda: se compara con la x local (x − borde izq.).
   482→      // v1.19: el calado lo marca el LAYOUT (las salas traseras viven más allá de view.depth).
   483→      const [a, b] = this.rangeOf(home);
   484→      const x0 = home.cell.col * this.view.cellW;
   485→      const maxZ = Math.max(...geo.rooms.map((r) => r.rect.z + r.rect.d));
   486→      const clampX = (x: number) => Math.min(b - this.view.margin, Math.max(a + this.view.margin, x));
   487→      const clampZ = (z: number) => Math.min(maxZ - this.view.margin, Math.max(this.view.margin, z));
   488→      const cur = roomAt(geo, this.x - x0, this.z);
   489→      const nx = clampX(this.x + dx);
   490→      if (walkable(geo, cur, nx - x0, this.z)) this.x = nx;
   491→      const cur2 = roomAt(geo, this.x - x0, this.z);
   492→      const nz = clampZ(this.z + dz);
   493→      if (walkable(geo, cur2, this.x - x0, nz)) this.z = nz;
   494→      return;
   495→    }
   496→    // Pasillo (v1.19): DOS carriles. A/D avanza en X; W/S, si no arrancan escalada (no hay
   497→    // pozo delante o ya estás en el límite), cambian de carril: W hacia la fachada (interior,
   498→    // junto a puertas y avisos), S hacia la barandilla (exterior, vía de adelantamiento).
   499→    const lanes = corridorLanes(this.view);
   500→    const dirZ = (input.up ? 1 : 0) - (input.down ? 1 : 0);
   501→    let climbing = false;
   502→    if (dirZ > 0) climbing = this.startClimb(1);
   503→    else if (dirZ < 0) climbing = this.startClimb(-1);
   504→    if (!climbing && dirZ !== 0) {
   505→      this.z = Math.min(lanes.maxZ, Math.max(lanes.minZ, this.z + dirZ * this.view.walkSpeed * speedMul * dt));
   506→    }
   507→    this.x = Math.min(this.width - this.view.margin, Math.max(this.view.margin, this.x + dir * this.view.walkSpeed * speedMul * dt));
   508→    this.sync();
   509→    const door = this.doorInFront()?.id ?? null;
   510→    if (door !== this.approached) {
   511→      this.approached = door;
   512→      if (door) this.exec({ type: 'approach', resident: this.actorId, block: door });
   513→    }
   514→  }

   516→  /** Vecino que está a tu lado en el pasillo (o en tu misma vivienda). */
   517→  residentNearby(): string | null {
   518→    const r = this.world.residents[this.actorId]!;
   519→    for (const other of Object.values(this.world.residents)) {
   520→      if (other.id === r.id || !other.present || other.location === null) continue;
   521→      const loc = this.world.blocks[other.location];
   522→      if (!loc) continue;
   523→      const myZone = this.mode === 'inside' ? this.insideBlock : this.zone().id;
   524→      const theirZone = loc.kind === 'dwelling' ? (this.mode === 'inside' && this.insideBlock === loc.id ? loc.id : null) : loc.id;
   525→      // Circulación compartida: en pasillo, un vecino en el pasillo o el núcleo de ESTA planta
   526→      // está "a tu lado" aunque zone() te asigne el núcleo (capa main) y a él el pasillo.
   527→      const sharedCirculation = this.mode === 'corridor' && loc.cell.floor === this.floor && loc.kind !== 'dwelling';
   528→      if (theirZone !== myZone && !sharedCirculation) continue;
   529→      // En circulación, la referencia es el centro del bloque donde está (pasillo ancho → su centro).
   530→      const theirX = (loc.cell.col + loc.size.w / 2) * this.view.cellW;
   531→      if (Math.abs(theirX - this.x) < this.view.nearRadius) return other.id;
   532→    }
   533→    return null;
   534→  }

   536→  /** Objetivo de interacción actual según posición (para el prompt y la tecla E). */
   537→  interactTarget(): InteractTarget | null {
   538→    if (this.climb) return null;
   539→    // Apilamiento 3D (v1.19): dentro del trastero propio la única salida es la puerta.
   540→    if (this.rearCell) return { kind: 'rearExit' };
   541→    if (this.mode === 'inside') {
   542→      const b = this.world.blocks[this.insideBlock!]!;
   543→      // Trastero propio (v1.19): delante de la puerta de TU celda → E abre el swap de bloques.
   544→      // La puerta se compara en Z DE MUNDO (la z del jugador es local al pasillo interior).
   545→      if (this.content.blocks[b.defId]!.features?.innerCorridor) {
   546→        const cell = this.ownCellDoor(b);
   547→        const worldZ = this.frontZ + this.z;
   548→        if (cell && Math.abs(this.x - cell.x) < this.view.itemRadius * 0.75 && Math.abs(worldZ - cell.z) < this.view.itemRadius) {
   549→          return { kind: 'storageCell' };
   550→        }
   551→        return { kind: 'door', blockId: this.insideBlock! };
   552→      }
   553→      if (b.ownerId === this.actorId) {
   554→        // Interactivos de la estancia actual (portátil/cama): posición 2D del catálogo, no offsets.
   555→        const geo = this.geometryOf(b);
   556→        if (geo) {
   557→          const room = roomAt(geo, this.x - b.cell.col * this.view.cellW, this.z);
   558→          const rdef = this.content.rooms[room.defId];
   559→          for (const it of rdef?.interactables ?? []) {
   560→            const wx = b.cell.col * this.view.cellW + room.rect.x + it.pos[0];
   561→            const wz = room.rect.z + it.pos[1];
   562→            if (Math.hypot(this.x - wx, this.z - wz) < this.view.itemRadius) return { kind: it.kind };
   563→          }
   564→        }
   565→      }
   566→      return { kind: 'door', blockId: this.insideBlock! };
   567→    }
   568→    // Prioridad (v1.18): las fijaciones del portal (buzones/tablón, proximidad exacta) van
   569→    // delante de la puerta — el portal ahora también es puerta (approach de misiones) y sin
   570→    // esto su propia fachada se comería la E del tablón. Después: puerta → vecino.
   571→    // v1.19: desde el carril EXTERIOR nada de esto salta (ni buzones ni tablón ni puertas).
   572→    if (!this.onOuterLane()) {
   573→      const zoneFirst = this.zone();
   574→      const [za, zz] = this.rangeOf(zoneFirst);
   575→      if (this.content.blocks[zoneFirst.defId]!.features?.mailboxes && Math.abs(this.x - (za + 1.2)) < this.view.itemRadius) return { kind: 'mailbox' };
   576→      if (this.content.blocks[zoneFirst.defId]!.features?.noticeboard && Math.abs(this.x - (zz - 0.8)) < this.view.itemRadius) return { kind: 'noticeboard' };
   577→      const door = this.doorInFront();
   578→      if (door) return { kind: 'door', blockId: door.id };
   579→    }
   580→    const neighbor = this.residentNearby();
   581→    if (neighbor) return { kind: 'resident', residentId: neighbor };
   582→    return null;
   583→  }

   585→  /** Puerta (mundo) de la celda del JUGADOR en un bloque de trasteros por columna; null si
   586→   *  el bloque no es el de su letra. La celda es la de la vivienda del jugador. */
   587→  private ownCellDoor(storageBlock: BlockInstance): { x: number; z: number; side: 'left' | 'right'; floor: number } | null {
   588→    const me = this.world.residents[this.actorId]!;
   589→    const home = this.world.blocks[me.home];
   590→    if (!home) return null;
   591→    const tag = home.id.slice(home.id.lastIndexOf('_') + 1);
   592→    if (!storageBlock.id.endsWith(`_${tag}`)) return null;
   593→    const floors = Object.values(this.world.blocks)
   594→      .filter((d) => d.kind === 'dwelling' && d.id.endsWith(`_${tag}`))
   595→      .map((d) => d.cell.floor)
   596→      .sort((a, z) => a - z);
   597→    const pt = storageDoorPoint(this.view, storageBlock, floors, home.cell.floor);
   598→    return pt ? { ...pt, floor: home.cell.floor } : null;
   599→  }

   601→  /** Tecla de interacción: usa el objeto cercano (portátil/cama/buzón/tablón/vecino) o resuelve la puerta. */
   602→  interact(): InteractResult {
   603→    if (this.climb) return { type: 'none' };
   604→    // Apilamiento 3D (v1.19): dentro del trastero propio, E es SALIR (el controlador
   605→    // deshará el swap de bloques; el sim jamás supo que entraste).
   606→    if (this.rearCell) {
   607→      const block = this.rearCell.block;
   608→      this.rearCell = null;
   609→      this.rearHold = false;
   610→      return { type: 'rearExit', block };
   611→    }
   612→    const t = this.interactTarget();
   613→    if (t?.kind === 'laptop') {
   614→      this.exec({ type: 'work', actor: this.actorId });
   615→      return { type: 'used', kind: 'laptop' };
   616→    }
   617→    if (t?.kind === 'bed') {
   618→      this.exec({ type: 'sleep', actor: this.actorId });
   619→      return { type: 'used', kind: 'bed' };
   620→    }
   621→    if (t?.kind === 'mailbox') return { type: 'used', kind: 'mailbox' }; // lectura: texto del catálogo en la presentación
   622→    if (t?.kind === 'noticeboard') return { type: 'used', kind: 'noticeboard' };
   623→    if (t?.kind === 'resident') {
   624→      const other = this.world.residents[t.residentId]!;
   625→      if (other.location) this.exec({ type: 'knock', block: other.location, actor: this.actorId });
   626→      return { type: 'used', kind: 'resident' };
   627→    }
   628→    if (this.mode === 'inside') {
   629→      const home = this.insideBlock!;
   630→      // Apilamiento 3D (v1.19): delante de la puerta de TU celda, E entra al trastero
   631→      // (swap de bloques) en vez de salir del bloque — solo sale quien no está en SU puerta.
   632→      const homeBlock = this.world.blocks[home]!;
   633→      if (this.content.blocks[homeBlock.defId]!.features?.innerCorridor) {
   634→        const cell = this.ownCellDoor(homeBlock);
   635→        const worldZ = this.frontZ + this.z;
   636→        if (cell && Math.abs(this.x - cell.x) < this.view.itemRadius * 0.75 && Math.abs(worldZ - cell.z) < this.view.itemRadius) {
   637→          this.rearCell = { block: homeBlock.id, side: cell.side, floor: cell.floor };
   638→          this.rearHold = true;
   639→          return { type: 'rearEnter', block: homeBlock.id };
   640→        }
   641→      }
   642→      const corridorId = this.corridor().id;
   643→      const res = this.exec({ type: 'move', resident: this.actorId, to: corridorId });
   644→      if (res && !res.ok) return { type: 'none' };
   645→      this.mode = 'corridor';
   646→      this.insideBlock = null;
   647→      this.z = this.view.corridorZ;
   648→      this.lastMoveTarget = corridorId; // ya está en vuelo este move
   649→      this.approached = home; // sigue delante de la misma puerta, sin repetir el aviso
   650→      return { type: 'exited', block: home };
   651→    }
   652→    const door = this.doorInFront();
   653→    if (!door) return { type: 'none' };
   654→    const res = this.exec({ type: 'move', resident: this.actorId, to: door.id });
   655→    // Rechazo = candado. En clásico lo dice el sim (res.ok false); en remoto el resultado llega
   656→    // después (res null), así que el candado visible de la puerta manda: NADIE entra aquí.
   657→    if (res ? !res.ok : door.access === 'locked') {
   658→      return { type: 'locked', block: door.id, reason: door.lockReason, price: door.availability?.price ?? null };
   659→    }
   660→    this.lastMoveTarget = door.id; // ya está en vuelo este move
   661→    if (door.kind === 'dwelling') {
   662→      this.landInside(door);
   663→    } else if (this.content.blocks[door.defId]!.features?.innerCorridor) {
   664→      // Trasteros por columna (v1.18): modo interior sobre su pasillo central — caminas
   665→      // hacia el fondo y tu celda queda a un lado (impares) o al otro (pares).
   666→      this.mode = 'inside';
   667→      this.insideBlock = door.id;
   668→      this.z = 0.45;
   669→      this.approached = door.id;
   670→    } else {
   671→      this.approached = door.id; // local común (bar/lavandería): te plantas en su puerta, sin modo interior
   672→    }
   673→    return { type: 'entered', block: door.id };
   674→  }

   676→  /**
   677→   * Colocación directa junto a la puerta de un bloque (QoL del editor multi-vivienda).
   678→   * No desbloquea nada: si la puerta está cerrada, el jugador simplemente se planta delante.
   679→   * La posición del sim se sincroniza recorriendo el camino legal del grafo de circulación
   680→   * (mismo recorrido que andar: portal → núcleo → planta), sin teletransportes prohibidos.
   681→   */
   682→  teleportTo(blockId: string): boolean {
   683→    const target = this.world.blocks[blockId];
   684→    if (!target || this.climb || this.stairPath) return false;
   685→    this.mode = 'corridor';
   686→    this.insideBlock = null;
   687→    this.rearCell = null;
   688→    this.rearHold = false;
   689→    this.z = this.view.corridorZ;
   690→    this.approached = null;
   691→    this.externalRequested = false;
   692→    this.floor = target.cell.floor;
   693→    this.x = this.centerOf(target);
   694→    for (let guard = 0; guard < 24; guard++) {
   695→      const zone = this.zone();
   696→      if (zone.id === this.location) break;
   697→      const next = nextStepTowards(this.world, this.content, this.location ?? zone.id, zone.id);
   698→      if (!next) break;
   699→      this.exec({ type: 'move', resident: this.actorId, to: next });
   700→    }
   701→    this.lastMoveTarget = null; // los move en vuelo los confirmará el estado; sin dedup pendiente
   702→    return true;
   703→  }

   705→  /**
   706→   * Reconciliación multijugador: impone la posición que dice el servidor (autoridad).
   707→   * Fija modo, planta y coordenada X al centro del bloque indicado; con null coloca en el
   708→   * pasillo de la planta actual. Resetea climb y el dedup de move.
   709→   */
   710→  forceLocation(blockId: string | null): void {
   711→    const target = blockId !== null ? this.world.blocks[blockId] : undefined;
   712→    this.climb = null;
   713→    this.stairPath = null;
   714→    this.externalRequested = false;
   715→    this.lastMoveTarget = null;
   716→    this.rearCell = null;
   717→    this.rearHold = false;
   718→    if (!target) {
   719→      const c = this.corridor(this.floor);
   720→      this.mode = 'corridor';
   721→      this.insideBlock = null;
   722→      this.z = this.view.corridorZ;
   723→      this.x = this.centerOf(c);
   724→      return;
   725→    }
   726→    this.floor = target.cell.floor;
   727→    this.x = this.centerOf(target);
   728→    if (target.kind === 'dwelling') {
   729→      this.landInside(target);
   730→    } else {
   731→      this.mode = 'corridor';
   732→      this.insideBlock = null;
   733→      this.z = this.view.corridorZ;
   734→    }
   735→  }
