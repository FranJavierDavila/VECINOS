    18→interface BlockView {
    19→  /** Cajas greybox estructurales (se ocultan cuando llega el modelo glTF; panel/candado NO van aquí). */
    20→  greybox: Mesh[];
    21→  /** Materiales que reciben glow de canales y tintado de estado (1 en greybox, N en modelos). */
    22→  mats: BlockMat[];
    23→  /** Emissive base por material (lo que paint() deja sin canales); lo usa el LOD 'glow'. */
    24→  baseEmissive: Color3[];
    25→  /** Color base por material (view.color en greybox; albedo capturado en modelos). */
    26→  baseColor: Color3[];
    27→  /** Fuerza del tintado de estado: 1 = color plano (greybox), 0..1 = mezcla sobre el arte (modelos). */
    28→  tintStrength: number;
    29→  /** Raíz del modelo glTF si este bloque ya lo lleva (null = greybox). */
    30→  model: TransformNode | null;
    31→  front: Mesh | null; // fachada que cierra la vivienda (candado o privacidad)
    32→  /** Puerta y picaporte de la fachada de privacidad (v1.17): la vivienda vacía se ve cerrada. */
    33→  door: Mesh | null;
    34→  handle: Mesh | null;
    35→  lock: Mesh | null;
    36→  lockBase: { x: number; y: number };
    37→  // ── LOD por distancia (rules.view.lod) ──
    38→  /** Centro X del bloque en metros (dx del LOD: |centerX − focus.x|). */
    39→  centerX: number;
    40→  /** Planta del bloque (dy del LOD: |floor − focus.floor| * view.cellH). */
    41→  floor: number;
    42→  /** Pared del fondo de la vivienda (la única malla que puede llevar edgesRendering) o null;
    43→   *  el EdgesRenderer activo solo existe en las paredes destacadas del jugador. */
    44→  edgesMesh: Mesh | null;
    45→  /** true mientras el LOD fuerza baseEmissive (al soltarlo se repinta el cálculo de paint una vez). */
    46→  glowForced: boolean;
    47→  /** true mientras el LOD oculta panel+candado; la visibilidad efectiva es la de paint() && !esto. */
    48→  lodHiddenPanels: boolean;
    49→  /** Segunda pared con edges (v1.19): la pared del fondo se parte si hay puerta a sala trasera. */
    50→  edgesMesh2: Mesh | null;
    51→  /** v1.24: decoración de la puerta de privacidad (marco + aplique) — se activa junto a view.door. */
    52→  doorDecor: Mesh[];
    53→  /** v1.25: color FIJO de la vivienda (pared gris clarito) — el tintado de estado no la repinta;
    54→   *  el emissive de canales (ruido/agua) sigue funcionando. null = tintado clásico. */
    55→  fixedDiffuse: Color3 | null;
    56→}

    62→  /** Tapas de los huecos de puerta que llevan a una estancia cerrada ("en obras"). */

    66→  /** Diorama v1.23: billboards con paralaje (pared/suelo/mobiliario) por estancia. */
    67→  dio: DioramaSet;
    68→  /** Materiales del diorama del bloque (tinte nocturno idéntico a suelos/lámparas). */
    69→  dioMats: StandardMaterial[];
    70→  /** Raíz de los quads de diorama (visibilidad espejo de la fachada en applyPanels). */

    74→/**
    75→ * Una caja por bloque (placeholder greybox del futuro prefab) y, si el manifiesto de assets
    76→ * (Content.assets.blocks) declara un modelo glTF para el tipo de bloque, el modelo en su lugar
    77→ * (con la caja oculta). TODA la geometría y los colores salen de content.rules.view +
    78→ * block.view + assets: sustituir el greybox por arte final no toca este código.
    79→ * Las viviendas se dibujan en corte (interior visible) SOLO mientras hay alguien dentro;
    80→ * vacías, la fachada se cierra con su puerta al pasillo (rules.view.privacy) — y las
    81→ * bloqueadas, con panel y candado como siempre.
    82→ */
    83→export function createBuilding(scene: Scene, world: World, content: Content) {
    84→  const { Color3, Color4, MeshBuilder, StandardMaterial, TransformNode } = B();
    85→  const v = content.rules.view;
    86→  const glow = Object.fromEntries(Object.entries(v.channelGlow).map(([k, hex]) => [k, Color3.FromHexString(hex)])) as Record<string, Color3>;
    87→  const views = new Map<string, BlockView>();
    88→  const shakes = new Map<string, number>();
    89→  /** Raíz de escena por bloque (v1.19): colgar aquí TODA malla del bloque permite el
    90→   *  desvanecimiento completo del bloque principal en el apilamiento 3D (swap de bloques). */
    91→  const roots = new Map<string, TransformNode>();
    92→  const attach = (blockId: string, mesh: Mesh): Mesh => {
    93→    const r = roots.get(blockId);
    94→    if (r) mesh.parent = r;
    95→    return mesh;
    96→  };
    97→  /** Interiores 2D (Fase C) por bloque: suelos de estancia y tapas de puertas cerradas. */
    98→  const interiors = new Map<string, InteriorView>();
    99→  /** Foco del LOD (posición del jugador): lo alimenta update() y lo consume paint(). */
   100→  let lastFocus: { x: number; floor: number } | null = null;
   101→  // Plano frontal de las estancias: la cara de la puerta. Con viviendas del doble de calado
   102→  // (v0.6-b), el extra crece hacia el FONDO (+Z) para no envolver el pasillo ni al núcleo.
   103→  // v1.19: los calados del catálogo YA incluyen el +1 módulo (rules.view.depthModule) —
   104→  // depth 4.2 en viviendas y commonDepth 2.8 en comunes son los valores finales.
   105→  const frontZ = v.roomZ - v.commonDepth / 2;
   106→  /** Calado por bloque: viviendas 2× (mobiliario interior); naves del cierre (v1.16) y el
   107→   *  tejado (con aleros) a calado de vivienda; bloques comunes, el de siempre. */
   108→  const depthOf = (b: BlockInstance): number => {
   109→    if (b.kind === 'dwelling') return v.depth;
   110→    if (b.defId === 'trasteros_planta' || b.defId === 'bajocubierta') return v.depth;
   111→    if (b.defId === v.trasteros.blockDefId) return v.depth * v.trasteros.blockDepthFactor;
   112→    if (b.defId === 'tejado') return v.depth + v.roof.overhang * 2;
   113→    return v.commonDepth;
   114→  };
   115→  const mat = (name: string) => new StandardMaterial(name, scene);
   116→  /** Texturas de puerta cacheadas por fichero (v1.24→v1.25): FULL-BLEED de verdad. El arte
   117→   *  llega con pared/suelo cocinados alrededor de la hoja: flood-fill contra los colores de
   118→   *  las esquinas (stripCorners) + recorte a la caja del contenido (alphaBoundsOf) → la
   119→   *  textura ES la hoja y desaparecen los elementos antiguos que se veían alrededor. */
   120→  const doorArtCache = new Map<string, DynamicTexture>();
   121→  const doorArtTex = (sc: Scene, file: string): DynamicTexture => {
   122→    const hit = doorArtCache.get(file);
   123→    if (hit) return hit;
   124→    const dt = new (B().DynamicTexture)(`puerta_tex_${doorArtCache.size}`, { width: 384, height: 768 }, sc, true);
   125→    doorArtCache.set(file, dt);
   126→    const img = new Image();
   127→    img.onload = () => {
   128→      const big = document.createElement('canvas');
   129→      big.width = img.naturalWidth;
   130→      big.height = img.naturalHeight;
   131→      const bx = big.getContext('2d', { willReadFrequently: true });
   132→      if (!bx) return;
   133→      bx.drawImage(img, 0, 0);
   134→      stripCorners(big, 0.14);
   135→      const b = alphaBoundsOf(big);
   136→      const dctx = dt.getContext() as unknown as CanvasRenderingContext2D;
   137→      dctx.clearRect(0, 0, 384, 768);
   138→      if (b) dctx.drawImage(big, b.x0, b.y0, b.x1 - b.x0, b.y1 - b.y0, 0, 0, 384, 768);
   139→      else dctx.drawImage(img, 0, 0, img.naturalWidth, img.naturalHeight, 0, 0, 384, 768);
   140→      dt.update(true);
   141→    };
   142→    img.src = file;
   143→    return dt;
   144→  };
   145→  /** Candado DIBUJADO (v1.25): silueta real de candado en un DynamicTexture — nada de cajas
   146→   *  3D. Colores del catálogo (lockColor); grillete de acero fijo. MISMO PATRÓN que los
   147→   *  sprites del diorama: se recorta el quad a la caja del contenido (alphaBounds) — el
   148→   *  canvas subido no puede confiar en el alpha fuera del contenido (fondo a negro). */
   149→  let lockTex: DynamicTexture | null = null;
   150→  let lockTexAr = 0.8;
   151→  const padlockTex = (): DynamicTexture => {
   152→    if (lockTex) return lockTex;
   153→    const dt = new (B().DynamicTexture)('candado_tex', { width: 128, height: 128 }, scene, true);
   154→    dt.hasAlpha = true;
   155→    const c = document.createElement('canvas');
   156→    c.width = 128;
   157→    c.height = 128;
   158→    const ctx = c.getContext('2d', { willReadFrequently: true });
   159→    if (!ctx) return dt;
   160→    ctx.clearRect(0, 0, 128, 128);
   161→    const body = Color3.FromHexString(v.lockColor).toHexString();
   162→    // Grillete: arco de acero.
   163→    ctx.strokeStyle = '#8d939c';
   164→    ctx.lineWidth = 11;
   165→    ctx.beginPath();
   166→    ctx.arc(64, 52, 26, Math.PI, 0);
   167→    ctx.stroke();
   168→    // Cuerpo de latón con borde.
   169→    ctx.fillStyle = body;
   170→    ctx.strokeStyle = 'rgba(60,40,10,0.9)';
   171→    ctx.lineWidth = 4;
   172→    const r = 12;
   173→    ctx.beginPath();
   174→    ctx.moveTo(24, 124);
   175→    ctx.lineTo(24, 66 + r);
   176→    ctx.arcTo(24, 66, 24 + r, 66, r);
   177→    ctx.lineTo(104 - r, 66);
   178→    ctx.arcTo(104, 66, 104, 66 + r, r);
   179→    ctx.lineTo(104, 124);
   180→    ctx.closePath();
   181→    ctx.fill();
   182→    ctx.stroke();
   183→    // Ojo de la cerradura.
   184→    ctx.fillStyle = 'rgba(50,32,8,0.95)';
   185→    ctx.beginPath();
   186→    ctx.arc(64, 90, 8, 0, Math.PI * 2);
   187→    ctx.fill();
   188→    ctx.fillRect(60, 90, 8, 20);
   189→    // Recorte a la caja del contenido (patrón diorama) + volcado a la textura cuadrada.
   190→    const b = alphaBoundsOf(c);
   191→    const dctx = dt.getContext() as unknown as CanvasRenderingContext2D;
   192→    dctx.clearRect(0, 0, 128, 128);
   193→    if (b) {
   194→      const bw = Math.max(1, b.x1 - b.x0);
   195→      const bh = Math.max(1, b.y1 - b.y0);
   196→      lockTexAr = bw / bh;
   197→      dctx.drawImage(c, b.x0, b.y0, bw, bh, 0, 0, 128, 128);
   198→    } else {
   199→      dctx.drawImage(c, 0, 0);
   200→    }
   201→    dt.update(true);
   202→    lockTex = dt;
   203→    return dt;
   204→  };
   205→  /** Apliques de las puertas de vivienda: el foco cálido amaanece con la noche (update). */
   206→  const doorLampMats: { mat: StandardMaterial; hex: string }[] = [];
   207→  /** Dioramas interiores (v1.23): null sin catálogo interiors.json — greybox como siempre. */
   208→  const diro: DioramaDirector | null = createDioramaDirector(scene, content);
   209→  const box = (name: string, w: number, h: number, d: number, x: number, y: number, z: number, m: StandardMaterial) => {
   210→    const mesh = MeshBuilder.CreateBox(name, { width: w, height: h, depth: d }, scene);
   211→    mesh.position.set(x, y, z);
   212→    mesh.material = m;
   213→    return mesh;
   214→  };
   215→
   216→  for (const b of Object.values(world.blocks)) {
   217→    const m = mat(`${b.id}_mat`);
   218→    const cx = (b.cell.col + b.size.w / 2) * v.cellW;
   219→    const bottom = b.cell.floor * v.cellH;
   220→    const w = b.size.w * v.cellW - v.gapW;
   221→    const h = v.cellH - v.gapH;
   222→    const d = depthOf(b);
   223→    const cz = frontZ + d / 2;
   224→    const view: BlockView = {
   225→      greybox: [],
   226→      mats: [m],
   227→      baseEmissive: [m.emissiveColor.clone()],
   228→      baseColor: [Color3.FromHexString(content.blocks[b.defId]!.view?.color ?? '#cccccc')],
   229→      tintStrength: 1,
   230→      model: null,
   231→      front: null,
   232→      door: null,
   233→      handle: null,
   234→      lock: null,
   235→      lockBase: { x: cx, y: bottom + v.cellH / 2 - 0.2 },
   236→      centerX: cx,
   237→      floor: b.cell.floor,
   238→      edgesMesh: null,
   239→      edgesMesh2: null,
   240→      glowForced: false,
   241→      lodHiddenPanels: false,
   242→      doorDecor: [],
   243→      fixedDiffuse: null,
   244→    };
   245→    const root = new TransformNode(`${b.id}_root`, scene);
   246→    roots.set(b.id, root);
   247→    if (b.layer === 'corridor') {
   248→      // Pasillo exterior (data-driven): tira transitable saliente hacia la calle, con el borde
   249→      // trasero a ras de fachada (frontZ) y barandilla baja en el borde exterior — sitio de
   250→      // sobra para que los personajes se crucen o charlen sin bailar el vacío.
   251→      const d = v.corridorDepth;
   252→      view.greybox.push(box(b.id, b.size.w * v.cellW, v.slab * 2, d, cx, bottom + v.slab, frontZ - d / 2, m));
   253→      if (v.corridorRail > 0) {
   254→        view.greybox.push(
   255→          box(`${b.id}_barandilla`, b.size.w * v.cellW, v.corridorRail, v.wall, cx, bottom + v.slab * 2 + v.corridorRail / 2, frontZ - d, m),
   256→        );
   257→      }
   258→    } else if (b.kind === 'dwelling') {
   259→      const y = bottom + v.cellH / 2;
   260→      const pv = v.privacy;
   261→      // v1.25 — acabado de la vivienda: pared GRIS CLARITO fija (el tintado de estado no la
   262→      // repinta; el glow de canales sigue) + ZÓCALO gris de 25 cm desde el suelo + marcos de
   263→      // puerta DEL COLOR del zócalo. Todo del catálogo (rules.view.privacy).
   264→      view.fixedDiffuse = Color3.FromHexString(pv.wallColor ?? '#dbd8d2');
   265→      // ── Puerta a sala TRASERA (v1.19, apilamiento 3D): la pared del fondo se parte con el
   266→      // hueco del layout (doorway hacia una estancia más profunda que el bloque principal).
   267→      const layoutId0 = content.blocks[b.defId]?.container?.layout;
   268→      const layout0 = layoutId0 ? content.layouts[layoutId0] : undefined;
   269→      const rearKeys = new Set((layout0?.rooms ?? []).filter((r) => r.rect && r.rect.z + r.rect.d > v.depth + 0.01).map((r) => r.key));
   270→      const rearDoor = (layout0?.doorways ?? []).find((dw) => dw.axis === 'x' && rearKeys.has(dw.b));
   271→      const x0w = cx - w / 2;
   272→      const x1w = cx + w / 2;
   273→      if (rearDoor) {
   274→        const gapA = Math.max(x0w, x0w + rearDoor.at - rearDoor.width / 2);
   275→        const gapB = Math.min(x1w, x0w + rearDoor.at + rearDoor.width / 2);
   276→        const doorH = Math.min(h - 0.15, 1.95);
   277→        const lintelH = h - doorH;
   278→        view.edgesMesh = box(`${b.id}_fondo_i`, gapA - x0w, h, v.wall, (x0w + gapA) / 2, y, frontZ + d, m);
   279→        view.edgesMesh2 = box(`${b.id}_fondo_d`, x1w - gapB, h, v.wall, (gapB + x1w) / 2, y, frontZ + d, m);
   280→        view.greybox.push(
   281→          view.edgesMesh,
   282→          view.edgesMesh2,
   283→          box(`${b.id}_fondo_dintel`, gapB - gapA, lintelH, v.wall, (gapA + gapB) / 2, y + h / 2 - lintelH / 2, frontZ + d, m),
   284→          box(`${b.id}_izq`, v.wall, h, d, cx - w / 2, y, cz, m),
   285→          box(`${b.id}_der`, v.wall, h, d, cx + w / 2, y, cz, m),
   286→          box(`${b.id}_suelo`, w, v.wall, d, cx, bottom + v.slab + v.wall / 2, cz, m),
   287→          box(`${b.id}_techo`, w, v.wall, d, cx, bottom + v.cellH - v.slab - v.wall / 2, cz, m),
   288→        );
   289→      } else {
   290→        view.edgesMesh = box(`${b.id}_fondo`, w, h, v.wall, cx, y, frontZ + d, m);
   291→        view.greybox.push(
   292→          view.edgesMesh,
   293→          box(`${b.id}_izq`, v.wall, h, d, cx - w / 2, y, cz, m),
   294→          box(`${b.id}_der`, v.wall, h, d, cx + w / 2, y, cz, m),
   295→          box(`${b.id}_suelo`, w, v.wall, d, cx, bottom + v.slab + v.wall / 2, cz, m),
   296→          box(`${b.id}_techo`, w, v.wall, d, cx, bottom + v.cellH - v.slab - v.wall / 2, cz, m),
   297→        );
   298→      }
   299→      view.front = box(`${b.id}_puerta`, w, h, v.wall, cx, y, frontZ, m);
   300→      // ── ZÓCALO (v1.25): banda gris de ~25 cm desde el suelo. Continúa en el MARCO de la
   301→      //  puerta (mismo material): la pieza que perimetra la puerta es del color del zócalo.
   302→      const skH = pv.skirtingH ?? 0.25;
   303→      const skMat = mat(`${b.id}_zocalo_mat`);
   304→      skMat.diffuseColor = Color3.FromHexString(pv.skirtingColor ?? '#a3a19b');
   305→      skMat.specularColor = new Color3(0.02, 0.02, 0.02);
   306→      const ySk = bottom + v.slab + v.wall + skH / 2;
   307→      view.greybox.push(
   308→        attach(b.id, box(`${b.id}_zocalo_frente`, w, skH, v.wall + 0.02, cx, ySk, frontZ, skMat)),
   309→        attach(b.id, box(`${b.id}_zocalo_izq`, v.wall + 0.02, skH, d - 2 * v.wall, cx - w / 2 + v.wall / 2, ySk, cz, skMat)),
   310→        attach(b.id, box(`${b.id}_zocalo_der`, v.wall + 0.02, skH, d - 2 * v.wall, cx + w / 2 - v.wall / 2, ySk, cz, skMat)),
   311→      );
   312→      // Candado DIBUJADO (v1.25): quad con silueta real de candado — los candados en forma
   313→      // de caja desaparecen. Cuelga de la puerta de privacidad y sigue sacudiéndose.
   314→      const lh = pv.lockH ?? 0.28;
   315→      const lw = lh * Math.max(0.5, Math.min(1.2, lockTexAr));
   316→      const lockMat = mat(`${b.id}_candado_mat`);
   317→      lockMat.diffuseTexture = padlockTex();
   318→      lockMat.emissiveColor = Color3.FromHexString(v.lockColor).scale(0.22);
   319→      lockMat.specularColor = new Color3(0, 0, 0);
   320→      lockMat.transparencyMode = 2;
   321→      lockMat.alphaCutOff = 0.35;
   322→      view.lock = MeshBuilder.CreatePlane(`${b.id}_candado`, { width: lw, height: lh }, scene);
   323→      view.lock.material = lockMat;
   324→      view.lock.isPickable = false;
   325→      view.lock.position.set(cx - pv.doorW / 2 - 0.32, view.lockBase.y, frontZ - v.wall / 2 - 0.07);
   326→      view.lock.setEnabled(false);
   327→
   328→      // ── Privacidad (v1.24): la puerta es un PLANO con arte real (full-bleed), marco 3D
   329→      // pequeño del color de la puerta que SÍ contornea la hoja, y un aplique de luz a un
   330→      // lado (brilla de noche). La placa numerada sube POR ENCIMA del marco (sin taparse).
   331→      // Se activan/desactivan cada frame según haya alguien dentro (propietario, visita…
   332→      // o un ladrón); medidas y colores del catálogo (rules.view.privacy).
   333→      const doorZ = frontZ - v.wall / 2 - 0.035;
   334→      view.door = MeshBuilder.CreatePlane(`${b.id}_puerta_priv`, { width: pv.doorW, height: pv.doorH }, scene);
   335→      view.door.position.set(cx, bottom + v.slab + v.wall + pv.doorH / 2, doorZ);
   336→      const doorMat = mat(`${b.id}_puerta_priv_mat`);
   337→      doorMat.specularColor = new Color3(0, 0, 0);
   338→      // Arte: una variante por hash del id (3 puertas del catálogo); textura cacheada por fichero.
   339→      const artFile = pv.doorArts?.length ? pv.doorArts[Math.floor(hash01(b.id, 3) * pv.doorArts.length) % pv.doorArts.length]! : null;
   340→      if (artFile) doorMat.diffuseTexture = doorArtTex(scene, artFile);
   341→      else doorMat.diffuseColor = Color3.FromHexString(pv.doorColor);
   342→      view.door.material = doorMat;
   343→      view.door.isPickable = false;
   344→      view.door.setEnabled(false);
   345→      // Marco pequeño DEL COLOR DEL ZÓCALO (v1.25: continuidad del zócalo — antes, del color
   346→      // de la puerta): dos jambas + dintel ABRAZANDO la hoja.
   347→      const fw = pv.frameW ?? 0.075;
   348→      const fd = pv.frameDepth ?? 0.1;

  1520→  /**
  1521→   * LOD por distancia (rules.view.lod). Se ejecuta SIEMPRE al final de update(): con foco degrada
  1522→   * cada bloque según su distancia, y sin foco (o con lod.enabled=false) restaura el estado completo.
  1523→   * Idempotente y sin allocations por frame: solo aritmética, booleanos y asignaciones.
  1524→   *
  1525→   * Métrica (Vista · LOD por distancia): dx = |centroX del bloque − focus.x| en metros y
  1526→   * dy = |planta del bloque − focus.floor| * view.cellH en metros; dist = Math.hypot(dx, dy).
  1527→   * Nivel efectivo = todos los niveles de lod.levels cuyo `dist` ≤ distBloque (el `hide` es
  1528→   * acumulado: un flag se oculta si cualquier nivel alcanzado lo lista; sin nivel → todo visible).
  1529→   */
  1530→  function applyLod(focus: { x: number; floor: number } | null) {
  1531→    const lod = v.lod;
  1532→    for (const id in world.blocks) {
  1533→      const b = world.blocks[id]!;
  1534→      const view = views.get(id)!;
  1535→      let hideEdges = false;
  1536→      let hideGlow = false;
  1537→      let hidePanels = false;
  1538→      if (lod.enabled && focus) {
  1539→        const dx = Math.abs(view.centerX - focus.x);
  1540→        const dy = Math.abs(view.floor - focus.floor) * v.cellH;
  1541→        const dist = Math.hypot(dx, dy);
  1542→        for (let i = 0; i < lod.levels.length; i++) {
  1543→          const level = lod.levels[i];
  1544→          if (level.dist > dist) continue;
  1545→          for (let j = 0; j < level.hide.length; j++) {
  1546→            const flag = level.hide[j];
  1547→            if (flag === 'edges') hideEdges = true;
  1548→            else if (flag === 'glow') hideGlow = true;
  1549→            else if (flag === 'panels') hidePanels = true;
  1550→          }
  1551→        }
  1552→      }
  1553→      // 'edges': interruptor nativo barato — el renderingGroup consulta edgesRenderer.isEnabled cada
  1554→      // frame, así que nunca se destruye ni recrea el EdgesRenderer (disable/enableEdgesRendering sí
  1555→      // lo harían: el primero lo dispone y el segundo recomputaría las aristas). El grosor/color

  1625→    /**
  1626→     * dt en segundos; `focus` es el punto de interés (posición del jugador) para el
  1627→     * LOD por distancia (rules.view.lod): applyLod() lo aplica SIEMPRE al final del frame

  1634→    update(dt: number, focus?: { x: number; floor: number }, inside?: string | null, pos?: { x: number; z: number; insideBlock: string | null }, env?: { daylight: number; movers?: { x: number; y: number; z: number }[] }) {
  1635→      if (focus) lastFocus = focus;
  1636→      if (inside !== undefined) playerInside = inside;

  1638→      // con su puerta al pasillo y se abre en cuanto alguien entra (propietario, visita…
  1639→      // o un ladrón). Vive en update() y no en paint() para reaccionar incluso con el
  1640→      // mundo congelado (modal abierto) y sin esperar al siguiente tick.
  1641→      const occSet = computeOccupancy();
  1642→      if (privacyOn) {
  1643→        for (const b of Object.values(world.blocks)) {
  1644→          if (b.kind !== 'dwelling') continue;

  1648→      // LUCES INTERIORES NOCTURNAS (v1.21): al caer el sol, las estancias que se ven abiertas
  1649→      // (alguien dentro) se ILUMINAN salvo que su inquilino duerma — entonces, a oscuras.

  1652→        const day = env?.daylight ?? 1;
  1653→        const litWorld = day < nightCfg.dayCutoff;
  1654→        // Apliques de las puertas (v1.24): foco cálido al anochecer, apagados de día.
  1655→        const lampGlow = litWorld ? 1.0 : 0.06;
  1656→        for (const { mat: lm, hex } of doorLampMats) lm.emissiveColor = Color3.FromHexString(hex).scale(lampGlow);
  1657→        // PUERTAS DE ESCALERA (v1.24): se abren antes de que alguien entre o salga (cualquier
  1658→        // mover cerca del umbral) y la ILUMINACIÓN COMPARTIDA de TODOS los cubos se enciende
  1659→        // de noche si algún vecino se mueve por las escaleras (un solo nivel para todos).
  1660→        const movers = env?.movers ?? [];
  1661→        const al = v.stairs.activityLight;
  1662→        for (const d of stairDoors) {
  1663→          let near = false;
  1664→          for (const mv of movers) {
  1665→            if (
  1666→              Math.abs(mv.x - d.doorX) < 0.85 &&
  1667→              Math.abs(mv.z - d.doorZ) < stF.openAhead &&
  1668→              Math.abs(mv.y - d.floorY) < v.cellH * 0.6
  1669→            ) {

  1672→            }

  1674→          const next = d.openT + (near ? dt : -dt * 0.8) * stF.openSpeed;
  1675→          d.openT = Math.min(1, Math.max(0, next));
  1676→          if (d.openT !== (near ? 1 : 0) || d.hinge.rotation.y !== 0) {
  1677→            // Bisagras a la izquierda (vista pasillo = x mayor): girar +Y mete la hoja

  1684→          let busy = false;
  1685→          if (litWorld) {
  1686→            for (const mv of movers) {
  1687→              for (const fp of stairFootprints) {

  1689→                  mv.x >= fp.x0 &&
  1690→                  mv.x <= fp.x1 &&
  1691→                  mv.z > frontZ + 0.3 &&
  1692→                  mv.z < frontZ + v.commonDepth - 0.1 &&
  1693→                  Math.abs(mv.y - fp.floorY) < v.cellH * 0.6
  1694→                ) {

  1696→                  break;
  1697→                }
  1698→              }
  1699→              if (busy) break;
  1700→            }
  1701→          }
  1702→          const next = stairLightLevel + (busy ? dt : -dt) / Math.max(0.4, al.fade);
  1703→          stairLightLevel = Math.min(1, Math.max(0, next));
