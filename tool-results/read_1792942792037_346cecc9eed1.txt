    33→    33→  /** Pared del fondo de la vivienda (la única malla que puede llevar edgesRendering) o null;
    34→    34→   *  el EdgesRenderer activo solo existe en las paredes destacadas del jugador. */
    35→    35→  edgesMesh: Mesh | null;
    36→    36→  /** true mientras el LOD fuerza baseEmissive (al soltarlo se repinta el cálculo de paint una vez). */
    37→    37→  glowForced: boolean;
    38→    38→  /** true mientras el LOD oculta panel+candado; la visibilidad efectiva es la de paint() && !esto. */
    39→    39→  lodHiddenPanels: boolean;
    40→    40→}
    41→    41→
    42→    42→/** Interior 2D de una vivienda (Fase C): suelo por estancia + tapa de puerta hacia estancia cerrada. */
    43→    43→interface InteriorView {
    44→    44→  /** Suelo tintado por estancia (color del catálogo; lockedColor si está cerrada). */
    45→    45→  floors: { roomKey: string; defId: string; mesh: Mesh; mat: StandardMaterial }[];
    46→    46→  /** Tapas de los huecos de puerta que llevan a una estancia cerrada ("en obras"). */
    47→    47→  covers: { mesh: Mesh; rooms: [string, string] }[];
    48→    48→}
    49→    49→
    50→    50→/**
    51→    51→ * Una caja por bloque (placeholder greybox del futuro prefab) y, si el manifiesto de assets
    52→    52→ * (Content.assets.blocks) declara un modelo glTF para el tipo de bloque, el modelo en su lugar
    53→    53→ * (con la caja oculta). TODA la geometría y los colores salen de content.rules.view +
    54→    54→ * block.view + assets: sustituir el greybox por arte final no toca este código.
    55→    55→ * Las viviendas se dibujan en corte (interior visible) SOLO mientras hay alguien dentro;
    56→    56→ * vacías, la fachada se cierra con su puerta al pasillo (rules.view.privacy) — y las
    57→    57→ * bloqueadas, con panel y candado como siempre.
    58→    58→ */
    59→    59→export function createBuilding(scene: Scene, world: World, content: Content) {
    60→    60→  const { Color3, Color4, MeshBuilder, StandardMaterial } = B();
    61→    61→  const v = content.rules.view;
    62→    62→  const glow = Object.fromEntries(Object.entries(v.channelGlow).map(([k, hex]) => [k, Color3.FromHexString(hex)])) as Record<string, Color3>;
    63→    63→  const views = new Map<string, BlockView>();
    64→    64→  const shakes = new Map<string, number>();
    65→    65→  /** Interiores 2D (Fase C) por bloque: suelos de estancia y tapas de puertas cerradas. */
    66→    66→  const interiors = new Map<string, InteriorView>();
    67→    67→  /** Foco del LOD (posición del jugador): lo alimenta update() y lo consume paint(). */
    68→    68→  let lastFocus: { x: number; floor: number } | null = null;
    69→    69→  // Plano frontal de las estancias: la cara de la puerta. Con viviendas del doble de calado
    70→    70→  // (v0.6-b), el extra crece hacia el FONDO (+Z) para no envolver el pasillo ni al núcleo.
    71→    71→  const frontZ = v.roomZ - v.commonDepth / 2;
    72→    72→  /** Calado por bloque: viviendas 2× (mobiliario interior); naves del cierre (v1.16) y el
    73→    73→   *  tejado (con aleros) a calado de vivienda; bloques comunes, el de siempre. */
    74→    74→  const depthOf = (b: BlockInstance): number => {
    75→    75→    if (b.kind === 'dwelling') return v.depth;
    76→    76→    if (b.defId === 'trasteros_planta' || b.defId === 'bajocubierta') return v.depth;
    77→    77→    if (b.defId === v.trasteros.blockDefId) return v.depth * v.trasteros.blockDepthFactor;
    78→    78→    if (b.defId === 'tejado') return v.depth + v.roof.overhang * 2;
    79→    79→    return v.commonDepth;
    80→    80→  };
    81→    81→  const mat = (name: string) => new StandardMaterial(name, scene);
    82→    82→  const box = (name: string, w: number, h: number, d: number, x: number, y: number, z: number, m: StandardMaterial) => {
    83→    83→    const mesh = MeshBuilder.CreateBox(name, { width: w, height: h, depth: d }, scene);
    84→    84→    mesh.position.set(x, y, z);
    85→    85→    mesh.material = m;
    86→    86→    return mesh;
    87→    87→  };
    88→    88→
    89→    89→  for (const b of Object.values(world.blocks)) {
    90→    90→    const m = mat(`${b.id}_mat`);
    91→    91→    const cx = (b.cell.col + b.size.w / 2) * v.cellW;
    92→    92→    const bottom = b.cell.floor * v.cellH;
    93→    93→    const w = b.size.w * v.cellW - v.gapW;
    94→    94→    const h = v.cellH - v.gapH;
    95→    95→    const d = depthOf(b);
    96→    96→    const cz = frontZ + d / 2;
    97→    97→    const view: BlockView = {
    98→    98→      greybox: [],
    99→    99→      mats: [m],
   100→   100→      baseEmissive: [m.emissiveColor.clone()],
   101→   101→      baseColor: [Color3.FromHexString(content.blocks[b.defId]!.view?.color ?? '#cccccc')],
   102→   102→      tintStrength: 1,
   103→   103→      model: null,
   104→   104→      front: null,
   105→   105→      door: null,
   106→   106→      handle: null,
   107→   107→      lock: null,
   108→   108→      lockBase: { x: cx, y: bottom + v.cellH / 2 - 0.2 },
   109→   109→      centerX: cx,
   110→   110→      floor: b.cell.floor,
   111→   111→      edgesMesh: null,
   112→   112→      glowForced: false,
   113→   113→      lodHiddenPanels: false,
   114→   114→    };
   115→   115→    if (b.layer === 'corridor') {
   116→   116→      // Pasillo exterior (data-driven): tira transitable saliente hacia la calle, con el borde
   117→   117→      // trasero a ras de fachada (frontZ) y barandilla baja en el borde exterior — sitio de
   118→   118→      // sobra para que los personajes se crucen o charlen sin bailar el vacío.
   119→   119→      const d = v.corridorDepth;
   120→   120→      view.greybox.push(box(b.id, b.size.w * v.cellW, v.slab * 2, d, cx, bottom + v.slab, frontZ - d / 2, m));
   121→   121→      if (v.corridorRail > 0) {
   122→   122→        view.greybox.push(
   123→   123→          box(`${b.id}_barandilla`, b.size.w * v.cellW, v.corridorRail, v.wall, cx, bottom + v.slab * 2 + v.corridorRail / 2, frontZ - d, m),
   124→   124→        );
   125→   125→      }
   126→   126→    } else if (b.kind === 'dwelling') {
   127→   127→      const y = bottom + v.cellH / 2;
   128→   128→      view.edgesMesh = box(`${b.id}_fondo`, w, h, v.wall, cx, y, frontZ + d, m);
   129→   129→      view.greybox.push(
   130→   130→        view.edgesMesh,
   131→   131→        box(`${b.id}_izq`, v.wall, h, d, cx - w / 2, y, cz, m),
   132→   132→        box(`${b.id}_der`, v.wall, h, d, cx + w / 2, y, cz, m),
   133→   133→        box(`${b.id}_suelo`, w, v.wall, d, cx, bottom + v.slab + v.wall / 2, cz, m),
   134→   134→        box(`${b.id}_techo`, w, v.wall, d, cx, bottom + v.cellH - v.slab - v.wall / 2, cz, m),
   135→   135→      );
   136→   136→      view.front = box(`${b.id}_puerta`, w, h, v.wall, cx, y, frontZ, m);
   137→   137→      view.lock = box(`${b.id}_candado`, 0.3, 0.38, 0.12, cx, view.lockBase.y, frontZ - 0.1, mat(`${b.id}_candado_mat`));
   138→   138→      (view.lock.material as StandardMaterial).diffuseColor = Color3.FromHexString(v.lockColor);
   139→   139→
   140→   140→      // ── Privacidad (v1.17): puerta y picaporte sobre la fachada de la vivienda vacía.
   141→   141→      // Se activan/desactivan cada frame según haya alguien dentro (propietario, visita…
   142→   142→      // o un ladrón); medidas y colores del catálogo (rules.view.privacy).
   143→   143→      const pv = v.privacy;
   144→   144→      view.door = box(
   145→   145→        `${b.id}_puerta_priv`,
   146→   146→        pv.doorW,
   147→   147→        pv.doorH,
   148→   148→        0.06,
   149→   149→        cx,
   150→   150→        bottom + v.slab + v.wall + pv.doorH / 2,
   151→   151→        frontZ - v.wall / 2 - 0.035,
   152→   152→        mat(`${b.id}_puerta_priv_mat`),
   153→   153→      );
   154→   154→      (view.door.material as StandardMaterial).diffuseColor = Color3.FromHexString(pv.doorColor);
   155→   155→      view.handle = box(
   156→   156→        `${b.id}_picaporte`,
   157→   157→        0.07,
   158→   158→        0.07,
   159→   159→        0.05,
   160→   160→        cx + pv.doorW / 2 - 0.14,
   161→   161→        bottom + v.slab + v.wall + pv.doorH * 0.55,
   162→   162→        frontZ - v.wall / 2 - 0.07,
   163→   163→        mat(`${b.id}_picaporte_mat`),
   164→   164→      );
   165→   165→      (view.handle.material as StandardMaterial).diffuseColor = Color3.FromHexString(pv.handleColor);
   166→   166→      view.door.setEnabled(false);
   167→   167→      view.handle.setEnabled(false);
   168→   168→
   169→   169→      // ── Cartel numerado (v1.6): rótulo sobre la puerta con el identificador de la vivienda
   170→   170→      // (blk_p3_c → "3C", blockLabel). Textura dinámica con colores/fuente del catálogo
   171→   171→      // (rules.view.plaque); se añade a greybox porque la vivienda usa modelo OVERLAY (el
   172→   172→      // greybox no se oculta) y no entra en mats: es señalización, no tinte ni glow.
   173→   173→      const pl = v.plaque;
   174→   174→      const plate = MeshBuilder.CreatePlane(`${b.id}_cartel`, { width: pl.w, height: pl.h, sideOrientation: B().Mesh.DOUBLESIDE }, scene);
   175→   175→      const tex = new (B().DynamicTexture)(`${b.id}_cartel_tex`, { width: 256, height: 96 }, scene, true);
   176→   176→      const ctx = tex.getContext() as unknown as CanvasRenderingContext2D;
   177→   177→      ctx.fillStyle = pl.bg;
   178→   178→      ctx.fillRect(0, 0, 256, 96);
   179→   179→      ctx.font = `bold ${pl.fontPx}px Arial, sans-serif`;
   180→   180→      ctx.fillStyle = pl.text;
   181→   181→      ctx.textAlign = 'center';
   182→   182→      ctx.textBaseline = 'middle';
   183→   183→      ctx.fillText(blockLabel(b.id), 128, 52);
   184→   184→      // update(true): el flag es invertY. Con false el canvas se sube SIN voltear y el texto
   185→   185→      // sale CABEZA ABAJO (la "A" se ve como "V"; B/C/D pasan por simétricas y esquivan el ojo).
   186→   186→      tex.update(true);
   187→   187→      const pmat = new StandardMaterial(`${b.id}_cartel_mat`, scene);
   188→   188→      pmat.diffuseTexture = tex;
   189→   189→      pmat.emissiveColor = Color3.FromHexString(pl.bg).scale(0.55); // legible en fachada norte
   190→   190→      pmat.specularColor = new Color3(0, 0, 0);
   191→   191→      plate.material = pmat;
   192→   192→      plate.position.set(cx, bottom + pl.y, frontZ + pl.z);
   193→   193→      view.greybox.push(plate);
   194→   194→
   195→   195→      // ── Interior 2D (Fase C): tabiques con huecos, suelo por estancia y tapas "en obras". ──
   196→   196→      // TODO sale del layout del catálogo (rects + doorways); sin rects no hay interior 2D.
   197→   197→      const layoutId = content.blocks[b.defId]?.container?.layout;
   198→   198→      const layout = layoutId ? content.layouts[layoutId] : undefined;
   199→   199→      const geoRooms = (layout?.rooms ?? []).filter((r) => r.rect !== undefined);
   200→   200→      if (geoRooms.length > 0 && layout) {
   201→   201→        const interior: InteriorView = { floors: [], covers: [] };
   202→   202→        const x0 = b.cell.col * v.cellW;
   203→   203→        const wallH = v.cellH - 2 * v.slab - 2 * v.wall;
   204→   204→        const partMat = mat(`${b.id}_tabique_mat`);
   205→   205→        partMat.diffuseColor = Color3.FromHexString('#d8d2c4');
   206→   206→        // Suelo tintado por estancia (plancha fina sobre el suelo, alpha suave).
   207→   207→        for (const lr of geoRooms) {
   208→   208→          const r = lr.rect!;
   209→   209→          const fmat = mat(`${b.id}_${lr.key}_suelo_mat`);
   210→   210→          fmat.diffuseColor = Color3.FromHexString(content.rooms[lr.defId]?.view?.color ?? '#cccccc');
   211→   211→          fmat.alpha = 0.32;
   212→   212→          const fm = box(`${b.id}_${lr.key}_suelo`, r.w, 0.02, r.d, x0 + r.x + r.w / 2, bottom + v.slab + v.wall + 0.02, frontZ + r.z + r.d / 2, fmat);
   213→   213→          interior.floors.push({ roomKey: lr.key, defId: lr.defId, mesh: fm, mat: fmat });
   214→   214→        }
   215→   215→        // Tabiques: para cada par de estancias que comparten borde, pared entre ambas.
   216→   216→        // Los doorways del catálogo recortan el hueco (y añaden tapa visible si alguna de las
   217→   217→        // dos estancias está cerrada). Sin hueco declarado → tabique sólido (tampoco se puede
   218→   218→        // pasar: walkable() exige doorway). Tabiques según X a media altura para no tapar la vista.
   219→   219→        const eps = 0.01;
   220→   220→        const seg = (a: number, b: number, along: 'x' | 'z', at: number, height: number, yC: number, name: string) => {
   221→   221→          const len = b - a;
   222→   222→          if (len <= 0.01) return;
   223→   223→          view.greybox.push(
   224→   224→            along === 'z'
   225→   225→              ? box(name, v.wall, height, len, at, yC, (a + b) / 2, partMat)
   226→   226→              : box(name, len, height, v.wall, (a + b) / 2, yC, at, partMat),
   227→   227→          );
   228→   228→        };
   229→   229→        geoRooms.forEach((lr, i) => {
   230→   230→          geoRooms.forEach((other, j) => {
   231→   231→            if (j <= i) return;
   232→   232→            const ra = lr.rect!;
   233→   233→            const rb = other.rect!;
   234→   234→            const door = (layout.doorways ?? []).find(
   235→   235→              (dw) => (dw.a === lr.key && dw.b === other.key) || (dw.a === other.key && dw.b === lr.key),
   236→   236→            );
   237→   237→            // Borde vertical compartido → tabique que corre según Z (se cruza andando en X).
   238→   238→            const touchX = Math.abs(ra.x + ra.w - rb.x) < eps || Math.abs(rb.x + rb.w - ra.x) < eps;
   239→   239→            const overlapZ = Math.min(ra.z + ra.d, rb.z + rb.d) - Math.max(ra.z, rb.z);
   240→   240→            if (touchX && overlapZ > eps) {
   241→   241→              const wx = x0 + Math.max(ra.x, rb.x);
   242→   242→              const z0 = frontZ + Math.max(ra.z, rb.z);
   243→   243→              const z1 = frontZ + Math.min(ra.z + ra.d, rb.z + rb.d);
   244→   244→              const yC = bottom + v.slab + v.wall + wallH / 2;
   245→   245→              if (door && door.axis === 'z') {
   246→   246→                const g0 = Math.max(z0, frontZ + door.at - door.width / 2);
   247→   247→                const g1 = Math.min(z1, frontZ + door.at + door.width / 2);
   248→   248→                seg(z0, g0, 'z', wx, wallH, yC, `${b.id}_tab_${i}_${j}_i`);
   249→   249→                seg(g1, z1, 'z', wx, wallH, yC, `${b.id}_tab_${i}_${j}_d`);
   250→   250→                const cover = box(`${b.id}_tapa_${i}_${j}`, v.wall * 0.8, wallH, g1 - g0, wx, yC, frontZ + door.at, mat(`${b.id}_tapa_${i}_${j}_mat`));
   251→   251→                (cover.material as StandardMaterial).diffuseColor = Color3.FromHexString(v.lockedColor);
   252→   252→                cover.setEnabled(false);
   253→   253→                interior.covers.push({ mesh: cover, rooms: [lr.key, other.key] });
   254→   254→              } else {
   255→   255→                seg(z0, z1, 'z', wx, wallH, yC, `${b.id}_tab_${i}_${j}`);
   256→   256→              }
   257→   257→            }
   258→   258→            // Borde horizontal compartido → tabique que corre según X (se cruza andando en Z).
   259→   259→            const touchZ = Math.abs(ra.z + ra.d - rb.z) < eps || Math.abs(rb.z + rb.d - ra.z) < eps;
   260→   260→            const overlapX = Math.min(ra.x + ra.w, rb.x + rb.w) - Math.max(ra.x, rb.x);
   261→   261→            if (touchZ && overlapX > eps) {
   262→   262→              const wz = frontZ + Math.max(ra.z, rb.z);
   263→   263→              const xa = x0 + Math.max(ra.x, rb.x);
   264→   264→              const xb = x0 + Math.min(ra.x + ra.w, rb.x + rb.w);
   265→   265→              const halfH = wallH * 0.45;
   266→   266→              const yHalf = bottom + v.slab + v.wall + halfH / 2;
   267→   267→              if (door && door.axis === 'x') {
   268→   268→                const g0 = Math.max(xa, x0 + door.at - door.width / 2);
   269→   269→                const g1 = Math.min(xb, x0 + door.at + door.width / 2);
   270→   270→                seg(xa, g0, 'x', wz, halfH, yHalf, `${b.id}_tab_${i}_${j}_i`);
   271→   271→                seg(g1, xb, 'x', wz, halfH, yHalf, `${b.id}_tab_${i}_${j}_d`);
   272→   272→                const cover = box(`${b.id}_tapa_${i}_${j}`, g1 - g0, halfH, v.wall * 0.8, x0 + door.at, yHalf, wz, mat(`${b.id}_tapa_${i}_${j}_mat`));
   273→   273→                (cover.material as StandardMaterial).diffuseColor = Color3.FromHexString(v.lockedColor);
   274→   274→                cover.setEnabled(false);
   275→   275→                interior.covers.push({ mesh: cover, rooms: [lr.key, other.key] });
   276→   276→              } else {
   277→   277→                seg(xa, xb, 'x', wz, halfH, yHalf, `${b.id}_tab_${i}_${j}`);
   278→   278→              }
   279→   279→            }
   280→   280→          });
   281→   281→        });
   282→   282→        interiors.set(b.id, interior);
   283→   283→      }
   284→   284→    } else if (b.defId === 'trasteros_planta' || b.defId === 'bajocubierta' || b.defId === v.trasteros.blockDefId) {
   285→   285→      // Nave de trasteros / sala de instalaciones (v1.16): forjado + trasera. El arte glTF
   286→   286→      // añade pilares (trasteros) o maquinaria (bajocubierta); las estanterías, trastos y
   287→   287→      // carteles numerados los levanta SIEMPRE el runtime desde rules.view.trasteros.
   288→   288→      // El bloque por COLUMNA (v1.18) comparte estructura: sus compartimentos los levanta
   289→   289→      // el runtime más abajo (pasillo central + celdas cerradas impares/pares).
   290→   290→      view.greybox.push(
   291→   291→        box(`${b.id}_forjado`, w, v.slab * 2, d, cx, bottom + v.slab, frontZ + d / 2, m),
   292→   292→        box(`${b.id}_trasera`, w, h, v.wall, cx, bottom + v.cellH / 2, frontZ + d - v.wall / 2, m),
   293→   293→      );
   294→   294→    } else if (b.defId === 'tejado') {
   295→   295→      // Tejadillo (v1.16): fallback = canto de cerramiento (el glTF trae faldones, cumbrera,
   296→   296→      // chimenea y antena).
   297→   297→      view.greybox.push(box(`${b.id}_canto`, w, v.slab * 2, d, cx, bottom + v.slab, frontZ + d / 2, m));
   298→   298→    } else {
   299→   299→      view.greybox.push(box(b.id, w, h, d, cx, bottom + v.cellH / 2, cz, m));
   300→   300→    }
   301→   301→    views.set(b.id, view);
   302→   302→  }
   303→   303→
   304→   304→  // ── Forjado interior CONTINUO del cubo de escalera (v1.18): a nivel de entrada de cada
   305→   305→  //  planta, una losa que cubre la huella completa del cubo (spanW celdas). La authoría glTF
   306→   306→  //  solo trae el forjado de la puerta; sin esta losa se camina (y se ve el vacío) entre la
   307→   307→  //  puerta y los tramos. Malla propia que SOBREVIVE a la carga del modelo (no está en el
   308→   308→  //  greybox, que se oculta) — 5 mm por debajo del forjado del arte: cero z-fighting.
   309→   309→  const stCfg = v.stairs.floorSlab;
   310→   310→  if (stCfg.enabled) {
   311→   311→    const stMat = mat('escalera_forjado_mat');
   312→   312→    stMat.diffuseColor = Color3.FromHexString(stCfg.color);
   313→   313→    for (const b of Object.values(world.blocks)) {
   314→   314→      const f = content.blocks[b.defId]!.features;
   315→   315→      if (!f || f.stairs !== true) continue;
   316→   316→      const spanW = v.stairs.spanW;
   317→   317→      const sw = spanW * v.cellW - v.gapW;
   318→   318→      const sx = b.cell.col * v.cellW + sw / 2;
   319→   319→      box(`${b.id}_forjado_interior`, sw, stCfg.thickness, v.commonDepth, sx, b.cell.floor * v.cellH + stCfg.topAt - stCfg.thickness / 2 - 0.005, frontZ + v.commonDepth / 2, stMat);
   320→   320→    }
   321→   321→  }
   322→   322→
   323→   323→  // ── Arte final (opcional): modelo glTF por TIPO de bloque (manifiesto assets.blocks). ──
   324→   324→  // Cada instancia se clona del contenedor cacheado y escala a la celda real; si carga, la
   325→   325→  // caja greybox estructural se oculta (el panel/candado de viviendas sigue siendo greybox).
   326→   326→  const blockAssets = content.assets?.blocks;
   327→   327→  if (blockAssets && content.assets) {
   328→   328→    for (const b of Object.values(world.blocks)) {
   329→   329→      const entry = blockAssets[b.defId];
   330→   330→      if (!entry) continue;
   331→   331→      void instantiateBlockAsset(scene, entry, content.assets.base).then((model) => {
