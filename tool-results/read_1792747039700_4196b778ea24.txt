    64→    64→      preloadSounds();

    68→    68→      }

    70→    70→      // 2) Motor de render (WebGPU con fallback WebGL2).

    73→    73→      window.addEventListener('resize', this.onResize);

   299→   299→  /** (Re)crea el mundo y la escena. Con `state` restaura una partida guardada. */
   300→   300→  private async restart(state?: World): Promise<void> {
   301→   301→    if (!this.engine || !this.canvas || !this._content) return;
   302→   302→    const { Color4, DirectionalLight, HemisphericLight, Scene, UniversalCamera, Vector3 } = B();
   303→   303→    const view = this.content.rules.view;
   304→   304→    useGameStore.getState().resetUi();
   305→   305→
   306→   306→    this.world = state ?? createWorld(this.content, this.activeScenarioId());
   307→   307→    this.scenarioOverrides = this.content.scenarios[this.world.scenarioId ?? '']?.minigameOverrides ?? {};
   308→   308→    // Guardados antiguos sin `entry`: la entrada es la zona común de planta baja.
   309→   309→    if (!this.world.entry) {
   310→   310→      this.world.entry = Object.values(this.world.blocks).find((b) => b.cell.floor === 0 && b.kind !== 'dwelling')?.id ?? Object.keys(this.world.blocks)[0]!;
   311→   311→    }
   312→   312→    // Migración de estancias de guardados v0.1: params y ranuras de muebles según catálogo.
   313→   313→    for (const b of Object.values(this.world.blocks)) migrateRooms(this.content, b.rooms);
   314→   314→    this.logic = new PlayerLogic(this.world, this.content);
   315→   315→
   316→   316→    this.scene?.dispose();
   317→   317→    this.scene = new Scene(this.engine);
   318→   318→    this.scene.clearColor = new Color4(view.sky.r, view.sky.g, view.sky.b, 1);
   319→   319→    new HemisphericLight('ambient', new Vector3(...view.ambient.dir), this.scene).intensity = view.ambient.intensity;
   320→   320→    new DirectionalLight('sun', new Vector3(...view.sun.dir), this.scene).intensity = view.sun.intensity;
   321→   321→
   322→   322→    this.building = createBuilding(this.scene, this.world, this.content);
   323→   323→    this.actors = createActors(this.scene, this.world, this.content);
   324→   324→    const blocks = Object.values(this.world.blocks);
   325→   325→    this.rig = new CameraRig(
   326→   326→      new UniversalCamera('cam', new Vector3(0, 0, -30), this.scene),
   327→   327→      this.engine,
   328→   328→      Math.max(...blocks.map((b) => b.cell.col + b.size.w)),
   329→   329→      Math.max(...blocks.map((b) => b.cell.floor)) + 1,
   330→   330→      view.cellW,
   331→   331→      view.cellH,
   332→   332→    );
   333→   333→    this.rig.mode = useGameStore.getState().camMode;
   334→   334→
   335→   335→    this.clockAcc = 0;
   336→   336→    this.keys.clear();
   337→   337→    this.phoneLines = [];
   338→   338→    this.phoneMission = '';
   339→   339→    this.lastHudSync = 0;
   340→   340→    this.pump();
   341→   341→  }

   349→   349→  private frame = (): void => {
   350→   350→    if (!this.engine || !this.scene || !this.world || !this.logic || !this.building || !this.actors || !this.rig) return;
   351→   351→    const dt = Math.min(0.1, this.engine.getDeltaTime() / 1000);
   352→   352→    const store = useGameStore.getState();
   353→   353→    const frozen = selectModalOpen(store) || store.paused;
   354→   354→
   355→   355→    let ticked = false;
   356→   356→    if (!frozen) {
   357→   357→      this.logic.update(dt, this.readInput());
   358→   358→      this.clockAcc += dt * store.speed;
   359→   359→      let guard = 0;
   360→   360→      while (this.clockAcc >= 1 && guard < 60) {
   361→   361→        tick(this.world, this.content);
   362→   362→        this.clockAcc -= 1;
   363→   363→        ticked = true;
   364→   364→        guard += 1;
   365→   365→      }
   366→   366→    }
   367→   367→
   368→   368→    if (this.world.events.length > 0) this.handleEvents();
   369→   369→    if (ticked) this.paintBuilding();
   370→   370→
   371→   371→    const now = performance.now();
   372→   372→    if (now - this.lastHudSync > 200) {
   373→   373→      this.lastHudSync = now;
   374→   374→      this.syncHud();
   375→   375→    }
   376→   376→
   377→   377→    const pos = this.logic.position();
   378→   378→    this.actors.update(dt, pos);
   379→   379→    this.building.update(dt);
   380→   380→    this.rig.update(dt, pos);
   381→   381→    this.scene.render();
   382→   382→  };

   413→   413→  private handleEvents(): void {
   414→   414→    if (!this.world) return;
   415→   415→    const store = useGameStore.getState();
   416→   416→    for (const e of drainEvents(this.world)) {
   417→   417→      const d = e.data as Record<string, unknown>;
   418→   418→      eventSound(e.type, typeof d.type === 'string' ? d.type : undefined);
   419→   419→      switch (e.type) {
   420→   420→        case 'step_started':
   421→   421→          if (d.type === 'phone_call') this.openPhone(String(d.mission), String(d.dialog));
   422→   422→          break;
   423→   423→        case 'choice_required': {
   424→   424→          const choices = (d.choices as { id: string; label: string; hint?: string | null }[] | undefined) ?? [];
   425→   425→          store.setChoices({ mission: String(d.mission), title: String(d.title), choices: choices.map((c) => ({ id: c.id, label: c.label, hint: c.hint ?? null })) });
   426→   426→          store.pushLog(`⚖️ Decisión pendiente: ${String(d.title)}`, 'info');
   427→   427→          break;
   428→   428→        }
   429→   429→        case 'door_locked': {
   430→   430→          this.building?.shake(String(d.block));
   431→   431→          const price = typeof d.price === 'number' ? d.price : null;
   432→   432→          store.pushToast(`${blockLabel(String(d.block))}: puerta con candado${d.reason === 'for_sale' && price !== null ? ` (a la venta por ${price} €)` : ''}`, 'info');
   433→   433→          break;
   434→   434→        }
   435→   435→        case 'neighbor_arrived':
   436→   436→          store.pushToast(`Nuevo vecino en ${blockLabel(String(d.block))}: ${this.nameOf(d.resident)}`, 'good');
   437→   437→          store.pushLog(`${this.nameOf(d.resident)} se muda a ${blockLabel(String(d.block))}. El edificio ya no es lo que era. Era mejor.`, 'info');
   438→   438→          break;
   439→   439→        case 'mission_activated':
   440→   440→          store.pushLog(`📋 Nueva misión: ${String(d.title)}`, 'info');
   441→   441→          break;
   442→   442→        case 'mission_completed':
   443→   443→          store.pushToast(`Misión completada: ${this.titleOf(d.mission)}`, 'good');
   444→   444→          store.pushLog(`✔ Misión completada: ${this.titleOf(d.mission)}`, 'good');
   445→   445→          break;
   446→   446→        case 'mission_failed':
   447→   447→          store.pushToast(`Misión fallida: ${this.titleOf(d.mission)}`, 'bad');
   448→   448→          store.pushLog(`✘ Misión fallida: ${this.titleOf(d.mission)}`, 'bad');
   449→   449→          break;
   450→   450→        case 'block_unlocked':
   451→   451→          store.pushToast(`${blockLabel(String(d.block))} desbloqueado`, 'good');
   452→   452→          break;
   453→   453→        case 'block_bought':
   454→   454→          store.pushLog(`🔑 ${blockLabel(String(d.block))} comprado por ${String(d.price)} €`, 'good');
   455→   455→          break;
   456→   456→        case 'block_for_sale':
   457→   457→          store.pushLog(`🪧 Cartel de "SE VENDE" en ${blockLabel(String(d.block))} (${String(d.price)} €)`, 'info');
   458→   458→          break;
   459→   459→        case 'noise_complaint': // texto data-driven: lo pinta el uiLog del catálogo (case mantenido por compatibilidad de eventos antiguos)
   460→   460→          if (d.uiLog === true) break;
   461→   461→          store.pushLog(`🔊 Ruido por encima del límite en ${blockLabel(String(d.block))}. Alguien aprieta los dientes.`, 'bad');
   462→   462→          break;
   463→   463→        case 'leak_detected':
   464→   464→          if (d.uiLog === true) break;
   465→   465→          store.pushLog(`💧 Goteo detectado en ${blockLabel(String(d.block))}. El cubo del salón hace ojitos.`, 'bad');
   466→   466→          break;
   467→   467→        case 'furniture_placed':
   468→   468→          store.pushLog(`${String(d.emoji ?? '📦')} ${String(d.name)} colocado en ${blockLabel(String(d.block))} (${String(d.room)}).`, 'info');
   469→   469→          break;
   470→   470→        case 'furniture_removed':
   471→   471→          store.pushLog(`${String(d.emoji ?? '📦')} ${String(d.name)} retirado de ${blockLabel(String(d.block))} (+${String(d.refund)} €).`, 'info');
   472→   472→          break;
   473→   473→        case 'room_renovated':
   474→   474→          store.pushLog(`🧱 Reforma terminada en ${blockLabel(String(d.block))}. Hay estancia nueva que amueblar.`, 'good');
   475→   475→          break;
   476→   476→        case 'weather_changed':
   477→   477→          store.pushLog(`El cielo cambia: ${this.content.rules.labels.weather[String(d.state)] ?? String(d.state)}`, 'info');
   478→   478→          break;
   479→   479→        case 'rent_collected':
   480→   480→          store.pushToast(`+${String(d.amount)} € de alquiler (${blockLabel(String(d.block))})`, 'good');
   481→   481→          break;
   482→   482→        case 'work_done':
   483→   483→          store.pushLog(`💻 Cerrada una "reunión que podía ser un correo": +${String(d.amount)} €`, 'info');
   484→   484→          break;
   485→   485→        case 'activity_started':
   486→   486→          if (d.activity !== 'idle') store.pushLog(`${this.nameOf(d.resident)} ${this.activityLabel(d.resident, String(d.activity))}`, 'info');
   487→   487→          break;
   488→   488→        case 'junta_requested': {
   489→   489→          const agenda = (d.agenda as { id: string; title: string; description: string; cost: number }[] | undefined) ?? [];
   490→   490→          store.setJunta(agenda.map((i) => ({ id: i.id, title: i.title, description: i.description, cost: i.cost })));
   491→   491→          break;
   492→   492→        }
   493→   493→        case 'junta_result':
   494→   494→          store.pushToast(
   495→   495→            d.approved ? `Aprobado: ${String(d.title)} (${String(d.yes)} sí / ${String(d.no)} no)` : `Rechazado: ${String(d.title)} (${String(d.yes)} sí / ${String(d.no)} no)`,
   496→   496→            d.approved ? ('good' as Tone) : ('bad' as Tone),
   497→   497→          );
   498→   498→          break;
   499→   499→        case 'minigame_requested': {
   500→   500→          const def = this.content.minigames[String(d.minigame)];
   501→   501→          if (!def) break;
   502→   502→          // Overrides numéricos del escenario activo (QA/E2E): solo config, nunca efectos ni textos.
   503→   503→          const ov = this.scenarioOverrides[def.id];
   504→   504→          const effective = ov ? ({ ...def, config: { ...def.config, ...ov } } as MinigameDef) : def;
   505→   505→          store.setMinigame({ ...effective, emoji: effective.emoji ?? '🎮', instruction: effective.instruction ?? null });
   506→   506→          break;
   507→   507→        }
   508→   508→        case 'minigame_finished': {
   509→   509→          const name = this.content.minigames[String(d.minigame)]?.name ?? String(d.minigame);
   510→   510→          store.pushLog(d.success ? `🎮 Minijuego superado: ${name}` : `🎮 Minijuego fallido: ${name}`, d.success ? 'good' : 'bad');
   511→   511→          break;
   512→   512→        }
   513→   513→        case 'ui_toast':
   514→   514→          store.pushToast(String(d.text), (d.tone as Tone | undefined) ?? 'info');
   515→   515→          break;
   516→   516→        case 'finale':
   517→   517→          store.setFinale(String(d.text));
   518→   518→          break;
   519→   519→        case 'funds_changed': {
   520→   520→          const amount = typeof d.amount === 'number' ? d.amount : 0;
   521→   521→          if ((d.to === 'player' || d.to === 'community') && amount !== 0) {
   522→   522→            store.pushLog(amount > 0 ? `+${amount} € (fondos)` : `${amount} € (fondos)`, amount > 0 ? 'good' : 'bad');
   523→   523→          }
   524→   524→          break;
   525→   525→        }
   526→   526→        default: {
   527→   527→          // Umbral de canal con texto del catálogo (odors, wifi y futuros canales): totalmente data-driven.
   528→   528→          if (d.uiLog === true && typeof d.message === 'string') {
   529→   529→            store.pushLog(String(d.message).replace('{block}', blockLabel(String(d.block))), (d.tone as Tone | undefined) ?? 'info');
   530→   530→          }
   531→   531→          break; // resident_moved, param_changed, activity_ended... demasiado ruido para el registro
   532→   532→        }
   533→   533→      }
   534→   534→    }
   535→   535→  }

   556→   556→  private objectiveUi(): { title: string; text: string } | null {
   557→   557→    if (!this.world) return null;
   558→   558→    const o = activeObjective(this.world, this.content);
   559→   559→    if (!o) return null;
   560→   560→    if (o.state === 'awaiting_choice') return { title: o.title, text: 'El edificio contiene la respiración. Decide qué hacer.' };
   561→   561→    const s = o.step!;
   562→   562→    const left = o.untilTick !== null ? ` · quedan ${Math.max(0, o.untilTick - this.world.tick)} min` : '';
   563→   563→    switch (s.type) {
   564→   564→      case 'phone_call':
   565→   565→        return { title: o.title, text: 'Suena el teléfono. Atiéndelo.' };
   566→   566→      case 'reach':
   567→   567→        return { title: o.title, text: `Llega a la puerta (${blockLabel(s.block ?? '')}).` };
   568→   568→      case 'enter':
   569→   569→        return { title: o.title, text: 'Entra en tu piso con E.' };
   570→   570→      case 'catch_in_act':
   571→   571→        return { title: o.title, text: `Pilla al responsable in fraganti${left}.` };
   572→   572→      default:
   573→   573→        return { title: o.title, text: o.title };
   574→   574→    }
   575→   575→  }
   576→   576→
   577→   577→  private promptText(): string | null {
   578→   578→    const logic = this.logic;
   579→   579→    const world = this.world;
   580→   580→    if (!logic || !world) return null;
   581→   581→    const p = logic.position();
   582→   582→    if (p.mode === 'inside') return 'E · salir a la escalera';
   583→   583→    if (p.climbing) return null;
   584→   584→    const door = logic.doorInFront();
   585→   585→    if (door) return door.access === 'locked' ? `E · ${blockLabel(door.id)}: candado` : `E · entrar en ${blockLabel(door.id)}`;
   586→   586→    const zone = logic.zone();
   587→   587→    if (this.content.blocks[zone.defId]!.features?.stairs) {
   588→   588→      return world.blocks[zone.id]!.params.elevator_status === 'ok' ? 'W/S · ascensor' : 'W/S · escalera (el ascensor no funciona)';
   589→   589→    }
   590→   590→    return null;
   591→   591→  }

   593→   593→  /** Viviendas del jugador para el editor de interiores (foto serializable, posiciones ya en %). */
   594→   594→  private homesUi() {
   595→   595→    const world = this.world!;
   596→   596→    const cellH = this.content.rules.view.cellH;
   597→   597→    return Object.values(world.blocks)
   598→   598→      .filter((b) => b.kind === 'dwelling' && b.ownerId === world.playerId)
   599→   599→      .sort((a, b) => a.id.localeCompare(b.id))
   600→   600→      .map((b) => ({
   601→   601→        id: b.id,
   602→   602→        label: blockLabel(b.id),
   603→   603→        rooms: b.rooms.map((room) => {
   604→   604→          const rdef = this.content.rooms[room.defId];
   605→   605→          const slots = (rdef?.furnitureSlots ?? []).map((s) => {
   606→   606→            const itemId = room.furniture[s.id] ?? '';
   607→   607→            const fdef = itemId ? this.content.furniture[itemId] : undefined;
   608→   608→            return {
   609→   609→              id: s.id,
   610→   610→              kind: s.kind,
   611→   611→              left: Math.min(92, Math.max(2, (s.pos[0] / Math.max(0.001, rdef?.size.w ?? 1)) * 100)),
   612→   612→              bottom: Math.min(76, Math.max(0, (s.pos[1] / Math.max(0.001, cellH)) * 100)),
   613→   613→              item: fdef ? { id: fdef.id, name: fdef.name, emoji: fdef.emoji ?? '📦', color: fdef.view.color } : null,
   614→   614→            };
   615→   615→          });
   616→   616→          return { key: room.key, label: rdef?.label ?? room.key, access: room.access, slots };
   617→   617→        }),
   618→   618→      }));
   619→   619→  }

   621→   621→  private syncHud(): void {
   622→   622→    const world = this.world;
   623→   623→    if (!world) return;
   624→   624→    const player = world.residents[world.playerId]!;
   625→   625→    const home = player.home;
   626→   626→    const params = visibleParams(world, this.content, home).map((p) => ({ name: p.name, label: p.def.label ?? p.name, value: String(p.value) }));
   627→   627→    const missions = world.missions.map((m) => {
   628→   628→      const def = this.content.missions[m.id]!;
   629→   629→      const step = m.state === 'active' ? def.steps[m.step] : undefined;
   630→   630→      return {
   631→   631→        id: m.id,
   632→   632→        title: def.title,
   633→   633→        state: m.state,
   634→   634→        hint: step?.hint ?? null,
   635→   635→        remaining: m.until !== null ? `quedan ${Math.max(0, m.until - world.tick)} min` : null,
   636→   636→      };
   637→   637→    });
   638→   638→    const travel = travelMinutes(buildCirculation(world, this.content), world.entry, home);
   639→   639→    const elevatorBlock = Object.values(world.blocks).find((b) => b.params.elevator_status !== undefined);
   640→   640→    useGameStore.getState().syncHud({
   641→   641→      clock: { ...world.clock },
   642→   642→      weather: world.weather.state,
   643→   643→      funds: world.community.funds,
   644→   644→      wallet: player.wallet,
   645→   645→      reputation: player.reputation,
   646→   646→      elevator: String(elevatorBlock?.params.elevator_status ?? 'ok'),
   647→   647→      homeLabel: blockLabel(home),
   648→   648→      travelText: Number.isFinite(travel) ? `del portal a tu puerta: ${travel} min` : null,
   649→   649→      objective: this.objectiveUi(),
   650→   650→      prompt: this.promptText(),
   651→   651→      params,
   652→   652→      missions,
   653→   653→      homes: this.homesUi(),
   654→   654→      furnitureConfig: { refundRatio: this.content.rules.furniture.refundRatio, roomUnlockPrice: this.content.rules.furniture.roomUnlock.price },
   655→   655→    });
   656→   656→  }
