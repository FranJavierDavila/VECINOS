    72→  async init(canvas: HTMLCanvasElement): Promise<void> {
    73→    if (this.engine || this.initializing) return;
    74→    this.initializing = true;
    75→    this.canvas = canvas;
    76→
    77→    try {
    78→      // 1) Contenido externo + validación (los errores de referencias se avisan, no abortan).
    79→      const { content, issues } = await loadContent();
    80→      this._content = content;
    81→      this.cutscenePlayer = new CutscenePlayer(content);
    82→      preloadSounds();
    83→      useGameStore.setState({ touchHint: content.rules.touch.hint });
    84→      if (issues.length > 0) {
    85→        useGameStore.getState().pushToast(`Contenido inválido: ${issues.length} errores de validación`, 'bad');
    86→      }
    87→
    88→      // 2) Motor de render (WebGPU con fallback WebGL2). La calidad la elige el jugador en Opciones.
    89→      const { engine, backend } = await createEngine(canvas);
    90→      this.engine = engine;
    91→      engine.setHardwareScalingLevel(QUALITY_SCALE[loadOptions().quality]);
    92→      window.addEventListener('resize', this.onResize);
    93→      await this.restart();
    94→      useGameStore.getState().markReady(backend);
    95→      startGamepadPolling();
    96→      engine.runRenderLoop(this.frame);
    97→      // Gancho de depuración/pruebas E2E (no lo usa la UI).
    98→      (window as unknown as { __vecinos?: unknown }).__vecinos = { store: useGameStore, controller: this };
    99→    } catch (err) {
   100→      console.error('No se pudo arrancar el juego (contenido o motor de render)', err);
   101→      useGameStore.getState().setEngineError(true);
   102→    }
   103→  }

   118→  onKeyDown(code: string): void {
   119→    // Cutscene en curso: E/Espacio/Enter/Escape saltan el espectáculo; nada más pasa.
   120→    if (this.cutscenePlayer?.active) {
   121→      if (code === 'KeyE' || code === 'Space' || code === 'Enter' || code === 'Escape') this.cutsceneSkip();
   122→      return;
   123→    }
   124→    if (this.keys.has(code)) return;
   125→    this.keys.add(code);
   126→    const store = useGameStore.getState();
   127→    if (selectModalOpen(store)) return; // los modales gestionan sus propias teclas
   128→    if (code === 'KeyE') {
   129→      this.interact();
   130→      return;
   131→    }
   132→    const cam = CAM_BY_KEY[code];
   133→    if (cam) this.setCamMode(cam);
   134→  }

   326→  private connectNet(): void {
   327→    this.net = new NetSession({
   328→      onState: (s) => this.applyRemoteState(s),
   329→      onParticipants: (p) => useGameStore.getState().setMpParticipants(p),
   330→      onJoined: (j) => {
   331→        this.myResidentId = j.yourResidentId;
   332→        this.isHost = j.isHost;
   333→        useGameStore.getState().setMpSession({ code: j.code, isHost: j.isHost, myResidentId: j.yourResidentId });
   334→        useGameStore.getState().setMpParticipants(j.participants);
   335→      },
   336→      onError: (m) => {
   337→        useGameStore.getState().pushToast(m, 'bad');
   338→      },
   339→      onStatus: (st) => {
   340→        if (!this.net) return;
   341→        if (st === 'connected' && useGameStore.getState().mpStatus === 'connecting') return; // el ack decide
   342→        if (st === 'disconnected' && useGameStore.getState().mpStatus === 'connected') {
   343→          useGameStore.getState().setMpStatus('error');
   344→          useGameStore.getState().setMpError('Conexión perdida con el servidor de la comunidad.');
   345→        }
   346→      },
   347→    });
   348→  }

   361→  /** Adopta el estado del servidor (autoridad). Primera vez: reinicia la escena con ese mundo. */
   362→  private applyRemoteState(s: SessionStatePayload): void {
   363→    if (s.seq <= this.appliedSeq) return;
   364→    this.appliedSeq = s.seq;
   365→    if (!this.world || this.world.scenarioId !== s.world.scenarioId) {
   366→      void this.restart(s.world).then(() => {
   367→        if (!this.world) return;
   368→        this.world.events = s.events.slice();
   369→        this.handleEvents();
   370→        this.reconcile();
   371→        this.syncHud();
   372→      });
   373→      return;
   374→    }
   375→    // Aplica en sitio conservando la identidad del objeto (PlayerLogic/building/actors referencian el mismo World).
   376→    Object.assign(this.world, s.world);
   377→    this.world.events = s.events.slice();
   378→    this.handleEvents();
   379→    this.paintBuilding();
   380→    this.reconcile();
   381→    this.syncHud();
   382→  }

   384→  /** Corrige desvíos de posición del avatar local respecto al mundo del servidor (autoridad). */
   385→  private reconcile(): void {
   386→    if (!this.logic || !this.world || !this.net) return;
   387→    const me = this.world.residents[this.actorId()];
   388→    if (!me) return;
   389→    const pos = this.logic.position();
   390→    const locBlock = me.location ? this.world.blocks[me.location] : null;
   391→    if (locBlock) {
   392→      const insideOk = pos.mode === 'inside' && pos.insideBlock === locBlock.id;
   393→      const corridorOk = pos.mode === 'corridor' && !pos.insideBlock;
   394→      if (!(locBlock.kind === 'dwelling' ? insideOk : corridorOk) || Math.abs(pos.floor - locBlock.cell.floor) > 0.01) {
   395→        this.logic.forceLocation(locBlock.id);
   396→      }
   397→    } else if (pos.mode === 'inside') {
   398→      this.logic.forceLocation(null);
   399→    }
   400→  }

   402→  /** Siguiente frase del teléfono; al colgar responde la llamada en el sim y saca la siguiente de la cola. */
   403→  phoneNext(): void {
   404→    if (this.phoneLine < this.phoneLines.length - 1) {
   405→      this.phoneLine += 1;
   406→      this.showPhoneLine();
   407→      return;
   408→    }
   409→    const mission = this.phoneMission;
   410→    this.phoneLines = [];
   411→    this.phoneMission = '';
   412→    useGameStore.getState().setPhone(null);
   413→    this.actors?.director.setGesture(this.actorId(), null); // cuelgas: se suelta el teléfono
   414→    if (this.world && mission) {
   415→      this.sendCommand({ type: 'answer_call', mission });
   416→    }
   417→    const next = this.phoneQueue.shift();
   418→    if (next) this.openPhone(next.mission, next.dialogId);
   419→  }

   487→  async save(): Promise<void> {
   488→    if (!this.world) return;
   489→    const store = useGameStore.getState();
   490→    if (this.net) {
   491→      store.pushToast(this.content.rules.multiplayer.texts.remoteSaveDisabled, 'info');
   492→      return;
   493→    }
   494→    store.setSaveState('saving');
   495→    drainEvents(this.world); // la cola de eventos no entra en el guardado
   496→    try {
   497→      const res = await fetch('/api/save', {
   498→        method: 'POST',
   499→        headers: { 'Content-Type': 'application/json' },
   500→        body: JSON.stringify({ slot: 'auto', state: this.world }),
   501→      });
   502→      if (!res.ok) throw new Error(String(res.status));
   503→      store.setSaveState('saved', new Date().toLocaleTimeString('es-ES', { hour: '2-digit', minute: '2-digit' }));
   504→      store.pushToast('Partida guardada. El acta queda en el tablón (y en SQLite).', 'good');
   505→    } catch {
   506→      store.setSaveState('error');
   507→      store.pushToast('No se pudo guardar la partida. ¿El administrador se habrá ido a comer?', 'bad');
   508→    }
   509→  }

   514→      store.pushToast(this.content.rules.multiplayer.texts.remoteSaveDisabled, 'info');
   515→      return;
   516→    }
   517→    try {
   518→      const res = await fetch('/api/save?slot=auto');
   519→      const json = (await res.json()) as { state?: unknown };
   520→      if (!json.state || typeof json.state !== 'object') {
   521→        store.pushToast('Todavía no hay ninguna partida guardada.', 'info');
   522→        return;

   526→        return;
   527→      }
   528→      await this.restart(json.state as World);
   529→      store.pushToast('Partida cargada. El vecindario, exactamente como lo dejaste.', 'good');
   530→    } catch {
   531→      store.pushToast('No se pudo cargar la partida.', 'bad');
   532→    }

   546→  /** (Re)crea el mundo y la escena. Con `state` restaura una partida guardada. */
   547→  private async restart(state?: World): Promise<void> {
   548→    if (!this.engine || !this.canvas || !this._content) return;
   549→    // Ninguna cutscene sobrevive a un cambio de mundo: cola fuera y teatro desmontado.
   550→    this.cutsceneQueue = [];
   551→    this.cutscenePlayer?.abort();
   552→    const { Color4, DirectionalLight, HemisphericLight, Scene, UniversalCamera, Vector3 } = B();
   553→    const view = this.content.rules.view;
   554→    useGameStore.getState().resetUi();
   555→
   556→    this.world = state ?? createWorld(this.content, this.activeScenarioId());
   557→    this.scenarioOverrides = this.content.scenarios[this.world.scenarioId ?? '']?.minigameOverrides ?? {};
   558→    // Guardados antiguos sin `entry`: la entrada es la zona común de planta baja.
   559→    if (!this.world.entry) {
   560→      this.world.entry = Object.values(this.world.blocks).find((b) => b.cell.floor === 0 && b.kind !== 'dwelling')?.id ?? Object.keys(this.world.blocks)[0]!;
   561→    }
   562→    // Migración de estancias de guardados v0.1: params y ranuras de muebles según catálogo.
   563→    for (const b of Object.values(this.world.blocks)) migrateRooms(this.content, b.rooms);
   564→    this.logic = new PlayerLogic(this.world, this.content, {
   565→      actorId: this.myResidentId ?? undefined,
   566→      send: (cmd) => this.sendCommand(cmd),
   567→    });
   568→
   569→    this.scene?.dispose();
   570→    this.scene = new Scene(this.engine);
   571→    this.scene.clearColor = new Color4(view.sky.r, view.sky.g, view.sky.b, 1);
   572→    new HemisphericLight('ambient', new Vector3(...view.ambient.dir), this.scene).intensity = view.ambient.intensity;
   573→    new DirectionalLight('sun', new Vector3(...view.sun.dir), this.scene).intensity = view.sun.intensity;
   574→
   575→    this.building = createBuilding(this.scene, this.world, this.content);
   576→    this.actors = createActors(this.scene, this.world, this.content);
   577→    const blocks = Object.values(this.world.blocks);
   578→    this.rig = new CameraRig(
   579→      new UniversalCamera('cam', new Vector3(0, 0, -30), this.scene),
   580→      this.engine,
   581→      Math.max(...blocks.map((b) => b.cell.col + b.size.w)),
   582→      Math.max(...blocks.map((b) => b.cell.floor)) + 1,
   583→      view.cellW,
   584→      view.cellH,
   585→    );
   586→    // Re-sincronización de vista (v0.6-b): la cámara sigue la posición del mundo (restaurado o
   587→    // nuevo), no la preferencia anterior. Dentro de una vivienda → estancia; en pasillo, la
   588→    // preferencia salvo que fuera 'estancia' (ahí estás en el pasillo, no en una estancia).
   589→    const pos = this.logic.position();
   590→    const store = useGameStore.getState();
   591→    const wanted: CamMode = pos.mode === 'inside' ? 'estancia' : store.camMode === 'estancia' ? 'planta' : store.camMode;
   592→    this.rig.mode = wanted;
   593→    store.setCamMode(wanted);
   594→
   595→    this.clockAcc = 0;
   596→    this.keys.clear();
   597→    this.phoneLines = [];
   598→    this.phoneMission = '';
   599→    this.phoneQueue = [];
   600→    this.lastHudSync = 0;
   601→
   602→    // Re-llamada (Fase F): si la partida montada tenía pasos de teléfono pendientes, vuelven a
   603→    // sonar. La cola de openPhone ordena el resto sin que ninguna llamada se pierda.
   604→    if (this.world) {
   605→      for (const m of this.world.missions) {
   606→        if (m.state !== 'active') continue;
   607→        const step = this.content.missions[m.id]?.steps[m.step];
   608→        if (step?.type === 'phone_call') this.openPhone(m.id, step.dialog);
   609→      }
   610→    }
   611→
   612→    this.pump();
   613→  }

   619→  // ───────────────────────── Bucle de render ─────────────────────────
   620→
   621→  private frame = (): void => {
   622→    if (!this.engine || !this.scene || !this.world || !this.logic || !this.building || !this.actors || !this.rig) return;
   623→    const dt = Math.min(0.1, this.engine.getDeltaTime() / 1000);
   624→    const store = useGameStore.getState();
   625→    const frozen = selectModalOpen(store) || store.paused;
   626→
   627→    let ticked = false;
   628→    if (!frozen) {
   629→      this.logic.update(dt, this.readInput());
   630→      // En remoto el servidor es la autoridad del reloj: nada de ticks locales.
   631→      if (!this.net) {
   632→        this.clockAcc += dt * store.speed;
   633→        let guard = 0;
   634→        while (this.clockAcc >= 1 && guard < 60) {
   635→          tick(this.world, this.content);
   636→          this.clockAcc -= 1;
   637→          ticked = true;
   638→          guard += 1;
   639→        }
   640→      }
   641→    }
   642→
   643→    if (this.world.events.length > 0) this.handleEvents();
   644→    if (ticked) this.paintBuilding();
   645→
   646→    const now = performance.now();
   647→    if (now - this.lastHudSync > 200) {
   648→      this.lastHudSync = now;
   649→      useGameStore.getState().setFps(Math.round(this.engine.getFps()));
   650→      this.syncHud();
   651→    }
   652→
   653→    const pos = this.logic.position();
   654→    this.actors.update(dt, pos, this.actorId());
   655→    // Cutscene (Fase D): sus beats mueven cámara/actores DESPUÉS de los eventos y ANTES del render.
   656→    this.cutscenePlayer?.update(dt);
   657→    // Si la cutscene acaba y queda cola, encadena la siguiente aquí: no dependemos de que el sim
   658→    // genere otro evento para descolgar la siguiente (durante el teatro el sim está congelado).
   659→    if (this.cutsceneQueue.length > 0 && !this.cutscenePlayer?.active) this.pumpCutscene();
   660→    // LOD por distancia: solo con cámaras cercanas (planta/estancia); el plano general (edificio) muestra todo.
   661→    this.building.update(dt, useGameStore.getState().camMode === 'edificio' ? undefined : { x: pos.x, floor: pos.floor });
   662→    this.rig.update(dt, pos);
   663→    this.scene.render();
   664→  };

   837→  /** Si el evento encaja con el trigger de alguna cutscene del catálogo, se encola (máx. 3). */
   838→  private maybeCutscene(e: GameEvent): void {
   839→    const defs = Object.values(this.content.cutscenes);
   840→    if (defs.length === 0) return;
   841→    const d = e.data as Record<string, unknown>;
   842→    const hit = defs.find((c) => {
   843→      const t = c.trigger;
   844→      if (t.on !== e.type) return false;
   845→      if (t.mission !== undefined && t.mission !== String(d.mission ?? '')) return false;
   846→      if (t.step !== undefined && t.step !== String(d.step ?? '')) return false;
   847→      return true;
   848→    });
   849→    if (!hit) return;
   850→    if (this.cutsceneQueue.length >= 3) return; // nunca más de 3 esperando: el teatro no atravaca la partida
   851→    this.cutsceneQueue.push({
   852→      def: hit,
   853→      ctx: {
   854→        mission: typeof d.mission === 'string' ? d.mission : null,
   855→        resident: typeof d.resident === 'string' ? d.resident : null,
   856→        block: typeof d.block === 'string' ? d.block : null,
   857→      },
   858→    });
   859→  }

   861→  /** Lanza la siguiente cutscene en cola si no hay nada en pantalla (también un modal la pospone). */
   862→  private pumpCutscene(): void {
   863→    if (!this.cutscenePlayer || !this.world || !this.rig || !this.actors || !this.engine || !this.logic) return;
   864→    if (this.cutscenePlayer.active || this.cutsceneQueue.length === 0) return;
   865→    if (selectModalOpen(useGameStore.getState())) return; // no se roba la pantalla a un modal
   866→    const next = this.cutsceneQueue.shift()!;
   867→    this.cutscenePlayer.start(next.def, next.ctx, {
   868→      engine: this.engine,
   869→      rig: this.rig,
   870→      actors: this.actors.director,
   871→      world: this.world,
   872→      playerPos: () => this.logic!.position(),
   873→    });
   874→  }

   912→    const o = activeObjective(this.world, this.content);
   913→    if (!o) return null;
   914→    if (o.state === 'awaiting_choice') return { title: o.title, text: 'El edificio contiene la respiración. Decide qué hacer.' };
   915→    const s = o.step!;
   916→    const left = o.untilTick !== null ? ` · quedan ${Math.max(0, o.untilTick - this.world.tick)} min` : '';
   917→    switch (s.type) {
   918→      case 'phone_call':
   919→        return { title: o.title, text: 'Suena el teléfono. Atiéndelo.' };
   920→      case 'reach':
   921→        return { title: o.title, text: `Llega a la puerta (${this.blockName(s.block ?? '')}).` };
   922→      case 'enter':
   923→        return { title: o.title, text: 'Entra en tu piso con E.' };
   924→      case 'catch_in_act':
   925→        return { title: o.title, text: `Pilla al responsable in fraganti${left}.` };
   926→      default:
   927→        return { title: o.title, text: o.title };
   928→    }
   929→  }
   930→
   931→  private promptText(): string | null {
   932→    const logic = this.logic;
   933→    const world = this.world;
   934→    if (!logic || !world) return null;
   935→    if (logic.position().climbing) return null;
   936→    // El prompt es SIEMPRE el objetivo real de la tecla E (misma fuente, cero divergencias).
   937→    const t = logic.interactTarget();
   938→    if (t) {
   939→      const ic = this.content.rules.interact;
   940→      switch (t.kind) {
   941→        case 'door': {
   942→          const door = world.blocks[t.blockId]!;
   943→          if (logic.position().mode === 'inside') return ic.exit;
   944→          return door.access === 'locked'
   945→            ? ic.doorLocked.replace('{block}', this.blockName(door.id))
   946→            : ic.doorEnter.replace('{block}', this.blockName(door.id));
   947→        }
   948→        case 'resident':
   949→          return ic.greet.replace('{name}', this.nameOf(t.residentId));
   950→        case 'mailbox':
   951→          return ic.mailbox.prompt;
   952→        case 'noticeboard':
   953→          return ic.noticeboard.prompt;
   954→        case 'laptop':
   955→          return ic.laptop;
   956→        case 'bed':
   957→          return ic.bed;
   958→      }
   959→    }
   960→    const zone = logic.zone();
   961→    const feat = this.content.blocks[zone.defId]!.features;
   962→    if (feat?.stairs || feat?.elevator) {
   963→      const ic = this.content.rules.interact;
   964→      // Lo que dice el prompt es lo que W/S hace aquí: ascensor operativo manda; si no hay
   965→      // ascensor utilizable pero sí escalera, escalera; núcleo averiado sin escalera = aviso.
   966→      if (feat.elevator && world.blocks[zone.id]!.params.elevator_status === 'ok') return ic.elevator;
   967→      if (feat.stairs) return ic.stairs;
   968→      return ic.elevatorBroken;
   969→    }
   970→    return null;
   971→  }

   973→  /** Viviendas del jugador para el editor de interiores (foto serializable, posiciones ya en %). */
   974→  private homesUi() {
   975→    const world = this.world!;
   976→    const cellH = this.content.rules.view.cellH;
   977→    const mine = this.actorId();
   978→    return Object.values(world.blocks)
   979→      .filter((b) => b.kind === 'dwelling' && b.ownerId === mine)
   980→      .sort((a, b) => a.id.localeCompare(b.id))
   981→      .map((b) => ({
   982→        id: b.id,
   983→        label: blockLabel(b.id),
   984→        rooms: b.rooms.map((room) => {
   985→          const rdef = this.content.rooms[room.defId];
   986→          const slots = (rdef?.furnitureSlots ?? []).map((s) => {
   987→            const itemId = room.furniture[s.id] ?? '';
   988→            const fdef = itemId ? this.content.furniture[itemId] : undefined;
   989→            return {
   990→              id: s.id,
   991→              kind: s.kind,
   992→              left: Math.min(92, Math.max(2, (s.pos[0] / Math.max(0.001, rdef?.size.w ?? 1)) * 100)),
   993→              bottom: Math.min(76, Math.max(0, (s.pos[1] / Math.max(0.001, cellH)) * 100)),
   994→              item: fdef ? { id: fdef.id, name: fdef.name, emoji: fdef.emoji ?? '📦', color: fdef.view.color } : null,
   995→            };
   996→          });
   997→          return { key: room.key, label: rdef?.label ?? room.key, access: room.access, slots };
   998→        }),
   999→      }));
  1000→  }

  1002→  private syncHud(): void {
  1003→    const world = this.world;
  1004→    if (!world) return;
  1005→    const player = world.residents[this.actorId()] ?? world.residents[world.playerId];
  1006→    if (!player) return;
  1007→    const home = player.home;
  1008→    const params = visibleParams(world, this.content, home).map((p) => ({ name: p.name, label: p.def.label ?? p.name, value: String(p.value) }));
  1009→    const missions = world.missions.map((m) => {
  1010→      const def = this.content.missions[m.id]!;
  1011→      const step = m.state === 'active' ? def.steps[m.step] : undefined;
  1012→      return {
  1013→        id: m.id,
  1014→        title: def.title,
  1015→        state: m.state,
  1016→        hint: step?.hint ?? null,
  1017→        remaining: m.until !== null ? `quedan ${Math.max(0, m.until - world.tick)} min` : null,
  1018→      };
  1019→    });
  1020→    const travel = travelMinutes(buildCirculation(world, this.content), world.entry, home);
  1021→    const elevatorBlock = Object.values(world.blocks).find((b) => b.params.elevator_status !== undefined);
  1022→    useGameStore.getState().syncHud({
  1023→      clock: { ...world.clock },
  1024→      weather: world.weather.state,
  1025→      funds: world.community.funds,
  1026→      wallet: player.wallet,
  1027→      reputation: player.reputation,
  1028→      elevator: String(elevatorBlock?.params.elevator_status ?? 'ok'),
  1029→      homeLabel: blockLabel(home),
  1030→      travelText: Number.isFinite(travel) ? `del portal a tu puerta: ${travel} min` : null,
  1031→      objective: this.objectiveUi(),
  1032→      prompt: this.promptText(),
  1033→      params,
  1034→      missions,
  1035→      homes: this.homesUi(),
  1036→      furnitureConfig: { refundRatio: this.content.rules.furniture.refundRatio, roomUnlockPrice: this.content.rules.furniture.roomUnlock.price },
  1037→    });
  1038→  }
