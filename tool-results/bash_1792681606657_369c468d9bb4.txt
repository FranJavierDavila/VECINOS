import type { Content } from '@vecinos/content';
import type { World } from './types';
import { emit } from './events';
import { applyEffects } from './effects';
import { advanceMissions, completeMission, notify } from './missions';

export type Command =
  | { type: 'move'; resident: string; to: string }
  | { type: 'approach'; resident: string; block: string }
  | { type: 'answer_call'; mission: string }
  | { type: 'choose'; mission: string; choice: string };

export interface CommandResult {
  ok: boolean;
  reason?: string;
}

const ok = (): CommandResult => ({ ok: true });
const fail = (reason: string): CommandResult => ({ ok: false, reason });

/**
 * Reglas de circulación: en la misma planta se pasa siempre por el pasillo (nunca de vivienda a vivienda ni
 * de vivienda a zona común sin salir); entre plantas solo de núcleo a núcleo contiguo (escalera o ascensor).
 */
export function canMove(world: World, content: Content, from: string | null, to: string): boolean {
  const b = world.blocks[to];
  if (!b) return false;
  const a = from ? world.blocks[from] : undefined;
  if (!a) return true;
  if (a.cell.floor === b.cell.floor) {
    if (a.layer === 'corridor' || b.layer === 'corridor') return true;
    return a.kind !== 'dwelling' && b.kind !== 'dwelling';
  }
  const stairs = (id: string) => !!content.blocks[world.blocks[id]!.defId]!.features?.stairs;
  return stairs(a.id) && stairs(b.id) && Math.abs(a.cell.floor - b.cell.floor) === 1 && a.cell.col === b.cell.col;
}

function run(world: World, content: Content, cmd: Command): CommandResult {
  switch (cmd.type) {
    case 'move': {
      const r = world.residents[cmd.resident];
      const to = world.blocks[cmd.to];
      if (!r || !r.present) return fail('unknown_resident');
      if (!to) return fail('unknown_block');
      if (r.location === to.id) return ok();
      if (to.kind === 'dwelling' && to.access === 'locked') {
        emit(world, 'door_locked', { resident: r.id, block: to.id, reason: to.lockReason, price: to.availability?.price ?? null });
        return fail('locked');
      }
      if (!canMove(world, content, r.location, to.id)) {
        emit(world, 'move_rejected', { resident: r.id, from: r.location, to: to.id });
        return fail('unreachable');
      }
      const from = r.location;
      r.location = to.id;
      emit(world, 'block_entered', { resident: r.id, block: to.id, from });
      notify(world, content, { type: 'block_entered', resident: r.id, block: to.id });
      return ok();
    }
    case 'approach': {
      const r = world.residents[cmd.resident];
      const b = world.blocks[cmd.block];
      if (!r || !r.present || !b) return fail('unknown_target');
      if (b.kind !== 'dwelling') return fail('not_a_door');
      const here = r.location ? world.blocks[r.location] : undefined;
      if (!here || here.cell.floor !== b.cell.floor) return fail('not_on_floor');
      emit(world, 'door_approached', { resident: r.id, block: b.id });
      notify(world, content, { type: 'door_approached', resident: r.id, block: b.id });
      return ok();
    }
    case 'answer_call': {
      const m = world.missions.find((x) => x.id === cmd.mission && x.state === 'active');
      if (!m || content.missions.id]!.steps.step]?.type !== 'phone_call') return fail('no_call');
      emit(world, 'call_answered', { mission: m.id });
      notify(world, content, { type: 'call_answered', mission: m.id });
      return ok();
    }
    case 'choose': {
      const m = world.missions.find((x) => x.id === cmd.mission && x.state === 'awaiting_choice');
      const choice = m && content.missions.id]!.choices?.find((c) => c.id === cmd.choice);
      if (!m || !choice) return fail('no_choice');
      m.choice = choice.id;
      emit(world, 'choice_made', { mission: m.id, choice: choice.id });
      applyEffects(world, content, choice.effects, { roles: m.roles });
      completeMission(world, content, m);
      return ok();
    }
  }
}

/** Punto de entrada único para las órdenes del jugador (o de la IA): valida, aplica y avanza las misiones. */
export function dispatch(world: World, content: Content, cmd: Command): CommandResult {
  const result = run(world, content, cmd);
  advanceMissions(world, content);
  return result;
}

import type { Content, Effect } from '@vecinos/content';
import type { BlockInstance, World } from './types';
import { emit } from './events';
import { instantiateRooms } from './rooms';
import { newMission } from './missions';

export interface EffectContext {
  /** Roles de la misión en curso (p. ej. kid → res_nino_01). */
  roles?: Record<string, string>;
}

/** Bloques a los que se refiere `ref`: id exacto, o todos los de un defId o un alignGroup. */
export function resolveBlocks(world: World, content: Content, ref: string): BlockInstance[] {
  const exact = world.blocks[ref];
  if (exact) return [exact];
  return Object.values(world.blocks).filter((b) => b.defId === ref || content.blocks[b.defId]?.alignGroup === ref);
}

function resolveActor(world: World, ref: string, ctx: EffectContext): string | null {
  if (ref === 'player') return world.playerId;
  const role = ctx.roles?.[ref];
  if (role) return role;
  return world.residents[ref] ? ref : null;
}

/** Único punto por el que misiones y decisiones modifican el mundo. */
export function applyEffect(world: World, content: Content, e: Effect, ctx: EffectContext = {}): void {
  switch (e.op) {
    case 'setParam': {
      const [ref, param] = e.param ? [e.target, e.param] : (e.target.split('.') as [string, string]);
      for (const b of resolveBlocks(world, content, ref)) {
        const pd = content.blocks[b.defId]!.params[param];
        if (!pd) continue;
        let next = e.value ?? (Number(b.params[param] ?? pd.default) + (e.delta ?? 0));
        if (pd.type === 'number' && typeof next === 'number') next = Math.min(pd.max ?? Infinity, Math.max(pd.min ?? -Infinity, next));
        b.params[param] = next;
        emit(world, 'param_changed', { block: b.id, param, value: next });
      }
      return;
    }
    case 'unlock': {
      const b = world.blocks[e.target];
      if (!b) return emit(world, 'effect_unresolved', { op: e.op, target: e.target });
      if (e.availability) {
        b.availability = e.availability;
        b.lockReason = 'for_sale';
        emit(world, 'block_for_sale', { block: b.id, price: e.availability.price });
      } else {
        const wasLocked = b.access === 'locked';
        b.access = 'unlocked';
        b.lockReason = null;
        b.availability = null;
        if (b.kind === 'dwelling' && b.rooms.length === 0) b.rooms = instantiateRooms(content, b.defId);
        if (wasLocked) emit(world, 'block_unlocked', { block: b.id });
      }
      b.occupancy = e.occupancy ?? (e.resident ? 'occupied' : b.occupancy);
      const r = e.resident ? world.residents[e.resident] : undefined;
      if (r) {
        r.present = true;
        r.location = b.id;
        b.ownerId ??= r.id;
        emit(world, 'neighbor_arrived', { resident: r.id, block: b.id, arrival: e.arrival ?? null });
      }
      return;
    }
    case 'addRelation': {
      const a = resolveActor(world, e.a, ctx);
      const b = resolveActor(world, e.b, ctx);
      const res = a ? world.residents[a] : undefined;
      if (!a || !b || !res) return emit(world, 'effect_unresolved', { op: e.op, a: e.a, b: e.b });
      let rel = res.relations.find((x) => x.with === b);
      if (!rel) res.relations.push((rel = { with: b, affinity: 0, trust: 0 }));
      rel.affinity += e.affinity ?? 0;
      rel.trust += e.trust ?? 0;
      return emit(world, 'relation_changed', { a, b, affinity: rel.affinity, trust: rel.trust });
    }
    case 'startMission': {
      if (world.missions.some((m) => m.id === e.id)) return;
      world.missions.push(newMission(e.id, world.tick + (e.delayMinutes ?? 0)));
      return emit(world, 'mission_started', { mission: e.id, startAtTick: world.tick + (e.delayMinutes ?? 0) });
    }
    case 'addFunds': {
      if (e.to === 'community') world.community.funds += e.amount;
      else {
        const id = resolveActor(world, e.to, ctx);
        if (!id) return emit(world, 'effect_unresolved', { op: e.op, to: e.to });
        world.residents[id]!.wallet += e.amount;
      }
      return emit(world, 'funds_changed', { to: e.to, amount: e.amount });
    }
    case 'revealParams': {
      if (e.tier) world.progress.revealedTier = Math.max(world.progress.revealedTier, e.tier);
      for (const p of e.params ?? []) {
        if (!world.progress.revealedParams.includes(p)) world.progress.revealedParams.push(p);
        world.progress.hiddenParams = world.progress.hiddenParams.filter((x) => x !== p);
      }
      return emit(world, 'params_revealed', { tier: e.tier ?? null, params: e.params ?? [] });
    }
    case 'hideParams': {
      for (const p of e.params ?? []) if (!world.progress.hiddenParams.includes(p)) world.progress.hiddenParams.push(p);
      if (e.tier) world.progress.revealedTier = Math.min(world.progress.revealedTier, e.tier - 1);
      return emit(world, 'params_hidden', { tier: e.tier ?? null, params: e.params ?? [] });
    }
    case 'grantFavor': {
      const from = resolveActor(world, e.from, ctx) ?? e.from;
      world.favors[from] = (world.favors[from] ?? 0) + e.count;
      return emit(world, 'favor_granted', { from, count: e.count });
    }
    // Estos efectos los consume la presentación o un sistema futuro; el núcleo solo avisa.
    case 'openJunta':
      return emit(world, 'junta_requested', { agenda: e.agenda });
    case 'startMinigame':
      return emit(world, 'minigame_requested', { minigame: e.id });
    default: {
      const never: never = e;
      return never;
    }
  }
}

export function applyEffects(world: World, content: Content, effects: Effect[], ctx: EffectContext = {}): void {
  for (const e of effects) applyEffect(world, content, e, ctx);
}

import type { Content } from '@vecinos/content';
import type { World } from './types';

export const STAIRS_COST = 3;
export const ELEVATOR_COST = 1;
const WALK_COST = 1;

export interface Graph {
  nodes: string[];
  edges: Map<string, Map<string, number>>;
}

function link(g: Graph, a: string, b: string, cost: number) {
  for (const [x, y] of [[a, b], [b, a]] as const) {
    if (!g.edges.has(x)) g.edges.set(x, new Map());
    const prev = g.edges.get(x)!.get(y);
    g.edges.get(x)!.set(y, prev === undefined ? cost : Math.min(prev, cost));
  }
}

const elevatorWorks = (world: World, content: Content, id: string) => {
  const b = world.blocks[id]!;
  return content.blocks[b.defId]!.features?.elevator === true && b.params.elevator_status === 'ok';
};

/**
 * Grafo de circulación (nodos = bloques, aristas = minutos de viaje):
 * cada bloque de la capa principal se conecta con el pasillo de su planta; los núcleos se conectan
 * entre plantas contiguas por escalera (siempre) o por ascensor (solo si funciona en ambas plantas).
 */
export function buildCirculation(world: World, content: Content): Graph {
  const blocks = Object.values(world.blocks);
  const g: Graph = { nodes: blocks.map((b) => b.id), edges: new Map() };
  const corridors = blocks.filter((b) => b.layer === 'corridor' && content.blocks[b.defId]!.traversable);
  for (const c of corridors)
    for (const b of blocks) if (b.layer === 'main' && b.cell.floor === c.cell.floor) link(g, c.id, b.id, WALK_COST);

  const cores = blocks.filter((b) => content.blocks[b.defId]!.features?.stairs).sort((a, b) => a.cell.floor - b.cell.floor);
  for (let i = 0; i + 1 < cores.length; i++) {
    const a = cores[i]!;
    const b = cores[i + 1]!;
    if (b.cell.floor !== a.cell.floor + 1 || b.cell.col !== a.cell.col) continue;
    const useElevator = elevatorWorks(world, content, a.id) && elevatorWorks(world, content, b.id);
    link(g, a.id, b.id, useElevator ? ELEVATOR_COST : STAIRS_COST);
  }
  return g;
}

/** Minutos de juego para ir de un bloque a otro (Dijkstra); Infinity si no hay camino. */
export function travelMinutes(g: Graph, from: string, to: string): number {
  const dist = new Map<string, number>(g.nodes.map((n) => [n, Infinity]));
  dist.set(from, 0);
  const open = new Set(g.nodes);
  while (open.size) {
    let u: string | null = null;
    for (const n of open) if (u === null || dist.get(n)! < dist.get(u)!) u = n;
    if (u === null || dist.get(u) === Infinity) break;
    open.delete(u);
    if (u === to) break;
    for (const [v, w] of g.edges.get(u) ?? []) dist.set(v, Math.min(dist.get(v)!, dist.get(u)! + w));
  }
  return dist.get(to) ?? Infinity;
}

/** Reglas de validación del edificio (apartado 6 del modelo de datos). Devuelve la lista de problemas. */
export function validateBuilding(world: World, content: Content): string[] {
  const issues: string[] = [];
  const blocks = Object.values(world.blocks);
  const cores = blocks.filter((b) => content.blocks[b.defId]!.alignGroup === 'nucleo');
  if (new Set(cores.map((b) => b.cell.col)).size > 1) issues.push('los núcleos no están alineados en la misma columna');
  for (const d of blocks.filter((b) => b.kind === 'dwelling'))
    if (!blocks.some((b) => b.layer === 'corridor' && b.cell.floor === d.cell.floor)) issues.push(`${d.id} no tiene pasillo en su planta`);
  const portal = blocks.find((b) => b.defId === 'portal');
  if (!portal) issues.push('el edificio no tiene portal');
  else {
    // Alcanzable por escalera aunque el ascensor esté roto: el coste solo importa para el tiempo de viaje.
    const g = buildCirculation(world, content);
    for (const c of blocks.filter((b) => b.layer === 'corridor'))
      if (!Number.isFinite(travelMinutes(g, portal.id, c.id))) issues.push(`${c.id} no es alcanzable desde el portal`);
  }
  return issues;
}

import type { Content, Direction } from '@vecinos/content';
import type { World } from './types';
import { emit } from './events';
import { applyEffect } from './effects';

export type Emissions = Record<string, Record<string, number>>; // canal → bloque → intensidad

interface Neighbor {
  id: string;
  dir: Direction;
}

const DIR_ORDER: Direction[] = ['horizontal', 'up', 'down', 'diagonal'];

/** Vecindad por adyacencia, vista desde cada bloque emisor (up = el vecino está encima). */
export function adjacency(world: World): Map<string, Neighbor[]> {
  const blocks = Object.values(world.blocks).sort((a, b) => a.id.localeCompare(b.id));
  const mains = blocks.filter((b) => b.layer === 'main');
  const cells = (b: (typeof blocks)[number]) => {
    const out: [number, number][] = [];
    for (let dc = 0; dc < b.size.w; dc++) for (let df = 0; df < b.size.h; df++) out.push([b.cell.col + dc, b.cell.floor + df]);
    return out;
  };
  const rel = (a: (typeof blocks)[number], b: (typeof blocks)[number]): Direction | null => {
    const found = new Set<Direction>();
    for (const [ca, fa] of cells(a))
      for (const [cb, fb] of cells(b)) {
        const dc = cb - ca;
        const df = fb - fa;
        if (df === 0 && Math.abs(dc) === 1) found.add('horizontal');
        else if (dc === 0 && df === 1) found.add('up');
        else if (dc === 0 && df === -1) found.add('down');
        else if (Math.abs(dc) === 1 && Math.abs(df) === 1) found.add('diagonal');
      }
    return DIR_ORDER.find((d) => found.has(d)) ?? null;
  };

  const out = new Map<string, Neighbor[]>();
  for (const b of blocks) out.set(b.id, []);
  for (const a of mains)
    for (const b of mains) {
      if (a.id === b.id) continue;
      const dir = rel(a, b);
      if (dir) out.get(a.id)!.push({ id: b.id, dir });
    }
  for (const c of blocks.filter((b) => b.layer === 'corridor'))
    for (const m of mains.filter((b) => b.cell.floor === c.cell.floor)) {
      out.get(m.id)!.push({ id: c.id, dir: 'corridor' });
      out.get(c.id)!.push({ id: m.id, dir: 'corridor' });
    }
  return out;
}

/** Emisiones de ruido, olores... de las actividades en curso de los residentes (en su vivienda). */
export function emissionsFromActivities(world: World, content: Content, active: { resident: string; activity: string }[]): Emissions {
  const out: Emissions = {};
  for (const { resident, activity } of active) {
    const r = world.residents[resident];
    const emits = r && content.archetypes[r.archetype]?.activities[activity]?.emits;
    if (!r || !r.present || !emits) continue;
    for (const [channel, v] of Object.entries(emits)) {
      const bucket = (out[channel] ??= {});
      bucket[r.home] = Math.max(bucket[r.home] ?? 0, v);
    }
  }
  return out;
}

const round3 = (n: number) => Math.round(n * 1000) / 1000;

/**
 * Un paso de propagación: cada canal decae, recibe las emisiones del tick (atenuadas por el receptor)
 * y dispara su evento al cruzar el umbral hacia arriba. Orden estable por id, así que es determinista.
 */
export function propagate(world: World, content: Content, emissions: Emissions): void {
  const nb = adjacency(world);
  for (const rule of Object.values(content.influences).sort((a, b) => a.id.localeCompare(b.id))) {
    const state = (world.channels[rule.channel] ??= {});
    const before = { ...state };
    for (const id of Object.keys(state)) {
      state[id] = state[id]! * (1 - rule.decayPerTick);
      if (state[id]! < 1e-6) delete state[id];
    }
    const em = emissions[rule.channel] ?? {};
    for (const emitter of Object.keys(em).sort()) {
      for (const n of nb.get(emitter) ?? []) {
        const factor = rule.spread[n.dir];
        if (!factor) continue;
        let att = 1;
        if (rule.attenuatedBy) {
          const p = world.blocks[n.id]!.params[rule.attenuatedBy];
          if (typeof p === 'number') att = 1 - Math.min(1, Math.max(0, p));
        }
        const value = em[emitter]! * factor * att;
        if (value > (state[n.id] ?? 0)) state[n.id] = value;
      }
    }
    if (!rule.onThreshold) continue;
    for (const id of Object.keys(state).sort()) {
      if ((before[id] ?? 0) >= rule.threshold || state[id]! < rule.threshold) continue;
      emit(world, rule.onThreshold.emitEvent, {
        block: id,
        channel: rule.channel,
        value: round3(state[id]!),
        occupied: world.blocks[id]!.occupancy === 'occupied',
      });
      const sp = rule.onThreshold.setParam;
      if (sp) applyEffect(world, content, { op: 'setParam', target: id, param: sp.param, delta: sp.delta });
    }
  }
}

import type { Condition, Content, MissionDef } from '@vecinos/content';
import type { ActiveMission, World } from './types';
import { emit } from './events';
import { applyEffects, resolveBlocks } from './effects';

type Step = MissionDef['steps'][number];
const byId = <T extends { id: string }>(a: T, b: T) => a.id.localeCompare(b.id);

export function newMission(id: string, startAtTick: number): ActiveMission {
  return { id, state: 'pending', startAtTick, step: 0, roles: {}, until: null, choice: null };
}

/** Enlaza los roles de una misión con residentes presentes: por arquetipo y por parentesco (`parentOf`). */
export function bindRoles(world: World, def: MissionDef): { roles: Record<string, string>; complete: boolean } {
  const roles: Record<string, string> = {};
  const taken = new Set<string>();
  const present = Object.values(world.residents).filter((r) => r.present).sort(byId);
  const defs = Object.entries(def.roles ?? {});
  for (const [name, r] of defs) {
    if (!r.archetype) continue;
    const c = present.find((x) => x.archetype === r.archetype && !taken.has(x.id));
    if (c) {
      roles[name] = c.id;
      taken.add(c.id);
    }
  }
  for (const [name, r] of defs) {
    if (r.relation !== 'parentOf' || !r.of || !roles[r.of]) continue;
    const kid = roles[r.of]!;
    const p = present.find((x) => x.parentOf.includes(kid));
    if (p) roles[name] = p.id;
  }
  return { roles, complete: defs.every(([name]) => name in roles) };
}

const resolveRef = (world: World, ref: string) => (ref === '$home' ? world.residents[world.playerId]!.home : ref);
const currentStep = (content: Content, m: ActiveMission): Step | undefined => content.missions.id]?.steps.step];

export function conditionHolds(world: World, content: Content, c: Condition, roles: Record<string, string>): boolean {
  if ('param' in c) {
    const [ref, p] = c.param.split('.') as [string, string];
    const blocks = resolveBlocks(world, content, ref);
    return blocks.length > 0 && blocks.every((b) => b.params[p] === c.eq);
  }
  if ('role' in c) return c.exists === (c.role in roles);
  if ('mission' in c) {
    const m = world.missions.find((x) => x.id === c.mission);
    return c.state === 'done' ? m?.state === 'done' : m?.state === 'active' || m?.state === 'awaiting_choice';
  }
  const h, mm] = c.time.after.split(':').map(Number) as [number, number];
  return world.clock.minute >= hh * 60 + mm;
}

function restoreLocations(world: World, m: ActiveMission) {
  for (const id of Object.values(m.roles)) {
    const r = world.residents[id];
    if (r && r.present && id !== world.playerId) r.location = r.home;
  }
}

function enterStep(world: World, content: Content, m: ActiveMission) {
  const step = currentStep(content, m)!;
  const base = { mission: m.id, step: step.id, type: step.type };
  switch (step.type) {
    case 'phone_call':
      return emit(world, 'step_started', { ...base, dialog: step.dialog });
    case 'reach':
      return emit(world, 'step_started', { ...base, block: resolveRef(world, step.target.block) });
    case 'enter':
      return emit(world, 'step_started', { ...base, block: resolveRef(world, step.target) });
    case 'catch_in_act': {
      const kidId = m.roles[step.target];
      const kid = kidId ? world.residents[kidId] : undefined;
      const homeFloor = kid ? world.blocks[kid.home]?.cell.floor : undefined;
      const candidates = resolveBlocks(world, content, step.place.split('.')[0]!).sort(byId);
      const block = candidates.find((b) => b.cell.floor === homeFloor) ?? candidates[0];
      m.until = world.tick + step.windowMinutes;
      if (kid && block) kid.location = block.id;
      return emit(world, 'step_started', { ...base, resident: kidId ?? null, block: block?.id ?? null, untilTick: m.until });
    }
  }
}

function activate(world: World, content: Content, m: ActiveMission) {
  const def = content.missions.id]!;
  m.roles = { ...bindRoles(world, def).roles, ...m.roles };
  m.state = 'active';
  m.step = 0;
  m.until = null;
  emit(world, 'mission_activated', { mission: m.id });
  enterStep(world, content, m);
}

export function completeMission(world: World, content: Content, m: ActiveMission) {
  const def = content.missions.id]!;
  m.state = 'done';
  applyEffects(world, content, def.onComplete ?? [], { roles: m.roles });
  restoreLocations(world, m);
  emit(world, 'mission_completed', { mission: m.id });
}

function failMission(world: World, m: ActiveMission, reason: string) {
  m.state = 'failed';
  restoreLocations(world, m);
  emit(world, 'mission_failed', { mission: m.id, reason });
}

function completeStep(world: World, content: Content, m: ActiveMission) {
  const def = content.missions.id]!;
  emit(world, 'step_completed', { mission: m.id, step: def.steps.step]!.id });
  m.step += 1;
  m.until = null;
  if (m.step < def.steps.length) return enterStep(world, content, m);
  if (def.choices?.length) {
    m.state = 'awaiting_choice';
    return emit(world, 'choice_required', { mission: m.id, choices: def.choices.map((c) => ({ id: c.id, label: c.label })) });
  }
  completeMission(world, content, m);
}

export type MissionSignal =
  | { type: 'door_approached' | 'block_entered'; resident: string; block: string }
  | { type: 'call_answered'; mission: string };

/** Las órdenes del jugador avisan aquí: si encajan con el paso actual de una misión activa, el paso se completa. */
export function notify(world: World, content: Content, sig: MissionSignal) {
  for (const m of world.missions) {
    if (m.state !== 'active') continue;
    const step = currentStep(content, m);
    if (!step) continue;
    const byPlayer = sig.type !== 'call_answered' && sig.resident === world.playerId;
    let hit = false;
    switch (step.type) {
      case 'phone_call':
        hit = sig.type === 'call_answered' && sig.mission === m.id;
        break;
      case 'reach':
        hit = sig.type === 'door_approached' && byPlayer && sig.block === resolveRef(world, step.target.block);
        break;
      case 'enter':
        hit = sig.type === 'block_entered' && byPlayer && sig.block === resolveRef(world, step.target);
        break;
      case 'catch_in_act': {
        const kid = m.roles[step.target];
        hit = sig.type === 'block_entered' && byPlayer && !!kid && world.residents[kid]?.location === sig.block && (m.until === null || world.tick <= m.until);
        break;
      }
    }
    if (hit) completeStep(world, content, m);
  }
}

function advanceOnce(world: World, content: Content): boolean {
  let changed = false;
  for (const m of world.missions) {
    if (m.state === 'pending' && world.tick >= m.startAtTick) {
      activate(world, content, m);
      changed = true;
    } else if (m.state === 'active') {
      const step = currentStep(content, m);
      if (step?.type === 'catch_in_act' && m.until !== null && world.tick > m.until) {
        failMission(world, m, 'timeout');
        changed = true;
      }
    }
  }
  for (const def of Object.values(content.missions).sort(byId)) {
    if (!def.trigger || world.missions.some((x) => x.id === def.id)) continue;
    const { roles, complete } = bindRoles(world, def);
    if (!complete) continue;
    const holds = 'all' in def.trigger ? def.trigger.all.every((c) => conditionHolds(world, content, c, roles)) : def.trigger.any.some((c) => conditionHolds(world, content, c, roles));
    if (!holds) continue;
    const m = newMission(def.id, world.tick);
    m.roles = roles;
    world.missions.push(m);
    activate(world, content, m);
    changed = true;
  }
  return changed;
}

/** Activa misiones pendientes o con disparador cumplido y caduca las que se pasaron de plazo. */
export function advanceMissions(world: World, content: Content): void {
  for (let i = 0; i < 4; i++) if (!advanceOnce(world, content)) break;
}

import type { World } from './types';

/** mulberry32 con el estado guardado en el mundo: misma semilla y mismas llamadas, mismos resultados. */
export function nextRandom(world: Pick<World, 'rngState'>): number {
  world.rngState = (world.rngState + 0x6d2b79f5) | 0;
  let t = world.rngState;
  t = Math.imul(t ^ (t >>> 15), t | 1);
  t ^= t + Math.imul(t ^ (t >>> 7), t | 61);
  return ((t ^ (t >>> 14)) >>> 0) / 4294967296;
}

/**
 * Parámetros que el jugador ve de un bloque: los de nivel <= nivel revelado, más los revelados uno a uno,
 * menos los ocultados. La simulación calcula todos igualmente; esto solo afecta a la interfaz.
 * Un parámetro sin `tier` queda oculto hasta que se revele por nombre.
 */
export function visibleParams(world: World, content: Content, blockId: string): VisibleParam[] {
  const b = world.blocks[blockId];
  if (!b) return [];
  const def = content.blocks[b.defId]!;
  const p = world.progress;
  return Object.entries(def.params)
    .filter(([name, d]) => {
      if (p.hiddenParams.includes(name)) return false;
      return (d.tier !== undefined && d.tier <= p.revealedTier) || p.revealedParams.includes(name);
    })
    .map(([name, d]) => ({ name, value: b.params[name] ?? d.default, def: d }));
}

export interface Objective {
  mission: string;
  title: string;
  state: 'active' | 'awaiting_choice';
  step: { id: string; type: string; block?: string; dialog?: string } | null;
  untilTick: number | null;
}

/** Objetivo que se muestra al jugador: la primera misión en curso. */
export function activeObjective(world: World, content: Content): Objective | null {
  const m = world.missions.find((x) => x.state === 'active' || x.state === 'awaiting_choice');
  if (!m) return null;
  const def = content.missions.id]!;
  const s = m.state === 'active' ? def.steps.step] : undefined;
  const home = world.residents[world.playerId]!.home;
  const block = s?.type === 'reach' ? s.target.block : s?.type === 'enter' ? s.target : undefined;
  return {
    mission: m.id,
    title: def.title,
    state: m.state as 'active' | 'awaiting_choice',
    step: s ? { id: s.id, type: s.type, block: block === '$home' ? home : block, dialog: s.type === 'phone_call' ? s.dialog : undefined } : null,
    untilTick: m.until,
  };
}

/** Avanza un minuto de juego. Las emisiones del tick las aportan las actividades de los residentes. */
export function tick(world: World, content: Content, emissions: Emissions = {}): void {
  world.tick += 1;
  world.clock.minute += 1;
  if (world.clock.minute >= 1440) {
    world.clock.minute = 0;
    world.clock.day += 1;
  }
  propagate(world, content, emissions);
  advanceMissions(world, content);
}

import type { Content } from '@vecinos/content';
import { expandTemplate } from '@vecinos/content';
import type { BlockInstance, World } from './types';
import { nextRandom } from './rng';
import { instantiateRooms } from './rooms';
import { advanceMissions, newMission } from './missions';

/** Crea una partida nueva a partir de un escenario. La vivienda del jugador se elige con la semilla. */
export function createWorld(content: Content, scenarioId: string, overrides: { seed?: number } = {}): World {
  const scenario = content.scenarios[scenarioId];
  if (!scenario) throw new Error(`Escenario desconocido: ${scenarioId}`);
  const template = content.templates[scenario.building.template]!;
  const seed = overrides.seed ?? scenario.world.seed;

  const world: World = {
    schemaVersion: 1,
    scenarioId,
    seed,
    rngState: seed,
    tick: 0,
    clock: { ...scenario.world.clock },
    playerId: scenario.player.residentId,
    community: { ...scenario.building.community },
    blocks: {},
    residents: {},
    channels: {},
    progress: { revealedTier: scenario.progress.revealedTier, revealedParams: [], hiddenParams: [] },
    missions: [],
    favors: {},
    events: [],
  };

  for (const b of expandTemplate(template, content.blocks)) {
    const def = content.blocks[b.defId]!;
    const params = Object.fromEntries(Object.entries(def.params).map(([k, p]) => [k, p.default]));
    Object.assign(params, scenario.building.stateByDef[b.defId]?.params ?? {});
    const dwelling = def.kind === 'dwelling';
    const inst: BlockInstance = {
      id: b.id,
      defId: b.defId,
      kind: def.kind,
      layer: b.layer,
      cell: b.cell,
      size: b.size,
      access: dwelling ? 'locked' : 'unlocked',
      lockReason: dwelling ? 'for_sale' : null,
      occupancy: 'vacant',
      ownerId: null,
      availability: null,
      params,
      rooms: [],
    };
    world.blocks[b.id] = inst;
  }

  // Vecinos del escenario: existen, pero aún no han llegado.
  for (const r of scenario.residents) {
    world.residents[r.id] = {
      id: r.id,
      archetype: r.archetype,
      controller: r.controller,
      home: r.home,
      present: r.present,
      location: r.present ? r.home : null,
      parentOf: [...r.parentOf],
      wallet: r.wallet,
      reputation: r.reputation,
      relations: r.relations.map((x) => ({ ...x })),
    };
  }

  // El jugador: su vivienda se elige al azar (con la semilla) entre las candidatas.
  const candidates = scenario.player.homeCandidates;
  const homeId = candidates[Math.floor(nextRandom(world) * candidates.length)]!;
  const home = world.blocksomeId]!;
  home.access = 'unlocked';
  home.lockReason = null;
  home.occupancy = 'occupied';
  home.ownerId = scenario.player.residentId;
  home.rooms = instantiateRooms(content, home.defId);
  world.residents[scenario.player.residentId] = {
    id: scenario.player.residentId,
    archetype: scenario.player.archetype,
    controller: { type: 'human', playerId: 'p1' },
    home: homeId,
    present: true,
    location: scenario.player.startBlock,
    parentOf: [],
    wallet: scenario.player.wallet,
    reputation: scenario.player.reputation,
    relations: [],
  };

  for (const m of scenario.startMissions) world.missions.push(newMission(m, 0));
  advanceMissions(world, content); // el teléfono del tutorial empieza a sonar desde el primer tick
  return world;
}
