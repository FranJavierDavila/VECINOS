    66→/** Convierte un color hex (#rgb / #rrggbb) en rgba con la opacidad dada. Si no es hex, lo devuelve tal cual. */
    67→function withAlpha(color: string, alpha: number): string {
    68→  const match = /^#([0-9a-f]{3}|[0-9a-f]{6})$/i.exec(color.trim());
    69→  if (!match) return color;
    70→  let hex = match[1]!;
    71→  if (hex.length === 3) hex = hex.split('').map((c) => c + c).join('');
    72→  const value = Number.parseInt(hex, 16);
    73→  const r = (value >> 16) & 0xff;
    74→  const g = (value >> 8) & 0xff;
    75→  const b = value & 0xff;
    76→  return `rgba(${r}, ${g}, ${b}, ${alpha})`;
    77→}

   323→export function InteriorEditorPanel() {
   324→  const homes = useGameStore(selectHomes);
   325→  const wallet = useGameStore((s) => s.wallet);
   326→  const furnitureConfig = useGameStore(selectFurnitureConfig);
   327→
   328→  const [catalog, setCatalog] = useState<CatalogData | null>(null);
   329→  const [catalogError, setCatalogError] = useState(false);
   330→  const [homeId, setHomeId] = useState<string | null>(null);
   331→  const [roomKey, setRoomKey] = useState<string | null>(null);
   332→  /** Ranura abierta (vacía → catálogo; ocupada → fila de retirada). */
   333→  const [selected, setSelected] = useState<{ roomKey: string; slotId: string } | null>(null);
   334→
   335→  // Estado de drag & drop: lo visible para render (fantasma + ranura bajo el puntero) y la sesión en refs.
   336→  const [drag, setDrag] = useState<DragState | null>(null);
   337→  const [hovered, setHovered] = useState<HoveredSlot | null>(null);
   338→  const dragRef = useRef<DragSession | null>(null);
   339→  const hoveredRef = useRef<HoveredSlot | null>(null);
   340→  const ghostRef = useRef<HTMLDivElement | null>(null);
   341→  /** Id del temporizador de long-press (separa de la sesión para no mutarla). */
   342→  const longPressTimerRef = useRef<number | null>(null);
   343→  /** Quita los listeners de ventana añadidos al empezar la sesión (misma identidad de funciones). */
   344→  const detachWindowListenersRef = useRef<(() => void) | null>(null);
   345→
   346→  // El catálogo se cachea como promesa en loadContent(); una sola llamada por montaje.
   347→  useEffect(() => {
   348→    let cancelled = false;
   349→    loadContent()
   350→      .then(({ content }) => {
   351→        if (cancelled) return;
   352→        const channels: Record<string, ChannelInfo> = {};
   353→        for (const influence of Object.values(content.influences)) {
   354→          channels[influence.channel] = { label: influence.label, emoji: influence.emoji };
   355→        }
   356→        setCatalog({ furniture: content.furniture, channels });
   357→      })
   358→      .catch(() => {
   359→        if (!cancelled) setCatalogError(true);
   360→      });
   361→    return () => {
   362→      cancelled = true;
   363→    };
   364→  }, []);
   365→
   366→  // Selecciones estables entre refrescos del HUD: si el id guardado ya no existe, se cae a la primera.
   367→  const home: HomeUi | null = homes.find((h) => h.id === homeId) ?? homes[0] ?? null;
   368→  const room: HomeRoomUi | null =
   369→    home?.rooms.find((r) => r.key === roomKey) ?? home?.rooms[0] ?? null;
   370→  const activeSlot: SlotUi | null =
   371→    selected && room && selected.roomKey === room.key
   372→      ? (room.slots.find((s) => s.id === selected.slotId) ?? null)
   373→      : null;
   374→
   375→  /**
   376→   * Muebles compatibles con la ranura vacía abierta, ordenados por precio. Sin useMemo a propósito:
   377→   * el React Compiler no puede preservar esa memoización manual junto al motor de drag imperativo
   378→   * (refs + listeners de ventana) y `preserve-manual-memoization` lo marca como error; el cálculo
   379→   * es trivial (catálogo de una veintena de items) y sin memo manual el compilador decide él solo.
   380→   */
   381→  const compatible: FurnitureDef[] =
   382→    catalog && activeSlot && !activeSlot.item
   383→      ? Object.values(catalog.furniture)
   384→          .filter((f) => f.slots.includes(activeSlot.kind))
   385→          .sort((a, b) => a.price - b.price)
   386→      : [];
   387→
   388→  const refundOf = (item: PlacedItemUi): number => {
   389→    const price = catalog?.furniture[item.id]?.price ?? 0;
   390→    return Math.round(price * furnitureConfig.refundRatio);
   391→  };
   392→
   393→  const handleSlotSelect = (roomKeyValue: string, slot: SlotUi) => {
   394→    setSelected((prev) =>
   395→      prev && prev.roomKey === roomKeyValue && prev.slotId === slot.id
   396→        ? null
   397→        : { roomKey: roomKeyValue, slotId: slot.id },
   398→    );
   399→  };
   400→
   401→  const handlePlace = (item: FurnitureDef) => {
   402→    if (!home || !room || !activeSlot) return;
   403→    getController()?.placeFurniture(home.id, room.key, activeSlot.id, item.id);
   404→    setSelected(null);
   405→  };
   406→
   407→  const handleRemove = () => {
   408→    if (!home || !room || !activeSlot) return;
   409→    getController()?.removeFurniture(home.id, room.key, activeSlot.id);
   410→    setSelected(null);
   411→  };
   412→
   413→  const handleRenovate = (key: string) => {
   414→    if (!home) return;
   415→    getController()?.renovateRoom(home.id, key);
   416→  };
   417→
   418→  // ── Drag & drop: sesión en refs + listeners de ventana imperativos (sin estados en efectos) ──
   419→
   420→  const activateDrag = () => {
   421→    const session = dragRef.current;
   422→    if (!session || session.active) return;
   423→    const item = catalog?.furniture[session.itemId];
   424→    if (!item) return;
   425→    dragRef.current = { ...session, active: true };
   426→    setDrag({ item, x: session.x, y: session.y });
   427→    if (session.pointerType === 'touch') {
   428→      // Congela el scroll del navegador durante el arrastre táctil (listener no pasivo).
   429→      document.addEventListener('touchmove', preventTouchScroll, { passive: false });
   430→    }
   431→  };
   432→
   433→  /** Ranura bajo el puntero vía elementFromPoint; solo re-renderiza al cruzar entre ranuras. */
   434→  const updateHoveredSlot = (x: number, y: number) => {
   435→    const target = document.elementFromPoint(x, y);
   436→    const slotEl = target?.closest('[data-slot-id]') ?? null;
   437→    const next: HoveredSlot | null = slotEl
   438→      ? {
   439→          roomKey: slotEl.getAttribute('data-slot-room') ?? '',
   440→          slotId: slotEl.getAttribute('data-slot-id') ?? '',
   441→          kind: slotEl.getAttribute('data-slot-kind') ?? '',
   442→        }
   443→      : null;
   444→    const prev = hoveredRef.current;
   445→    if (prev?.roomKey === next?.roomKey && prev?.slotId === next?.slotId) return;
   446→    hoveredRef.current = next;
   447→    setHovered(next);
   448→  };
   449→
   450→  /** Suelta listeners, temporizador, clases de body y estado de drag. Silencioso si no había sesión. */
   451→  const endDrag = () => {
   452→    detachWindowListenersRef.current?.();
   453→    detachWindowListenersRef.current = null;
   454→    document.removeEventListener('touchmove', preventTouchScroll);
   455→    document.body.classList.remove('select-none');
   456→    if (longPressTimerRef.current != null) window.clearTimeout(longPressTimerRef.current);
   457→    longPressTimerRef.current = null;
   458→    dragRef.current = null;
   459→    hoveredRef.current = null;
   460→    setDrag(null);
   461→    setHovered(null);
   462→  };
   463→
   464→  const handleDragPointerMove = (event: PointerEvent) => {
   465→    const session = dragRef.current;
   466→    if (!session || event.pointerId !== session.pointerId) return;
   467→    // Sesión inmutable: la posición se actualiza reemplazando el objeto, nunca mutándolo.
   468→    dragRef.current = { ...session, x: event.clientX, y: event.clientY };
   469→
   470→    if (!session.active) {
   471→      const dist = Math.hypot(event.clientX - session.startX, event.clientY - session.startY);
   472→      if (session.pointerType === 'touch') {
   473→        // Movimiento apreciable antes del long-press: era un scroll, se cancela el drag.
   474→        if (dist > TOUCH_CANCEL_PX) endDrag();
   475→        return;
   476→      }
   477→      if (dist > DRAG_THRESHOLD_MOUSE_PX) activateDrag();
   478→      return;
   479→    }
   480→
   481→    // Drag activo: el fantasma sigue al puntero (imperativo, sin re-render por frame).
   482→    const ghost = ghostRef.current;
   483→    if (ghost) {
   484→      ghost.style.left = `${event.clientX}px`;
   485→      ghost.style.top = `${event.clientY}px`;
   486→    }
   487→    updateHoveredSlot(event.clientX, event.clientY);
   488→  };
   489→
   490→  const handleDragPointerUp = (event: PointerEvent) => {
   491→    const session = dragRef.current;
   492→    if (!session || event.pointerId !== session.pointerId) return;
   493→    if (session.active) {
   494→      const target = hoveredRef.current;
   495→      // El sim valida (ranura equivocada, ocupada, sin fondos...) y muestra sus toasts;
   496→      // el cliente no duplica reglas. Sin ranura bajo el puntero: cancela en silencio.
   497→      if (target) {
   498→        getController()?.placeFurniture(session.homeId, target.roomKey, target.slotId, session.itemId);
   499→      }
   500→    }
   501→    endDrag();
   502→  };
   503→
   504→  const handleDragPointerCancel = (event: PointerEvent) => {
   505→    const session = dragRef.current;
   506→    if (!session || event.pointerId !== session.pointerId) return;
   507→    endDrag();
   508→  };
   509→
   510→  const startPotentialDrag = (event: ReactPointerEvent<HTMLDivElement>, item: FurnitureDef) => {
   511→    if (dragRef.current) return; // ya hay una sesión en curso
   512→    if (!event.isPrimary) return;
   513→    if (event.pointerType === 'mouse' && event.button !== 0) return;
   514→
   515→    const session: DragSession = {
   516→      itemId: item.id,
   517→      homeId: home?.id ?? '',
   518→      pointerId: event.pointerId,
   519→      pointerType: event.pointerType,
   520→      startX: event.clientX,
   521→      startY: event.clientY,
   522→      x: event.clientX,
   523→      y: event.clientY,
   524→      active: false,
   525→    };
   526→    dragRef.current = session;
   527→    // Evita selección de texto/magnifier desde el primer frame del gesto (se quita al terminar).
   528→    document.body.classList.add('select-none');
   529→
   530→    window.addEventListener('pointermove', handleDragPointerMove);
   531→    window.addEventListener('pointerup', handleDragPointerUp);
   532→    window.addEventListener('pointercancel', handleDragPointerCancel);
   533→    detachWindowListenersRef.current = () => {
   534→      window.removeEventListener('pointermove', handleDragPointerMove);
   535→      window.removeEventListener('pointerup', handleDragPointerUp);
   536→      window.removeEventListener('pointercancel', handleDragPointerCancel);
   537→    };
   538→
   539→    if (event.pointerType === 'touch') {
   540→      longPressTimerRef.current = window.setTimeout(() => {
   541→        const current = dragRef.current;
   542→        if (!current || current.pointerId !== session.pointerId) return;
   543→        longPressTimerRef.current = null;
   544→        activateDrag();
   545→      }, LONG_PRESS_MS);
   546→    }
   547→  };
   548→
   549→  const handleCardContextMenu = (event: ReactMouseEvent<HTMLDivElement>) => {
   550→    // El long-press en móvil no debe abrir el menú contextual a mitad de drag.
   551→    if (dragRef.current) event.preventDefault();
   552→  };
   553→
   554→  // Desmontaje a mitad de drag: descolgar todo sin tocar estados (los setState serían no-op).
   555→  useEffect(() => {
   556→    return () => {
   557→      detachWindowListenersRef.current?.();
   558→      detachWindowListenersRef.current = null;
   559→      document.removeEventListener('touchmove', preventTouchScroll);
   560→      document.body.classList.remove('select-none');
   561→      if (longPressTimerRef.current != null) window.clearTimeout(longPressTimerRef.current);
   562→      dragRef.current = null;
   563→      hoveredRef.current = null;
   564→    };
   565→  }, []);
   566→
   567→  if (homes.length === 0) {
   568→    return (
   569→      <Card className="gap-3 py-4">
   570→        <CardHeader className="px-4">
   571→          <CardTitle className="flex items-center gap-2 text-sm">
   572→            <Sofa className="size-4 text-amber-500" aria-hidden />
   573→            Mi piso
   574→          </CardTitle>
   575→        </CardHeader>
   576→        <CardContent className="px-4">
   577→          <p className="text-sm text-muted-foreground">
   578→            Aún no tienes piso. El portal, de momento, es tuyo.
   579→          </p>
   580→        </CardContent>
   581→      </Card>
   582→    );
   583→  }
   584→
   585→  return (
   586→    <>
   587→      <Card className="gap-3 py-4">
   588→        <CardHeader className="px-4">
   589→          <CardTitle className="flex items-center gap-2 text-sm">
   590→            <Sofa className="size-4 text-amber-500" aria-hidden />
   591→            Mi piso
   592→          </CardTitle>
   593→          <CardDescription>Amuebla, desamuebla y reforma. Como el BGAs, pero con candado.</CardDescription>
   594→          <CardAction>
   595→            <Badge
   596→              variant="outline"
   597→              title="Tu cartera"
   598→              className="border-amber-500/30 bg-amber-500/15 font-mono text-amber-600 dark:text-amber-400"
   599→            >
   600→              {wallet} €
   601→            </Badge>
   602→          </CardAction>
   603→        </CardHeader>
   604→        <CardContent className="px-4">
   605→          {catalogError && (
   606→            <p className="text-sm text-muted-foreground">
   607→              El catálogo de muebles no ha bajado al portal. Prueba de nuevo en un rato.
   608→            </p>
   609→          )}
   610→
   611→          {!catalog && !catalogError && (
   612→            <p className="flex items-center gap-2 text-sm text-muted-foreground" role="status">
   613→              <Loader2 className="size-4 animate-spin" aria-hidden />
   614→              Subiendo el catálogo del barrio...
   615→            </p>
   616→          )}
   617→
   618→          {catalog && home && (
   619→            <div className="flex flex-col gap-3">
   620→              {homes.length > 1 && (
   621→                <Select
   622→                  value={home.id}
   623→                  onValueChange={(id) => {
   624→                    setHomeId(id);
   625→                    setRoomKey(null);
   626→                    setSelected(null);
   627→                  }}
   628→                >
   629→                  <SelectTrigger className="w-full sm:w-56" aria-label="Elegir vivienda">
   630→                    <SelectValue placeholder="Elige vivienda" />
   631→                  </SelectTrigger>
   632→                  <SelectContent>
   633→                    {homes.map((h) => (
   634→                      <SelectItem key={h.id} value={h.id}>
   635→                        {h.label}
   636→                      </SelectItem>
   637→                    ))}
   638→                  </SelectContent>
   639→                </Select>
   640→              )}
   641→
   642→              <Tabs
   643→                value={room?.key ?? ''}
   644→                onValueChange={(key) => {
   645→                  setRoomKey(key);
   646→                  setSelected(null);
   647→                }}
   648→              >
   649→                <TabsList className="h-auto w-full flex-wrap justify-start">
   650→                  {home.rooms.map((r) => (
   651→                    <TabsTrigger
   652→                      key={r.key}
   653→                      value={r.key}
   654→                      aria-label={r.access === 'locked' ? `${r.label} (cerrada)` : r.label}
   655→                    >
   656→                      {r.label}
   657→                      {r.access === 'locked' && <Lock className="size-3" aria-hidden />}
   658→                    </TabsTrigger>
   659→                  ))}
   660→                </TabsList>
   661→
   662→                {home.rooms.map((r) => (
   663→                  <TabsContent key={r.key} value={r.key} className="flex flex-col gap-3">
   664→                    {r.access === 'locked' ? (
   665→                      <LockedRoomBody
   666→                        roomLabel={r.label}
   667→                        price={furnitureConfig.roomUnlockPrice}
   668→                        affordable={wallet >= furnitureConfig.roomUnlockPrice}
   669→                        onRenovate={() => handleRenovate(r.key)}
   670→                      />
   671→                    ) : (
   672→                      <>
   673→                        {/* Plano de la estancia con las ranuras posicionadas (objetivos del drag) */}
   674→                        <div className="relative h-40 w-full overflow-hidden rounded-lg border bg-muted/50 sm:h-48">
   675→                          {r.slots.length === 0 ? (
   676→                            <p className="absolute inset-0 flex items-center justify-center px-4 text-center text-xs text-muted-foreground">
   677→                              Esta estancia no tiene ranuras. Minimalismo forzado.
   678→                            </p>
   679→                          ) : (
   680→                            r.slots.map((slot) => {
   681→                              const compatibleWithDrag =
   682→                                drag !== null && drag.item.slots.includes(slot.kind);
   683→                              const isEmpty = slot.item === null;
   684→                              const isHoveredTarget =
   685→                                hovered !== null &&
   686→                                hovered.roomKey === r.key &&
   687→                                hovered.slotId === slot.id;
   688→
   689→                              let dropHint: DropHint = null;
   690→                              if (drag !== null) {
   691→                                if (compatibleWithDrag && isEmpty) dropHint = 'ok';
   692→                                else if (!compatibleWithDrag) dropHint = 'no';
   693→                              }
   694→                              const hoverHint: DropHint =
   695→                                drag !== null && isHoveredTarget
   696→                                  ? compatibleWithDrag && isEmpty
   697→                                    ? 'ok'
   698→                                    : 'no'
   699→                                  : null;
   700→
   701→                              return (
   702→                                <SlotChip
   703→                                  key={slot.id}
   704→                                  slot={slot}
   705→                                  roomKey={r.key}
   706→                                  selected={activeSlot?.id === slot.id}
   707→                                  dropHint={dropHint}
   708→                                  hoverHint={hoverHint}
   709→                                  onSelect={() => handleSlotSelect(r.key, slot)}
   710→                                />
   711→                              );
   712→                            })
   713→                          )}
   714→                        </div>
   715→
   716→                        {/* Ranura ocupada: retirada con reembolso */}
   717→                        {activeSlot?.item && (
   718→                          <div className="flex items-center justify-between gap-2 rounded-md border bg-background px-3 py-2">
   719→                            <p className="min-w-0 truncate text-sm">
   720→                              <span aria-hidden>{activeSlot.item.emoji}</span> {activeSlot.item.name}
   721→                            </p>
   722→                            <Button
   723→                              variant="outline"
   724→                              size="sm"
   725→                              onClick={handleRemove}
   726→                              aria-label={`Quitar ${activeSlot.item.name} y recuperar ${refundOf(activeSlot.item)} euros`}
   727→                              className="shrink-0 border-rose-500/40 text-rose-600 hover:bg-rose-500/10 hover:text-rose-700 dark:text-rose-400"
   728→                            >
   729→                              <Trash2 className="size-4" aria-hidden />
   730→                              Quitar (+{refundOf(activeSlot.item)} €)
   731→                            </Button>
   732→                          </div>
   733→                        )}
   734→
   735→                        {/* Ranura vacía seleccionada: catálogo filtrado por tipo de ranura */}
   736→                        {activeSlot && !activeSlot.item && (
   737→                          <div className="flex flex-col gap-2">
   738→                            <p className="text-xs font-medium tracking-wide text-muted-foreground uppercase">
   739→                              Catálogo · ranura «{activeSlot.kind}»
   740→                            </p>
   741→                            <ScrollArea className="max-h-72">
   742→                              {compatible.length === 0 ? (
   743→                                <p className="px-1 py-3 text-sm text-muted-foreground">
   744→                                  Nada del catálogo encaja aquí. Ni el mueble de la suegra.
   745→                                </p>
   746→                              ) : (
   747→                                <div className="grid grid-cols-1 gap-2 pr-2 sm:grid-cols-2">
   748→                                  {compatible.map((item) => (
   749→                                    <CatalogItemCard
   750→                                      key={item.id}
   751→                                      item={item}
   752→                                      channels={catalog.channels}
   753→                                      affordable={wallet >= item.price}
   754→                                      onPlace={() => handlePlace(item)}
   755→                                      onDragStart={(event) => startPotentialDrag(event, item)}
   756→                                      onContextMenu={handleCardContextMenu}
   757→                                    />
   758→                                  ))}
   759→                                </div>
   760→                              )}
   761→                              <ScrollBar className="w-1.5 [&_[data-slot=scroll-area-thumb]]:bg-amber-500/40" />
   762→                            </ScrollArea>
   763→                          </div>
   764→                        )}
   765→                      </>
   766→                    )}
   767→                  </TabsContent>
   768→                ))}
   769→              </Tabs>
   770→            </div>
   771→          )}
   772→        </CardContent>
   773→      </Card>
   774→
   775→      {/* Fantasma del drag: fijo, sin eventos de puntero y oculto para lectores de pantalla */}
   776→      {drag && (
   777→        <div
   778→          ref={ghostRef}
   779→          aria-hidden
   780→          style={{ left: drag.x, top: drag.y }}
   781→          className="pointer-events-none fixed z-50 flex -translate-x-1/2 -translate-y-1/2 flex-col items-center gap-1"
   782→        >
   783→          <div
   784→            className="flex max-w-40 rotate-2 scale-105 items-center gap-1.5 rounded-full px-3 py-1.5 text-xs font-medium text-white shadow-xl ring-2 ring-amber-500"
   785→            style={{ backgroundColor: withAlpha(drag.item.view.color, 0.9) }}
   786→          >
   787→            {drag.item.emoji && <span aria-hidden>{drag.item.emoji}</span>}
   788→            <span className="truncate">{drag.item.name}</span>
   789→          </div>
   790→          <span className="rounded-full bg-neutral-900/80 px-2 py-0.5 text-[10px] text-amber-200">
   791→            {drag.item.price} €
   792→          </span>
   793→        </div>
   794→      )}
   795→    </>
   796→  );
   797→}
