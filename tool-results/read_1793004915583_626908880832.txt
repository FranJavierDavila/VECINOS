  1025→  function color(b: BlockInstance): Color3 {
  1026→    const base = Color3.FromHexString(content.blocks[b.defId]!.view?.color ?? '#cccccc');
  1027→    if (b.kind === 'dwelling' && b.access === 'locked') return Color3.FromHexString(v.lockedColor);

  1041→  /** Emissive del bloque: canales de influencia → emissive según color del canal en el catálogo. Se acumulan. */

  1054→  /** Color difuso/albedo por material: greybox plano (tintStrength 1) o mezcla sobre el arte. */

  1109→  /**
  1110→   * LOD por distancia (rules.view.lod). Se ejecuta SIEMPRE al final de update(): con foco degrada
  1111→   * cada bloque según su distancia, y sin foco (o con lod.enabled=false) restaura el estado completo.
  1112→   * Idempotente y sin allocations por frame: solo aritmética, booleanos y asignaciones.
  1113→   *
  1114→   * Métrica (Vista · LOD por distancia): dx = |centroX del bloque − focus.x| en metros y
  1115→   * dy = |planta del bloque − focus.floor| * view.cellH en metros; dist = Math.hypot(dx, dy).
  1116→   * Nivel efectivo = todos los niveles de lod.levels cuyo `dist` ≤ distBloque (el `hide` es
  1117→   * acumulado: un flag se oculta si cualquier nivel alcanzado lo lista; sin nivel → todo visible).
  1118→   */
  1119→  function applyLod(focus: { x: number; floor: number } | null) {
  1120→    const lod = v.lod;
  1121→    for (const id in world.blocks) {
  1122→      const b = world.blocks[id]!;
  1123→      const view = views.get(id)!;
  1124→      let hideEdges = false;
  1125→      let hideGlow = false;
  1126→      let hidePanels = false;
  1127→      if (lod.enabled && focus) {
  1128→        const dx = Math.abs(view.centerX - focus.x);
  1129→        const dy = Math.abs(view.floor - focus.floor) * v.cellH;
  1130→        const dist = Math.hypot(dx, dy);
  1131→        for (let i = 0; i < lod.levels.length; i++) {
  1132→          const level = lod.levels[i];
  1133→          if (level.dist > dist) continue;
  1134→          for (let j = 0; j < level.hide.length; j++) {
  1135→            const flag = level.hide[j];
  1136→            if (flag === 'edges') hideEdges = true;
  1137→            else if (flag === 'glow') hideGlow = true;
  1138→            else if (flag === 'panels') hidePanels = true;
  1139→          }
  1140→        }
  1141→      }
  1142→      // 'edges': interruptor nativo barato — el renderingGroup consulta edgesRenderer.isEnabled cada
  1143→      // frame, así que nunca se destruye ni recrea el EdgesRenderer (disable/enableEdgesRendering sí
  1144→      // lo harían: el primero lo dispone y el segundo recomputaría las aristas). El grosor/color
  1145→      // viven en la malla y sobreviven al apagado. Los modelos glTF no tienen edges (edgesMesh null).
  1146→      if (view.edgesMesh?.edgesRenderer) view.edgesMesh.edgesRenderer.isEnabled = !hideEdges;
  1147→      if (view.edgesMesh2?.edgesRenderer) view.edgesMesh2.edgesRenderer.isEnabled = !hideEdges;
  1148→      // 'glow': pisa el emissive CADA frame (paint() pudo repintar los canales este tick); al soltarlo,
  1149→      // una única llamada repinta el cálculo de paint para no esperar al próximo tick.
  1150→      if (hideGlow) {
  1151→        view.glowForced = true;
  1152→        for (let i = 0; i < view.mats.length; i++) view.mats[i]!.emissiveColor = view.baseEmissive[i]!;
  1153→      } else if (view.glowForced) {
  1154→        view.glowForced = false;
  1155→        applyEmissive(view, b);
  1156→      }
  1157→      // 'panels': solo gira la bandera (la visibilidad la calcula applyPanels); en la transición se
  1158→      // aplica ya para no esperar al próximo paint(), que reproducirá el mismo resultado.
  1159→      if (view.lodHiddenPanels !== hidePanels) {
  1160→        view.lodHiddenPanels = hidePanels;
  1161→        applyPanels(view, b);
  1162→      }
  1163→    }
  1164→  }

  1190→    /** Sacude el candado de una puerta cerrada. */
  1191→    shake(blockId: string) {
  1192→      shakes.set(blockId, 0.4);
  1193→    },
  1194→    /**
  1195→     * Apilamiento 3D (v1.19): fase del bloque trasero de `blockId`.
  1196→     *  · 'idle'   — bloque inactivo (todo como siempre).
  1197→     *  · 'peek'   — el jugador se acerca a la puerta del fondo: el bloque trasero asoma por
  1198→     *               el hueco (transparencia → visible) y la hoja se abre.
  1199→     *  · 'inside' — cruzaste: el bloque principal SE DESVANECE y el trasero queda en primer
  1200→     *               plano con su contorno rojo.
  1201→     */
  1202→    setRearPhase(blockId: string, phase: 'idle' | 'peek' | 'inside'): void {
  1203→      const r = rears.get(blockId);
  1204→      if (r) r.phase = phase;
  1205→    },
  1206→    /** Rects de obstáculos del interior del trastero propio (v1.21): estanterías y trastos
  1207→     *  registrados al levantar la geometría — player-logic los esquiva al moverse dentro. */
  1208→    cellObstacles(): { x0: number; z0: number; w: number; d: number }[] {
  1209→      return cellObstacleRects;
  1210→    },
  1211→    hasRear(blockId: string): boolean {
  1212→      return rears.has(blockId);
  1213→    },
  1214→    /**
  1215→     * dt en segundos; `focus` es el punto de interés (posición del jugador) para el
  1216→     * LOD por distancia (rules.view.lod): applyLod() lo aplica SIEMPRE al final del frame
  1217→     * (con focus undefined, p. ej. cámara 'edificio', restaura el estado completo).
  1218→     * `inside` es el bloque donde está el jugador ahora (pos.insideBlock): alimenta la
  1219→     * privacidad y re-evalúa las fachadas CADA frame — un vecino que llega a casa abre
  1220→     * su interior al instante, sin esperar al siguiente tick.
  1221→     * `pos` (v1.19) alimenta los carteles guía de trasteros y el apilamiento 3D.
  1222→     */
  1223→    update(dt: number, focus?: { x: number; floor: number }, inside?: string | null, pos?: { x: number; z: number; insideBlock: string | null }, env?: { daylight: number }) {
  1224→      if (focus) lastFocus = focus;
  1225→      if (inside !== undefined) playerInside = inside;
  1226→      // PRIVACIDAD (v1.17) cada frame: la fachada de una vivienda sin nadie dentro se cierra
  1227→      // con su puerta al pasillo y se abre en cuanto alguien entra (propietario, visita…
  1228→      // o un ladrón). Vive en update() y no en paint() para reaccionar incluso con el
  1229→      // mundo congelado (modal abierto) y sin esperar al siguiente tick.
  1230→      const occSet = computeOccupancy();
  1231→      if (privacyOn) {
  1232→        for (const b of Object.values(world.blocks)) {
  1233→          if (b.kind !== 'dwelling') continue;
  1234→          applyPanels(views.get(b.id)!, b, occSet);
  1235→        }
  1236→      }
  1237→      // LUCES INTERIORES NOCTURNAS (v1.21): al caer el sol, las estancias que se ven abiertas
  1238→      // (alguien dentro) se ILUMINAN salvo que su inquilino duerma — entonces, a oscuras.
  1239→      // Colores cacheados: cero allocations por frame. De día, lámparas apagadas.
  1240→      {
  1241→        const day = env?.daylight ?? 1;
  1242→        const litWorld = day < 0.4;
  1243→        for (const [blockId, interior] of interiors) {
  1244→          let lit = false;
  1245→          if (litWorld && occSet.has(blockId)) {
  1246→            lit = true;
  1247→            for (const r of Object.values(world.residents)) {
  1248→              if (r.present && r.location === blockId && r.activity === 'sleep') {
  1249→                lit = false;
  1250→                break;
  1251→              }
  1252→            }
  1253→          }
  1254→          for (const lamp of interior.lamps) {
  1255→            lamp.mesh.setEnabled(lit);
  1256→            if (lit) lamp.mat.emissiveColor = nightLampLit;
  1257→          }
  1258→          for (const f of interior.floors) f.mat.emissiveColor = lit ? nightFloorGlow : nightDark;
  1259→          // Diorama (v1.23): el arte ilustrado sigue la misma luz que suelos y lámparas.
  1260→          for (const m of interior.dioMats) m.emissiveColor = lit ? nightDioGlow : nightDark;
  1261→        }
  1262→      }
  1263→      // CARTELES GUÍA (v1.19): visibles solo con el jugador dentro del bloque de trasteros;
  1264→      // texto = los trasteros que tienes a cada lado según la profundidad del pasillo.
  1265→      const sgG = colBlocks.length > 0 ? storageGeomOf(v) : null;
  1266→      for (const [blkId, sign] of guideSigns) {
  1267→        const active = !!pos && pos.insideBlock === blkId && !!sgG;
  1268→        const plate = scene.getMeshByName(`${blkId}_guia`);
  1269→        const frame = scene.getMeshByName(`${blkId}_guia_marco`);
  1270→        plate?.setEnabled(active);
  1271→        frame?.setEnabled(active);
  1272→        if (!active || !sgG) continue;
  1273→        const localZ = pos!.z; // en modo 'inside' de trasteros, z es profundidad local
  1274→        const usable = sgG.depth - 2 * sgG.wallT;
  1275→        const perSide = new Map<string, number>();
  1276→        for (const u of sign.plan) perSide.set(u.side, (perSide.get(u.side) ?? 0) + 1);
  1277→        const labelOf = (side: 'left' | 'right'): string => {
  1278→          const count = perSide.get(side) ?? 0;
  1279→          if (count === 0) return tr.emptyLabel;
  1280→          const idx = Math.max(0, Math.min(count - 1, Math.floor((localZ - sgG.wallT) / (usable / count))));
  1281→          const unit = sign.plan.find((u) => u.side === side && u.index === idx);
  1282→          return unit ? blockLabel(`blk_p${unit.floor}_${sign.ownTag}`) : tr.emptyLabel;
  1283→        };
  1284→        const left = labelOf('left');
  1285→        const right = labelOf('right');
  1286→        const key = `${left}|${right}`;
  1287→        if (sign.last !== key) {
  1288→          sign.last = key;
  1289→          const ctx = sign.tex.getContext() as unknown as CanvasRenderingContext2D;
  1290→          ctx.fillStyle = tr.guide.bg;
  1291→          ctx.fillRect(0, 0, 256, 60);
  1292→          ctx.font = 'bold 30px Arial, sans-serif';
  1293→          ctx.textBaseline = 'middle';
  1294→          ctx.fillStyle = tr.guide.text;
  1295→          ctx.textAlign = 'left';
  1296→          ctx.fillText('◀', 10, 32);
  1297→          ctx.textAlign = 'center';
  1298→          const ownLeft = left.includes(sign.ownTag.toUpperCase());
  1299→          const ownRight = right.includes(sign.ownTag.toUpperCase());
  1300→          ctx.fillStyle = ownLeft ? tr.ownBg : tr.guide.text;
  1301→          ctx.fillText((ownLeft ? '★ ' : '') + left, 88, 32);
  1302→          ctx.fillStyle = ownRight ? tr.ownBg : tr.guide.text;
  1303→          ctx.fillText((ownRight ? '★ ' : '') + right, 176, 32);
  1304→          ctx.fillStyle = tr.guide.text;
  1305→          ctx.textAlign = 'right';
  1306→          ctx.fillText('▶', 248, 32);
  1307→          sign.tex.update(true);
  1308→        }
  1309→      }
  1310→      // FACHADA DE TRASTERO (v1.21): mientras el jugador camina el pasillo central de un
  1311→      // bloque, su fachada se desvanece (corte de muñecas) y la cámara frontal ve TODO el
  1312→      // pasillo hasta el fondo. Al salir, la fachada vuelve. Fundido del catálogo.
  1313→      for (const [blkId, parts] of storageFacades) {
  1314→        const target = pos && pos.insideBlock === blkId ? 0 : 1;
  1315→        let alpha = facadeAlphas.get(blkId) ?? 1;
  1316→        if (alpha !== target) {
  1317→          alpha += Math.sign(target - alpha) * Math.min(Math.abs(target - alpha), dt / Math.max(0.05, v.rearBlock.fadeSeconds));
  1318→          if (Math.abs(target - alpha) < 0.01) alpha = target;
  1319→          facadeAlphas.set(blkId, alpha);
  1320→          for (const m of parts) {
  1321→            m.visibility = alpha;
  1322→            m.setEnabled(alpha > 0.02);
  1323→          }
  1324→        }
  1325→      }
  1326→      // APILAMIENTO 3D (v1.19): avanzar los fundidos de los bloques traseros.
  1327→      for (const r of rears.values()) {
  1328→        const mainT = r.phase === 'inside' ? 0 : 1;
  1329→        const rearT = r.phase === 'idle' ? 0 : 1;
  1330→        // v1.20: la PARED DIVISORIA del bloque trasero desaparece al entrar (fase 'inside').
  1331→        // Se evalúa SIEMPRE (incluso con los fundidos ya asentados): es un cambio de fase,
  1332→        // no de alpha — el early-continue de abajo no la vería.
  1333→        const frontOpen = r.phase === 'inside';
  1334→        for (const f of r.frontParts) f.setEnabled(!frontOpen);
  1335→        if (r.settled && r.mainAlpha === mainT && r.rearAlpha === rearT) continue;
  1336→        const mRate = dt / Math.max(0.05, rb.fadeSeconds);
  1337→        const rRate = dt / Math.max(0.05, rb.peekSeconds);
  1338→        r.mainAlpha += Math.sign(mainT - r.mainAlpha) * Math.min(Math.abs(mainT - r.mainAlpha), mRate);
  1339→        r.rearAlpha += Math.sign(rearT - r.rearAlpha) * Math.min(Math.abs(rearT - r.rearAlpha), rRate);
  1340→        r.settled = r.mainAlpha === mainT && r.rearAlpha === rearT;
  1341→        const mainKids = r.mainRoot.getChildMeshes();
  1342→        for (const k of mainKids) k.visibility = r.mainAlpha;
  1343→        r.mainRoot.setEnabled(r.mainAlpha > 0.002);
  1344→        const rearKids = r.node.getChildMeshes();
  1345→        for (const k of rearKids) k.visibility = r.rearAlpha;
  1346→        r.node.setEnabled(r.rearAlpha > 0.002);
  1347→        for (const o of r.outline) o.setEnabled(r.mainAlpha < 0.55 && r.rearAlpha > 0.4);
  1348→        if (r.doorLeaf) r.doorLeaf.setEnabled(r.rearAlpha < 0.2);
  1349→      }
  1350→      for (const [id, left] of shakes) {
  1351→        const view = views.get(id)!;
  1352→        const t = left - dt;
  1353→        if (t <= 0) {
  1354→          shakes.delete(id);
  1355→          view.lock?.position.set(view.lockBase.x, view.lockBase.y, view.lock.position.z);
  1356→        } else {
  1357→          shakes.set(id, t);
  1358→          view.lock?.position.set(view.lockBase.x + Math.sin(t * 60) * 0.06, view.lockBase.y, view.lock.position.z);
  1359→        }
  1360→      }
  1361→      // LOD SIEMPRE al final del frame: con foco aplica el nivel efectivo por bloque,
  1362→      // sin foco restaura edges/emissive/paneles al estado completo.
  1363→      applyLod(focus ?? null);
  1364→    },
  1365→  };
