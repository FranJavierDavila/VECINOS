     1→import type { Color3, Color4, Mesh, Scene, StandardMaterial, TransformNode } from '@babylonjs/core';
     2→import { blockLabel } from '@/game/core/types';
     3→import type { BlockInstance, Content, World } from '@/game/core/types';
     4→import { hash01 } from '@/game/interior';
     5→import { storageCorridorX, storageGeomOf, storageUnitPlan, storageUnitZ } from '@/game/storage';
     6→import { B } from './babylon';
     7→import { instantiateBlockAsset, type BlockMat } from './assets';
     8→
     9→interface BlockView {
    10→  /** Cajas greybox estructurales (se ocultan cuando llega el modelo glTF; panel/candado NO van aquí). */
    11→  greybox: Mesh[];
    12→  /** Materiales que reciben glow de canales y tintado de estado (1 en greybox, N en modelos). */
    13→  mats: BlockMat[];
    14→  /** Emissive base por material (lo que paint() deja sin canales); lo usa el LOD 'glow'. */
    15→  baseEmissive: Color3[];
    16→  /** Color base por material (view.color en greybox; albedo capturado en modelos). */
    17→  baseColor: Color3[];
    18→  /** Fuerza del tintado de estado: 1 = color plano (greybox), 0..1 = mezcla sobre el arte (modelos). */
    19→  tintStrength: number;
    20→  /** Raíz del modelo glTF si este bloque ya lo lleva (null = greybox). */
    21→  model: TransformNode | null;
    22→  front: Mesh | null; // fachada que cierra la vivienda (candado o privacidad)
    23→  /** Puerta y picaporte de la fachada de privacidad (v1.17): la vivienda vacía se ve cerrada. */
    24→  door: Mesh | null;
    25→  handle: Mesh | null;
    26→  lock: Mesh | null;
    27→  lockBase: { x: number; y: number };
    28→  // ── LOD por distancia (rules.view.lod) ──
    29→  /** Centro X del bloque en metros (dx del LOD: |centerX − focus.x|). */
    30→  centerX: number;
    31→  /** Planta del bloque (dy del LOD: |floor − focus.floor| * view.cellH). */
    32→  floor: number;
    33→  /** Pared del fondo de la vivienda (la única malla que puede llevar edgesRendering) o null;
    34→   *  el EdgesRenderer activo solo existe en las paredes destacadas del jugador. */
    35→  edgesMesh: Mesh | null;
    36→  /** true mientras el LOD fuerza baseEmissive (al soltarlo se repinta el cálculo de paint una vez). */
    37→  glowForced: boolean;
    38→  /** true mientras el LOD oculta panel+candado; la visibilidad efectiva es la de paint() && !esto. */
    39→  lodHiddenPanels: boolean;
    40→}
    41→
    42→/** Interior 2D de una vivienda (Fase C): suelo por estancia + tapa de puerta hacia estancia cerrada. */
    43→interface InteriorView {
    44→  /** Suelo tintado por estancia (color del catálogo; lockedColor si está cerrada). */
    45→  floors: { roomKey: string; defId: string; mesh: Mesh; mat: StandardMaterial }[];
    46→  /** Tapas de los huecos de puerta que llevan a una estancia cerrada ("en obras"). */
    47→  covers: { mesh: Mesh; rooms: [string, string] }[];
    48→}
    49→
    50→/**
    51→ * Una caja por bloque (placeholder greybox del futuro prefab) y, si el manifiesto de assets
    52→ * (Content.assets.blocks) declara un modelo glTF para el tipo de bloque, el modelo en su lugar
    53→ * (con la caja oculta). TODA la geometría y los colores salen de content.rules.view +
    54→ * block.view + assets: sustituir el greybox por arte final no toca este código.
    55→ * Las viviendas se dibujan en corte (interior visible) SOLO mientras hay alguien dentro;
    56→ * vacías, la fachada se cierra con su puerta al pasillo (rules.view.privacy) — y las
    57→ * bloqueadas, con panel y candado como siempre.
    58→ */
    59→export function createBuilding(scene: Scene, world: World, content: Content) {
    60→  const { Color3, Color4, MeshBuilder, StandardMaterial } = B();
    61→  const v = content.rules.view;
    62→  const glow = Object.fromEntries(Object.entries(v.channelGlow).map(([k, hex]) => [k, Color3.FromHexString(hex)])) as Record<string, Color3>;
    63→  const views = new Map<string, BlockView>();
    64→  const shakes = new Map<string, number>();
    65→  /** Interiores 2D (Fase C) por bloque: suelos de estancia y tapas de puertas cerradas. */
    66→  const interiors = new Map<string, InteriorView>();
    67→  /** Foco del LOD (posición del jugador): lo alimenta update() y lo consume paint(). */
    68→  let lastFocus: { x: number; floor: number } | null = null;
    69→  // Plano frontal de las estancias: la cara de la puerta. Con viviendas del doble de calado
    70→  // (v0.6-b), el extra crece hacia el FONDO (+Z) para no envolver el pasillo ni al núcleo.
    71→  const frontZ = v.roomZ - v.commonDepth / 2;
    72→  /** Calado por bloque: viviendas 2× (mobiliario interior); naves del cierre (v1.16) y el
    73→   *  tejado (con aleros) a calado de vivienda; bloques comunes, el de siempre. */
    74→  const depthOf = (b: BlockInstance): number => {
    75→    if (b.kind === 'dwelling') return v.depth;
    76→    if (b.defId === 'trasteros_planta' || b.defId === 'bajocubierta') return v.depth;
    77→    if (b.defId === v.trasteros.blockDefId) return v.depth * v.trasteros.blockDepthFactor;
    78→    if (b.defId === 'tejado') return v.depth + v.roof.overhang * 2;
    79→    return v.commonDepth;
    80→  };
    81→  const mat = (name: string) => new StandardMaterial(name, scene);
    82→  const box = (name: string, w: number, h: number, d: number, x: number, y: number, z: number, m: StandardMaterial) => {
    83→    const mesh = MeshBuilder.CreateBox(name, { width: w, height: h, depth: d }, scene);
    84→    mesh.position.set(x, y, z);
    85→    mesh.material = m;
    86→    return mesh;
    87→  };
    88→
    89→  for (const b of Object.values(world.blocks)) {
    90→    const m = mat(`${b.id}_mat`);
    91→    const cx = (b.cell.col + b.size.w / 2) * v.cellW;
    92→    const bottom = b.cell.floor * v.cellH;
    93→    const w = b.size.w * v.cellW - v.gapW;
    94→    const h = v.cellH - v.gapH;
    95→    const d = depthOf(b);
    96→    const cz = frontZ + d / 2;
    97→    const view: BlockView = {
    98→      greybox: [],
    99→      mats: [m],
   100→      baseEmissive: [m.emissiveColor.clone()],
   101→      baseColor: [Color3.FromHexString(content.blocks[b.defId]!.view?.color ?? '#cccccc')],
   102→      tintStrength: 1,
   103→      model: null,
   104→      front: null,
   105→      door: null,
   106→      handle: null,
   107→      lock: null,
   108→      lockBase: { x: cx, y: bottom + v.cellH / 2 - 0.2 },
   109→      centerX: cx,
   110→      floor: b.cell.floor,
   111→      edgesMesh: null,
   112→      glowForced: false,
   113→      lodHiddenPanels: false,
   114→    };
   115→    if (b.layer === 'corridor') {
   116→      // Pasillo exterior (data-driven): tira transitable saliente hacia la calle, con el borde
   117→      // trasero a ras de fachada (frontZ) y barandilla baja en el borde exterior — sitio de
   118→      // sobra para que los personajes se crucen o charlen sin bailar el vacío.
   119→      const d = v.corridorDepth;
   120→      view.greybox.push(box(b.id, b.size.w * v.cellW, v.slab * 2, d, cx, bottom + v.slab, frontZ - d / 2, m));
   121→      if (v.corridorRail > 0) {
   122→        view.greybox.push(
   123→          box(`${b.id}_barandilla`, b.size.w * v.cellW, v.corridorRail, v.wall, cx, bottom + v.slab * 2 + v.corridorRail / 2, frontZ - d, m),
   124→        );
   125→      }
   126→    } else if (b.kind === 'dwelling') {
   127→      const y = bottom + v.cellH / 2;
   128→      view.edgesMesh = box(`${b.id}_fondo`, w, h, v.wall, cx, y, frontZ + d, m);
   129→      view.greybox.push(
   130→        view.edgesMesh,
   131→        box(`${b.id}_izq`, v.wall, h, d, cx - w / 2, y, cz, m),
   132→        box(`${b.id}_der`, v.wall, h, d, cx + w / 2, y, cz, m),
   133→        box(`${b.id}_suelo`, w, v.wall, d, cx, bottom + v.slab + v.wall / 2, cz, m),
   134→        box(`${b.id}_techo`, w, v.wall, d, cx, bottom + v.cellH - v.slab - v.wall / 2, cz, m),
   135→      );
   136→      view.front = box(`${b.id}_puerta`, w, h, v.wall, cx, y, frontZ, m);
   137→      view.lock = box(`${b.id}_candado`, 0.3, 0.38, 0.12, cx, view.lockBase.y, frontZ - 0.1, mat(`${b.id}_candado_mat`));
   138→      (view.lock.material as StandardMaterial).diffuseColor = Color3.FromHexString(v.lockColor);
   139→
   140→      // ── Privacidad (v1.17): puerta y picaporte sobre la fachada de la vivienda vacía.
   141→      // Se activan/desactivan cada frame según haya alguien dentro (propietario, visita…
   142→      // o un ladrón); medidas y colores del catálogo (rules.view.privacy).
   143→      const pv = v.privacy;
   144→      view.door = box(
   145→        `${b.id}_puerta_priv`,
   146→        pv.doorW,
   147→        pv.doorH,
   148→        0.06,
   149→        cx,
   150→        bottom + v.slab + v.wall + pv.doorH / 2,
   151→        frontZ - v.wall / 2 - 0.035,
   152→        mat(`${b.id}_puerta_priv_mat`),
   153→      );
   154→      (view.door.material as StandardMaterial).diffuseColor = Color3.FromHexString(pv.doorColor);
   155→      view.handle = box(
   156→        `${b.id}_picaporte`,
   157→        0.07,
   158→        0.07,
   159→        0.05,
   160→        cx + pv.doorW / 2 - 0.14,
   161→        bottom + v.slab + v.wall + pv.doorH * 0.55,
   162→        frontZ - v.wall / 2 - 0.07,
   163→        mat(`${b.id}_picaporte_mat`),
   164→      );
   165→      (view.handle.material as StandardMaterial).diffuseColor = Color3.FromHexString(pv.handleColor);
   166→      view.door.setEnabled(false);
   167→      view.handle.setEnabled(false);
   168→
   169→      // ── Cartel numerado (v1.6): rótulo sobre la puerta con el identificador de la vivienda
   170→      // (blk_p3_c → "3C", blockLabel). Textura dinámica con colores/fuente del catálogo
   171→      // (rules.view.plaque); se añade a greybox porque la vivienda usa modelo OVERLAY (el
   172→      // greybox no se oculta) y no entra en mats: es señalización, no tinte ni glow.
   173→      const pl = v.plaque;
   174→      const plate = MeshBuilder.CreatePlane(`${b.id}_cartel`, { width: pl.w, height: pl.h, sideOrientation: B().Mesh.DOUBLESIDE }, scene);
   175→      const tex = new (B().DynamicTexture)(`${b.id}_cartel_tex`, { width: 256, height: 96 }, scene, true);
   176→      const ctx = tex.getContext() as unknown as CanvasRenderingContext2D;
   177→      ctx.fillStyle = pl.bg;
   178→      ctx.fillRect(0, 0, 256, 96);
   179→      ctx.font = `bold ${pl.fontPx}px Arial, sans-serif`;
   180→      ctx.fillStyle = pl.text;
   181→      ctx.textAlign = 'center';
   182→      ctx.textBaseline = 'middle';
   183→      ctx.fillText(blockLabel(b.id), 128, 52);
   184→      // update(true): el flag es invertY. Con false el canvas se sube SIN voltear y el texto
   185→      // sale CABEZA ABAJO (la "A" se ve como "V"; B/C/D pasan por simétricas y esquivan el ojo).
   186→      tex.update(true);
   187→      const pmat = new StandardMaterial(`${b.id}_cartel_mat`, scene);
   188→      pmat.diffuseTexture = tex;
   189→      pmat.emissiveColor = Color3.FromHexString(pl.bg).scale(0.55); // legible en fachada norte
   190→      pmat.specularColor = new Color3(0, 0, 0);
   191→      plate.material = pmat;
   192→      plate.position.set(cx, bottom + pl.y, frontZ + pl.z);
   193→      view.greybox.push(plate);
   194→
   195→      // ── Interior 2D (Fase C): tabiques con huecos, suelo por estancia y tapas "en obras". ──
   196→      // TODO sale del layout del catálogo (rects + doorways); sin rects no hay interior 2D.
   197→      const layoutId = content.blocks[b.defId]?.container?.layout;
   198→      const layout = layoutId ? content.layouts[layoutId] : undefined;
   199→      const geoRooms = (layout?.rooms ?? []).filter((r) => r.rect !== undefined);
   200→      if (geoRooms.length > 0 && layout) {
   201→        const interior: InteriorView = { floors: [], covers: [] };
   202→        const x0 = b.cell.col * v.cellW;
   203→        const wallH = v.cellH - 2 * v.slab - 2 * v.wall;
   204→        const partMat = mat(`${b.id}_tabique_mat`);
   205→        partMat.diffuseColor = Color3.FromHexString('#d8d2c4');
   206→        // Suelo tintado por estancia (plancha fina sobre el suelo, alpha suave).
   207→        for (const lr of geoRooms) {
   208→          const r = lr.rect!;
   209→          const fmat = mat(`${b.id}_${lr.key}_suelo_mat`);
   210→          fmat.diffuseColor = Color3.FromHexString(content.rooms[lr.defId]?.view?.color ?? '#cccccc');
   211→          fmat.alpha = 0.32;
   212→          const fm = box(`${b.id}_${lr.key}_suelo`, r.w, 0.02, r.d, x0 + r.x + r.w / 2, bottom + v.slab + v.wall + 0.02, frontZ + r.z + r.d / 2, fmat);
   213→          interior.floors.push({ roomKey: lr.key, defId: lr.defId, mesh: fm, mat: fmat });
   214→        }
   215→        // Tabiques: para cada par de estancias que comparten borde, pared entre ambas.
   216→        // Los doorways del catálogo recortan el hueco (y añaden tapa visible si alguna de las
   217→        // dos estancias está cerrada). Sin hueco declarado → tabique sólido (tampoco se puede
   218→        // pasar: walkable() exige doorway). Tabiques según X a media altura para no tapar la vista.
   219→        const eps = 0.01;
   220→        const seg = (a: number, b: number, along: 'x' | 'z', at: number, height: number, yC: number, name: string) => {
   221→          const len = b - a;
   222→          if (len <= 0.01) return;
   223→          view.greybox.push(
   224→            along === 'z'
   225→              ? box(name, v.wall, height, len, at, yC, (a + b) / 2, partMat)
   226→              : box(name, len, height, v.wall, (a + b) / 2, yC, at, partMat),
   227→          );
   228→        };
   229→        geoRooms.forEach((lr, i) => {
   230→          geoRooms.forEach((other, j) => {
   231→            if (j <= i) return;
   232→            const ra = lr.rect!;
   233→            const rb = other.rect!;
   234→            const door = (layout.doorways ?? []).find(
   235→              (dw) => (dw.a === lr.key && dw.b === other.key) || (dw.a === other.key && dw.b === lr.key),
   236→            );
   237→            // Borde vertical compartido → tabique que corre según Z (se cruza andando en X).
   238→            const touchX = Math.abs(ra.x + ra.w - rb.x) < eps || Math.abs(rb.x + rb.w - ra.x) < eps;
   239→            const overlapZ = Math.min(ra.z + ra.d, rb.z + rb.d) - Math.max(ra.z, rb.z);
   240→            if (touchX && overlapZ > eps) {
   241→              const wx = x0 + Math.max(ra.x, rb.x);
   242→              const z0 = frontZ + Math.max(ra.z, rb.z);
   243→              const z1 = frontZ + Math.min(ra.z + ra.d, rb.z + rb.d);
   244→              const yC = bottom + v.slab + v.wall + wallH / 2;
   245→              if (door && door.axis === 'z') {
   246→                const g0 = Math.max(z0, frontZ + door.at - door.width / 2);
   247→                const g1 = Math.min(z1, frontZ + door.at + door.width / 2);
   248→                seg(z0, g0, 'z', wx, wallH, yC, `${b.id}_tab_${i}_${j}_i`);
   249→                seg(g1, z1, 'z', wx, wallH, yC, `${b.id}_tab_${i}_${j}_d`);
   250→                const cover = box(`${b.id}_tapa_${i}_${j}`, v.wall * 0.8, wallH, g1 - g0, wx, yC, frontZ + door.at, mat(`${b.id}_tapa_${i}_${j}_mat`));
   251→                (cover.material as StandardMaterial).diffuseColor = Color3.FromHexString(v.lockedColor);
   252→                cover.setEnabled(false);
   253→                interior.covers.push({ mesh: cover, rooms: [lr.key, other.key] });
   254→              } else {
   255→                seg(z0, z1, 'z', wx, wallH, yC, `${b.id}_tab_${i}_${j}`);
   256→              }
   257→            }
   258→            // Borde horizontal compartido → tabique que corre según X (se cruza andando en Z).
   259→            const touchZ = Math.abs(ra.z + ra.d - rb.z) < eps || Math.abs(rb.z + rb.d - ra.z) < eps;
   260→            const overlapX = Math.min(ra.x + ra.w, rb.x + rb.w) - Math.max(ra.x, rb.x);
   261→            if (touchZ && overlapX > eps) {
   262→              const wz = frontZ + Math.max(ra.z, rb.z);
   263→              const xa = x0 + Math.max(ra.x, rb.x);
   264→              const xb = x0 + Math.min(ra.x + ra.w, rb.x + rb.w);
   265→              const halfH = wallH * 0.45;
   266→              const yHalf = bottom + v.slab + v.wall + halfH / 2;
   267→              if (door && door.axis === 'x') {
   268→                const g0 = Math.max(xa, x0 + door.at - door.width / 2);
   269→                const g1 = Math.min(xb, x0 + door.at + door.width / 2);
   270→                seg(xa, g0, 'x', wz, halfH, yHalf, `${b.id}_tab_${i}_${j}_i`);
   271→                seg(g1, xb, 'x', wz, halfH, yHalf, `${b.id}_tab_${i}_${j}_d`);
   272→                const cover = box(`${b.id}_tapa_${i}_${j}`, g1 - g0, halfH, v.wall * 0.8, x0 + door.at, yHalf, wz, mat(`${b.id}_tapa_${i}_${j}_mat`));
   273→                (cover.material as StandardMaterial).diffuseColor = Color3.FromHexString(v.lockedColor);
   274→                cover.setEnabled(false);
   275→                interior.covers.push({ mesh: cover, rooms: [lr.key, other.key] });
   276→              } else {
   277→                seg(xa, xb, 'x', wz, halfH, yHalf, `${b.id}_tab_${i}_${j}`);
   278→              }
   279→            }
   280→          });
   281→        });
   282→        interiors.set(b.id, interior);
   283→      }
   284→    } else if (b.defId === 'trasteros_planta' || b.defId === 'bajocubierta' || b.defId === v.trasteros.blockDefId) {
   285→      // Nave de trasteros / sala de instalaciones (v1.16): forjado + trasera. El arte glTF
   286→      // añade pilares (trasteros) o maquinaria (bajocubierta); las estanterías, trastos y
   287→      // carteles numerados los levanta SIEMPRE el runtime desde rules.view.trasteros.
   288→      // El bloque por COLUMNA (v1.18) comparte estructura: sus compartimentos los levanta
   289→      // el runtime más abajo (pasillo central + celdas cerradas impares/pares).
   290→      view.greybox.push(
   291→        box(`${b.id}_forjado`, w, v.slab * 2, d, cx, bottom + v.slab, frontZ + d / 2, m),
   292→        box(`${b.id}_trasera`, w, h, v.wall, cx, bottom + v.cellH / 2, frontZ + d - v.wall / 2, m),
   293→      );
   294→    } else if (b.defId === 'tejado') {
   295→      // Tejadillo (v1.16): fallback = canto de cerramiento (el glTF trae faldones, cumbrera,
   296→      // chimenea y antena).
   297→      view.greybox.push(box(`${b.id}_canto`, w, v.slab * 2, d, cx, bottom + v.slab, frontZ + d / 2, m));
   298→    } else {
   299→      view.greybox.push(box(b.id, w, h, d, cx, bottom + v.cellH / 2, cz, m));
   300→    }
   301→    views.set(b.id, view);
   302→  }
   303→
   304→  // ── Forjado interior CONTINUO del cubo de escalera (v1.18): a nivel de entrada de cada
   305→  //  planta, una losa que cubre la huella completa del cubo (spanW celdas). La authoría glTF
   306→  //  solo trae el forjado de la puerta; sin esta losa se camina (y se ve el vacío) entre la
   307→  //  puerta y los tramos. Malla propia que SOBREVIVE a la carga del modelo (no está en el
   308→  //  greybox, que se oculta) — 5 mm por debajo del forjado del arte: cero z-fighting.
   309→  const stCfg = v.stairs.floorSlab;
   310→  if (stCfg.enabled) {
   311→    const stMat = mat('escalera_forjado_mat');
   312→    stMat.diffuseColor = Color3.FromHexString(stCfg.color);
   313→    for (const b of Object.values(world.blocks)) {
   314→      const f = content.blocks[b.defId]!.features;
   315→      if (!f || f.stairs !== true) continue;
   316→      const spanW = v.stairs.spanW;
   317→      const sw = spanW * v.cellW - v.gapW;
   318→      const sx = b.cell.col * v.cellW + sw / 2;
   319→      box(`${b.id}_forjado_interior`, sw, stCfg.thickness, v.commonDepth, sx, b.cell.floor * v.cellH + stCfg.topAt - stCfg.thickness / 2 - 0.005, frontZ + v.commonDepth / 2, stMat);
   320→    }
   321→  }
   322→
   323→  // ── Arte final (opcional): modelo glTF por TIPO de bloque (manifiesto assets.blocks). ──
   324→  // Cada instancia se clona del contenedor cacheado y escala a la celda real; si carga, la
   325→  // caja greybox estructural se oculta (el panel/candado de viviendas sigue siendo greybox).
   326→  const blockAssets = content.assets?.blocks;
   327→  if (blockAssets && content.assets) {
   328→    for (const b of Object.values(world.blocks)) {
   329→      const entry = blockAssets[b.defId];
   330→      if (!entry) continue;
   331→      void instantiateBlockAsset(scene, entry, content.assets.base).then((model) => {
   332→        // La escena pudo morir mientras cargaba (restart/Cargar partida): no tocar vistas viejas.
   333→        if (model && scene.isDisposed) {
   334→          model.root.dispose(false, false);
   335→          return;
   336→        }
   337→        if (!model) return; // fallback greybox permanente
   338→        const view = views.get(b.id);
   339→        if (!view || view.model) {
   340→          model.root.dispose(false, false); // bloque ya servido: no dejar instancias sueltas
   341→          return;
   342→        }
   343→        // Dimensiones reales de la celda: el pasillo (tira saliente) tiene geometría propia,
   344→        // espejo exacto de su greybox (ancho completo, canto slab*2, calado corridorDepth).

   644→    const x0 = b.cell.col * v.cellW;
   645→    (layout?.rooms ?? []).forEach((layoutRoom, roomIdx) => {
   646→      const rdef = content.rooms[layoutRoom.defId];
   647→      (rdef?.furnitureSlots ?? []).forEach((slot, slotIdx) => {
   648→        let x: number;
   649→        let z: number;
   650→        const rect = geoRooms[roomIdx]?.rect;
   651→        if (rect) {
   652→          // Fase C: metros locales al rect de la estancia (0,0 = esquina frontal-izquierda).
   653→          x = x0 + rect.x + slot.pos[0];
   654→          z = frontZ + rect.z + slot.pos[1];
   655→        } else {
   656→          // Fallback 1D (layout sin rects): bandas como siempre.
   657→          const w = b.size.w * v.cellW - v.gapW;
   658→          const bandW = w / Math.max(1, layout?.rooms.length ?? 1);
   659→          const roomW = rdef?.size.w ?? 1;
   660→          x = x0 + roomIdx * bandW + (slot.pos[0] / Math.max(0.001, roomW)) * bandW;
   661→          z = frontZ + 0.35 + slotIdx * 0.32;
   662→        }
   663→        const fm = mat(`${b.id}_${layoutRoom.key}_${slot.id}_furn`);
   664→        const mesh = box(`${b.id}_${layoutRoom.key}_${slot.id}_furn`, 1, 1, 1, x, bottom + v.slab, z, fm);
   665→        mesh.setEnabled(false);
   666→        furnViews.set(`${b.id}:${layoutRoom.key}:${slot.id}`, { mesh, mat: fm, x, z, bottom, y: slot.y ?? null });
   667→      });
   668→    });
   669→  }
   670→
   671→  function color(b: BlockInstance): Color3 {
   672→    const base = Color3.FromHexString(content.blocks[b.defId]!.view?.color ?? '#cccccc');
   673→    if (b.kind === 'dwelling' && b.access === 'locked') return Color3.FromHexString(v.lockedColor);
   674→    if (b.kind === 'dwelling' && b.occupancy !== 'occupied') return Color3.Lerp(base, Color3.White(), v.vacantTint);
   675→    if (b.params.elevator_status !== undefined && b.params.elevator_status !== 'ok') return Color3.FromHexString(v.brokenColor);
   676→    return base;
   677→  }
   678→
   679→  /** Color de ESTADO (null = sin estado especial). El greybox lo pinta plano; los modelos lo mezclan. */
   680→  function stateTint(b: BlockInstance): Color3 | null {
   681→    if (b.kind === 'dwelling' && b.access === 'locked') return Color3.FromHexString(v.lockedColor);
   682→    if (b.kind === 'dwelling' && b.occupancy !== 'occupied') return Color3.Lerp(Color3.FromHexString(content.blocks[b.defId]!.view?.color ?? '#cccccc'), Color3.White(), v.vacantTint);
   683→    if (b.params.elevator_status !== undefined && b.params.elevator_status !== 'ok') return Color3.FromHexString(v.brokenColor);
   684→    return null;
   685→  }
   686→
   687→  /** Emissive del bloque: canales de influencia → emissive según color del canal en el catálogo. Se acumulan. */
   688→  function applyEmissive(view: BlockView, b: BlockInstance) {
   689→    let e = new Color3(0, 0, 0);
   690→    for (const [channel, value] of Object.entries(world.channels)) {
   691→      const level = value[b.id];
   692→      if (!level) continue;
   693→      const g = glow[channel];
   694→      if (!g) continue;
   695→      e = e.add(g.scale(Math.min(1, level)));
   696→    }
   697→    for (const m of view.mats) m.emissiveColor = e;
   698→  }
   699→
   700→  /** Color difuso/albedo por material: greybox plano (tintStrength 1) o mezcla sobre el arte. */
   701→  function applyTint(view: BlockView, b: BlockInstance) {
   702→    const state = color(b);
   703→    const override = stateTint(b);
   704→    for (let i = 0; i < view.mats.length; i++) {
   705→      const m = view.mats[i]!;
   706→      const base = view.baseColor[i]!;
   707→      const c = view.tintStrength >= 1 ? state : override ? Color3.Lerp(base, override, view.tintStrength) : base;
   708→      if (m.diffuseColor) m.diffuseColor = c;
   709→      else if (m.albedoColor) m.albedoColor = c;
   710→    }
   711→  }
   712→
   713→  // ── Privacidad de viviendas (v1.17): conjunto de bloques con alguien dentro (el jugador
   714→  //  vía playerInside + cualquier residente presente cuyo location sea el bloque). Se
   715→  //  recalcula por frame en update(): la fachada reacciona al instante cuando un vecino
   716→  //  llega a casa o el jugador entra, sin esperar al siguiente tick.
   717→  let playerInside: string | null = null;
   718→  const privacyOn = v.privacy.enabled;
   719→  const computeOccupancy = (): Set<string> => {
   720→    const occ = new Set<string>();
   721→    if (playerInside) occ.add(playerInside);
   722→    for (const r of Object.values(world.residents)) {
   723→      if (r.present && r.location) occ.add(r.location);
   724→    }
   725→    return occ;
   726→  };
   727→
   728→  /** Visibilidad efectiva de fachada/puerta/candado: candado solo si access='locked';
   729→   *  fachada cerrada si está bloqueada O si la privacidad la cierra (nadie dentro).
   730→   *  paint() la fija por el acceso y el LOD puede ocultarlos encima. */
   731→  function applyPanels(view: BlockView, b: BlockInstance, occ?: Set<string>) {
   732→    const locked = b.kind === 'dwelling' && b.access === 'locked';
   733→    let privacyClosed = false;
   734→    if (privacyOn && b.kind === 'dwelling') {
   735→      const set = occ ?? computeOccupancy();
   736→      privacyClosed = !set.has(b.id);
   737→    }
   738→    const wall = (locked || privacyClosed) && !view.lodHiddenPanels;
   739→    view.front?.setEnabled(wall);
   740→    view.door?.setEnabled(wall);
   741→    view.handle?.setEnabled(wall);
   742→    view.lock?.setEnabled(locked && !view.lodHiddenPanels);
   743→  }
   744→
   745→  /**
   746→   * LOD por distancia (rules.view.lod). Se ejecuta SIEMPRE al final de update(): con foco degrada
   747→   * cada bloque según su distancia, y sin foco (o con lod.enabled=false) restaura el estado completo.
   748→   * Idempotente y sin allocations por frame: solo aritmética, booleanos y asignaciones.
   749→   *
   750→   * Métrica (Vista · LOD por distancia): dx = |centroX del bloque − focus.x| en metros y
   751→   * dy = |planta del bloque − focus.floor| * view.cellH en metros; dist = Math.hypot(dx, dy).
   752→   * Nivel efectivo = todos los niveles de lod.levels cuyo `dist` ≤ distBloque (el `hide` es
   753→   * acumulado: un flag se oculta si cualquier nivel alcanzado lo lista; sin nivel → todo visible).
   754→   */
   755→  function applyLod(focus: { x: number; floor: number } | null) {
   756→    const lod = v.lod;
   757→    for (const id in world.blocks) {
   758→      const b = world.blocks[id]!;
   759→      const view = views.get(id)!;
   760→      let hideEdges = false;
   761→      let hideGlow = false;
   762→      let hidePanels = false;
   763→      if (lod.enabled && focus) {
   764→        const dx = Math.abs(view.centerX - focus.x);
   765→        const dy = Math.abs(view.floor - focus.floor) * v.cellH;
   766→        const dist = Math.hypot(dx, dy);
   767→        for (let i = 0; i < lod.levels.length; i++) {
   768→          const level = lod.levels[i];
   769→          if (level.dist > dist) continue;
   770→          for (let j = 0; j < level.hide.length; j++) {
   771→            const flag = level.hide[j];
   772→            if (flag === 'edges') hideEdges = true;
   773→            else if (flag === 'glow') hideGlow = true;
   774→            else if (flag === 'panels') hidePanels = true;
   775→          }
   776→        }
   777→      }
   778→      // 'edges': interruptor nativo barato — el renderingGroup consulta edgesRenderer.isEnabled cada
   779→      // frame, así que nunca se destruye ni recrea el EdgesRenderer (disable/enableEdgesRendering sí
   780→      // lo harían: el primero lo dispone y el segundo recomputaría las aristas). El grosor/color
   781→      // viven en la malla y sobreviven al apagado. Los modelos glTF no tienen edges (edgesMesh null).
   782→      if (view.edgesMesh?.edgesRenderer) view.edgesMesh.edgesRenderer.isEnabled = !hideEdges;
   783→      // 'glow': pisa el emissive CADA frame (paint() pudo repintar los canales este tick); al soltarlo,
   784→      // una única llamada repinta el cálculo de paint para no esperar al próximo tick.
   785→      if (hideGlow) {
   786→        view.glowForced = true;
   787→        for (let i = 0; i < view.mats.length; i++) view.mats[i]!.emissiveColor = view.baseEmissive[i]!;
   788→      } else if (view.glowForced) {
   789→        view.glowForced = false;
   790→        applyEmissive(view, b);
   791→      }
   792→      // 'panels': solo gira la bandera (la visibilidad la calcula applyPanels); en la transición se
   793→      // aplica ya para no esperar al próximo paint(), que reproducirá el mismo resultado.
   794→      if (view.lodHiddenPanels !== hidePanels) {
   795→        view.lodHiddenPanels = hidePanels;
   796→        applyPanels(view, b);
   797→      }
   798→    }
   799→  }
   800→
   801→  return {
   802→    paint() {
   803→      for (const b of Object.values(world.blocks)) {
   804→        const view = views.get(b.id)!;
   805→        applyTint(view, b);
   806→        applyEmissive(view, b);
   807→        applyPanels(view, b);
   808→      }
   809→      // Interiores 2D (Fase C): suelo de cada estancia según su acceso + tapas de puertas cerradas.
   810→      for (const [blockId, interior] of interiors) {
   811→        const block = world.blocks[blockId];
   812→        if (!block) continue;
   813→        for (const f of interior.floors) {
   814→          const room = block.rooms.find((r) => r.key === f.roomKey);
   815→          const locked = room ? room.access !== 'unlocked' : true;
   816→          const hex = locked ? v.lockedColor : content.rooms[f.defId]?.view?.color ?? '#cccccc';
   817→          f.mat.diffuseColor = Color3.FromHexString(hex);
   818→        }
   819→        for (const c of interior.covers) {
   820→          const anyLocked = c.rooms.some((key) => block.rooms.find((r) => r.key === key)?.access !== 'unlocked');
   821→          c.mesh.setEnabled(anyLocked);
   822→        }
   823→      }
   824→      // Muebles del editor de interiores: visibles/escala/color según furniture.view (catálogo).
   825→      for (const [key, fv] of furnViews) {
   826→        const [blockId, roomKey, slotId] = key.split(':');
   827→        const room = world.blocks[blockId ?? '']?.rooms.find((r) => r.key === roomKey);
   828→        const itemId = room?.furniture[slotId ?? ''] ?? '';
   829→        const fdef = itemId ? content.furniture[itemId] : undefined;
   830→        if (!fdef) {
   831→          fv.mesh.setEnabled(false);
   832→          continue;
   833→        }
   834→        const [iw, ih, idz] = fdef.view.size;
   835→        fv.mesh.setEnabled(true);
   836→        fv.mesh.scaling.set(iw, ih, idz);
   837→        // Con y de anclaje (muebles de pared) cuelga a esa altura; si no, apoyado en el suelo.
   838→        const yBase = fv.y !== null ? fv.bottom + v.slab + v.wall + fv.y : fv.bottom + v.slab + v.wall;
   839→        fv.mesh.position.set(fv.x, yBase + ih / 2, fv.z);
   840→        fv.mat.diffuseColor = Color3.FromHexString(fdef.view.color);
   841→      }
   842→    },
   843→    /** Sacude el candado de una puerta cerrada. */
   844→    shake(blockId: string) {
   845→      shakes.set(blockId, 0.4);
   846→    },
   847→    /**
   848→     * dt en segundos; `focus` es el punto de interés (posición del jugador) para el
   849→     * LOD por distancia (rules.view.lod): applyLod() lo aplica SIEMPRE al final del frame
   850→     * (con focus undefined, p. ej. cámara 'edificio', restaura el estado completo).
   851→     * `inside` es el bloque donde está el jugador ahora (pos.insideBlock): alimenta la
   852→     * privacidad y re-evalúa las fachadas CADA frame — un vecino que llega a casa abre
   853→     * su interior al instante, sin esperar al siguiente tick.
   854→     */
   855→    update(dt: number, focus?: { x: number; floor: number }, inside?: string | null) {
   856→      if (focus) lastFocus = focus;
   857→      if (inside !== undefined) playerInside = inside;
   858→      // PRIVACIDAD (v1.17) cada frame: la fachada de una vivienda sin nadie dentro se cierra
   859→      // con su puerta al pasillo y se abre en cuanto alguien entra (propietario, visita…
   860→      // o un ladrón). Vive en update() y no en paint() para reaccionar incluso con el
   861→      // mundo congelado (modal abierto) y sin esperar al siguiente tick.
   862→      if (privacyOn) {
   863→        const occ = computeOccupancy();
   864→        for (const b of Object.values(world.blocks)) {
   865→          if (b.kind !== 'dwelling') continue;
   866→          applyPanels(views.get(b.id)!, b, occ);
   867→        }
   868→      }
   869→      for (const [id, left] of shakes) {
   870→        const view = views.get(id)!;
   871→        const t = left - dt;
   872→        if (t <= 0) {
   873→          shakes.delete(id);
   874→          view.lock?.position.set(view.lockBase.x, view.lockBase.y, view.lock.position.z);
   875→        } else {
   876→          shakes.set(id, t);
   877→          view.lock?.position.set(view.lockBase.x + Math.sin(t * 60) * 0.06, view.lockBase.y, view.lock.position.z);
   878→        }
   879→      }
   880→      // LOD SIEMPRE al final del frame: con foco aplica el nivel efectivo por bloque,
   881→      // sin foco restaura edges/emissive/paneles al estado completo.
   882→      applyLod(focus ?? null);
   883→    },
   884→  };
   885→}
