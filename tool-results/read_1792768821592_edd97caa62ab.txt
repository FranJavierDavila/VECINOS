   258→   258→  private connectNet(): void {

   260→   260→      onState: (s) => this.applyRemoteState(s),

   263→   263→        this.myResidentId = j.yourResidentId;
   264→   264→        this.isHost = j.isHost;
   265→   265→        useGameStore.getState().setMpSession({ code: j.code, isHost: j.isHost, myResidentId: j.yourResidentId });
   266→   266→        useGameStore.getState().setMpParticipants(j.participants);
   267→   267→      },
   268→   268→      onError: (m) => {
   269→   269→        useGameStore.getState().pushToast(m, 'bad');
   270→   270→      },
   271→   271→      onStatus: (st) => {
   272→   272→        if (!this.net) return;
   273→   273→        if (st === 'connected' && useGameStore.getState().mpStatus === 'connecting') return; // el ack decide
   274→   274→        if (st === 'disconnected' && useGameStore.getState().mpStatus === 'connected') {
   275→   275→          useGameStore.getState().setMpStatus('error');
   276→   276→          useGameStore.getState().setMpError('Conexión perdida con el servidor de la comunidad.');

   278→   278→      },
   279→   279→    });

   293→   293→  /** Adopta el estado del servidor (autoridad). Primera vez: reinicia la escena con ese mundo. */
   294→   294→  private applyRemoteState(s: SessionStatePayload): void {
   295→   295→    if (s.seq <= this.appliedSeq) return;
   296→   296→    this.appliedSeq = s.seq;
   297→   297→    if (!this.world || this.world.scenarioId !== s.world.scenarioId) {
   298→   298→      void this.restart(s.world).then(() => {
   299→   299→        if (!this.world) return;
   300→   300→        this.world.events = s.events.slice();
   301→   301→        this.handleEvents();
   302→   302→        this.reconcile();
   303→   303→        this.syncHud();
   304→   304→      });
   305→   305→      return;
   306→   306→    }
   307→   307→    // Aplica en sitio conservando la identidad del objeto (PlayerLogic/building/actors referencian el mismo World).
   308→   308→    Object.assign(this.world, s.world);
   309→   309→    this.world.events = s.events.slice();
   310→   310→    this.handleEvents();
   311→   311→    this.paintBuilding();
   312→   312→    this.reconcile();
   313→   313→    this.syncHud();
   314→   314→  }

   316→   316→  /** Corrige desvíos de posición del avatar local respecto al mundo del servidor (autoridad). */
   317→   317→  private reconcile(): void {
   318→   318→    if (!this.logic || !this.world || !this.net) return;
   319→   319→    const me = this.world.residents[this.actorId()];
   320→   320→    if (!me) return;
   321→   321→    const pos = this.logic.position();
   322→   322→    const locBlock = me.location ? this.world.blocks[me.location] : null;
   323→   323→    if (locBlock) {
   324→   324→      const insideOk = pos.mode === 'inside' && pos.insideBlock === locBlock.id;
   325→   325→      const corridorOk = pos.mode === 'corridor' && !pos.insideBlock;
   326→   326→      if (!(locBlock.kind === 'dwelling' ? insideOk : corridorOk) || Math.abs(pos.floor - locBlock.cell.floor) > 0.01) {
   327→   327→        this.logic.forceLocation(locBlock.id);
   328→   328→      }
   329→   329→    } else if (pos.mode === 'inside') {
   330→   330→      this.logic.forceLocation(null);
   331→   331→    }
   332→   332→  }

   414→   414→  async save(): Promise<void> {
   415→   415→    if (!this.world) return;
   416→   416→    const store = useGameStore.getState();
   417→   417→    if (this.net) {
   418→   418→      store.pushToast(this.content.rules.multiplayer.texts.remoteSaveDisabled, 'info');
   419→   419→      return;
   420→   420→    }
   421→   421→    store.setSaveState('saving');
   422→   422→    drainEvents(this.world); // la cola de eventos no entra en el guardado
   423→   423→    try {
   424→   424→      const res = await fetch('/api/save', {
   425→   425→        method: 'POST',
   426→   426→        headers: { 'Content-Type': 'application/json' },
   427→   427→        body: JSON.stringify({ slot: 'auto', state: this.world }),
   428→   428→      });
   429→   429→      if (!res.ok) throw new Error(String(res.status));
   430→   430→      store.setSaveState('saved', new Date().toLocaleTimeString('es-ES', { hour: '2-digit', minute: '2-digit' }));

   526→   526→  private frame = (): void => {
   527→   527→    if (!this.engine || !this.scene || !this.world || !this.logic || !this.building || !this.actors || !this.rig) return;
   528→   528→    const dt = Math.min(0.1, this.engine.getDeltaTime() / 1000);
   529→   529→    const store = useGameStore.getState();
   530→   530→    const frozen = selectModalOpen(store) || store.paused;
   531→   531→
   532→   532→    let ticked = false;
   533→   533→    if (!frozen) {
   534→   534→      this.logic.update(dt, this.readInput());
   535→   535→      // En remoto el servidor es la autoridad del reloj: nada de ticks locales.
   536→   536→      if (!this.net) {
   537→   537→        this.clockAcc += dt * store.speed;
   538→   538→        let guard = 0;
   539→   539→        while (this.clockAcc >= 1 && guard < 60) {
   540→   540→          tick(this.world, this.content);
   541→   541→          this.clockAcc -= 1;
   542→   542→          ticked = true;
   543→   543→          guard += 1;
   544→   544→        }
   545→   545→      }
   546→   546→    }
   547→   547→
   548→   548→    if (this.world.events.length > 0) this.handleEvents();
   549→   549→    if (ticked) this.paintBuilding();
   550→   550→
   551→   551→    const now = performance.now();
   552→   552→    if (now - this.lastHudSync > 200) {
   553→   553→      this.lastHudSync = now;
   554→   554→      this.syncHud();
   555→   555→    }
   556→   556→
   557→   557→    const pos = this.logic.position();
   558→   558→    this.actors.update(dt, pos, this.actorId());
   559→   559→    // LOD por distancia: solo con cámaras cercanas (planta/estancia); el plano general (edificio) muestra todo.
   560→   560→    this.building.update(dt, useGameStore.getState().camMode === 'edificio' ? undefined : { x: pos.x, floor: pos.floor });
   561→   561→    this.rig.update(dt, pos);
   562→   562→    this.scene.render();
   563→   563→  };

   800→   800→              bottom: Math.min(76, Math.max(0, (s.pos[1] / Math.max(0.001, cellH)) * 100)),

   802→   802→            };
   803→   803→          });
   804→   804→          return { key: room.key, label: rdef?.label ?? room.key, access: room.access, slots };
   805→   805→        }),
   806→   806→      }));

   809→   809→  private syncHud(): void {
   810→   810→    const world = this.world;
   811→   811→    if (!world) return;
   812→   812→    const player = world.residents[this.actorId()] ?? world.residents[world.playerId];
   813→   813→    if (!player) return;
   814→   814→    const home = player.home;
   815→   815→    const params = visibleParams(world, this.content, home).map((p) => ({ name: p.name, label: p.def.label ?? p.name, value: String(p.value) }));
   816→   816→    const missions = world.missions.map((m) => {
   817→   817→      const def = this.content.missions[m.id]!;
   818→   818→      const step = m.state === 'active' ? def.steps[m.step] : undefined;
   819→   819→      return {
   820→   820→        id: m.id,
   821→   821→        title: def.title,
   822→   822→        state: m.state,
   823→   823→        hint: step?.hint ?? null,
   824→   824→        remaining: m.until !== null ? `quedan ${Math.max(0, m.until - world.tick)} min` : null,
   825→   825→      };
   826→   826→    });
   827→   827→    const travel = travelMinutes(buildCirculation(world, this.content), world.entry, home);
   828→   828→    const elevatorBlock = Object.values(world.blocks).find((b) => b.params.elevator_status !== undefined);
   829→   829→    useGameStore.getState().syncHud({
   830→   830→      clock: { ...world.clock },
   831→   831→      weather: world.weather.state,
   832→   832→      funds: world.community.funds,
   833→   833→      wallet: player.wallet,
   834→   834→      reputation: player.reputation,
   835→   835→      elevator: String(elevatorBlock?.params.elevator_status ?? 'ok'),
   836→   836→      homeLabel: blockLabel(home),
   837→   837→      travelText: Number.isFinite(travel) ? `del portal a tu puerta: ${travel} min` : null,
   838→   838→      objective: this.objectiveUi(),
   839→   839→      prompt: this.promptText(),
   840→   840→      params,
   841→   841→      missions,
   842→   842→      homes: this.homesUi(),
   843→   843→      furnitureConfig: { refundRatio: this.content.rules.furniture.refundRatio, roomUnlockPrice: this.content.rules.furniture.roomUnlock.price },
   844→   844→    });
   845→   845→  }
