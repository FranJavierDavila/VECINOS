    79→  async init(canvas: HTMLCanvasElement): Promise<void> {
    80→    if (this.engine || this.initializing) return;
    81→    this.initializing = true;
    82→    this.canvas = canvas;
    83→
    84→    try {
    85→      // 1) Contenido externo + validación (los errores de referencias se avisan, no abortan).
    86→      const { content, issues } = await loadContent();
    87→      this._content = content;
    88→      this.cutscenePlayer = new CutscenePlayer(content);
    89→      preloadSounds();
    90→      useGameStore.setState({ touchHint: content.rules.touch.hint });
    91→      if (issues.length > 0) {
    92→        useGameStore.getState().pushToast(`Contenido inválido: ${issues.length} errores de validación`, 'bad');
    93→      }
    94→
    95→      // 2) Motor de render (WebGPU con fallback WebGL2). La calidad la elige el jugador en Opciones.
    96→      const { engine, backend } = await createEngine(canvas);
    97→      this.engine = engine;
    98→      engine.setHardwareScalingLevel(QUALITY_SCALE[loadOptions().quality]);
    99→      window.addEventListener('resize', this.onResize);
   100→      await this.restart();
   101→      useGameStore.getState().markReady(backend);
   102→      startGamepadPolling();
   103→      engine.runRenderLoop(this.frame);
   104→      // Gancho de depuración/pruebas E2E (no lo usa la UI).
   105→      (window as unknown as { __vecinos?: unknown }).__vecinos = { store: useGameStore, controller: this };
   106→    } catch (err) {
   107→      console.error('No se pudo arrancar el juego (contenido o motor de render)', err);
   108→      useGameStore.getState().setEngineError(true);
   109→    }
   110→  }

   127→  onKeyDown(code: string): void {
   128→    // Cutscene en curso: E/Espacio/Enter/Escape saltan el espectáculo; nada más pasa.
   129→    if (this.cutscenePlayer?.active) {
   130→      if (code === 'KeyE' || code === 'Space' || code === 'Enter' || code === 'Escape') this.cutsceneSkip();
   131→      return;
   132→    }
   133→    if (this.keys.has(code)) return;
   134→    this.keys.add(code);
   135→    const store = useGameStore.getState();
   136→    if (selectModalOpen(store)) return; // los modales gestionan sus propias teclas
   137→    if (code === 'KeyE') {
   138→      this.interact();
   139→      return;
   140→    }
   141→    const cam = CAM_BY_KEY[code];
   142→    if (cam) this.setCamMode(cam);
   143→  }

   221→  /** Interacción (tecla E o botón virtual ✋): usar el objeto cercano, entrar o salir. */
   222→  interact(): void {
   223→    // El ✋ táctil también salta la cutscene (es la única acción habilitada mientras dura).
   224→    if (this.cutscenePlayer?.active) {
   225→      this.cutsceneSkip();
   226→      return;
   227→    }
   228→    if (!this.logic) return;
   229→    // Ascensor (v1.9): si el prompt anuncia ascensor (sin objetivo de E delante), la E abre el
   230→    // panel de destino. Misma condición que promptText → lo que ves es lo que la E ejecuta.
   231→    if (!this.logic.interactTarget()) {
   232→      const panel = this.logic.elevatorPanelTarget();
   233→      if (panel) {
   234→        useGameStore.getState().setElevatorPanel(panel);
   235→        return;
   236→      }
   237→    }
   238→    const r = this.logic.interact();
   239→    if (r.type === 'entered') this.setCamMode('estancia');
   240→    else if (r.type === 'exited') this.setCamMode('planta');
   241→    else if (r.type === 'used') this.presentInteraction(r.kind);
   242→    this.pump();
   243→  }

   360→  private connectNet(): void {
   361→    this.net = new NetSession({
   362→      onState: (s) => this.applyRemoteState(s),
   363→      onParticipants: (p) => useGameStore.getState().setMpParticipants(p),
   364→      onJoined: (j) => {
   365→        this.myResidentId = j.yourResidentId;
   366→        this.isHost = j.isHost;
   367→        useGameStore.getState().setMpSession({ code: j.code, isHost: j.isHost, myResidentId: j.yourResidentId });
   368→        useGameStore.getState().setMpParticipants(j.participants);
   369→      },
   370→      onError: (m) => {
   371→        useGameStore.getState().pushToast(m, 'bad');
   372→      },
   373→      onStatus: (st) => {
   374→        if (!this.net) return;
   375→        if (st === 'connected' && useGameStore.getState().mpStatus === 'connecting') return; // el ack decide
   376→        if (st === 'disconnected' && useGameStore.getState().mpStatus === 'connected') {
   377→          useGameStore.getState().setMpStatus('error');
   378→          useGameStore.getState().setMpError('Conexión perdida con el servidor de la comunidad.');
   379→        }
   380→      },
   381→    });
   382→  }

   384→  private onHostJoinResult(res: { ok: boolean; code?: string; isHost?: boolean; yourResidentId?: string; error?: string }): void {

   389→      return;
   390→    }
   391→    store.setMpStatus('connected');
   392→    // onJoined ya habrá fijado myResidentId/isHost/código con session:joined.
   393→  }

   395→  /** Adopta el estado del servidor (autoridad). Primera vez: reinicia la escena con ese mundo. */
   396→  private applyRemoteState(s: SessionStatePayload): void {
   397→    if (s.seq <= this.appliedSeq) return;
   398→    this.appliedSeq = s.seq;
   399→    if (!this.world || this.world.scenarioId !== s.world.scenarioId) {
   400→      void this.restart(s.world).then(() => {
   401→        if (!this.world) return;
   402→        this.world.events = s.events.slice();
   403→        this.handleEvents();
   404→        this.reconcile();
   405→        this.syncHud();
   406→      });
   407→      return;
   408→    }
   409→    // Aplica en sitio conservando la identidad del objeto (PlayerLogic/building/actors referencian el mismo World).
   410→    Object.assign(this.world, s.world);
   411→    this.world.events = s.events.slice();
   412→    this.handleEvents();
   413→    this.paintBuilding();
   414→    this.reconcile();
   415→    this.syncHud();
   416→  }

   418→  /** Corrige desvíos de posición del avatar local respecto al mundo del servidor (autoridad). */
   419→  private reconcile(): void {
   420→    if (!this.logic || !this.world || !this.net) return;
   421→    const me = this.world.residents[this.actorId()];
   422→    if (!me) return;
   423→    const pos = this.logic.position();
   424→    const locBlock = me.location ? this.world.blocks[me.location] : null;
   425→    if (locBlock) {
   426→      const insideOk = pos.mode === 'inside' && pos.insideBlock === locBlock.id;
   427→      const corridorOk = pos.mode === 'corridor' && !pos.insideBlock;
   428→      if (!(locBlock.kind === 'dwelling' ? insideOk : corridorOk) || Math.abs(pos.floor - locBlock.cell.floor) > 0.01) {
   429→        this.logic.forceLocation(locBlock.id);
   430→      }
   431→    } else if (pos.mode === 'inside') {
   432→      this.logic.forceLocation(null);
   433→    }
   434→  }

   436→  /** Siguiente frase del teléfono; al colgar responde la llamada en el sim y saca la siguiente de la cola. */
   437→  phoneNext(): void {
   438→    if (this.phoneLine < this.phoneLines.length - 1) {
   439→      this.phoneLine += 1;
   440→      this.showPhoneLine();
   441→      return;
   442→    }
   443→    const mission = this.phoneMission;
   444→    this.phoneLines = [];
   445→    this.phoneMission = '';
   446→    useGameStore.getState().setPhone(null);
   447→    this.actors?.director.setGesture(this.actorId(), null); // cuelgas: se suelta el teléfono
   448→    if (this.world && mission) {
   449→      this.sendCommand({ type: 'answer_call', mission });
   450→    }
   451→    const next = this.phoneQueue.shift();
   452→    if (next) this.openPhone(next.mission, next.dialogId);
   453→  }

   477→  placeFurniture(block: string, room: string, slot: string, item: string): void {

   580→  /** (Re)crea el mundo y la escena. Con `state` restaura una partida guardada. */
   581→  private async restart(state?: World): Promise<void> {
   582→    if (!this.engine || !this.canvas || !this._content) return;
   583→    // Ninguna cutscene sobrevive a un cambio de mundo: cola fuera y teatro desmontado.
   584→    this.cutsceneQueue = [];
   585→    this.cutscenePlayer?.abort();
   586→    const { Color4, DirectionalLight, HemisphericLight, Scene, UniversalCamera, Vector3 } = B();
   587→    const view = this.content.rules.view;
   588→    useGameStore.getState().resetUi();
   589→
   590→    this.world = state ?? createWorld(this.content, this.activeScenarioId());
   591→    this.scenarioOverrides = this.content.scenarios[this.world.scenarioId ?? '']?.minigameOverrides ?? {};
   592→    // Guardados antiguos sin `entry`: la entrada es la zona común de planta baja.
   593→    if (!this.world.entry) {
   594→      this.world.entry = Object.values(this.world.blocks).find((b) => b.cell.floor === 0 && b.kind !== 'dwelling')?.id ?? Object.keys(this.world.blocks)[0]!;
   595→    }
   596→    // Migración de estancias de guardados v0.1: params y ranuras de muebles según catálogo.
   597→    for (const b of Object.values(this.world.blocks)) migrateRooms(this.content, b.rooms);
   598→    this.logic = new PlayerLogic(this.world, this.content, {
   599→      actorId: this.myResidentId ?? undefined,
   600→      send: (cmd) => this.sendCommand(cmd),
   601→    });
   602→
   603→    this.scene?.dispose();
   604→    this.scene = new Scene(this.engine);
   605→    this.scene.clearColor = new Color4(view.sky.r, view.sky.g, view.sky.b, 1);
   606→    new HemisphericLight('ambient', new Vector3(...view.ambient.dir), this.scene).intensity = view.ambient.intensity;
   607→    new DirectionalLight('sun', new Vector3(...view.sun.dir), this.scene).intensity = view.sun.intensity;
   608→
   609→    this.building = createBuilding(this.scene, this.world, this.content);
   610→    this.elevators = createElevators(this.scene, this.world, this.content);
   611→    this.actors = createActors(this.scene, this.world, this.content, this.elevators);
   612→    // Sonda de QA (SOLO dev): escena y controlador accesibles para el E2E automatizado.
   613→    // En producción no se expone nada (la partida queda encapsulada igual que siempre).
   614→    if (process.env.NODE_ENV !== 'production') {
   615→      (window as unknown as { __vecinosQA?: { scene: unknown; controller: unknown; elevators: unknown } }).__vecinosQA = {
   616→        scene: this.scene,
   617→        controller: this,
   618→        elevators: this.elevators,
   619→      };
   620→    }
   621→    // Bocadillos de expresión (v1.10): capa DOM sobre el canvas, recreada con el mundo
   622→    // (referencia al World vivo). Sin emotes.json en el catálogo, director nulo sin capa.
   623→    this.bubbles?.dispose();
   624→    const stage = this.canvas?.parentElement;
   625→    this.bubbles = stage && this.actors ? createBubbles(stage, this.scene, this.engine, this.world, this.content, this.actors.director) : null;
   626→    const blocks = Object.values(this.world.blocks);
   627→    // El cubo de escalera (v1.4) sobresale spanW−1 celdas por la derecha del edificio:
   628→    // el encuadre lo incluye para que la torre no se corte en cámara.
   629→    const hasStairs = blocks.some((b) => this.content.blocks[b.defId]?.features?.stairs === true);
   630→    const cols = Math.max(...blocks.map((b) => b.cell.col + b.size.w)) + (hasStairs ? this.content.rules.view.stairs.spanW - 1 : 0);
   631→    this.rig = new CameraRig(
   632→      new UniversalCamera('cam', new Vector3(0, 0, -30), this.scene),
   633→      this.engine,
   634→      cols,
   635→      Math.max(...blocks.map((b) => b.cell.floor)) + 1,
   636→      view.cellW,
   637→      view.cellH,
   638→    );
   639→    // Re-sincronización de vista (v0.6-b): la cámara sigue la posición del mundo (restaurado o
   640→    // nuevo), no la preferencia anterior. Dentro de una vivienda → estancia; en pasillo, la
   641→    // preferencia salvo que fuera 'estancia' (ahí estás en el pasillo, no en una estancia).
   642→    const pos = this.logic.position();
   643→    const store = useGameStore.getState();
   644→    const wanted: CamMode = pos.mode === 'inside' ? 'estancia' : store.camMode === 'estancia' ? 'planta' : store.camMode;
   645→    this.rig.mode = wanted;
   646→    store.setCamMode(wanted);
   647→
   648→    this.clockAcc = 0;
   649→    this.keys.clear();
   650→    this.phoneLines = [];
   651→    this.phoneMission = '';
   652→    this.phoneQueue = [];
   653→    this.lastHudSync = 0;
   654→
   655→    // Re-llamada (Fase F): si la partida montada tenía pasos de teléfono pendientes, vuelven a
   656→    // sonar. La cola de openPhone ordena el resto sin que ninguna llamada se pierda.
   657→    if (this.world) {
   658→      for (const m of this.world.missions) {
   659→        if (m.state !== 'active') continue;
   660→        const step = this.content.missions[m.id]?.steps[m.step];
   661→        if (step?.type === 'phone_call') this.openPhone(m.id, step.dialog);
   662→      }
   663→    }
   664→
   665→    this.pump();
   666→  }

   674→  private frame = (): void => {
   675→    if (!this.engine || !this.scene || !this.world || !this.logic || !this.building || !this.actors || !this.rig || !this.elevators) return;
   676→    const dt = Math.min(0.1, this.engine.getDeltaTime() / 1000);
   677→    const store = useGameStore.getState();
   678→    const frozen = selectModalOpen(store) || store.paused;
   679→
   680→    let ticked = false;
   681→    if (!frozen) {
   682→      this.logic.update(dt, this.readInput());
   683→      // Ascensor (v1.5): el director mueve cabinas y puertas; el jugador llama con W/S en el
   684→      // núcleo y su escalada EXTERNA la gobierna la cabina (espera → embarco → viaje → salida).
   685→      this.elevators.update(dt);
   686→      const climb = this.logic.climbInfo();
   687→      if (climb) {
   688→        if (!this.logic.externalRequested) {
   689→          const zone = this.logic.zone();
   690→          this.elevators.requestRide(this.actorId(), zone.cell.col, climb.from, climb.to);
   691→          this.logic.externalRequested = true;
   692→        }
   693→        const st = this.elevators.status(this.actorId());
   694→        if (!st) {
   695→          this.logic.finishClimbAtNearest(); // fuera de servicio: aterriza en la planta más cercana (sin teletransporte)
   696→          this.logic.externalRequested = false;
   697→          this.playerBoardT = 0;
   698→        } else {
   699→          this.logic.setClimbProgress(st.phase === 'riding' ? st.progress : 0);
   700→          const inside = st.phase === 'boarding' || st.phase === 'riding';
   701→          this.logic.setClimbZ(inside ? this.content.rules.view.roomZ - this.content.rules.view.commonDepth / 2 + this.content.rules.view.elevator.rideZ : null);
   702→          // El jugador "da un paso" dentro de la cabina un instante tras abrirse la puerta
   703→          // (el zOverride ya lo acerca); avisar al director para que cierre y viaje.
   704→          if (st.phase === 'boarding') {
   705→            this.playerBoardT += dt;
   706→            if (this.playerBoardT > 1.0) this.elevators.notifyBoarded(this.actorId());
   707→          } else {
   708→            this.playerBoardT = 0;
   709→          }
   710→          if (st.phase === 'exiting') {
   711→            this.logic.finishClimb();
   712→            this.logic.externalRequested = false;
   713→          }
   714→        }
   715→      } else if (this.logic.externalRequested) {
   716→        this.logic.externalRequested = false;
   717→      }
   718→      // En remoto el servidor es la autoridad del reloj: nada de ticks locales.
   719→      if (!this.net) {
   720→        this.clockAcc += dt * store.speed;
   721→        let guard = 0;
   722→        while (this.clockAcc >= 1 && guard < 60) {
   723→          tick(this.world, this.content);
   724→          this.clockAcc -= 1;
   725→          ticked = true;
   726→          guard += 1;
   727→        }
   728→      }
   729→    }
   730→
   731→    if (this.world.events.length > 0) this.handleEvents();
   732→    if (ticked) this.paintBuilding();
   733→
   734→    const now = performance.now();
   735→    if (now - this.lastHudSync > 200) {
   736→      this.lastHudSync = now;
   737→      useGameStore.getState().setFps(Math.round(this.engine.getFps()));
   738→      this.syncHud();
   739→    }
   740→
   741→    const pos = this.logic.position();
   742→    this.actors.update(dt, pos, this.actorId());
   743→    // Cutscene (Fase D): sus beats mueven cámara/actores DESPUÉS de los eventos y ANTES del render.
   744→    this.cutscenePlayer?.update(dt);
   745→    // Bocadillos (v1.10): después de actores y cámara — proyectan la posición FINAL del frame.
   746→    this.bubbles?.update(dt, this.actorId());
   747→    // Si la cutscene acaba y queda cola, encadena la siguiente aquí: no dependemos de que el sim
   748→    // genere otro evento para descolgar la siguiente (durante el teatro el sim está congelado).
   749→    if (this.cutsceneQueue.length > 0 && !this.cutscenePlayer?.active) this.pumpCutscene();
   750→    // LOD por distancia: solo con cámaras cercanas (planta/estancia); el plano general (edificio) muestra todo.
   751→    this.building.update(dt, useGameStore.getState().camMode === 'edificio' ? undefined : { x: pos.x, floor: pos.floor });
   752→    this.rig.update(dt, pos);
   753→    this.scene.render();
   754→  };

   896→        }
   897→        case 'minigame_finished': {
   898→          const name = this.content.minigames[String(d.minigame)]?.name ?? String(d.minigame);
   899→          store.pushLog(d.success ? `🎮 Minijuego superado: ${name}` : `🎮 Minijuego fallido: ${name}`, d.success ? 'good' : 'bad');
   900→          break;
   901→        }
   902→        case 'ui_toast':
   903→          store.pushToast(String(d.text), (d.tone as Tone | undefined) ?? 'info');
   904→          break;
   905→        case 'finale':
   906→          store.setFinale(String(d.text));
   907→          break;
   908→        case 'funds_changed': {
   909→          const amount = typeof d.amount === 'number' ? d.amount : 0;
   910→          if ((d.to === 'player' || d.to === 'community') && amount !== 0) {
   911→            store.pushLog(amount > 0 ? `+${amount} € (fondos)` : `${amount} € (fondos)`, amount > 0 ? 'good' : 'bad');
   912→          }
   913→          break;
   914→        }
   915→        default: {
   916→          // Umbral de canal con texto del catálogo (odors, wifi y futuros canales): totalmente data-driven.
   917→          if (d.uiLog === true && typeof d.message === 'string') {
   918→            store.pushLog(String(d.message).replace('{block}', blockLabel(String(d.block))), (d.tone as Tone | undefined) ?? 'info');
   919→          }
   920→          break; // resident_moved, param_changed, activity_ended... demasiado ruido para el registro
   921→        }
   922→      }
   923→      // Cutscenes (Fase D): después del reparto, ¿alguna quiere este evento como disparador?
   924→      this.maybeCutscene(e);
   925→      // Bocadillos (v1.10): reacciones expresivas al evento según emotes.json del catálogo.
   926→      this.bubbles?.onEvent(e);
   927→    }
   928→    this.pumpCutscene();
   929→  }
   930→
   931→  // ───────────────────────── Cutscenes (Fase D) ─────────────────────────
   932→
   933→  /** Si el evento encaja con el trigger de alguna cutscene del catálogo, se encola (máx. 3). */
   934→  private maybeCutscene(e: GameEvent): void {
   935→    const defs = Object.values(this.content.cutscenes);
   936→    if (defs.length === 0) return;
   937→    const d = e.data as Record<string, unknown>;
   938→    const hit = defs.find((c) => {
   939→      const t = c.trigger;
   940→      if (t.on !== e.type) return false;
   941→      if (t.mission !== undefined && t.mission !== String(d.mission ?? '')) return false;
   942→      if (t.step !== undefined && t.step !== String(d.step ?? '')) return false;
   943→      return true;
   944→    });
   945→    if (!hit) return;
   946→    if (this.cutsceneQueue.length >= 3) return; // nunca más de 3 esperando: el teatro no atravaca la partida
   947→    this.cutsceneQueue.push({
   948→      def: hit,
   949→      ctx: {
   950→        mission: typeof d.mission === 'string' ? d.mission : null,
   951→        resident: typeof d.resident === 'string' ? d.resident : null,
   952→        block: typeof d.block === 'string' ? d.block : null,
   953→      },
   954→    });
   955→  }
   956→
   957→  /** Lanza la siguiente cutscene en cola si no hay nada en pantalla (también un modal la pospone). */
   958→  private pumpCutscene(): void {
   959→    if (!this.cutscenePlayer || !this.world || !this.rig || !this.actors || !this.engine || !this.logic) return;
   960→    if (this.cutscenePlayer.active || this.cutsceneQueue.length === 0) return;
   961→    if (selectModalOpen(useGameStore.getState())) return; // no se roba la pantalla a un modal
   962→    const next = this.cutsceneQueue.shift()!;
   963→    this.cutscenePlayer.start(next.def, next.ctx, {
   964→      engine: this.engine,
   965→      rig: this.rig,
   966→      actors: this.actors.director,
   967→      world: this.world,
   968→      playerPos: () => this.logic!.position(),
   969→    });
   970→  }

   974→  private openPhone(mission: string, dialogId: string): void {
   975→    const dlg = this.content.dialogs[dialogId];
   976→    if (!dlg || dlg.lines.length === 0) return;
   977→    // Si ya hay una llamada en pantalla, la nueva espera en cola (dedupe por misión).
   978→    if (this.phoneMission) {
   979→      if (this.phoneMission !== mission && !this.phoneQueue.some((q) => q.mission === mission)) {
   980→        this.phoneQueue.push({ mission, dialogId });
   981→      }
   982→      return;
   983→    }
   984→    this.phoneMission = mission;
   985→    this.phoneLines = dlg.lines;
   986→    this.phoneLine = 0;
   987→    this.actors?.director.setGesture(this.actorId(), 'phone'); // el jugador atiende con el móvil a la oreja
   988→    this.showPhoneLine();
   989→  }
   990→
   991→  private showPhoneLine(): void {
   992→    const line = this.phoneLines[this.phoneLine];
   993→    if (!line) return;
   994→    useGameStore.getState().setPhone({ mission: this.phoneMission, speaker: line.speaker, text: line.text, isLast: this.phoneLine === this.phoneLines.length - 1 });
   995→  }

   999→  /** Nombre visible de un bloque: label del catálogo (data-driven) o etiqueta de vivienda (1B). */
  1000→  private blockName(id: string): string {
  1001→    const b = this.world?.blocks[id];
  1002→    const defLabel = b ? this.content.blocks[b.defId]?.label : undefined;
  1003→    return defLabel ?? blockLabel(id);
  1004→  }

  1006→  private objectiveUi(): { title: string; text: string } | null {
  1007→    if (!this.world) return null;
  1008→    const o = activeObjective(this.world, this.content);
  1009→    if (!o) return null;
  1010→    if (o.state === 'awaiting_choice') return { title: o.title, text: 'El edificio contiene la respiración. Decide qué hacer.' };
  1011→    const s = o.step!;
  1012→    const left = o.untilTick !== null ? ` · quedan ${Math.max(0, o.untilTick - this.world.tick)} min` : '';
  1013→    switch (s.type) {
  1014→      case 'phone_call':
  1015→        return { title: o.title, text: 'Suena el teléfono. Atiéndelo.' };
  1016→      case 'reach':
  1017→        return { title: o.title, text: `Llega a la puerta (${this.blockName(s.block ?? '')}).` };
  1018→      case 'enter':
  1019→        return { title: o.title, text: 'Entra en tu piso con E.' };
  1020→      case 'catch_in_act':
  1021→        return { title: o.title, text: `Pilla al responsable in fraganti${left}.` };
  1022→      default:
  1023→        return { title: o.title, text: o.title };
  1024→    }
  1025→  }
  1026→
  1027→  private promptText(): string | null {
  1028→    const logic = this.logic;
  1029→    const world = this.world;
  1030→    if (!logic || !world) return null;
  1031→    if (logic.position().climbing) return null;
  1032→    // El prompt es SIEMPRE el objetivo real de la tecla E (misma fuente, cero divergencias).
  1033→    const t = logic.interactTarget();
  1034→    if (t) {
  1035→      const ic = this.content.rules.interact;
  1036→      switch (t.kind) {
  1037→        case 'door': {
  1038→          const door = world.blocks[t.blockId]!;
  1039→          if (logic.position().mode === 'inside') return ic.exit;
  1040→          return door.access === 'locked'
  1041→            ? ic.doorLocked.replace('{block}', this.blockName(door.id))
  1042→            : ic.doorEnter.replace('{block}', this.blockName(door.id));
  1043→        }
  1044→        case 'resident':
  1045→          return ic.greet.replace('{name}', this.nameOf(t.residentId));
  1046→        case 'mailbox':
  1047→          return ic.mailbox.prompt;
  1048→        case 'noticeboard':
  1049→          return ic.noticeboard.prompt;
  1050→        case 'laptop':
  1051→          return ic.laptop;
  1052→        case 'bed':
  1053→          return ic.bed;
  1054→      }
  1055→    }
  1056→    const zone = logic.zone();
  1057→    const feat = this.content.blocks[zone.defId]!.features;
  1058→    if (feat?.stairs || feat?.elevator) {
  1059→      const ic = this.content.rules.interact;
  1060→      // Lo que dice el prompt es lo que W/S hace aquí: ascensor operativo manda; si no hay
  1061→      // ascensor utilizable pero sí escalera, escalera; núcleo averiado sin escalera = aviso.
  1062→      if (feat.elevator && world.blocks[zone.id]!.params.elevator_status === 'ok') return ic.elevator;
  1063→      if (feat.stairs) return ic.stairs;
  1064→      return ic.elevatorBroken;
  1065→    }
  1066→    return null;
  1067→  }

  1069→  /** Viviendas del jugador para el editor de interiores (foto serializable, posiciones ya en %). */
  1070→  private homesUi() {
  1071→    const world = this.world!;
  1072→    const cellH = this.content.rules.view.cellH;
  1073→    const mine = this.actorId();
  1074→    return Object.values(world.blocks)
  1075→      .filter((b) => b.kind === 'dwelling' && b.ownerId === mine)
  1076→      .sort((a, b) => a.id.localeCompare(b.id))
  1077→      .map((b) => ({
  1078→        id: b.id,
  1079→        label: blockLabel(b.id),
  1080→        rooms: b.rooms.map((room) => {
  1081→          const rdef = this.content.rooms[room.defId];
  1082→          const slots = (rdef?.furnitureSlots ?? []).map((s) => {
  1083→            const itemId = room.furniture[s.id] ?? '';
  1084→            const fdef = itemId ? this.content.furniture[itemId] : undefined;
  1085→            return {
  1086→              id: s.id,
  1087→              kind: s.kind,
  1088→              left: Math.min(92, Math.max(2, (s.pos[0] / Math.max(0.001, rdef?.size.w ?? 1)) * 100)),
  1089→              bottom: Math.min(76, Math.max(0, (s.pos[1] / Math.max(0.001, cellH)) * 100)),
  1090→              item: fdef ? { id: fdef.id, name: fdef.name, emoji: fdef.emoji ?? '📦', color: fdef.view.color } : null,
  1091→            };
  1092→          });
  1093→          return { key: room.key, label: rdef?.label ?? room.key, access: room.access, slots };
  1094→        }),
  1095→      }));
  1096→  }

  1098→  private syncHud(): void {
  1099→    const world = this.world;
  1100→    if (!world) return;
  1101→    const player = world.residents[this.actorId()] ?? world.residents[world.playerId];
  1102→    if (!player) return;
  1103→    const home = player.home;
  1104→    const params = visibleParams(world, this.content, home).map((p) => ({ name: p.name, label: p.def.label ?? p.name, value: String(p.value) }));
  1105→    const missions = world.missions.map((m) => {
  1106→      const def = this.content.missions[m.id]!;
  1107→      const step = m.state === 'active' ? def.steps[m.step] : undefined;
  1108→      return {
  1109→        id: m.id,
  1110→        title: def.title,
  1111→        state: m.state,
  1112→        hint: step?.hint ?? null,
  1113→        remaining: m.until !== null ? `quedan ${Math.max(0, m.until - world.tick)} min` : null,
  1114→      };
  1115→    });
  1116→    const travel = travelMinutes(buildCirculation(world, this.content), world.entry, home);
  1117→    const elevatorBlock = Object.values(world.blocks).find((b) => b.params.elevator_status !== undefined);
  1118→    useGameStore.getState().syncHud({
  1119→      clock: { ...world.clock },
  1120→      weather: world.weather.state,
  1121→      funds: world.community.funds,
  1122→      wallet: player.wallet,
  1123→      reputation: player.reputation,
  1124→      elevator: String(elevatorBlock?.params.elevator_status ?? 'ok'),
  1125→      homeLabel: blockLabel(home),
  1126→      travelText: Number.isFinite(travel) ? `del portal a tu puerta: ${travel} min` : null,
  1127→      objective: this.objectiveUi(),
  1128→      prompt: this.promptText(),
  1129→      params,
  1130→      missions,
  1131→      homes: this.homesUi(),
  1132→      furnitureConfig: { refundRatio: this.content.rules.furniture.refundRatio, roomUnlockPrice: this.content.rules.furniture.roomUnlock.price },
  1133→    });
  1134→  }
